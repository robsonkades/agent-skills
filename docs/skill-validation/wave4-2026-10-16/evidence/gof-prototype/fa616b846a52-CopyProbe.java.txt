import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.IdentityHashMap;
import java.util.List;
import java.util.Map;
import java.util.Properties;
import java.util.SortedMap;
import java.util.TreeMap;

public final class CopyProbe {
    static int assertions;
    static void check(boolean condition, String reason) {
        assertions++;
        if (!condition) throw new AssertionError(reason);
    }

    public static void main(String[] args) throws Exception {
        Properties settings = new Properties();
        try (InputStream in = Files.newInputStream(Path.of(args[0], "application.properties"))) {
            settings.load(in);
        }
        boolean expectedAlias = settings.getProperty("copy.context").equals("operation");
        GraphCopier copier = new GraphCopier();
        GraphCopier.Copier selected = copier.configured(settings);
        var source = DraftEditor.configuredDraft();
        var first = selected.copy(source);
        var second = selected.copy(source);
        check(first != source, "fresh draft");
        check(first.left() != source.left() && first.right() != source.right(), "fresh roots");
        check(first.left().children() != source.left().children(), "fresh owned list");
        check(first.left().value() == source.left().value(), "immutable payload may be shared");
        check(first.left().children().get(0) != source.left().children().get(0), "fresh shared node");
        check(first.left().children().get(0).children().get(0) == first.left(), "cycle retained");
        var equal1 = first.left().children().get(1);
        var equal2 = first.left().children().get(2);
        check(equal1.equals(equal2), "equals-equal fixture nodes");
        check(equal1 != equal2, "distinct equals-equal source nodes stay distinct");
        check(first.left().children().get(0) != second.left().children().get(0), "fresh context for next operation");
        boolean actualAlias = first.left().children().get(0) == first.right().children().get(0);
        check(actualAlias == expectedAlias, "selected scope has expected alias outcome");
        DraftEditor.appendSharedNote(first);
        check(DraftEditor.notesVisibleFromRight(first) == (expectedAlias ? 2 : 1), "consumer sees shared update only with correct context");
        check(source.left().children().get(0).children().size() == 1, "source graph untouched");
        check(second.left().children().get(0).children().size() == 1, "other copy untouched");

        check(copier.copy(null, new IdentityHashMap<>(), 0) == null, "null node");
        var deep = new GraphCopier.Node("0");
        var cursor = deep;
        for (int i = 1; i <= 65; i++) {
            var next = new GraphCopier.Node(Integer.toString(i));
            cursor.add(next);
            cursor = next;
        }
        try { copier.copy(deep, new IdentityHashMap<>(), 0); throw new AssertionError("missing depth guard"); }
        catch (IllegalArgumentException expected) { check(expected.getMessage().contains("depth"), "bounded depth failure"); }

        var wide = new GraphCopier.Node("root");
        for (int i = 0; i < 10_000; i++) wide.add(new GraphCopier.Node(Integer.toString(i)));
        try { copier.copy(wide, new IdentityHashMap<>(), 0); throw new AssertionError("missing node guard"); }
        catch (IllegalArgumentException expected) { check(expected.getMessage().contains("node budget"), "bounded node failure"); }

        var identityKeys = new IdentityHashMap<GraphCopier.Node, String>();
        identityKeys.put(equal1, "one");
        identityKeys.put(equal2, "two");
        check(identityKeys.size() == 2, "identity map preserves both nodes");
        check(new HashMap<>(identityKeys).size() == 1, "negative equal-map control collapses distinct nodes");

        SortedMap<String, Integer> sorted = new TreeMap<>(Comparator.reverseOrder());
        sorted.put("a", 1); sorted.put("z", 2);
        check(new TreeMap<>(sorted).firstKey().equals("z"), "SortedMap overload preserves comparator");
        Map<String, Integer> widened = sorted;
        check(new TreeMap<>(widened).firstKey().equals("a"), "Map overload changes to natural ordering");

        var original = new Child();
        original.values.add("value");
        int constructed = Parent.constructed;
        var shallow = original.shallow();
        check(shallow != original && shallow.getClass() == Child.class, "super.clone preserves subtype");
        check(Parent.constructed == constructed, "clone bypasses constructor");
        check(shallow.values == original.values, "default field copying aliases mutable list");
        var repaired = original.clone();
        check(repaired.getClass() == Child.class, "adequate compatible public clone retains subtype");
        check(repaired.values != original.values, "adequate clone repairs owned mutable list");
        repaired.values.add("copy-only");
        check(original.values.size() == 1, "adequate clone preserves source ownership");

        System.out.println("Java=" + System.getProperty("java.version"));
        System.out.println("context=" + settings.getProperty("copy.context"));
        System.out.println("crossRootAlias=" + actualAlias);
        System.out.println("originalContract=" + (actualAlias ? "satisfied for tested graph" : "violated: shared node duplicated across roots"));
        System.out.println("assertions=" + assertions);
        System.out.println("LIMIT: bounded in-memory Java checks; no concurrency/ORM/persistence execution or agent evaluation.");
    }

    static class Parent implements Cloneable {
        static int constructed;
        List<String> values = new ArrayList<>();
        Parent() { constructed++; }
        public Parent shallow() throws CloneNotSupportedException { return (Parent) super.clone(); }
        public Parent clone() throws CloneNotSupportedException {
            Parent result = (Parent) super.clone();
            result.values = new ArrayList<>(values);
            return result;
        }
    }
    static final class Child extends Parent {}
}
