import assert from "node:assert/strict";
import fs from "node:fs";
import path from "node:path";
import {fileURLToPath} from "node:url";
import {spawnSync} from "node:child_process";
import {createHash} from "node:crypto";
const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),"..");
const repo="C:/git/agent-skills", owned="skills/gof-prototype";
const evidence=path.join(root,"evidence");
fs.mkdirSync(evidence,{recursive:true});
const records=[];
function run(label,executable,args) {
 const result=spawnSync(executable,args,{cwd:repo,encoding:"utf8",windowsHide:true});
 const stdoutFile=path.join(evidence,label+".stdout.txt"),stderrFile=path.join(evidence,label+".stderr.txt");
 fs.writeFileSync(stdoutFile,result.stdout??"");fs.writeFileSync(stderrFile,result.stderr??"");
 const record={label,executable,args,workingDirectory:repo,exitCode:result.status,signal:result.signal,error:result.error?.message??null,stdoutFile,stderrFile};
 records.push(record);fs.writeFileSync(path.join(evidence,"commands.json"),JSON.stringify(records,null,2)+"\n");
 assert.equal(result.status,0,JSON.stringify(record));
 return result.stdout;
}
const a=path.join(root,"inputs/case-a"),b=path.join(root,"inputs/case-b");
assert.deepEqual(fs.readdirSync(a).sort(),fs.readdirSync(b).sort());
const differences=fs.readdirSync(a).filter(f=>fs.readFileSync(path.join(a,f),"utf8")!==fs.readFileSync(path.join(b,f),"utf8"));
assert.deepEqual(differences,["application.properties"]);
assert.equal(fs.readFileSync(path.join(b,"application.properties"),"utf8"),fs.readFileSync(path.join(a,"application.properties"),"utf8").replace("field","operation"));
const ref=fs.readFileSync(path.join(repo,owned,"references/copying-in-java.md"),"utf8");
const snippet=ref.match(/\x60\x60\x60java\r?\n(Node copy\(Node n, IdentityHashMap<Node, Node> seen, int depth\) \{[\s\S]*?)\x60\x60\x60/)[1].trim().replaceAll("\r\n","\n");
const javaSource=fs.readFileSync(path.join(a,"GraphCopier.java"),"utf8");
const embedded=javaSource.slice(javaSource.indexOf("    Node copy("),javaSource.lastIndexOf("\n}")).split("\n").map(l=>l.slice(4)).join("\n").trim();
assert.equal(embedded,snippet,"fixture uses the exact shipped graph copy method");
fs.writeFileSync(path.join(evidence,"pair-and-snippet-check.json"),JSON.stringify({identicalFiles:4,changedFile:"application.properties",difference:"copy.context=field versus operation",shippedGraphMethodByteEquivalentAfterIndentation:true},null,2)+"\n");
const jdk="C:/Users/robso/.jdks/temurin-25.0.3/bin";
for(const id of ["case-a","case-b"]) {
 const classes=path.join(root,"classes",id);fs.mkdirSync(classes,{recursive:true});
 run(id+"-compile",jdk+"/javac.exe",["--release","17","-d",classes,path.join(root,"inputs",id,"GraphCopier.java"),path.join(root,"inputs",id,"DraftEditor.java"),path.join(root,"evaluator","CopyProbe.java")]);
 run(id+"-probe",jdk+"/java.exe",["-cp",classes,"CopyProbe",path.join(root,"inputs",id)]);
}
run("structural",process.execPath,["--input-type=module","--eval",fs.readFileSync(path.join(root,"evaluator","structural-source.txt"),"utf8")]);
run("format",process.execPath,["node_modules/prettier/bin/prettier.cjs","--check",owned+"/SKILL.md",owned+"/skill.yaml",owned+"/references/copying-in-java.md",owned+"/references/worked-example.md"]);
run("diff-check","git",["diff","--check","--",owned]);
run("final-diff","git",["diff","--",owned]);
run("final-status","git",["status","--short","--",owned]);
run("staged-diff","git",["diff","--cached","--",owned]);
const hash=p=>createHash("sha256").update(fs.readFileSync(p)).digest("hex");
const hashes=[];
for(const file of ["SKILL.md","skill.yaml","references/copying-in-java.md","references/worked-example.md"]) {
 const source=path.join(repo,owned,file),final=path.join(evidence,"final",file),committed=path.join(evidence,"committed",file);
 fs.mkdirSync(path.dirname(final),{recursive:true});fs.mkdirSync(path.dirname(committed),{recursive:true});
 fs.copyFileSync(source,final);
 fs.writeFileSync(committed,run("committed-"+file.replaceAll("/","_"),"git",["show","HEAD:"+owned+"/"+file]));
 hashes.push({path:owned+"/"+file,sha256:hash(source)});
}
fs.writeFileSync(path.join(evidence,"final-hashes.json"),JSON.stringify(hashes,null,2)+"\n");
console.log(JSON.stringify({commands:records.length,allPassed:true,finalHashes:hashes,evidenceDirectory:evidence,subprocessesRemaining:0},null,2));
