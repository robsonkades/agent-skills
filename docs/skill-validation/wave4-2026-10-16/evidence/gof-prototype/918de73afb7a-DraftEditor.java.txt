public final class DraftEditor {
    public static GraphCopier.Draft configuredDraft() {
        var shared = new GraphCopier.Node("shared");
        var left = new GraphCopier.Node("left");
        var right = new GraphCopier.Node("right");
        left.add(shared);
        right.add(shared);
        shared.add(left);
        left.add(new GraphCopier.Node("same-value"));
        left.add(new GraphCopier.Node("same-value"));
        return new GraphCopier.Draft(left, right);
    }

    public static void appendSharedNote(GraphCopier.Draft working) {
        working.left().children().get(0).add(new GraphCopier.Node("note"));
    }

    public static int notesVisibleFromRight(GraphCopier.Draft working) {
        return working.right().children().get(0).children().size();
    }
}
