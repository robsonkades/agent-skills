import java.util.ArrayList;
import java.util.IdentityHashMap;
import java.util.List;
import java.util.Objects;
import java.util.Properties;

public final class GraphCopier {
    public record Draft(Node left, Node right) {}

    public static final class Node {
        private final String value;
        private final List<Node> children = new ArrayList<>();
        public Node(String value) { this.value = Objects.requireNonNull(value); }
        public String value() { return value; }
        public List<Node> children() { return children; }
        public void add(Node node) { children.add(node); }
        public boolean equals(Object other) {
            return other instanceof Node node && value.equals(node.value);
        }
        public int hashCode() { return value.hashCode(); }
    }

    public interface Copier {
        Draft copy(Draft source);
    }

    public Copier configured(Properties config) {
        return switch (config.getProperty("copy.context")) {
            case "field" -> source -> new Draft(
                copy(source.left(), new IdentityHashMap<>(), 0),
                copy(source.right(), new IdentityHashMap<>(), 0));
            case "operation" -> source -> {
                var seen = new IdentityHashMap<Node, Node>();
                return new Draft(copy(source.left(), seen, 0), copy(source.right(), seen, 0));
            };
            default -> throw new IllegalArgumentException("Unsupported copy.context");
        };
    }

    Node copy(Node n, IdentityHashMap<Node, Node> seen, int depth) {
        if (n == null) return null;
        if (depth > 64) throw new IllegalArgumentException("copy depth exceeded");
        Node existing = seen.get(n);
        if (existing != null) return existing;
        if (seen.size() >= 10_000) throw new IllegalArgumentException("copy node budget exceeded");
        Node copy = new Node(n.value()); // value must be immutable or copied by its own policy
        seen.put(n, copy);                    // register BEFORE recursing
        n.children().forEach(c -> copy.add(copy(c, seen, depth + 1)));
        return copy;
    }
}
