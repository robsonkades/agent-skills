function run(label, executable, args, expectedExit = 0, diagnostic) {
  const result = spawnSync(executable, args, { cwd: work, env, encoding: 'utf8', timeout: 30000 });
  const entry = { label, executable, args, cwd: work, expectedExit, actualExit: result.status, stdout: result.stdout, stderr: result.stderr, error: result.error?.message, signal: result.signal };
  commands.push(entry);
  fs.writeFileSync(path.join(work, 'results.json'), JSON.stringify({ work, commands }, null, 2));
  assert.equal(result.error, undefined, label);
  assert.equal(result.status, expectedExit, `${label}: ${result.stderr}`);
  if (diagnostic) assert.match(result.stdout + result.stderr, diagnostic, label);
}
function write(relative, text) {
  const target = path.join(work, relative);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, text);
}
