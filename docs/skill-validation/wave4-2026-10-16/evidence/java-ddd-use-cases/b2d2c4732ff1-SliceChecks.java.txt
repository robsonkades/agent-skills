package fixture;
import com.example.application.*;
import com.example.application.order.submit.*;
import com.example.domain.order.*;
import com.example.domain.tenant.TenantID;
public class SliceChecks {
    static final TenantID A = new TenantID("A"), B = new TenantID("B");
    static final OrderID ID = OrderID.from("order-1");
    static int checks, scenarios;
    static class Denied extends RuntimeException {}
    static class Access implements SubmitOrderAccess {
        boolean actionDenied, resourceDenied;
        public void requireActionAllowed(Actor actor) { if (actionDenied) throw new Denied(); }
        public void requireResourceAllowed(Actor actor, Order order) { if (resourceDenied || !actor.tenantId().equals(order.getTenant())) throw new Denied(); }
    }
    static class Context implements ActorContextProvider {
        Actor actor = new Actor("trusted", A);
        public Actor current() { return actor; }
    }
    static class Fixture {
        FakeGateway gateway = new FakeGateway();
        Access access = new Access();
        Context context = new Context();
        DefaultSubmitOrderUseCase useCase = new DefaultSubmitOrderUseCase(gateway, context, access);
        Fixture() { gateway.put(new Order(ID, A, 4, Order.Status.DRAFT, false)); }
        SubmitOrderOutput submit(long version) { return useCase.execute(SubmitOrderCommand.with(ID.value(), version)); }
    }
    static void require(boolean condition, String message) { checks++; if (!condition) throw new AssertionError(message); }
    static void rejects(Class<? extends Throwable> expected, Runnable action) {
        checks++;
        try { action.run(); } catch (Throwable failure) {
            if (expected.isInstance(failure)) return;
            throw new AssertionError("expected " + expected.getSimpleName() + " but got " + failure, failure);
        }
        throw new AssertionError("expected " + expected.getSimpleName());
    }
    static void untouched(Fixture f, String message) {
        require(f.gateway.writes == 0, message + ": wrote storage");
        require(f.gateway.stored(A, ID).getStatus() == Order.Status.DRAFT, message + ": changed stored state");
    }
    public static void main(String[] args) {
        { scenarios++; Fixture f = new Fixture(); SubmitOrderOutput out = f.submit(4);
          require(out.orderId().equals(ID.value()) && out.status().equals("SUBMITTED") && out.version() == 5, "success output");
          require(f.gateway.writes == 1 && f.gateway.stored(A, ID).getVersion() == 5, "conditional write"); }
        { scenarios++; Fixture f = new Fixture(); f.gateway.rows.clear(); rejects(NotFoundException.class, () -> f.submit(4)); require(f.gateway.updates == 0 && f.gateway.rows.isEmpty(), "unknown ID created"); }
        { scenarios++; Fixture f = new Fixture(); f.context.actor = new Actor("other", B); rejects(NotFoundException.class, () -> f.submit(4)); untouched(f, "foreign tenant"); }
        { scenarios++; Fixture f = new Fixture(); f.access.actionDenied = true; rejects(Denied.class, () -> f.submit(4)); require(f.gateway.reads == 0, "denied action read"); untouched(f, "action denial"); }
        { scenarios++; Fixture f = new Fixture(); f.access.resourceDenied = true; rejects(Denied.class, () -> f.submit(4)); untouched(f, "resource denial"); }
        { scenarios++; Fixture f = new Fixture(); rejects(StaleOrderException.class, () -> f.submit(3)); require(f.gateway.updates == 0, "early conflict wrote"); untouched(f, "stale command"); }
        for (Order.Status status : new Order.Status[] { Order.Status.CANCELLED, Order.Status.SUBMITTED }) {
          scenarios++; Fixture f = new Fixture(); f.gateway.put(new Order(ID, A, 4, status, false)); rejects(IllegalStateException.class, () -> f.submit(4)); require(f.gateway.updates == 0 && f.gateway.stored(A, ID).getStatus() == status, "rejected transition changed state"); }
        { scenarios++; Fixture f = new Fixture(); f.gateway.put(new Order(ID, A, 4, Order.Status.DRAFT, true)); rejects(IllegalStateException.class, () -> f.submit(4)); untouched(f, "empty draft"); }
        { scenarios++; Fixture f = new Fixture(); f.gateway.mode = FakeGateway.Mode.LOAD_OUTAGE; rejects(FakeGateway.Unavailable.class, () -> f.submit(4)); untouched(f, "load outage"); }
        { scenarios++; Fixture f = new Fixture(); f.gateway.mode = FakeGateway.Mode.UPDATE_OUTAGE; rejects(FakeGateway.Unavailable.class, () -> f.submit(4)); untouched(f, "update failure leaked"); }
        { scenarios++; Fixture f = new Fixture(); f.gateway.mode = FakeGateway.Mode.ADVANCE_BEFORE_UPDATE; rejects(StaleOrderException.class, () -> f.submit(4)); untouched(f, "race conflict"); require(f.gateway.stored(A, ID).getVersion() == 5, "lost competing version"); }
        { scenarios++; Fixture f = new Fixture(); f.gateway.mode = FakeGateway.Mode.DELETE_BEFORE_UPDATE; rejects(StaleOrderException.class, () -> f.submit(4)); require(f.gateway.rows.isEmpty() && f.gateway.writes == 0, "recreated deleted order"); }
        { scenarios++; Fixture f = new Fixture(); rejects(NullPointerException.class, () -> f.useCase.execute(null)); rejects(NullPointerException.class, () -> SubmitOrderCommand.with(null, 4)); rejects(IllegalArgumentException.class, () -> SubmitOrderCommand.with(ID.value(), -1)); untouched(f, "invalid command"); }
        { scenarios++; Fixture f = new Fixture(); f.submit(4); f.gateway.put(new Order(ID, B, 4, Order.Status.DRAFT, false)); f.context.actor = new Actor("job", B); f.submit(4); require(f.gateway.writes == 2 && f.gateway.stored(B, ID).getStatus() == Order.Status.SUBMITTED, "retained invocation tenant"); }
        System.out.println("Slice checks passed: " + scenarios + " scenarios, " + checks + " assertions");
    }
}
