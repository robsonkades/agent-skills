import { writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';
const root = dirname(fileURLToPath(import.meta.url));
const urls = [
  'https://jakarta.ee/specifications/persistence/3.2/jakarta-persistence-spec-3.2',
  'https://jakarta.ee/specifications/persistence/3.2/jakarta-persistence-spec-3.2.html',
  'https://raw.githubusercontent.com/spring-projects/spring-data-jpa/3.5.2/spring-data-jpa/src/main/java/org/springframework/data/jpa/domain/Specification.java',
  'https://raw.githubusercontent.com/spring-projects/spring-data-jpa/4.0.0/spring-data-jpa/src/main/java/org/springframework/data/jpa/domain/Specification.java',
  'https://www.postgresql.org/docs/17/functions-comparisons.html',
  'https://www.postgresql.org/docs/17/queries-limit.html',
];
const results = await Promise.all(urls.map(async url => {
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(20000) });
    const body = await response.text();
    return { url, finalUrl: response.url, status: response.status, title: body.match(/<title[^>]*>([\s\S]*?)<\/title>/i)?.[1]?.trim(), bodySha256: createHash('sha256').update(body).digest('hex') };
  } catch (error) { return { url, error: error.message }; }
}));
writeFileSync(resolve(root, 'source-status.json'), JSON.stringify({ consulted: '2026-10-16', results }, null, 2) + '\n');
console.log(JSON.stringify(results, null, 2));
if (results[1].status !== 200 || results.slice(2).some(result => result.status !== 200)) process.exitCode = 1;
