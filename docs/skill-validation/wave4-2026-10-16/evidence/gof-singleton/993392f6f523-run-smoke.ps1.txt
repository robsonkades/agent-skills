$ErrorActionPreference = 'Stop'
$root = $PSScriptRoot
$dependencies = Get-Content -Raw -LiteralPath "$root/fixture-dependencies.json" | ConvertFrom-Json
$classpath = ($dependencies.jars | ForEach-Object { if (-not (Test-Path -LiteralPath $_.path)) { throw "Missing cached dependency: $($_.path)" }; $_.path }) -join ';'
$output = Join-Path $root 'run-output'
New-Item -ItemType Directory -Force -Path "$output/classes", "$output/storage" | Out-Null
$source = @(Get-ChildItem -LiteralPath "$root/src" -Filter '*.java' | ForEach-Object { $_.FullName })
& "$($dependencies.jdk)/bin/javac.exe" --release 17 -cp $classpath -d "$output/classes" @source "$root/test/SmokeTest.java"
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
& "$($dependencies.jdk)/bin/java.exe" "-Dfixture.storage=$output/storage" -cp "$output/classes;$classpath" SmokeTest
exit $LASTEXITCODE
