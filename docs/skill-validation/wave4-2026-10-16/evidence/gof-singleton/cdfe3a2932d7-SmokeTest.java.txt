public final class SmokeTest {
    public static void main(String[] args) throws Exception {
        ArchiveClient managed;
        try (var context = Startup.start()) {
            managed = context.getBean(ArchiveClient.class);
            if (managed != context.getBean(ArchiveClient.class)) throw new AssertionError("lookup identity");
            if (context.getBeanNamesForType(ArchiveClient.class).length != 1) throw new AssertionError("definition count");
            if (!context.getBean(UploadService.class).client().isOpen()) throw new AssertionError("upload usable");
            if (!context.getBean(DownloadService.class).client().isOpen()) throw new AssertionError("download usable");
        }
        if (managed.isOpen() || managed.closeCalls() != 1) throw new AssertionError("managed shutdown");
        System.out.println("PASS five smoke assertions");
    }
}
