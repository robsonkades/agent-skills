import java.io.IOException;
import java.nio.channels.FileChannel;
import java.nio.file.Files;
import java.nio.file.Path;
import static java.nio.file.StandardOpenOption.WRITE;

public final class ArchiveClient implements AutoCloseable {
    private final FileChannel channel;
    private int closeCalls;

    public ArchiveClient(Path directory) throws IOException {
        Files.createDirectories(directory);
        Path file = Files.createTempFile(directory, "archive-client-", ".bin");
        channel = FileChannel.open(file, WRITE);
    }

    public boolean isOpen() { return channel.isOpen(); }
    public int closeCalls() { return closeCalls; }

    @Override
    public void close() throws IOException {
        closeCalls++;
        channel.close();
    }
}
