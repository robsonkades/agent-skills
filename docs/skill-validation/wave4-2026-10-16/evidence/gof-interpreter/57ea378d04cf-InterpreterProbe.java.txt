import java.util.*;
import java.util.function.Predicate;

public final class InterpreterProbe {
public sealed interface Filter permits Filter.And, Filter.Or, Filter.Not, Filter.Comparison {

    record And(Filter left, Filter right) implements Filter {
        public And { Objects.requireNonNull(left); Objects.requireNonNull(right); }
    }
    record Or(Filter left, Filter right) implements Filter {
        public Or { Objects.requireNonNull(left); Objects.requireNonNull(right); }
    }
    record Not(Filter inner) implements Filter {
        public Not { Objects.requireNonNull(inner); }
    }

    record Comparison(Field field, Operator operator, Value value) implements Filter {
        public Comparison {
            Objects.requireNonNull(field); Objects.requireNonNull(operator); Objects.requireNonNull(value);
        }
    }
}

public enum Operator { EQ, NEQ, GT, GTE, LT, LTE, CONTAINS }

static void checkStructure(Filter root) {
    record Pending(Filter node, int depth) {}
    var pending = new ArrayDeque<Pending>();
    pending.push(new Pending(Objects.requireNonNull(root), 1));
    int nodes = 0;
    while (!pending.isEmpty()) {
        var next = pending.pop();
        if (next.depth() > 16 || ++nodes > 100) {
            throw new IllegalArgumentException("filter exceeds depth/node budget");
        }
        int depth = next.depth() + 1;
        switch (next.node()) {
            case Filter.And a -> {
                pending.push(new Pending(a.right(), depth));
                pending.push(new Pending(a.left(), depth));
            }
            case Filter.Or o -> {
                pending.push(new Pending(o.right(), depth));
                pending.push(new Pending(o.left(), depth));
            }
            case Filter.Not n -> pending.push(new Pending(n.inner(), depth));
            case Filter.Comparison c -> { }
        }
    }
}

static boolean matches(Filter filter, Document doc) {
    return switch (filter) {
        case Filter.And a -> matches(a.left(), doc) && matches(a.right(), doc);
        case Filter.Or o -> matches(o.left(), doc) || matches(o.right(), doc);
        case Filter.Not n -> !matches(n.inner(), doc);
        case Filter.Comparison c -> compare(doc.valueOf(c.field()), c.operator(), c.value());
    };
}

static List<Issue> validate(Filter filter, Set<Field> permitted) {
    return switch (filter) {
        case Filter.And a -> concat(validate(a.left(), permitted), validate(a.right(), permitted));
        case Filter.Or o -> concat(validate(o.left(), permitted), validate(o.right(), permitted));
        case Filter.Not n -> validate(n.inner(), permitted);
        case Filter.Comparison c -> {
            if (!permitted.contains(c.field())) {
                yield List.of(Issue.forbiddenField(c.field()));
            }
            var issues = new ArrayList<Issue>();
            if (!c.field().supports(c.operator())) {
                issues.add(Issue.badOperator(c.field(), c.operator()));
            } else if (!c.field().accepts(c.operator(), c.value())) {
                issues.add(Issue.typeMismatch(c.field(), c.operator(), c.value()));
            }
            yield issues;
        }
    };
}

static Predicate<Document> compile(Filter filter) {
    return switch (filter) {
        case Filter.And a -> { var l = compile(a.left()); var r = compile(a.right());
                        yield d -> l.test(d) && r.test(d); }
        case Filter.Or o -> { var l = compile(o.left()); var r = compile(o.right());
                       yield d -> l.test(d) || r.test(d); }
        case Filter.Not n -> { var i = compile(n.inner()); yield d -> !i.test(d); }
        case Filter.Comparison c -> { var field = c.field(); var op = c.operator(); var v = c.value();
                               yield d -> compare(d.valueOf(field), op, v); }
    };
}

    sealed interface Value permits Text, Texts, Count { }
    record Text(String value) implements Value { }
    record Texts(List<String> values) implements Value {
        Texts { values = List.copyOf(values); }
    }
    record Count(int value) implements Value { }
    private static int hiddenMetadataCalls;
    private static int acceptsCalls;
    enum Field {
        STATUS, TAGS, SECRET;
        boolean supports(Operator op) {
            if (this == SECRET) { hiddenMetadataCalls++; throw new AssertionError("Hidden metadata read"); }
            return this == STATUS ? op == Operator.EQ || op == Operator.NEQ
                    : op == Operator.EQ || op == Operator.CONTAINS;
        }
        boolean accepts(Operator op, Value value) {
            if (this == SECRET) { hiddenMetadataCalls++; throw new AssertionError("Hidden metadata read"); }
            acceptsCalls++;
            return this == TAGS && op == Operator.EQ ? value instanceof Texts : value instanceof Text;
        }
    }
    record Document(Map<Field, Value> values) {
        Document { values = Map.copyOf(values); }
        Value valueOf(Field field) { return values.get(field); }
    }
    record Issue(String kind, Field field) {
        static Issue forbiddenField(Field field) { return new Issue("forbidden", field); }
        static Issue badOperator(Field field, Operator op) { return new Issue("operator", field); }
        static Issue typeMismatch(Field field, Operator op, Value value) { return new Issue("type", field); }
    }
    static List<Issue> concat(List<Issue> left, List<Issue> right) {
        var result = new ArrayList<Issue>(left);
        result.addAll(right);
        return result;
    }
    static final class EvaluationFailure extends RuntimeException { }
    static final class BudgetExceeded extends RuntimeException { }
    static boolean compare(Value actual, Operator op, Value expected) {
        if (expected instanceof Text t && t.value().equals("ERROR")) throw new EvaluationFailure();
        if (expected instanceof Text t && t.value().equals("BUDGET")) throw new BudgetExceeded();
        if (actual == null) return false;
        return switch (op) {
            case EQ -> actual.equals(expected);
            case NEQ -> !actual.equals(expected);
            default -> throw new IllegalArgumentException("Probe comparator covers EQ/NEQ only");
        };
    }
    private static int checks;
    static void check(boolean value, String message) {
        checks++;
        if (!value) throw new AssertionError(message);
    }
    static void expects(Class<? extends Throwable> type, Runnable work, String label) {
        checks++;
        try { work.run(); }
        catch (Throwable failure) {
            if (type.isInstance(failure)) return;
            throw new AssertionError(label + ": unexpected failure", failure);
        }
        throw new AssertionError(label + ": expected " + type.getSimpleName());
    }
    static Filter comparison(String value) {
        return new Filter.Comparison(Field.STATUS, Operator.EQ, new Text(value));
    }
    static Filter nots(Filter leaf, int count) {
        Filter tree = leaf;
        for (int i=0;i<count;i++) tree = new Filter.Not(tree);
        return tree;
    }
    static Filter shared(Filter leaf, int count) {
        Filter tree = leaf;
        for (int i=0;i<count;i++) tree = new Filter.And(tree, tree);
        return tree;
    }
    // Deliberately incorrect negative-control mutation, separate from extracted folds.
    static boolean masksFailure(Filter filter, Document doc) {
        return switch (filter) {
            case Filter.Not n -> !masksFailure(n.inner(), doc);
            default -> {
                try { yield matches(filter, doc); }
                catch (EvaluationFailure failure) { yield false; }
            }
        };
    }
    public static void main(String[] args) {
        var active = new Document(Map.of(Field.STATUS, new Text("ACTIVE")));
        var paused = new Document(Map.of(Field.STATUS, new Text("PAUSED")));
        var missing = new Document(Map.of());
        Filter yes = comparison("ACTIVE"), no = comparison("PAUSED");
        Filter error = comparison("ERROR"), budget = comparison("BUDGET");
        if (args.length > 0 && args[0].equals("mask-errors")) {
            expects(EvaluationFailure.class, () -> masksFailure(new Filter.Not(error), active),
                    "Masking failure must not turn NOT into an accepted decision");
            return;
        }
        checkStructure(nots(yes, 15));
        check(true, "Depth 16 accepted");
        expects(IllegalArgumentException.class, () -> checkStructure(nots(yes, 16)), "Depth 17 rejected");
        expects(IllegalArgumentException.class, () -> checkStructure(nots(yes, 5000)), "Deep direct AST rejected iteratively");
        checkStructure(shared(yes, 5));
        check(true, "63 occurrences accepted");
        expects(IllegalArgumentException.class, () -> checkStructure(shared(yes, 6)), "127 occurrences rejected despite only 7 distinct nodes");
        expects(NullPointerException.class, () -> new Filter.Not(null), "Null child rejected");

        check(matches(yes, active), "Equal operand true");
        check(!matches(yes, paused), "Different operand false");
        check(!matches(yes, missing), "Defined missing-value comparison false");
        check(matches(new Filter.Not(yes), missing), "Defined missing value differs from failure");
        expects(EvaluationFailure.class, () -> matches(new Filter.Not(error), active), "NOT propagates reached failure");
        expects(EvaluationFailure.class, () -> compile(new Filter.Not(error)).test(active), "Compiled NOT propagates failure");
        check(!matches(new Filter.And(no, error), active), "AND skips unreached error");
        check(matches(new Filter.Or(yes, error), active), "OR skips unreached error");
        check(!compile(new Filter.And(no, error)).test(active), "Compiled AND skips unreached error");
        check(compile(new Filter.Or(yes, error)).test(active), "Compiled OR skips unreached error");
        expects(EvaluationFailure.class, () -> matches(new Filter.And(yes, error), active), "AND reaches error");
        expects(EvaluationFailure.class, () -> compile(new Filter.Or(no, error)).test(active), "Compiled OR reaches error");
        expects(BudgetExceeded.class, () -> matches(new Filter.Not(budget), active), "Budget signal preserved by NOT");
        expects(BudgetExceeded.class, () -> compile(new Filter.Not(budget)).test(active), "Compiled budget signal preserved");
        check(!matches(new Filter.And(no, budget), active), "Unreached budget primitive skipped");
        check(!compile(new Filter.And(no, budget)).test(active), "Compiled unreached budget primitive skipped");

        var permitted = Set.of(Field.STATUS, Field.TAGS);
        var forbidden = new Filter.Comparison(Field.SECRET, Operator.EQ, new Text("x"));
        var wrongType = new Filter.Comparison(Field.STATUS, Operator.EQ, new Count(3));
        var issues = validate(new Filter.And(new Filter.And(no, forbidden), wrongType), permitted);
        check(issues.size() == 2, "Validation includes unreached forbidden and independent type issue");
        check(issues.get(0).kind().equals("forbidden"), "Forbidden field classified first");
        check(issues.get(1).kind().equals("type"), "Allowed field type issue retained");
        check(hiddenMetadataCalls == 0, "No hidden operator or type metadata read");
        check(validate(new Filter.Comparison(Field.TAGS, Operator.EQ, new Texts(List.of("a"))), permitted).isEmpty(), "Collection equality takes collection");
        check(validate(new Filter.Comparison(Field.TAGS, Operator.CONTAINS, new Text("a")), permitted).isEmpty(), "Membership takes element");
        check(validate(new Filter.Comparison(Field.TAGS, Operator.EQ, new Text("a")), permitted).size() == 1, "Collection equality rejects element");
        check(validate(new Filter.Comparison(Field.TAGS, Operator.CONTAINS, new Texts(List.of("a"))), permitted).size() == 1, "Membership rejects collection");
        int before = acceptsCalls;
        check(validate(new Filter.Comparison(Field.STATUS, Operator.GT, new Count(3)), permitted).get(0).kind().equals("operator"), "Unsupported operator rejected");
        check(acceptsCalls == before, "No invented operand contract for unsupported operator");

        var compiled = compile(yes);
        check(compiled.test(active), "First context reads active document");
        check(!compiled.test(paused), "Shared closure reads second document");
        check(compiled.test(active), "Third call does not retain previous caller");
        check(compile(new Filter.Not(yes)).test(missing), "Compiled defined missing semantics retained");
        System.out.println("PASS " + checks + " checks of extracted AST/structure/validation/evaluation/closure code");
        System.out.println("Scope: fixture domain types; no parser, SQL, SpEL runtime, concurrency or performance validation.");
    }

}
