import java.util.Objects;
import java.util.regex.Pattern;

public final class WorkflowApi {
    private static final Pattern TAIL = Pattern.compile("\\s+then\\s+(accept|reject)\\s*");
    private final AtomParser parser = new AtomParser();

    public String execute(String line, AtomParser.Status actual) {
        Objects.requireNonNull(line);
        if (line.length() > 256) throw new IllegalArgumentException("input exceeds 256 characters");
        if (!line.startsWith("when ")) throw new IllegalArgumentException("expected when");
        var parsed = parser.parse(line.substring(5));
        var tail = TAIL.matcher(line.substring(5 + parsed.nextOffset()));
        if (!tail.matches()) throw new IllegalArgumentException("invalid workflow tail");
        return parsed.atom().matches(actual) ? tail.group(1) : "skip";
    }
}
