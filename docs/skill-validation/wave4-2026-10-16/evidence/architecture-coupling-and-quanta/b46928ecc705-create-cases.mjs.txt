const request = 'Review the supplied project evidence and revise coupling-map.md so it accurately describes release and runtime coupling for POST /quote and the named releases. Determine whether the current deployment boundaries can be retained. Keep the task to a scoped map and findings; do not implement a split, release gate or migration. const cases = [];
for (const [name, id, requiredCoordination] of [['case-a', 'AQ-C1', false], ['case-b', 'AQ-C2', true]]) {

  mkdirSync(workspace, { recursive: true });
  for (const [file, value] of Object.entries(common)) writeFileSync(resolve(workspace, file), JSON.stringify(value, null, 2) + '\n');

assert.equal(b.requiredCoordination, true);
delete a.requiredCoordination; delete b.requiredCoordination;
assert.deepEqual(a, b);
const criteria = JSON.parse(readFileSync(resolve(here, 'case-criteria.json'), 'utf8'));
assert.deepEqual(criteria.cases.map(c => c.id), cases.map(c => c.id));
const result = { status: 'PASS', cases, criteria: resolve(here, 'case-criteria.json'),

writeFileSync(resolve(here, 'case-fixture-results.json'), JSON.stringify(result, null, 2) + '\n');
