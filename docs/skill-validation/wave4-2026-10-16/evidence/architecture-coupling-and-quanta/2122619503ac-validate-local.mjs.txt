import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, existsSync, readdirSync, writeFileSync } from 'node:fs';
import { resolve, dirname, relative } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
const here = dirname(fileURLToPath(import.meta.url));
const repo = 'C:/git/agent-skills';
const skill = resolve(repo, 'skills/architecture-coupling-and-quanta');
const { parseManifest } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/skill-document.ts')));
const { manifest, issues } = parseManifest(readFileSync(resolve(skill, 'skill.yaml'), 'utf8'), { strict: true });
const document = parseSkillDocument(readFileSync(resolve(skill, 'SKILL.md'), 'utf8'));
assert.deepEqual(issues, []);
assert.equal(documentName(document), manifest.name);
assert.equal(documentDescription(document), manifest.description);
assert.equal(manifest.name, 'architecture-coupling-and-quanta');
assert.equal(manifest.version, '1.1.5');
for (const entry of manifest.files) assert.ok(existsSync(resolve(skill, entry)), entry);
const baseline = JSON.parse(readFileSync(resolve(here, 'baseline.json'), 'utf8').replace(/^\uFEFF/, ''));
const files = readdirSync(skill, { recursive: true, withFileTypes: true }).filter(e => e.isFile())
  .map(e => relative(skill, resolve(e.parentPath, e.name)).replaceAll('\\', '/')).sort();
assert.deepEqual(files, baseline.files.map(e => e.path).sort());
let relativeLinks = 0;
for (const file of baseline.files) {
  const full = resolve(skill, file.path);
  const bytes = readFileSync(full);
  assert.equal(createHash('sha256').update(bytes).digest('hex'), file.sha256, 'unchanged ' + file.path);
  if (!file.path.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\]\(([^\s)]+)\)/g)) {
    const target = match[1].replace(/^<|>$/g, '');
    if (/^(https?:|#)/.test(target)) continue;
    const [path, fragment] = target.split('#');
    const resolved = resolve(dirname(full), path);
    assert.ok(existsSync(resolved), target);
    if (fragment) {
      const headings = [...readFileSync(resolved, 'utf8').matchAll(/^#{1,6} (.+)$/gm)].map(m => m[1]
        .toLowerCase().replace(/[^\p{L}\p{N}_\- ]/gu, '').replaceAll(' ', '-'));
      assert.ok(headings.includes(fragment), target);
    }
    relativeLinks++;
  }
}
const referredOwners = ['distribution-boundaries', 'enterprise-architecture-smells', 'java-cohesion-coupling',
  'component-and-release-boundaries', 'architecture-fitness-functions', 'architecture-testing'];
const declared = new Set([...(manifest.dependencies ?? []).map(d => d.name), ...(manifest.suggests ?? [])]);
for (const owner of referredOwners) {
  assert.ok(declared.has(owner), 'declared direct referral ' + owner);
  assert.ok(existsSync(resolve(repo, 'skills', owner, 'SKILL.md')), owner);
}
const wave = JSON.parse(readFileSync(resolve(repo, 'docs/skill-validation/wave4-2026-10-16/baseline.json'), 'utf8'));
const wavePackage = wave.packages.find(p => p.name === manifest.name);
assert.equal(wavePackage.initialVersion, manifest.version);
for (const file of wavePackage.files) assert.equal(createHash('sha256').update(readFileSync(resolve(repo, file.path))).digest('hex'), file.sha256);
const result = { status: 'PASS', skill: manifest.name, version: manifest.version, strictManifestIssues: 0,
  metadataAgreement: true, packageFiles: files.length, relativeLinks, directReferralsChecked: referredOwners.length,
  assignmentAndWaveBaselineByteIdentical: true, changedFiles: [] };
writeFileSync(resolve(here, 'structure-results.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify(result, null, 2));
