function run(id, command, args, cwd = root) {
  const started = new Date().toISOString();
  const result = spawnSync(command, args, {cwd, encoding:'utf8', timeout:60000, windowsHide:true, maxBuffer:8*1024*1024});
  const stdout = result.stdout ?? '';
  const stderr = result.stderr ?? '';
  const log = path.join(logs, id + '.txt');
  fs.writeFileSync(log, 'command=' + JSON.stringify([command,...args]) + '\ncwd=' + cwd + '\nexit=' + result.status + '\nSTDOUT\n' + stdout + '\nSTDERR\n' + stderr);
  record({id, command, args, cwd, started, exitCode:result.status, signal:result.signal, error:result.error?.message ?? null, stdout, stderr, log});
  return result;
}
