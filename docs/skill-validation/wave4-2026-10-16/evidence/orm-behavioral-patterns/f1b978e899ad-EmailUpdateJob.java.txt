import java.nio.file.*;
import java.io.*;
import java.util.Properties;
import org.hibernate.Session;
import org.hibernate.Transaction;

public final class EmailUpdateJob {
    record Edit(long id, long expectedVersion, String email) {}
    static OrmFixture.Contact input(OrmFixture db, Edit edit, String adapter) {
        if ("editor".equals(adapter)) {
            OrmFixture.Contact contact = db.readContact(edit.id());
            if (contact.version != edit.expectedVersion())
                throw new jakarta.persistence.OptimisticLockException("Stale edit");
            contact.email = edit.email();
            return contact;
        }
        if ("form".equals(adapter)) {
            OrmFixture.Contact contact = new OrmFixture.Contact(edit.id(), edit.email());
            contact.version = edit.expectedVersion();
            return contact;
        }
        throw new IllegalArgumentException("Unknown adapter: " + adapter);
    }
    static void execute(OrmFixture db, OrmFixture.Contact contact) {
        db.tx(session -> {
            session.merge(contact);
            return null;
        });
    }
    public static void main(String[] args) throws Exception {
        Properties config = new Properties();
        try (Reader reader = Files.newBufferedReader(Path.of(args[0]))) { config.load(reader); }
        try (OrmFixture db = new OrmFixture()) {
            Edit edit = new Edit(7L, 0L, "updated@example.test");
            execute(db, input(db, edit, config.getProperty("input.adapter")));
            System.out.println("RESULT email=" + db.readContact(7).email + " phones=" + db.phoneCount(7));
        }
    }
}
