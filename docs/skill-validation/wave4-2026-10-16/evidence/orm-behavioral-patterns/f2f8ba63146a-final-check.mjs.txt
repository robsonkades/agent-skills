import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {fileURLToPath} from 'node:url';
import {parseManifest} from 'file:///C:/git/agent-skills/packages/core/src/domain/manifest.ts';
const root=path.dirname(fileURLToPath(import.meta.url));
const skill='C:/git/agent-skills/skills/orm-behavioral-patterns';
const baseline=JSON.parse(fs.readFileSync(path.join(root,'baseline.json'),'utf8'));
const checked=[];
for(const file of baseline.files) {
 const relative=path.relative(skill,file.path);
 const current=fs.readFileSync(path.join(skill,relative),'utf8');
 let expected=file.content;
 if(relative==='skill.yaml') expected=expected.replace('version: 1.3.4','version: 1.3.5');
 if(relative.replaceAll('\\','/')==='references/unit-of-work-and-identity-map.md')
  expected=expected.replace('https://jakarta.ee/specifications/persistence/3.2/jakarta-persistence-spec-3.2)',
                            'https://jakarta.ee/specifications/persistence/3.2/jakarta-persistence-spec-3.2.html)');
 assert.equal(current,expected,'Unexpected package alteration: '+relative);
 checked.push({relative,sha256:createHash('sha256').update(current).digest('hex')});
}
const {manifest}=parseManifest(fs.readFileSync(path.join(skill,'skill.yaml'),'utf8'));
for(const ref of [...manifest.dependencies.map(d=>d.name),...manifest.suggests])
 assert(fs.existsSync(path.join(skill,'..',ref,'SKILL.md')),'Missing referred skill '+ref);
const entry=fs.readFileSync(path.join(skill,'SKILL.md'),'utf8');
for(const name of ['lazy-load.md','unit-of-work-and-identity-map.md']) {
 assert(entry.includes(name),'Reference not routed from SKILL: '+name);
 assert(fs.existsSync(path.join(skill,'references',name)));
}
const a=fs.readFileSync(path.join(root,'cases/A.properties'),'utf8');
const b=fs.readFileSync(path.join(root,'cases/B.properties'),'utf8');
assert.equal(a.replace('editor','form'),b,'More than one case constraint changed');
const result={status:'PASS',exactBaselineTransform:true,files:checked,referrals:manifest.dependencies.length+manifest.suggests.length,routedReferences:2,pairedCaseOnlyDifference:'input.adapter'};
fs.writeFileSync(path.join(root,'final-check.json'),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result,null,2));
