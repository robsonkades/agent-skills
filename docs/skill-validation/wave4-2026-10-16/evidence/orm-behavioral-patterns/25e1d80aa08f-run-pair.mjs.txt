import fs from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
import {spawnSync} from 'node:child_process';
import assert from 'node:assert/strict';
const root=path.dirname(fileURLToPath(import.meta.url));
const bin='C:/Users/robso/.jdks/temurin-25.0.3/bin';
const cp=path.join(root,'classes')+path.delimiter+JSON.parse(fs.readFileSync(path.join(root,'dependencies.json'),'utf8')).jars.map(j=>j.file).join(path.delimiter);
const results=JSON.parse(fs.readFileSync(path.join(root,'results.json'),'utf8'));
function run(id,command,args){
 const r=spawnSync(command,args,{cwd:root,encoding:'utf8',windowsHide:true,timeout:60000});
 const log=path.join(root,'logs',id+'.txt');
 fs.writeFileSync(log,'command='+JSON.stringify([command,...args])+'\ncwd='+root+'\nexit='+r.status+'\nSTDOUT\n'+r.stdout+'\nSTDERR\n'+r.stderr);
 results.push({id,command,args,cwd:root,exitCode:r.status,stdout:r.stdout,stderr:r.stderr,signal:r.signal,error:r.error?.message??null,log});
 fs.writeFileSync(path.join(root,'results.json'),JSON.stringify(results,null,2)+'\n');
 assert.equal(r.status,0,id+': '+r.stderr);
 return r;
}
run('compile-pair-final',path.join(bin,'javac.exe'),['--release','17','-cp',cp,'-d',path.join(root,'classes'),path.join(root,'fixtures/EmailUpdateJob.java'),path.join(root,'fixtures/InputVersionCheck.java')]);
for(const [name,count] of [['A',1],['B',0]]){
 const r=run('case-'+name+'-final',path.join(bin,'java.exe'),['-cp',cp,'EmailUpdateJob',path.join(root,'cases',name+'.properties')]);
 assert.match(r.stdout,new RegExp('RESULT email=updated@example.test phones='+count));
}
const r=run('input-version-final',path.join(bin,'java.exe'),['-cp',cp,'InputVersionCheck']);
assert.match(r.stdout,/PASS both input adapters/);
console.log(JSON.stringify({status:'PASS',pair:'A phone preserved; B phone removed',version:'Both adapters reject stale input without changed stored state',commands:4}));
