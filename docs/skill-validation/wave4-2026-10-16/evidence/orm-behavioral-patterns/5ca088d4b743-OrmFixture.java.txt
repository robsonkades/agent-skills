import jakarta.persistence.*;
import org.hibernate.Session;
import org.hibernate.SessionFactory;
import org.hibernate.Transaction;
import org.hibernate.cfg.Configuration;
import java.util.*;
import java.util.function.Function;

public final class OrmFixture implements AutoCloseable {
    @Entity(name = "Contact")
    @Table(name = "contacts")
    public static class Contact {
        @Id public Long id;
        @Version public Long version;
        @Column(nullable = false, unique = true) public String email;
        @OneToMany(mappedBy = "contact", cascade = CascadeType.ALL, orphanRemoval = true)
        public List<Phone> phones = new ArrayList<>();
        public Contact() {}
        public Contact(long id, String email) { this.id = id; this.email = email; }
    }
    @Entity(name = "Phone")
    @Table(name = "phones")
    public static class Phone {
        @Id public Long id;
        @Column(nullable = false) public String number;
        @ManyToOne(fetch = FetchType.LAZY, optional = false) public Contact contact;
        public Phone() {}
        public Phone(long id, String number, Contact contact) {
            this.id = id; this.number = number; this.contact = contact;
        }
    }
    public final SessionFactory factory;
    public OrmFixture() {
        factory = new Configuration()
            .addAnnotatedClass(Contact.class).addAnnotatedClass(Phone.class)
            .setProperty("hibernate.connection.driver_class", "org.h2.Driver")
            .setProperty("hibernate.connection.url", "jdbc:h2:mem:orm_" + UUID.randomUUID())
            .setProperty("hibernate.connection.username", "sa")
            .setProperty("hibernate.connection.password", "")
            .setProperty("hibernate.hbm2ddl.auto", "create-drop")
            .setProperty("hibernate.generate_statistics", "true")
            .setProperty("hibernate.session.events.log", "false")
            .setProperty("hibernate.show_sql", "false")
            .setProperty("hibernate.cache.use_second_level_cache", "false")
            .setProperty("hibernate.cache.use_query_cache", "false")
            .setProperty("hibernate.enable_lazy_load_no_trans", "false")
            .buildSessionFactory();
        tx(s -> {
            for (long id = 7; id <= 9; id++) {
                Contact contact = new Contact(id, "contact" + id + "@example.test");
                contact.phones.add(new Phone(id * 10, "+55-555-000" + id, contact));
                s.persist(contact);
            }
            return null;
        });
    }
    public <T> T tx(Function<Session, T> action) {
        try (Session session = factory.openSession()) {
            Transaction tx = session.beginTransaction();
            try {
                T result = action.apply(session);
                tx.commit();
                return result;
            } catch (RuntimeException | Error error) {
                if (tx.isActive()) tx.rollback();
                throw error;
            }
        }
    }
    public Contact readContact(long id) { return tx(s -> s.find(Contact.class, id)); }
    public long phoneCount(long id) {
        return tx(s -> s.createQuery("select count(p) from Phone p where p.contact.id = :id", Long.class)
            .setParameter("id", id).getSingleResult());
    }
    @Override public void close() { factory.close(); }
}
