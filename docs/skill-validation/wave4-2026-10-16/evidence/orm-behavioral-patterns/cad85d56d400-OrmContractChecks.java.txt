import jakarta.persistence.*;
import org.hibernate.*;
import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;

public final class OrmContractChecks {
    static int assertions;
    static void check(boolean condition, String message) {
        assertions++;
        if (!condition) throw new AssertionError(message);
    }
    static void expect(Class<? extends Throwable> type, Runnable action, String message) {
        assertions++;
        try { action.run(); } catch (Throwable error) {
            if (type.isInstance(error)) return;
            throw new AssertionError(message + ": wrong exception " + error, error);
        }
        throw new AssertionError(message + ": no exception");
    }
    static void applyEmail(OrmFixture db, long id, long expectedVersion, String email) {
        db.tx(s -> {
            var contact = s.find(OrmFixture.Contact.class, id);
            if (contact.version != expectedVersion) throw new OptimisticLockException("Stale edit");
            contact.email = email;
            return null;
        });
    }
    public static void main(String[] args) {
        try (OrmFixture db = new OrmFixture()) {
            var stats = db.factory.getStatistics();
            stats.clear();
            db.tx(s -> {
                var first = s.find(OrmFixture.Contact.class, 7L);
                check(first == s.find(OrmFixture.Contact.class, 7L), "identity lookup");
                check(stats.getPrepareStatementCount() == 1, "second find avoids SQL in fixture");
                var queried = s.createQuery("from Contact where id = 7", OrmFixture.Contact.class).getSingleResult();
                check(first == queried, "query resolves managed identity");
                check(stats.getPrepareStatementCount() == 2, "query is not identity-map result caching");
                first.email = "dirty@example.test";
                return null;
            });
            check(db.readContact(7).email.equals("dirty@example.test"), "dirty checking commits without save");

            OrmFixture.Contact rolledBack;
            try (Session s = db.factory.openSession()) {
                var tx = s.beginTransaction();
                rolledBack = s.find(OrmFixture.Contact.class, 7L);
                rolledBack.email = "rollback@example.test";
                s.flush();
                check(s.createQuery("select email from Contact where id = 7", String.class)
                    .getSingleResult().equals("rollback@example.test"), "flushed row inside transaction");
                tx.rollback();
            }
            check(rolledBack.email.equals("rollback@example.test"), "rollback does not rewind Java");
            check(db.readContact(7).email.equals("dirty@example.test"), "fresh context after rollback");

            var detached = db.readContact(7);
            detached.email = "copied@example.test";
            db.tx(s -> {
                var managed = s.merge(detached);
                check(managed != detached && s.contains(managed) && !s.contains(detached), "merge ownership");
                detached.email = "late-detached@example.test";
                managed.email = "managed@example.test";
                return null;
            });
            check(db.readContact(7).email.equals("managed@example.test"), "returned merge target tracked");

            db.tx(s -> {
                var managed = s.find(OrmFixture.Contact.class, 7L);
                long version = managed.version;
                int count = s.createMutationQuery("update Contact set email = 'bulk@example.test' where id = 7").executeUpdate();
                check(count == 1 && managed.email.equals("managed@example.test"), "bulk leaves identity-map state stale");
                check(s.find(OrmFixture.Contact.class, 7L) == managed, "find keeps existing identity");
                s.refresh(managed);
                check(managed.email.equals("bulk@example.test"), "refresh reconciles fixture");
                check(managed.version == version, "ordinary bulk does not increment version");
                return null;
            });

            var lazy = db.readContact(7);
            check(!Hibernate.isInitialized(lazy.phones), "closed-input collection unfetched");
            db.tx(other -> {
                expect(LazyInitializationException.class, () -> lazy.phones.size(), "another active transaction cannot load old wrapper");
                check(other.find(OrmFixture.Contact.class, 7L).phones.size() == 1, "receiving context loads own graph");
                return null;
            });

            stats.clear();
            db.tx(s -> {
                var contacts = s.createQuery("from Contact order by id", OrmFixture.Contact.class).getResultList();
                check(contacts.size() == 3, "nonempty roots");
                check(contacts.stream().mapToInt(c -> c.phones.size()).sum() == 3, "all lazy data returned");
                return null;
            });
            check(stats.getPrepareStatementCount() == 4, "cold 1+N fixture");
            stats.clear();
            db.tx(s -> {
                var contacts = s.createQuery("from Contact c join fetch c.phones order by c.id", OrmFixture.Contact.class).getResultList();
                check(contacts.size() == 3 && contacts.stream().mapToInt(c -> c.phones.size()).sum() == 3, "bounded fetch returns complete data");
                return null;
            });
            check(stats.getPrepareStatementCount() == 1, "bounded fetch query fixture");

            db.tx(s -> {
                expect(NoResultException.class, () -> s.createQuery("from Contact where id = -1", OrmFixture.Contact.class).getSingleResult(), "no-result branch");
                check(!s.getTransaction().getRollbackOnly(), "caught no-result does not itself mark rollback");
                s.find(OrmFixture.Contact.class, 7L).email = "after-empty@example.test";
                return null;
            });
            check(db.readContact(7).email.equals("after-empty@example.test"), "commit after caught no-result");

            try (Session failed = db.factory.openSession()) {
                var tx = failed.beginTransaction();
                failed.find(OrmFixture.Contact.class, 7L).email = "contact8@example.test";
                expect(PersistenceException.class, failed::flush, "unique violation");
                check(tx.getRollbackOnly(), "failed flush marks rollback");
                failed.clear();
                check(tx.getRollbackOnly(), "clear does not reset failed transaction");
                tx.rollback();
            }
            check(db.readContact(7).email.equals("after-empty@example.test"), "failed write rolled back");

            long version = db.readContact(7).version;
            applyEmail(db, 7, version, "command@example.test");
            check(db.readContact(7).email.equals("command@example.test") && db.phoneCount(7) == 1, "explicit command preserves unrelated data");
            expect(OptimisticLockException.class, () -> applyEmail(db, 7, version, "stale@example.test"), "stale command");
            check(db.readContact(7).email.equals("command@example.test") && db.phoneCount(7) == 1, "stale command has no effect");
        }
        System.out.println("PASS assertions=" + assertions);
    }
}
