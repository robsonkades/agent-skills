import java.util.Properties;

final class ConfirmationController {
    private final DraftSession draft;
    private final DeliveryAction action;
    private DraftSession.Snapshot before;
    ConfirmationController(DraftSession draft, Properties configuration, DeliverySink sink) {
        this.draft = draft;
        this.action = RuntimeFactory.create(configuration, sink);
    }
    void confirm() {
        var candidate = draft.capture();
        action.confirm(draft);
        before = candidate;
    }
    void undoConfirm() {
        if (before == null) throw new IllegalStateException("nothing to undo");
        draft.restore(before);
        before = null;
    }
}
