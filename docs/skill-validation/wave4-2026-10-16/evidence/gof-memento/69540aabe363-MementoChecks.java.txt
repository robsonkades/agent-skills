import java.util.*;
import java.util.concurrent.atomic.*;

public final class MementoChecks {
    record Mark(String label) { }
    record LineItem(String id) { }
    record Adjustment(int amount) { }
    record SettlementOption(String name) { }
    record EditorView(String text, int caret, List<Mark> marks) { }
    record DraftView(List<LineItem> items, List<Adjustment> adjustments, Optional<SettlementOption> settlement, String notes) { }
    static <T> List<T> append(List<T> source, T element) {
        var copy = new ArrayList<T>(source); copy.add(element); return List.copyOf(copy);
    }
    static int assertions;
    static void check(boolean condition, String label) {
        assertions++; if (!condition) throw new AssertionError(label);
    }
    static void rejects(Runnable action, String label) {
        assertions++;
        try { action.run(); throw new AssertionError(label); }
        catch (IllegalArgumentException expected) { }
    }
static final class Editor {
    private String text = "";
    private int caret = 0;
    private List<Mark> marks = new ArrayList<>();
    void fixtureEdit(String text, int caret, List<Mark> marks) {
        this.text = text; this.caret = caret; this.marks = new ArrayList<>(marks);
    }
    EditorView observe() { return new EditorView(text, caret, List.copyOf(marks)); }
    void fixtureMutateList() { marks.add(new Mark("later")); }


    public sealed interface Snapshot permits State { }        // opaque to callers
    private final Object owner = new Object();
    private record State(Object owner, String text, int caret, List<Mark> marks) implements Snapshot {
        @Override public String toString() { return "Editor snapshot"; }
    }

    public Snapshot capture() { return new State(owner, text, caret, List.copyOf(marks)); }

    public void restore(Snapshot snapshot) {
        if (!(snapshot instanceof State state) || state.owner() != owner) {
            throw new IllegalArgumentException("foreign or null snapshot");
        }
        var restoredMarks = new ArrayList<>(state.marks()); // prepare before changing live fields
        this.text = state.text();
        this.caret = state.caret();
        this.marks = restoredMarks;
    }
}

static final class MutableDraft {
    void fixtureEdit(List<LineItem> i, List<Adjustment> a, Optional<SettlementOption> s, String n) {
        items = new ArrayList<>(i); adjustments = new ArrayList<>(a); settlement = s; notes = n;
    }
    DraftView observe() { return new DraftView(List.copyOf(items), List.copyOf(adjustments), settlement, notes); }
    void fixtureMutateLists() { items.add(new LineItem("later")); adjustments.add(new Adjustment(9)); }


    public sealed interface Snapshot permits State { }

    private final Object owner = new Object();

    private record State(Object owner, List<LineItem> items, List<Adjustment> adjustments,
                         Optional<SettlementOption> settlement, String notes) implements Snapshot {
        @Override public String toString() { return "ClaimDraft snapshot"; }
    }

    private List<LineItem> items = new ArrayList<>();
    private List<Adjustment> adjustments = new ArrayList<>();
    private Optional<SettlementOption> settlement = Optional.empty();
    private String notes = "";

    public Snapshot capture() {
        return new State(owner, List.copyOf(items), List.copyOf(adjustments), settlement, notes);
    }

    public void restore(Snapshot snapshot) {
        if (!(snapshot instanceof State state) || state.owner() != owner) {
            throw new IllegalArgumentException("foreign or null snapshot");
        }
        var restoredItems = new ArrayList<>(state.items());
        var restoredAdjustments = new ArrayList<>(state.adjustments());
        this.items = restoredItems;
        this.adjustments = restoredAdjustments;
        this.settlement = state.settlement();
        this.notes = state.notes();
    }
}

static final class ImmutableDraft {
    void fixtureEdit(List<LineItem> i, List<Adjustment> a, Optional<SettlementOption> s, String n) {
        state = new State(owner, List.copyOf(i), List.copyOf(a), s, n);
    }
    DraftView observe() {
        State observed = state;
        return new DraftView(observed.items(), observed.adjustments(), observed.settlement(), observed.notes());
    }
    void fixtureChangeNotes(String notes) { state = state.withNotes(notes); }


    public sealed interface Snapshot permits State { }
    private final Object owner = new Object();
    private volatile State state = new State(owner, List.of(), List.of(), Optional.empty(), "");

    private record State(Object owner, List<LineItem> items, List<Adjustment> adjustments,
                         Optional<SettlementOption> settlement, String notes) implements Snapshot {
        @Override public String toString() { return "ClaimDraft snapshot"; }
        State withNotes(String notes) { return new State(owner, items, adjustments, settlement, notes); }
        State withAdjustmentAdded(Adjustment a) {
            return new State(owner, items, append(adjustments, a), Optional.empty(), notes);
        }
    }

    public Snapshot capture() { return state; }                 // no copying at all
    public void restore(Snapshot s) {
        if (!(s instanceof State captured) || captured.owner() != owner) {
            throw new IllegalArgumentException("foreign or null snapshot");
        }
        state = captured; // intentional overwrite; serialize with edits when required
    }
}

    public static void main(String[] args) {
        var editor = new Editor();
        editor.fixtureEdit("private text", 3, List.of(new Mark("original")));
        var expectedEditor = editor.observe();
        var editorSnapshot = editor.capture();
        editor.fixtureEdit("changed", 1, List.of(new Mark("replacement")));
        editor.restore(editorSnapshot);
        check(editor.observe().equals(expectedEditor), "editor complete independent view round trip");
        editor.fixtureMutateList();
        editor.restore(editorSnapshot);
        check(editor.observe().equals(expectedEditor), "editor repeated restore isolates retained list");
        rejects(() -> editor.restore(null), "editor null rejected");
        rejects(() -> editor.restore(new Editor().capture()), "editor foreign owner rejected");
        check(editor.observe().equals(expectedEditor), "editor rejection leaves state unchanged");
        check(editorSnapshot.toString().equals("Editor snapshot"), "editor rendering redacted");

        var mutable = new MutableDraft();
        mutable.fixtureEdit(List.of(new LineItem("L1")), List.of(new Adjustment(4)), Optional.of(new SettlementOption("fixed")), "private notes");
        var expectedMutable = mutable.observe();
        var mutableSnapshot = mutable.capture();
        mutable.fixtureEdit(List.of(new LineItem("L2")), List.of(new Adjustment(7)), Optional.empty(), "changed");
        mutable.restore(mutableSnapshot);
        check(mutable.observe().equals(expectedMutable), "mutable draft complete independent view round trip");
        mutable.fixtureMutateLists();
        mutable.restore(mutableSnapshot);
        check(mutable.observe().equals(expectedMutable), "mutable draft repeated restore isolates both lists");
        rejects(() -> mutable.restore(null), "mutable draft null rejected");
        rejects(() -> mutable.restore(new MutableDraft().capture()), "mutable draft foreign owner rejected");
        check(mutable.observe().equals(expectedMutable), "mutable rejection leaves state unchanged");
        check(mutableSnapshot.toString().equals("ClaimDraft snapshot"), "mutable draft rendering redacted");

        var immutable = new ImmutableDraft();
        immutable.fixtureEdit(List.of(new LineItem("L1")), List.of(new Adjustment(4)), Optional.of(new SettlementOption("fixed")), "private notes");
        var expectedImmutable = immutable.observe();
        var immutableSnapshot = immutable.capture();
        check(immutableSnapshot == immutable.capture(), "immutable capture retains same state reference");
        immutable.fixtureEdit(List.of(new LineItem("L2")), List.of(new Adjustment(7)), Optional.empty(), "changed");
        immutable.restore(immutableSnapshot);
        check(immutable.observe().equals(expectedImmutable), "immutable draft complete independent view round trip");
        immutable.fixtureChangeNotes("later");
        immutable.restore(immutableSnapshot);
        check(immutable.observe().equals(expectedImmutable), "immutable draft repeated restore");
        rejects(() -> immutable.restore(null), "immutable draft null rejected");
        rejects(() -> immutable.restore(new ImmutableDraft().capture()), "immutable draft foreign owner rejected");
        check(immutable.observe().equals(expectedImmutable), "immutable rejection leaves state unchanged");
        check(immutableSnapshot.toString().equals("ClaimDraft snapshot"), "immutable rendering redacted");

        // Negative controls establish the stated assumptions, not defects in the qualified snippets.
        var leaf = new StringBuilder("A");
        var shallow = List.copyOf(List.of(leaf));
        leaf.append("B");
        check(shallow.get(0).toString().equals("AB"), "control: List.copyOf leaves mutable element aliased");
        var retainedMutableCapture = new ArrayList<>(List.of("A"));
        var restoredLive = retainedMutableCapture;
        restoredLive.add("B");
        check(retainedMutableCapture.size() == 2, "control: direct mutable restore changes retained capture");

        // Deterministic operation ordering illustrates ABA; this is not a concurrency stress test.
        Object a = new Object(), b = new Object(), c = new Object();
        var identity = new AtomicReference<Object>(a);
        var staleExpected = identity.get();
        identity.set(b); identity.set(a);
        check(identity.compareAndSet(staleExpected, c), "control: identity-only CAS accepts A-B-A");
        var stamped = new AtomicStampedReference<Object>(a, 0);
        int[] stamp = new int[1];
        var expectedReference = stamped.get(stamp);
        stamped.set(b, 1); stamped.set(a, 2);
        check(!stamped.compareAndSet(expectedReference, c, stamp[0], 3), "joint stamp comparison rejects stale transition");
        check(stamped.getReference() == a && stamped.getStamp() == 2, "stale rejection leaves pair unchanged");
        System.out.println("PASS " + assertions + " assertions; three extracted originator snippets; two alias controls and deterministic ABA controls");
    }
}
