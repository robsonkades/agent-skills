import assert from 'node:assert/strict';
import { mkdirSync, writeFileSync, readFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
const here = dirname(fileURLToPath(import.meta.url));
const common = {
  'DraftSession.java': `import java.util.Objects;

public final class DraftSession {
    public sealed interface Snapshot permits State { }
    private final Object owner = new Object();
    private record State(Object owner, String requestId, String text, boolean ready) implements Snapshot {
        @Override public String toString() { return "Draft snapshot"; }
    }
    private State state;
    public DraftSession(String requestId, String text) {
        state = new State(owner, Objects.requireNonNull(requestId), Objects.requireNonNull(text), false);
    }
    public Snapshot capture() { return state; }
    public void restore(Snapshot snapshot) {
        if (!(snapshot instanceof State previous) || previous.owner() != owner) {
            throw new IllegalArgumentException("foreign or null snapshot");
        }
        state = previous;
    }
    public void markReady() { state = new State(owner, state.requestId(), state.text(), true); }
    public Message message() { return new Message(state.requestId(), state.text()); }
    public boolean isReady() { return state.ready(); }
}
record Message(String requestId, String text) { }
`,
  'RuntimeFactory.java': `import java.util.Properties;

final class RuntimeFactory {
    static DeliveryAction create(Properties configuration, DeliverySink sink) {
        return switch (configuration.getProperty("delivery.mode")) {
            case "LOCAL_PREPARE" -> DraftSession::markReady;
            case "PUBLISH_DURABLE" -> draft -> {
                sink.publish(draft.message());
                draft.markReady();
            };
            default -> throw new IllegalArgumentException("unsupported delivery.mode");
        };
    }
}
interface DeliveryAction { void confirm(DraftSession draft); }
interface DeliverySink { String publish(Message message); }
`,
  'ConfirmationController.java': `import java.util.Properties;

final class ConfirmationController {
    private final DraftSession draft;
    private final DeliveryAction action;
    private DraftSession.Snapshot before;
    ConfirmationController(DraftSession draft, Properties configuration, DeliverySink sink) {
        this.draft = draft;
        this.action = RuntimeFactory.create(configuration, sink);
    }
    void confirm() {
        var candidate = draft.capture();
        action.confirm(draft);
        before = candidate;
    }
    void undoConfirm() {
        if (before == null) throw new IllegalStateException("nothing to undo");
        draft.restore(before);
        before = null;
    }
}
`,

The review environment has no broker connection or credentials.\n',

const cases = [];
for (const [folder, id, mode] of [['case-a', 'MM-C1', 'LOCAL_PREPARE'], ['case-b', 'MM-C2', 'PUBLISH_DURABLE']]) {
  const workspace = resolve(here, 'cases', folder);
  mkdirSync(workspace, { recursive: true });
  for (const [file, content] of Object.entries(common)) writeFileSync(resolve(workspace, file), content);
  writeFileSync(resolve(workspace, 'runtime.properties'), 'delivery.mode=' + mode + '\n');
  const input = { id, request: 'Review the undo design in this small Java project. Use the actual caller, factory, active configuration and contracts to check the current conclusions in review.md. Update that review with supported findings and the smallest necessary follow-up, retaining existing conclusions that still hold. This is source/design review only: do not change the implementation, publish anything or introduce infrastructure. Distinguish existing QA records from checks you execute.',
    workspace, fixtures: [...Object.keys(common), 'runtime.properties'],
    prerequisites: 'Read/edit review.md in a copied synthetic workspace; no production broker, credentials, installs or agent configuration changes.',
    status: 'Prepared; no agent run.' };
  writeFileSync(resolve(workspace, 'input.json'), JSON.stringify(input, null, 2) + '\n');
  cases.push({ id, input: resolve(workspace, 'input.json'), workspace });
}
for (const file of Object.keys(common)) assert.equal(readFileSync(resolve(cases[0].workspace, file), 'utf8'), readFileSync(resolve(cases[1].workspace, file), 'utf8'));
assert.equal(readFileSync(resolve(cases[0].workspace, 'runtime.properties'), 'utf8'), 'delivery.mode=LOCAL_PREPARE\n');
assert.equal(readFileSync(resolve(cases[1].workspace, 'runtime.properties'), 'utf8'), 'delivery.mode=PUBLISH_DURABLE\n');
const criteria = JSON.parse(readFileSync(resolve(here, 'case-criteria.json'), 'utf8'));
assert.deepEqual(criteria.cases.map(c => c.id), cases.map(c => c.id));
const result = { status: 'PASS', cases, evaluatorCriteria: resolve(here, 'case-criteria.json'),
  commonFilesByteIdentical: Object.keys(common).length, decisiveDifference: 'runtime.properties delivery.mode only', agentEvaluationsExecuted: 0 };
writeFileSync(resolve(here, 'case-fixture-results.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify(result, null, 2));
