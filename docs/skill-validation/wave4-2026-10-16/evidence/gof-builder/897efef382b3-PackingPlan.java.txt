import java.util.List;
import java.util.Objects;

public record PackingPlan(String label, List<String> items) {
    public PackingPlan {
        Objects.requireNonNull(label, "label");
        if (label.isBlank()) throw new IllegalArgumentException("label is blank");
        items = ProductLists.prepare(items);
        if (items.isEmpty()) throw new IllegalArgumentException("items is empty");
        for (String item : items) Objects.requireNonNull(item, "item");
    }
}
