import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Objects;

public final class ContractProbe {
    private static int checks;
    private static int failures;

    private static void check(String name, Runnable action) {
        checks++;
        try {
            action.run();
            System.out.println("PASS " + name);
        } catch (Throwable failure) {
            failures++;
            System.out.println("FAIL " + name + ": " + failure);
        }
    }

    private static void equal(Object expected, Object actual) {
        if (!Objects.equals(expected, actual)) {
            throw new AssertionError("expected " + expected + " but got " + actual);
        }
    }

    private static void expect(Class<? extends Throwable> type, Runnable action) {
        Throwable caught = null;
        try {
            action.run();
        } catch (Throwable failure) {
            caught = failure;
        }
        if (caught == null || !type.isInstance(caught)) {
            throw new AssertionError("expected " + type.getName() + " but got " + caught);
        }
    }

    public static void main(String[] args) {
        check("previous product survives builder reuse", () -> {
            PlanBuilder builder = new PlanBuilder().label("first").addItem("book");
            PackingPlan first = builder.build();
            PackingPlan second = builder.label("second").addItem("pen").build();
            equal("first", first.label());
            equal(List.of("book"), first.items());
            equal(List.of("book", "pen"), second.items());
        });
        check("public constructor isolates caller list", () -> {
            var input = new ArrayList<>(List.of("book"));
            PackingPlan plan = new PackingPlan("direct", input);
            input.add("pen");
            equal(List.of("book"), plan.items());
        });
        check("setter isolates caller list before build", () -> {
            var input = new ArrayList<>(List.of("book"));
            PlanBuilder builder = new PlanBuilder().label("setter").items(input);
            input.add("pen");
            equal(List.of("book"), builder.build().items());
        });
        check("product accessor rejects mutation", () -> {
            PackingPlan plan = new PlanBuilder().label("out").addItem("book").build();
            expect(UnsupportedOperationException.class, () -> plan.items().add("pen"));
            equal(List.of("book"), plan.items());
        });
        check("failed build retains state for correction", () -> {
            PlanBuilder builder = new PlanBuilder().label("retained");
            expect(IllegalArgumentException.class, builder::build);
            PackingPlan result = builder.addItem("book").build();
            equal("retained", result.label());
            equal(List.of("book"), result.items());
        });
        check("failed setters preserve previous state", () -> {
            PlanBuilder builder = new PlanBuilder().label("retained").addItem("book");
            expect(IllegalArgumentException.class, () -> builder.label(" "));
            expect(NullPointerException.class, () -> builder.items(Arrays.asList("pen", null)));
            expect(NullPointerException.class, () -> builder.addItem(null));
            equal("retained", builder.build().label());
            equal(List.of("book"), builder.build().items());
        });
        check("all public product paths enforce invariants", () -> {
            expect(IllegalArgumentException.class, () -> new PackingPlan(" ", List.of("book")));
            expect(IllegalArgumentException.class, () -> new PackingPlan("out", List.of()));
            expect(NullPointerException.class, () -> new PackingPlan("out", Arrays.asList("book", null)));
            expect(NullPointerException.class, () -> new PackingPlan(null, List.of("book")));
        });
        check("copyOf does not copy mutable elements", () -> {
            var element = new StringBuilder("before");
            var copied = List.copyOf(List.of(element));
            element.replace(0, element.length(), "after");
            equal("after", copied.get(0).toString());
        });
        System.out.println("SUMMARY checks=" + checks + " failures=" + failures);
        if (failures != 0) throw new AssertionError(failures + " contract checks failed");
    }
}
