import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

public final class PlanBuilder {
    private String label;
    private List<String> items = new ArrayList<>();

    public PlanBuilder label(String label) {
        Objects.requireNonNull(label, "label");
        if (label.isBlank()) throw new IllegalArgumentException("label is blank");
        this.label = label;
        return this;
    }

    public PlanBuilder items(List<String> items) {
        this.items = new ArrayList<>(List.copyOf(items));
        return this;
    }

    public PlanBuilder addItem(String item) {
        items.add(Objects.requireNonNull(item, "item"));
        return this;
    }

    public PackingPlan build() {
        return new PackingPlan(label, items);
    }
}
