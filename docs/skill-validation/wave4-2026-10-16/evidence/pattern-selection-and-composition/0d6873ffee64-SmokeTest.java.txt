import java.nio.file.Path;

public final class SmokeTest {
    private static int checks;
    private static void check(boolean condition, String label) {
        if (!condition) throw new AssertionError(label);
        checks++;
    }
    public static void main(String[] args) throws Exception {
        Path root = Path.of(args[0]);
        try (var projection = new AccountProjection("smoke", root.resolve("schema.sql"))) {
            var view = new AccountView(projection, root.resolve("queries/account-view.sql"));
            var first = new AccountEvent("evt-41", 7, 41, 7000);
            var second = new AccountEvent("evt-42", 7, 42, 9000);
            check(projection.handle(first), "First event handled");
            check(projection.handle(second), "Second event handled");
            check(!projection.handle(second), "Repeated message skipped");
            check(view.read(7).equals(new AccountView.State(7, 42, 9000)), "Panel result");
            System.out.println("SMOKE PASS assertions=" + checks + " result=" + view.read(7));
        }
    }
}
