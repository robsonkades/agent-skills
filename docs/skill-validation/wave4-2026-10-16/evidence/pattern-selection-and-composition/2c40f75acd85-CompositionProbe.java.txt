import java.nio.file.Path;
import java.sql.SQLException;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;

public final class CompositionProbe {
    private static int checks;
    private static void check(boolean condition, String label) {
        if (!condition) throw new AssertionError(label);
        checks++;
    }
    private static long scalar(AccountProjection projection, String sql) throws Exception {
        try (var connection = projection.open(); var statement = connection.createStatement();
                var result = statement.executeQuery(sql)) {
            if (!result.next()) throw new AssertionError("No scalar row");
            return result.getLong(1);
        }
    }
    private static int updateWithRevisionGuard(AccountProjection projection,
            AccountEvent event) throws Exception {
        try (var connection = projection.open(); var statement = connection.prepareStatement(
                "UPDATE account_latest SET revision = ?, balance_cents = ? WHERE account_id = ? AND revision < ?")) {
            statement.setLong(1, event.revision());
            statement.setLong(2, event.balanceCents());
            statement.setLong(3, event.accountId());
            statement.setLong(4, event.revision());
            return statement.executeUpdate();
        }
    }
    public static void main(String[] args) throws Exception {
        Path root = Path.of(args[0]);
        long expectedRevision = Long.parseLong(args[1]);
        long expectedBalance = expectedRevision == 41 ? 7000 : 9000;
        System.out.println("runtime=" + System.getProperty("java.runtime.version"));
        try (var projection = new AccountProjection("controlled", root.resolve("schema.sql"));
                var dispatcher = new BatchDispatcher()) {
            try (var connection = projection.open()) {
                System.out.println("database=" + connection.getMetaData().getDatabaseProductVersion());
            }
            var view = new AccountView(projection, root.resolve("queries/account-view.sql"));
            var first = new AccountEvent("evt-41", 7, 41, 7000);
            var second = new AccountEvent("evt-42", 7, 42, 9000);
            var secondCommitted = new CountDownLatch(1);
            var commitOrder = new CopyOnWriteArrayList<Long>();
            dispatcher.accept(List.of(first, second), event -> {
                if (event.revision() == 41) {
                    if (!secondCommitted.await(5, TimeUnit.SECONDS)) {
                        throw new AssertionError("Second worker did not finish");
                    }
                }
                try {
                    projection.handle(event);
                    commitOrder.add(event.revision());
                } finally {
                    if (event.revision() == 42) secondCommitted.countDown();
                }
            });
            check(commitOrder.equals(List.of(42L, 41L)), "Controlled reversed completion");
            var expected = new AccountView.State(7, expectedRevision, expectedBalance);
            check(view.read(7).equals(expected), "Actual deployed query result");
            check(scalar(projection, "SELECT COUNT(*) FROM handled_message") == 2, "Two committed IDs");
            check(scalar(projection, "SELECT COUNT(*) FROM account_history") == 2, "Two history rows");
            check(scalar(projection, "SELECT revision FROM account_latest WHERE account_id=7") == 41, "Compatibility table last-applied revision");
            check(scalar(projection, "SELECT MAX(revision) FROM account_history WHERE account_id=7") == 42, "Greatest authoritative revision retained");
            System.out.println("completion=" + commitOrder + " served=" + view.read(7)
                    + " rawLatest=41 historyMax=42 handled=2");
            check(!projection.handle(second), "Replay suppressed");
            check(view.read(7).equals(expected), "Replay does not repair stale order");
            check(scalar(projection, "SELECT COUNT(*) FROM handled_message") == 2, "Replay adds no identity");

            boolean rolledBack = false;
            try {
                projection.handle(new AccountEvent("invalid-43", 7, 43, -1));
            } catch (SQLException failure) {
                if (!"23513".equals(failure.getSQLState())) throw failure;
                rolledBack = true;
            }
            check(rolledBack, "Constraint failure observed");
            check(scalar(projection, "SELECT COUNT(*) FROM handled_message WHERE message_id='invalid-43'") == 0, "Failed effect retains no inbox record");
            check(scalar(projection, "SELECT COUNT(*) FROM account_history WHERE revision=43") == 0, "Failed effect retains no history");
            check(scalar(projection, "SELECT revision FROM account_latest WHERE account_id=7") == 41, "Failed effect changes no compatibility state");
            check(view.read(7).equals(expected), "Failed transaction changes no served result");
            check(projection.handle(new AccountEvent("evt-43", 7, 43, 11000)), "Subsequent valid event works");
            check(view.read(7).equals(new AccountView.State(7, 43, 11000)), "Subsequent revision visible");
            check(scalar(projection, "SELECT COUNT(*) FROM handled_message") == 3, "Three valid identities");
            check(scalar(projection, "SELECT COUNT(*) FROM account_history") == 3, "Three valid snapshots");
            System.out.println("duplicate_and_rollback_controls=PASS served_after_valid_next=" + view.read(7));
        }

        // Narrow technical control: complete snapshots, existing row, no transition effects.
        // This is not a replacement production consumer or an insert-on-missing protocol.
        try (var projection = new AccountProjection("guard_control", root.resolve("schema.sql"))) {
            projection.handle(new AccountEvent("seed", 8, 40, 10000));
            check(updateWithRevisionGuard(projection, new AccountEvent("g42", 8, 42, 9000)) == 1, "New complete snapshot accepted");
            check(updateWithRevisionGuard(projection, new AccountEvent("g41", 8, 41, 7000)) == 0, "Older complete snapshot refused");
            check(scalar(projection, "SELECT revision FROM account_latest WHERE account_id=8") == 42, "Guarded revision stays greatest");
            check(scalar(projection, "SELECT balance_cents FROM account_latest WHERE account_id=8") == 9000, "Guarded full value preserved");
        }
        // Arithmetic counterexample only: applying a latest-version-only rule to deltas
        // would discard a necessary earlier delta. No database/provider guarantee is inferred.
        long allDeltas = 20 + 10;
        long onlyNewestDelta = 20;
        check(allDeltas == 30, "Both deltas needed");
        check(onlyNewestDelta != allDeltas, "Latest-only is not a delta fix");
        System.out.println("PROBE PASS assertions=" + checks + " fixture_effects_only=true broker_tested=false");
    }
}
