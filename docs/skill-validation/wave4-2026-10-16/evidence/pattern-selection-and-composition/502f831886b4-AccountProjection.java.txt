import java.nio.file.Files;
import java.nio.file.Path;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;

public final class AccountProjection implements AutoCloseable {
    private final String url;

    public AccountProjection(String databaseName, Path schema) throws Exception {
        if (!databaseName.matches("[a-zA-Z0-9_]+")) {
            throw new IllegalArgumentException("Only a local in-memory database name is accepted");
        }
        url = "jdbc:h2:mem:" + databaseName + ";DB_CLOSE_DELAY=-1;LOCK_TIMEOUT=5000";
        try (var connection = open(); var statement = connection.createStatement()) {
            for (String sql : Files.readString(schema).split(";")) {
                if (!sql.isBlank()) statement.execute(sql);
            }
        }
    }

    public Connection open() throws SQLException {
        return DriverManager.getConnection(url, "sa", "");
    }

    public boolean handle(AccountEvent event) throws SQLException {
        try (var connection = open()) {
            connection.setAutoCommit(false);
            try {
                try (var seen = connection.prepareStatement(
                        "INSERT INTO handled_message(subscriber, message_id) VALUES ('panel', ?)")) {
                    seen.setString(1, event.messageId());
                    try {
                        seen.executeUpdate();
                    } catch (SQLException duplicate) {
                        if (!"23505".equals(duplicate.getSQLState())) throw duplicate;
                        connection.rollback();
                        return false;
                    }
                }
                try (var history = connection.prepareStatement(
                        "INSERT INTO account_history(account_id, revision, balance_cents) VALUES (?, ?, ?)")) {
                    bind(history, event);
                    history.executeUpdate();
                }
                try (var latest = connection.prepareStatement(
                        "MERGE INTO account_latest(account_id, revision, balance_cents) KEY(account_id) VALUES (?, ?, ?)")) {
                    bind(latest, event);
                    latest.executeUpdate();
                }
                connection.commit();
                return true;
            } catch (SQLException failure) {
                try {
                    connection.rollback();
                } catch (SQLException rollbackFailure) {
                    failure.addSuppressed(rollbackFailure);
                }
                throw failure;
            }
        }
    }

    private static void bind(java.sql.PreparedStatement statement, AccountEvent event) throws SQLException {
        statement.setLong(1, event.accountId());
        statement.setLong(2, event.revision());
        statement.setLong(3, event.balanceCents());
    }

    @Override public void close() throws SQLException {
        try (var connection = open(); var statement = connection.createStatement()) {
            statement.execute("SHUTDOWN");
        }
    }
}
