import java.nio.file.Files;
import java.nio.file.Path;

public final class AccountView {
    public record State(long accountId, long revision, long balanceCents) {}
    private final AccountProjection projection;
    private final String query;

    public AccountView(AccountProjection projection, Path deployedQuery) throws Exception {
        this.projection = projection;
        query = Files.readString(deployedQuery);
    }

    public State read(long accountId) throws Exception {
        try (var connection = projection.open(); var statement = connection.prepareStatement(query)) {
            statement.setLong(1, accountId);
            try (var result = statement.executeQuery()) {
                if (!result.next()) throw new IllegalStateException("No account state");
                return new State(result.getLong("account_id"), result.getLong("revision"),
                        result.getLong("balance_cents"));
            }
        }
    }
}
