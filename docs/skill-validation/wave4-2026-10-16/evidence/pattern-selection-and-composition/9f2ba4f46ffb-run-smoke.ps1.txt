param(
  [Parameter(Mandatory=$true)][string]$JdkBin,
  [Parameter(Mandatory=$true)][string]$H2Jar
)
$ErrorActionPreference = 'Stop'
$fixtureRoot = $PSScriptRoot
$classes = Join-Path $fixtureRoot '.build/smoke'
New-Item -ItemType Directory -Path $classes -Force | Out-Null
$javaSources = @((Get-ChildItem -LiteralPath (Join-Path $fixtureRoot 'src') -Filter '*.java').FullName)
$javaSources += Join-Path $fixtureRoot 'test/SmokeTest.java'
$compilerOutput = @(& (Join-Path $JdkBin 'javac.exe') --release 17 -cp $H2Jar -d $classes @javaSources 2>&1)
$compilerExit = $LASTEXITCODE
[IO.File]::WriteAllText((Join-Path $classes 'javac.log'), ($compilerOutput -join [Environment]::NewLine))
$compilerOutput | Write-Output
if ($compilerExit -ne 0) { exit $compilerExit }
& (Join-Path $JdkBin 'java.exe') -cp "$classes;$H2Jar" SmokeTest $fixtureRoot
exit $LASTEXITCODE
