import assert from 'node:assert/strict';
import { readdir, readFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const root = path.dirname(fileURLToPath(import.meta.url));
let checks = 0;
const equal = (left, right) => { assert.deepEqual(left, right); checks++; };
async function files(dir, prefix = '') {
  const entries = await readdir(dir, {withFileTypes:true});
  const names = [];
  for (const entry of entries) {
    const item = prefix + entry.name;
    if (entry.isDirectory()) names.push(...await files(path.join(dir, entry.name), item + '/'));
    else names.push(item);
  }
  return names.sort();
}
const numeric = path.join(root, 'cases/numeric');
const scale = path.join(root, 'cases/scale');
const numericFiles = await files(numeric);
const scaleFiles = await files(scale);
equal(numericFiles, scaleFiles);
equal(numericFiles.length, 8);
for (const name of numericFiles.filter(n => n !== 'requirements.md')) {
  equal(await readFile(path.join(numeric, name), 'utf8'),
        await readFile(path.join(scale, name), 'utf8'));
}
const numericRequirements = await readFile(path.join(numeric, 'requirements.md'), 'utf8');
const scaleRequirements = await readFile(path.join(scale, 'requirements.md'), 'utf8');
equal(numericRequirements.split('\n').slice(0, 4), scaleRequirements.split('\n').slice(0, 4));
assert.notEqual(numericRequirements, scaleRequirements);
checks++;
const criteria = JSON.parse(await readFile(path.join(root, 'case-criteria.json'), 'utf8'));
equal(criteria.map(c => c.id), ['measurement-numeric-equivalence', 'measurement-significant-scale']);
console.log(JSON.stringify({
  checks, result:'PASS',
  scope:'Only accepted scale significance differs; all seven ordinary task/source artifacts are byte-identical. This is fixture consistency, not agent behavior.'
}, null, 2));
