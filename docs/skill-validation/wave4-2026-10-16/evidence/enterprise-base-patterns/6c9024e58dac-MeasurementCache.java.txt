import java.util.HashMap;
import java.util.Map;
import java.util.Optional;

public final class MeasurementCache {
    private final Map<Measurement, String> results = new HashMap<>();
    public void remember(Measurement key, String result) { results.put(key, result); }
    public Optional<String> find(Measurement key) { return Optional.ofNullable(results.get(key)); }
    public int size() { return results.size(); }
}
