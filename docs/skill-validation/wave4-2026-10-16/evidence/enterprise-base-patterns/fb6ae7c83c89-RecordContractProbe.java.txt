import java.math.BigDecimal;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public final class RecordContractProbe {
    private static int checks;

    record ArrayKey(byte[] bytes) { }

    record CopyOnlyKey(byte[] bytes) {
        CopyOnlyKey { bytes = bytes.clone(); }
        @Override public byte[] bytes() { return bytes.clone(); }
    }

    record ContentKey(byte[] bytes) {
        ContentKey { bytes = bytes.clone(); }
        @Override public byte[] bytes() { return bytes.clone(); }
        @Override public boolean equals(Object other) {
            return other instanceof ContentKey key && Arrays.equals(bytes, key.bytes);
        }
        @Override public int hashCode() { return Arrays.hashCode(bytes); }
    }

    record ScaleKey(BigDecimal amount) { }
    record NumericKey(BigDecimal amount) {
        NumericKey { amount = amount.stripTrailingZeros(); }
    }

    private static void check(boolean condition, String message) {
        checks++;
        if (!condition) throw new AssertionError(message);
    }

    public static void main(String[] args) {
        if (args.length > 0 && args[0].equals("wrong-array-fix")) {
            var key = new CopyOnlyKey(new byte[] {1, 2});
            check(key.equals(new CopyOnlyKey(key.bytes())),
                    "Defensive copies alone fail the record copy/value contract");
            return;
        }
        if (args.length > 0 && args[0].equals("wrong-decimal-normalization")) {
            check(!new NumericKey(new BigDecimal("2.0")).equals(
                            new NumericKey(new BigDecimal("2.00"))),
                    "Unconditional normalization erases required significant scale");
            return;
        }

        var raw = new ArrayKey(new byte[] {1, 2});
        var rawEquivalent = new ArrayKey(new byte[] {1, 2});
        check(!raw.equals(rawEquivalent), "Array record does not supply content equality");
        check(Map.of(raw, "accepted").get(rawEquivalent) == null, "Independent array key misses");

        byte[] copiedInput = {1, 2};
        var copied = new CopyOnlyKey(copiedInput);
        copiedInput[0] = 9;
        check(copied.bytes()[0] == 1, "Copy-only constructor owns its input");
        byte[] copiedOutput = copied.bytes();
        copiedOutput[0] = 8;
        check(copied.bytes()[0] == 1, "Copy-only accessor protects its storage");
        check(!copied.equals(new CopyOnlyKey(copied.bytes())),
                "Ownership correction alone does not satisfy record copy invariant");

        byte[] contentInput = {1, 2};
        var content = new ContentKey(contentInput);
        var equivalent = new ContentKey(new byte[] {1, 2});
        var keys = new HashMap<ContentKey, String>();
        keys.put(content, "accepted");
        check(content.equals(equivalent), "Independent content values compare equal");
        check(content.hashCode() == equivalent.hashCode(), "Equal content values have equal hash");
        check("accepted".equals(keys.get(equivalent)), "Equivalent key finds entry");
        check(content.equals(new ContentKey(content.bytes())), "Content record copy invariant");
        contentInput[0] = 9;
        check("accepted".equals(keys.get(equivalent)), "Input mutation cannot alter key");
        byte[] contentOutput = content.bytes();
        contentOutput[0] = 8;
        check("accepted".equals(keys.get(equivalent)), "Accessor mutation cannot alter key");
        check(content.bytes()[0] == 1, "Owned content unchanged");
        check(!content.equals(new ContentKey(new byte[] {1, 3})), "Distinct content remains distinct");

        BigDecimal oneScale = new BigDecimal("2.0");
        BigDecimal twoScale = new BigDecimal("2.00");
        check(oneScale.compareTo(twoScale) == 0, "Same numerical value");
        var firstScale = new ScaleKey(oneScale);
        var secondScale = new ScaleKey(twoScale);
        check(!firstScale.equals(secondScale), "Generated equality preserves scale");
        var scaleEntries = new HashMap<ScaleKey, String>();
        scaleEntries.put(firstScale, "tenths");
        scaleEntries.put(secondScale, "hundredths");
        check(scaleEntries.size() == 2, "Significant precision entries remain distinct");
        check("tenths".equals(scaleEntries.get(new ScaleKey(new BigDecimal("2.0")))),
                "Same value and scale retrieves the correct entry");
        check(firstScale.equals(new ScaleKey(firstScale.amount())), "Scale record copy invariant");

        var firstNumeric = new NumericKey(oneScale);
        var secondNumeric = new NumericKey(twoScale);
        check(firstNumeric.equals(secondNumeric), "Numeric-only contract can normalize");
        check(firstNumeric.hashCode() == secondNumeric.hashCode(), "Normalized equality/hash agree");
        check("accepted".equals(Map.of(firstNumeric, "accepted").get(secondNumeric)),
                "Equivalent numeric key finds entry");
        check(firstNumeric.equals(new NumericKey(firstNumeric.amount())), "Normalized copy invariant");
        check(firstNumeric.amount().toPlainString().equals("2"), "Normalization changes wire representation");
        check(!firstNumeric.equals(new NumericKey(new BigDecimal("2.01"))), "Different quantity not merged");
        System.out.println("PASS " + checks + " focused contract checks");
        System.out.println("Scope: Java source compiled --release 17, executed on the available JDK; no agent evaluation.");
    }
}
