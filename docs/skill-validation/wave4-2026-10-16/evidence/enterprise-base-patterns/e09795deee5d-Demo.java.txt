import java.math.BigDecimal;

public final class Demo {
    public static void main(String[] args) {
        var first = new Measurement(new BigDecimal("2.0"), "kg");
        var second = new Measurement(new BigDecimal("2.00"), "kg");
        var cache = new MeasurementCache();
        cache.remember(first, "result-A");
        System.out.println("second lookup before insert = " + cache.find(second));
        cache.remember(second, "result-B");
        System.out.println("cache entries = " + cache.size());
        var reports = new ReportFormatter();
        System.out.println("first report = " + reports.render(first));
        System.out.println("second report = " + reports.render(second));
    }
}
