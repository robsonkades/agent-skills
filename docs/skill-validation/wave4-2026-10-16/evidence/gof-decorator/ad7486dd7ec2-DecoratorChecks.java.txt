import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CancellationException;
import java.util.concurrent.atomic.AtomicInteger;

public final class DecoratorChecks {
    private static int checks;
    private static final Sku SKU = new Sku("sku-1");
    private static final Price PRICE = new Price("9.99");

    private static final class TestDeadline implements Deadline {
        boolean expired;
        public boolean hasExpired() { return expired; }
    }
    private static final class RecordingPolicy implements RetryPolicy {
        final int limit;
        final List<Integer> waits = new ArrayList<>();
        final Runnable onSleep;
        RecordingPolicy(int limit) { this(limit, () -> {}); }
        RecordingPolicy(int limit, Runnable onSleep) {
            this.limit = limit;
            this.onSleep = onSleep;
        }
        public int maxAttempts() { return limit; }
        public void sleepBefore(int nextAttempt, Deadline deadline) {
            waits.add(nextAttempt);
            onSleep.run();
        }
    }
    private static void check(boolean value, String message) {
        checks++;
        if (!value) throw new AssertionError(message);
    }
    private static void sameFailure(Throwable expected, Runnable action) {
        try {
            action.run();
        } catch (Throwable actual) {
            check(actual == expected, "failure identity: expected " + expected
                    + ", got " + actual);
            return;
        }
        throw new AssertionError("expected failure " + expected);
    }
    private static <T extends Throwable> T failureType(Class<T> expected, Runnable action) {
        try {
            action.run();
        } catch (Throwable actual) {
            check(expected.isInstance(actual), "expected " + expected + ", got " + actual);
            return expected.cast(actual);
        }
        throw new AssertionError("expected failure type " + expected);
    }
    private static PriceLookup record(List<String> observations, PriceLookup delegate) {
        return (sku, deadline) -> {
            try {
                Price result = delegate.of(sku, deadline);
                observations.add("success");
                return result;
            } catch (RuntimeException failure) {
                observations.add("failure");
                throw failure;
            }
        };
    }
    public static void main(String[] args) {
        failureType(IllegalArgumentException.class, () -> new RetryingPriceLookup(
                (sku, d) -> PRICE, new RecordingPolicy(0)));
        failureType(NullPointerException.class, () -> new RetryingPriceLookup(
                null, new RecordingPolicy(1)));
        failureType(NullPointerException.class, () -> new RetryingPriceLookup(
                (sku, d) -> PRICE, null));
        {
            var calls = new AtomicInteger();
            var policy = new RecordingPolicy(3);
            var deadline = new TestDeadline();
            PriceLookup delegate = (sku, d) -> {
                check(sku == SKU && d == deadline, "forward original arguments");
                if (calls.incrementAndGet() < 3) throw new PricingUnavailable(sku);
                return PRICE;
            };
            Price result = new RetryingPriceLookup(delegate, policy).of(SKU, deadline);
            check(result == PRICE, "preserve returned object");
            check(calls.get() == 3, "success after three attempts");
            check(policy.waits.equals(List.of(2, 3)), "wait before attempts two and three");
        }
        {
            var calls = new AtomicInteger();
            var policy = new RecordingPolicy(3);
            var failure = new PricingUnavailable(SKU);
            PriceLookup delegate = (sku, d) -> { calls.incrementAndGet(); throw failure; };
            sameFailure(failure, () -> new RetryingPriceLookup(delegate, policy)
                    .of(SKU, new TestDeadline()));
            check(calls.get() == 3, "exhaustion uses exactly three attempts");
            check(policy.waits.equals(List.of(2, 3)), "no wait after final failure");
        }
        {
            var calls = new AtomicInteger();
            var policy = new RecordingPolicy(3);
            var failure = new PriceRejected(SKU, "unknown sku");
            PriceLookup delegate = (sku, d) -> { calls.incrementAndGet(); throw failure; };
            sameFailure(failure, () -> new RetryingPriceLookup(delegate, policy)
                    .of(SKU, new TestDeadline()));
            check(calls.get() == 1 && policy.waits.isEmpty(), "permanent failure not retried");
        }
        {
            var calls = new AtomicInteger();
            var policy = new RecordingPolicy(3);
            var deadline = new TestDeadline();
            deadline.expired = true;
            var failure = failureType(PricingDeadlineExceeded.class,
                    () -> new RetryingPriceLookup((sku, d) -> {
                        calls.incrementAndGet(); return PRICE;
                    }, policy).of(SKU, deadline));
            check(calls.get() == 0 && policy.waits.isEmpty(), "expired call has no attempt");
            check(failure.completedAttempts == 0, "initial expiry count");
        }
        {
            var calls = new AtomicInteger();
            var deadline = new TestDeadline();
            var policy = new RecordingPolicy(3, () -> deadline.expired = true);
            var failure = failureType(PricingDeadlineExceeded.class,
                    () -> new RetryingPriceLookup((sku, d) -> {
                        calls.incrementAndGet(); throw new PricingUnavailable(sku);
                    }, policy).of(SKU, deadline));
            check(calls.get() == 1 && policy.waits.equals(List.of(2)),
                    "expiry during backoff prevents next attempt");
            check(failure.completedAttempts == 1, "backoff expiry count");
        }
        {
            var calls = new AtomicInteger();
            var cancellation = new CancellationException("backoff cancelled");
            var policy = new RecordingPolicy(3, () -> { throw cancellation; });
            sameFailure(cancellation, () -> new RetryingPriceLookup((sku, d) -> {
                calls.incrementAndGet(); throw new PricingUnavailable(sku);
            }, policy).of(SKU, new TestDeadline()));
            check(calls.get() == 1 && policy.waits.equals(List.of(2)),
                    "backoff cancellation stops retries");
        }
        {
            var calls = new AtomicInteger();
            var cancellation = new CancellationException("transport cancelled");
            var policy = new RecordingPolicy(3);
            sameFailure(cancellation, () -> new RetryingPriceLookup((sku, d) -> {
                calls.incrementAndGet(); throw cancellation;
            }, policy).of(SKU, new TestDeadline()));
            check(calls.get() == 1 && policy.waits.isEmpty(),
                    "transport cancellation is not transient retry");
        }
        {
            var logical = new ArrayList<String>();
            var attempts = new ArrayList<String>();
            var callsA = new AtomicInteger();
            var callsB = new AtomicInteger();
            PriceLookup flakyA = (sku, d) -> {
                if (callsA.incrementAndGet() < 3) throw new PricingUnavailable(sku);
                return PRICE;
            };
            PriceLookup flakyB = (sku, d) -> {
                if (callsB.incrementAndGet() < 3) throw new PricingUnavailable(sku);
                return PRICE;
            };
            Price a = record(logical, new RetryingPriceLookup(flakyA,
                    new RecordingPolicy(3))).of(SKU, new TestDeadline());
            Price b = new RetryingPriceLookup(record(attempts, flakyB),
                    new RecordingPolicy(3)).of(SKU, new TestDeadline());
            check(a == PRICE && b == PRICE, "both arrangements preserve success");
            check(callsA.get() == 3 && callsB.get() == 3, "same underlying attempt count");
            check(logical.equals(List.of("success")), "outer observation sees one operation");
            check(attempts.equals(List.of("failure", "failure", "success")),
                    "inner observation sees each attempt");
            System.out.println("ORDER: outer=" + logical + "; inner=" + attempts
                    + "; delegate calls=" + callsA + "/" + callsB);
        }
        System.out.println("PASS: " + checks + " checks");
    }
}
