import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.Properties;

public final class ApplicationBinding implements AutoCloseable {
    private final boolean borrowed;
    private final List<TrackingParser> instances = new ArrayList<>();
    private final TrackingParser shared;
    private int calls;

    public ApplicationBinding(Path configuration) throws IOException {
        Properties properties = new Properties();
        try (var input = Files.newInputStream(configuration)) {
            properties.load(input);
        }
        String scope = properties.getProperty("scope");
        if (!"borrowed".equals(scope) && !"owned".equals(scope)) {
            throw new IllegalArgumentException("Unsupported scope: " + scope);
        }
        borrowed = "borrowed".equals(scope);
        shared = borrowed ? allocate() : null;
    }

    private TrackingParser allocate() {
        TrackingParser parser = new TrackingParser();
        instances.add(parser);
        return parser;
    }

    // borrowed=true: application owns the shared result until this binding closes.
    // borrowed=false: each result is transferred to the caller for one operation.
    public Parser createParser() {
        calls++;
        return borrowed ? shared : allocate();
    }

    public boolean borrowed() { return borrowed; }
    public int factoryCalls() { return calls; }
    public int instanceCount() { return instances.size(); }
    public int closeCount() { return instances.stream().mapToInt(TrackingParser::closes).sum(); }
    public boolean sharedClosed() { return shared != null && shared.closed(); }

    public void close() {
        if (shared != null) shared.close();
    }
}
