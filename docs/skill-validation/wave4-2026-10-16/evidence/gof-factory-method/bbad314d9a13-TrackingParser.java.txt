import java.io.IOException;

public final class TrackingParser implements Parser {
    private boolean closed;
    private int closes;

    public int parse(String input) throws IOException {
        if (closed) throw new IOException("parser is closed");
        return input.length();
    }

    public void close() {
        if (!closed) {
            closed = true;
            closes++;
        }
    }

    public boolean closed() { return closed; }
    public int closes() { return closes; }
}
