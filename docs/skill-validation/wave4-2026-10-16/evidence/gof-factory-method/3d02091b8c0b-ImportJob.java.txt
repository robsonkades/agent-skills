import java.io.IOException;
import java.util.Objects;
import java.util.function.IntConsumer;

// Published extension contract. Invoke createParser only after construction.
public abstract class ImportJob {
    // A successful call returns a non-null, per-run owned parser or lease.
    // The run closes it after synchronous parse/save; checked creation failures propagate.
    protected abstract Parser createParser() throws IOException;

    public final int run(String input, IntConsumer save) throws IOException {
        Objects.requireNonNull(save, "save");
        try (Parser parser = Objects.requireNonNull(createParser(), "factory returned null")) {
            int rows = parser.parse(input);
            save.accept(rows);
            return rows;
        }
    }
}
