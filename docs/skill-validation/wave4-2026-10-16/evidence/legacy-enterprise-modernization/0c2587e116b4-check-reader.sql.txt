-- Evaluator-only assertion helper. Runs with the SET ROLE selected by probe.sql.
\ir ../common/consumer-query.sql
DO $check$
DECLARE
    actual_ids integer[];
    expected_ids integer[] := string_to_array(current_setting('lm_w4.expected_ids'), ',')::integer[];
BEGIN
    SELECT array_agg(cod_cli ORDER BY cod_cli) INTO actual_ids FROM lm_w4.cliente;
    IF actual_ids IS DISTINCT FROM expected_ids THEN
        RAISE EXCEPTION 'role % observed %, expected %', current_user, actual_ids, expected_ids;
    END IF;
    BEGIN
        INSERT INTO lm_w4.cliente(cod_cli, tenant_role, nome_cli)
            VALUES (99, current_user::text, 'forbidden application write');
        RAISE EXCEPTION 'application write unexpectedly succeeded for %', current_user;
    EXCEPTION WHEN insufficient_privilege THEN
        NULL;
    END;
END
$check$;
