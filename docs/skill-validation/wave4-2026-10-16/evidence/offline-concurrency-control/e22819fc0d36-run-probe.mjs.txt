import fs from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
import {spawnSync} from 'node:child_process';
const cwd=path.dirname(fileURLToPath(import.meta.url));
const jdk='C:/Users/robso/.jdks/temurin-25.0.3/bin';
const h2='C:/Users/robso/.m2/repository/com/h2database/h2/2.4.240/h2-2.4.240.jar';
const commands=[
 {id:'01-javac-version',command:jdk+'/javac.exe',args:['-version']},
 {id:'02-java-version',command:jdk+'/java.exe',args:['-version']},
 {id:'03-compile',command:jdk+'/javac.exe',args:['--release','17','-Xlint:all','-Werror','-XDrawDiagnostics','-d','classes','OfflineProbe.java']},
 {id:'04-probe',command:jdk+'/java.exe',args:['-cp','classes;'+h2,'OfflineProbe']}
];
const checks=[];
for(const c of commands){
 const r=spawnSync(c.command,c.args,{cwd,encoding:'utf8',timeout:15000,windowsHide:true});
 const result={...c,cwd,exitCode:r.status,signal:r.signal,error:r.error?.message??null,stdout:r.stdout??'',stderr:r.stderr??''};
 checks.push(result);
 fs.writeFileSync(path.join(cwd,c.id+'.log'),JSON.stringify(result,null,2)+'\n');
 console.log(JSON.stringify(result));
 if(r.status!==0||r.error){process.exitCode=1;break;}
}
fs.writeFileSync(path.join(cwd,'technical-results.json'),JSON.stringify({checks,scope:'Actual H2 2.4.240 JDBC READ_COMMITTED transactions with controlled interleaving; not PostgreSQL/JPA or agent behavior evidence.'},null,2)+'\n');
