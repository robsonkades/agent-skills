import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import {spawnSync} from 'node:child_process';
const cwd='C:/git/agent-skills',root=cwd+'/skills/offline-concurrency-control';
const baseline='C:/Users/robso/AppData/Local/Temp/skills-wave4-validation-Ceekn9/baseline/offline-concurrency-control';
const commands=[
 ['status','git',['status','--short','--untracked-files=all','--ignored','--','skills/offline-concurrency-control']],
 ['staged','git',['diff','--cached','--','skills/offline-concurrency-control']],
 ['diff','git',['diff','HEAD','--','skills/offline-concurrency-control']],
 ['diff-check','git',['diff','--check','--','skills/offline-concurrency-control']],
 ['stat','git',['diff','--stat','--','skills/offline-concurrency-control']]
];
const checks=commands.map(([id,command,args])=>{const r=spawnSync(command,args,{cwd,encoding:'utf8',timeout:15000,windowsHide:true});return{id,command,args,cwd,exitCode:r.status,stdout:r.stdout,stderr:r.stderr,error:r.error?.message??null};});
const files=[];
const sha=b=>crypto.createHash('sha256').update(b).digest('hex').toUpperCase();
for(const e of fs.readdirSync(root,{recursive:true,withFileTypes:true})){
 if(!e.isFile())continue;
 const full=path.join(e.parentPath,e.name),relative=path.relative(root,full).replaceAll('\\','/');
 const bytes=fs.readFileSync(full),before=fs.readFileSync(path.join(baseline,relative));
 files.push({path:full.replaceAll('\\','/'),relative,bytes:bytes.length,sha256:sha(bytes),baselineSha256:sha(before),changedFromAssignment:!bytes.equals(before)});
}

const jar=fs.readFileSync(dependency);
console.log(JSON.stringify({checks,files,availableDependency:{path:dependency,bytes:jar.length,sha256:sha(jar),copied:false,installed:false},finalManifestVersion:fs.readFileSync(root+'/skill.yaml','utf8').match(/^version: (.+)$/m)[1]},null,2));
