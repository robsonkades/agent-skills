import java.sql.*;

public final class ReviewStore {
    public record Edit(long revision, int first, int second) {}
    @FunctionalInterface
    public interface SqlStep { void run() throws SQLException; }

    public static int readInt(Connection c, String sql) throws SQLException {
        try (Statement statement = c.createStatement()) {
            statement.setQueryTimeout(2);
            try (ResultSet rows = statement.executeQuery(sql)) {
                if (!rows.next()) throw new SQLException("missing row");
                return rows.getInt(1);
            }
        }
    }

    public static Edit loadEdit(Connection c, SqlStep betweenQueries) throws SQLException {
        long revision = readInt(c, "SELECT revision FROM review_root WHERE id = 1");
        int first = readInt(c, "SELECT amount FROM review_line WHERE id = 1");
        int second = readInt(c, "SELECT amount FROM review_line WHERE id = 2");
        betweenQueries.run();
        return new Edit(revision, first, second);
    }

    private static int update(Connection c, String sql, long... values) throws SQLException {
        try (PreparedStatement statement = c.prepareStatement(sql)) {
            statement.setQueryTimeout(2);
            for (int i = 0; i < values.length; i++) statement.setLong(i + 1, values[i]);
            return statement.executeUpdate();
        }
    }

    public static boolean submit(Connection c, Edit original, int secondRequested) throws SQLException {
        try {
            if (original.first() + secondRequested > 100) throw new IllegalArgumentException("limit");
            int changed = update(c,
                    "UPDATE review_root SET revision = revision + 1 WHERE id = 1 AND revision = ?",
                    original.revision());
            if (changed == 0) { c.rollback(); return false; }
            if (changed != 1) throw new SQLException("unexpected root count");
            if (update(c, "UPDATE review_line SET amount = ? WHERE id = 1", original.first()) != 1)
                throw new SQLException("missing first line");
            if (update(c, "UPDATE review_line SET amount = ? WHERE id = 2", secondRequested) != 1)
                throw new SQLException("missing second line");
            c.commit();
            return true;
        } catch (SQLException | RuntimeException | Error failure) {
            c.rollback();
            throw failure;
        }
    }

    public static void recalculate(Connection c) throws SQLException {
        try {
            if (update(c, "UPDATE review_root SET revision = revision + 1 WHERE id = 1 AND revision = ?", 1) != 1)
                throw new SQLException("job precondition");
            if (update(c, "UPDATE review_line SET amount = ? WHERE id = 1", 80) != 1)
                throw new SQLException("job row");
            c.commit();
        } catch (SQLException | RuntimeException | Error failure) {
            c.rollback();
            throw failure;
        }
    }
}
