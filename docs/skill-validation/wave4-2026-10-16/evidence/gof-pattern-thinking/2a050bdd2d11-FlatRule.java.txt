package example.pricing;
import java.math.BigDecimal;
import java.util.Objects;
public final class FlatRule implements PricingRule {
    private final BigDecimal unitPrice;
    public FlatRule(BigDecimal unitPrice) {
        this.unitPrice = Objects.requireNonNull(unitPrice);
        if (unitPrice.signum() < 0) throw new IllegalArgumentException("negative price");
    }
    public BigDecimal quote(int units) {
        if (units < 0) throw new IllegalArgumentException("negative quantity");
        return unitPrice.multiply(BigDecimal.valueOf(units));
    }
}
