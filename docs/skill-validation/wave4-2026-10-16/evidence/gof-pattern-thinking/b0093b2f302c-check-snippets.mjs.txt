import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';

const repo = 'C:/git/agent-skills';
const evidence = path.dirname(new URL(import.meta.url).pathname.replace(/^\/(?:([A-Za-z]):)/, '$1:'));
const work = fs.mkdtempSync(path.join(evidence, 'snippets-'));
const bin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const childEnv = { ...process.env };
for (const name of ['JAVA_TOOL_OPTIONS', 'JDK_JAVA_OPTIONS', '_JAVA_OPTIONS']) delete childEnv[name];
const commands = [];
function run(label, executable, args, expectedExit = 0, requiredDiagnostic) {
  const result = spawnSync(executable, args, { cwd: work, env: childEnv, encoding: 'utf8', timeout: 30000 });
  const entry = { label, executable, args, cwd: work, expectedExit, actualExit: result.status, signal: result.signal, stdout: result.stdout, stderr: result.stderr, error: result.error?.message };
  commands.push(entry);
  fs.writeFileSync(path.join(work, 'results.json'), JSON.stringify({ work, commands }, null, 2));
  assert.equal(result.error, undefined, label);
  assert.equal(result.status, expectedExit, `${label}: ${result.stderr}`);
  if (requiredDiagnostic) assert.match(result.stderr, requiredDiagnostic, label);
}
const java = path.join(bin, 'java.exe');
const javac = path.join(bin, 'javac.exe');
run('compiler-version', javac, ['-version']);
run('runtime-version', java, ['-version']);
const doc = fs.readFileSync(path.join(repo, 'skills/gof-pattern-thinking/references/alternatives-ladder.md'), 'utf8');
const snippets = [...doc.matchAll(/```java\r?\n([\s\S]*?)```/g)].map(match => match[1]);
assert.equal(snippets.length, 3);
function write(relative, text) {
  const target = path.join(work, relative);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, text);
  return target;
}

public class DiscountChecks {
    static int checks;
    static void require(boolean value) { checks++; if (!value) throw new AssertionError(); }
    static void rejects(Class<? extends Throwable> expected, Runnable action) {
        checks++;
        try { action.run(); } catch (Throwable actual) {
            if (expected.isInstance(actual)) return;
            throw new AssertionError(actual);
        }
        throw new AssertionError("expected " + expected.getName());
    }
    public static void main(String[] args) {
        Money subtotal = new Money(new BigDecimal("100.00"));
        Discount twenty = Discount.percentage("TWENTY", new BigDecimal("0.20"));
        require(twenty.code().equals("TWENTY"));
        require(twenty.rule().apply(subtotal).amount().compareTo(new BigDecimal("80")) == 0);
        require(Discount.none().rule().apply(subtotal) == subtotal);
        require(Discount.percentage("ZERO", BigDecimal.ZERO).rule().apply(subtotal).amount().compareTo(subtotal.amount()) == 0);
        require(Discount.percentage("ALL", BigDecimal.ONE).rule().apply(subtotal).amount().signum() == 0);
        rejects(IllegalArgumentException.class, () -> Discount.percentage("BAD", new BigDecimal("-0.01")));
        rejects(IllegalArgumentException.class, () -> Discount.percentage("BAD", new BigDecimal("1.01")));
        rejects(NullPointerException.class, () -> Discount.percentage("BAD", null));
        rejects(NullPointerException.class, () -> new Discount(null, twenty.rule()));
        rejects(NullPointerException.class, () -> new Discount("BAD", null));
        System.out.println("Discount checks passed: " + checks);

}
`);
