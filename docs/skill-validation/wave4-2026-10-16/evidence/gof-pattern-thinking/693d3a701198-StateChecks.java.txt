import java.time.*;
import java.util.Objects;
public class StateChecks {
sealed interface OrderState permits Draft, Paid, Shipped, Cancelled {}

static OrderState onPayment(OrderState current, Clock clock) {
    Objects.requireNonNull(clock);
    return switch (Objects.requireNonNull(current)) {
        case Draft d -> new Paid(clock.instant());
        case Paid p -> throw new IllegalTransition(p, "pay");
        case Shipped s -> throw new IllegalTransition(s, "pay");
        case Cancelled c -> throw new IllegalTransition(c, "pay");
    };
}

    record Draft() implements OrderState {}
    record Paid(Instant time) implements OrderState {}
    record Shipped() implements OrderState {}
    record Cancelled() implements OrderState {}
    static class IllegalTransition extends RuntimeException {
        IllegalTransition(OrderState state, String event) { super(state + ":" + event); }
    }
    static int checks;
    static void rejects(Class<? extends Throwable> expected, Runnable action) {
        checks++;
        try { action.run(); } catch (Throwable actual) {
            if (expected.isInstance(actual)) return;
            throw new AssertionError(actual);
        }
        throw new AssertionError("expected " + expected.getName());
    }
    public static void main(String[] args) {
        Instant now = Instant.parse("2026-10-16T12:00:00Z");
        Clock clock = Clock.fixed(now, ZoneOffset.UTC);
        if (!onPayment(new Draft(), clock).equals(new Paid(now))) throw new AssertionError();
        checks++;
        rejects(IllegalTransition.class, () -> onPayment(new Paid(now), clock));
        rejects(IllegalTransition.class, () -> onPayment(new Shipped(), clock));
        rejects(IllegalTransition.class, () -> onPayment(new Cancelled(), clock));
        rejects(NullPointerException.class, () -> onPayment(null, clock));
        rejects(NullPointerException.class, () -> onPayment(new Draft(), null));
        System.out.println("State checks passed: " + checks);
    }
}
