    record User(String id) { boolean isOwner(Order o) { return id.equals(o.ownerId()); } }
    record Viewer(String id) { boolean owns(Order o) { return id.equals(o.ownerId()); } }
    @interface GetMapping { String value(); }

    @interface RequestParam { String defaultValue(); }
    record ResponseEntity<T>(int status, T body) {
        static <T> ResponseEntity<T> ok(T body) { return new ResponseEntity<>(200, body); }
        static Builder status(int status) { return new Builder(status); }
        record Builder(int status) {
            <T> ResponseEntity<T> body(T body) { return new ResponseEntity<>(status, body); }
        }
    }
    interface Orders { Optional<Order> find(String id); }
    record Scenario(String name, Order order, String ownerId, String identity,

    record Observation(String result, List<String> effects) { }
    final List<String> effects = new ArrayList<>();
    Scenario scenario;
    final Orders orders = id -> {
        effects.add("orders:" + id);
        if (scenario.repositoryFails()) throw new IllegalStateException("repository unavailable");
        return Optional.ofNullable(scenario.order());
    };
    final OrderPresenter presenter = new OrderPresenter();
    User currentUser() {
        if (!scenario.identity().equals("snapshot")) effects.add("identity");
        if (scenario.identity().equals("throws")) throw new IllegalStateException("identity unavailable");
        return new User(scenario.ownerId());
    }
    static Viewer viewerOf(User user) { return new Viewer(user.id()); }
    static List<LineView> lineViews(Order order) { return List.of(new LineView("A-1")); }
    static Object problem(OrderView view) {
        return Map.of("error", view instanceof OrderView.NotFound ? "not found" : "forbidden");
    }
    static final class SummaryView {
        static OrderView.Summary of(Order order) {
            return new OrderView.Summary(order.id(), order.status().name(), order.total());
        }
    }
    static final class DetailView {
        static OrderView.Detailed of(Order order) {
            return new OrderView.Detailed(order.id(), order.status().name(), order.total(), lineViews(order));

    // EXTRACTED_ORDER_VIEW

    class Before {
        // EXTRACTED_BEFORE
    }
    class After {
        // EXTRACTED_AFTER
    }

    Observation observe(Scenario input, boolean after) {
        scenario = input;
        effects.clear();
        String result;
        try {
            ResponseEntity<?> response = after ? new After().get("order-7", input.detailed())
                                               : new Before().get("order-7", input.detailed());
            result = response.toString();
        } catch (RuntimeException failure) {
            result = failure.getClass().getSimpleName() + ":" + failure.getMessage();
        }
        return new Observation(result, List.copyOf(effects));
    }

    public static void main(String[] args) {
        var draft = new Order("order-7", Status.DRAFT, new BigDecimal("12.50"), "alice");
        var published = new Order("order-7", Status.PUBLISHED, new BigDecimal("12.50"), "alice");
        var fixtures = List.of(
            new Scenario("missing-unavailable-identity", null, "alice", "throws", false, false),
            new Scenario("published-unavailable-identity", published, "alice", "throws", false, false),
            new Scenario("missing-effectful-identity", null, "alice", "effectful", false, false),
            new Scenario("published-effectful-identity", published, "alice", "effectful", false, false),
            new Scenario("draft-owner", draft, "alice", "effectful", false, false),
            new Scenario("draft-nonowner", draft, "bob", "effectful", false, false),
            new Scenario("draft-identity-failure", draft, "alice", "throws", false, false),
            new Scenario("repository-fails-first", null, "alice", "throws", true, false),
            new Scenario("draft-detailed-owner", draft, "alice", "effectful", false, true),
            new Scenario("published-detailed", published, "alice", "effectful", false, true),
            new Scenario("missing-stable-snapshot", null, "alice", "snapshot", false, false),
            new Scenario("published-stable-snapshot", published, "alice", "snapshot", false, false)
        );
        ControllerFixture fixture = new ControllerFixture();
        int failed = 0;
        for (Scenario input : fixtures) {
            Observation before = fixture.observe(input, false);
            Observation after = fixture.observe(input, true);
            boolean match = before.equals(after);
            System.out.println((match ? "PASS " : "FAIL ") + input.name()
                    + "\nbefore=" + before + "\nafter=" + after);
            if (!match) failed++;
        }
        fixture.effects.clear();
        OrderView one = fixture.presenter.present("order-7", Optional.of(draft), CORE_OWNER_ARGUMENT, true);
        OrderView two = fixture.presenter.present("order-7", Optional.of(draft), CORE_OWNER_ARGUMENT, true);
        boolean pureExample = one.equals(two) && fixture.effects.isEmpty()
                && one instanceof OrderView.Detailed;
        System.out.println((pureExample ? "PASS " : "FAIL ") + "presenter-explicit-inputs");
        if (!pureExample) failed++;
        System.out.println("checks=13 failed=" + failed);
        if (failed != 0) System.exit(1);
    }
}
