import java.util.Map;
import java.util.Optional;

public class Smoke {
    public static void main(String[] args) throws Exception {
        var request = new Reports.Request("alice", new Reports.Viewer("alice"));
        var source = Reports.configured();
        var reports = Map.of("Q1", new Reports.Report("Q1", true, "alice", "Quarter one"));
        Reports.Repository repository = id -> Optional.ofNullable(reports.get(id));
        var actual = new Reports.ProposedEndpoint(repository, source).get("Q1", request);
        if (!actual.equals(new Reports.Response(200, "Quarter one"))) throw new AssertionError(actual);
        System.out.println("PASS owner-report-smoke");
    }
}
