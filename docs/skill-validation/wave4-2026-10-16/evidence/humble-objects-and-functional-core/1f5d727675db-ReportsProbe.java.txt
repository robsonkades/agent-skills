import java.util.List;
import java.util.Map;
import java.util.Optional;

public class ReportsProbe {
    record Scenario(String name, String id, Reports.Request request, boolean repositoryFails) { }
    static String observe(Reports.Repository repository, Reports.ViewerSource source,
                          Scenario scenario, boolean proposed) {
        try {
            Reports.Response response = proposed
                    ? new Reports.ProposedEndpoint(repository, source).get(scenario.id(), scenario.request())
                    : new Reports.ExistingEndpoint(repository, source).get(scenario.id(), scenario.request());
            return response.toString();
        } catch (RuntimeException failure) {
            return failure.getClass().getSimpleName() + ":" + failure.getMessage();
        }
    }
    public static void main(String[] args) throws Exception {
        var anonymous = new Reports.Request(null, new Reports.Viewer(null));
        var owner = new Reports.Request("alice", new Reports.Viewer("alice"));
        var other = new Reports.Request("bob", new Reports.Viewer("bob"));
        var source = Reports.configured();
        var reports = Map.of(
                "public", new Reports.Report("public", false, "alice", "Published"),
                "restricted", new Reports.Report("restricted", true, "alice", "Internal"));
        var scenarios = List.of(
                new Scenario("missing-anonymous", "missing", anonymous, false),
                new Scenario("public-anonymous", "public", anonymous, false),
                new Scenario("restricted-owner", "restricted", owner, false),
                new Scenario("restricted-nonowner", "restricted", other, false),
                new Scenario("restricted-anonymous", "restricted", anonymous, false),
                new Scenario("repository-failure", "missing", anonymous, true));
        int failed = 0;
        for (Scenario scenario : scenarios) {
            Reports.Repository repository = id -> {
                if (scenario.repositoryFails()) throw new IllegalStateException("Repository unavailable");
                return Optional.ofNullable(reports.get(id));
            };
            String existing = observe(repository, source, scenario, false);
            String proposed = observe(repository, source, scenario, true);
            boolean same = existing.equals(proposed);
            System.out.println((same ? "PASS " : "FAIL ") + scenario.name()
                    + "\nexisting=" + existing + "\nproposed=" + proposed);
            if (!same) failed++;
        }
        System.out.println("checks=6 failed=" + failed);
        if (failed != 0) System.exit(1);
    }
}
