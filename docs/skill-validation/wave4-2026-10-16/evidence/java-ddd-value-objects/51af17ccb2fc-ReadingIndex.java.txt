import java.util.HashSet;
import java.util.List;
import java.util.Set;

public final class ReadingIndex {
    public static Set<GaugeReading> distinct(List<ImportRow> rows) {
        Set<GaugeReading> readings = new HashSet<>();
        for (ImportRow row : rows) readings.add(row.toValue());
        return Set.copyOf(readings);
    }
}
