import java.math.BigDecimal;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public final class ContractProbe {
    private static int checks;
    private static void check(boolean ok, String message) {
        if (!ok) throw new AssertionError(message);
        checks++;
    }
    private static void rejects(Runnable action) {
        try { action.run(); }
        catch (IllegalArgumentException | NullPointerException expected) { checks++; return; }
        throw new AssertionError("invalid reading accepted");
    }
    public static void main(String[] args) {
        boolean numericOnly = args[0].equals("numeric");
        GaugeReading one = GaugeReading.parse("1.0", "bar");
        GaugeReading equalSameScale = GaugeReading.parse("01.0", "bar");
        GaugeReading otherScale = GaugeReading.parse("1.00", "bar");
        check(one.equals(equalSameScale) && equalSameScale.equals(one), "same-scale equality");
        check(one.hashCode() == equalSameScale.hashCode(), "equal values share hash");
        check(one.equals(otherScale) == numericOnly, "cross-scale equivalence must follow approved policy");
        check(!one.equals(GaugeReading.parse("1.0", "kPa")), "unit remains significant");
        check(!one.equals(GaugeReading.parse("1.5", "bar")), "different number remains distinct");
        Set<GaugeReading> keys = new HashSet<>(List.of(one, otherScale));
        check(keys.size() == (numericOnly ? 1 : 2), "hash collection respects equality");
        check(keys.contains(new GaugeReading(new BigDecimal("1.00"), "bar")), "canonical constructor agrees");
        check(ReadingIndex.distinct(List.of(new ImportRow("1.0", "bar"),
                new ImportRow("1.00", "bar"), new ImportRow("1.5", "bar"),
                new ImportRow("1.0", "kPa"))).size() == (numericOnly ? 3 : 4), "import deduplication");
        ImportRow raw = new ImportRow("0001.00", "bar");
        raw.toValue();
        check(raw.rawNumber().equals("0001.00"), "raw audit representation preserved");
        rejects(() -> GaugeReading.parse("1.0x", "bar"));
        rejects(() -> GaugeReading.parse("1e0", "bar"));
        rejects(() -> GaugeReading.parse(" 1.0", "bar"));
        rejects(() -> GaugeReading.parse("1.0", "BAR"));
        rejects(() -> new GaugeReading(new BigDecimal("-1"), "bar"));
        rejects(() -> new GaugeReading(new BigDecimal("1000.01"), "bar"));
        rejects(() -> new GaugeReading(null, "bar"));
        System.out.println("PASS " + checks + " value-contract checks for " + args[0]);
    }
}
