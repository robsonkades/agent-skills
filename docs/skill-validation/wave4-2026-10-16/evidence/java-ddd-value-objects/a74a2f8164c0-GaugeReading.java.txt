import java.math.BigDecimal;
import java.util.Objects;
import java.util.Set;
import java.util.regex.Pattern;

public record GaugeReading(BigDecimal value, String unit) {
    private static final Pattern NUMBER = Pattern.compile("[0-9]{1,4}(?:\\.[0-9]{1,3})?");
    private static final Set<String> UNITS = Set.of("bar", "kPa");

    public GaugeReading {
        Objects.requireNonNull(value, "value");
        Objects.requireNonNull(unit, "unit");
        value = value.stripTrailingZeros();
        if (!UNITS.contains(unit)) throw new IllegalArgumentException("unsupported unit");
        if (value.signum() < 0 || value.compareTo(new BigDecimal("1000")) > 0) {
            throw new IllegalArgumentException("reading outside range");
        }
    }

    public static GaugeReading parse(String raw, String unit) {
        Objects.requireNonNull(raw, "raw");
        if (raw.length() > 8 || !NUMBER.matcher(raw).matches()) {
            throw new IllegalArgumentException("invalid reading text");
        }
        return new GaugeReading(new BigDecimal(raw), unit);
    }
}
