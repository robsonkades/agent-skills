import java.nio.file.Files;
import java.nio.file.Path;
import java.sql.*;
import java.util.Properties;

public final class CaseProbe {
    record Change(int id, long expectedVersion, int observedAvailable, boolean permitted) { }

    static final class OperatorGateway {
        private final Connection connection;
        OperatorGateway(Connection connection) { this.connection = connection; }
        int available() throws SQLException {
            try (Statement s = connection.createStatement();
                 ResultSet r = s.executeQuery("select count(*) from operator_availability where available = true")) {
                r.next(); return r.getInt(1);
            }
        }
        long version(int id) throws SQLException {
            try (PreparedStatement s = connection.prepareStatement("select version from operator_availability where id = ?")) {
                s.setInt(1, id);
                try (ResultSet r = s.executeQuery()) {
                    if (!r.next()) throw new SQLException("Missing operator " + id);
                    return r.getLong(1);
                }
            }
        }
        int disable(int id, long expected) throws SQLException {
            try (PreparedStatement s = connection.prepareStatement(
                    "update operator_availability set available = false, version = version + 1 " +
                    "where id = ? and available = true and version = ?")) {
                s.setInt(1, id); s.setLong(2, expected); return s.executeUpdate();
            }
        }
    }
    static final class AvailabilityService {
        private final OperatorGateway gateway;
        private final int minimum;
        AvailabilityService(OperatorGateway gateway, int minimum) {
            this.gateway = gateway; this.minimum = minimum;
        }
        Change prepare(int id) throws SQLException {
            int count = gateway.available();
            return new Change(id, gateway.version(id), count, count > minimum);
        }
        int apply(Change change) throws SQLException {
            return change.permitted() ? gateway.disable(change.id(), change.expectedVersion()) : 0;
        }
    }
    public static void main(String[] args) throws Exception {
        Properties policy = new Properties();
        try (var in = Files.newInputStream(Path.of("operating-policy.properties"))) { policy.load(in); }
        int minimum = Integer.parseInt(policy.getProperty("minimum.available"));
        String url = "jdbc:h2:mem:case_probe;LOCK_TIMEOUT=1000";
        try (Connection setup = DriverManager.getConnection(url)) {
            for (String sql : Files.readString(Path.of("schema.sql")).split(";")) {
                if (!sql.isBlank()) try (Statement statement = setup.createStatement()) { statement.execute(sql); }
            }
            System.out.println("runtime=" + System.getProperty("java.runtime.version"));
            System.out.println("database=" + setup.getMetaData().getDatabaseProductVersion());
            try (Connection a = DriverManager.getConnection(url); Connection b = DriverManager.getConnection(url)) {
                a.setAutoCommit(false); b.setAutoCommit(false);
                a.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                b.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                AvailabilityService sa = new AvailabilityService(new OperatorGateway(a), minimum);
                AvailabilityService sb = new AvailabilityService(new OperatorGateway(b), minimum);
                Change ca = sa.prepare(1);
                Change cb = sb.prepare(2);
                System.out.println("isolationA=" + a.getTransactionIsolation() + " isolationB=" + b.getTransactionIsolation());
                System.out.println("observedA=" + ca.observedAvailable() + " observedB=" + cb.observedAvailable());
                int rowsA = sa.apply(ca);
                if (rowsA == 1) a.commit(); else a.rollback();
                int rowsB = sb.apply(cb);
                if (rowsB == 1) b.commit(); else b.rollback();
                System.out.println("rowsA=" + rowsA + " rowsB=" + rowsB);
            }
            OperatorGateway observer = new OperatorGateway(setup);
            System.out.println("available=" + observer.available());
            System.out.println("versionA=" + observer.version(1) + " versionB=" + observer.version(2));
        }
    }
}
