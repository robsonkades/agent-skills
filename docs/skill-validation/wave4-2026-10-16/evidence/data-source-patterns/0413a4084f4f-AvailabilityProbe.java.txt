import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;

/** Isolated JDBC interleaving fixture; not an ORM or PostgreSQL test. */
public final class AvailabilityProbe {
    record Decision(int operatorId, long expectedVersion, int observedAvailable,
                    boolean allowed, long expectedGuardVersion) { }
    record Outcome(int minimum, boolean sharedGuard, int firstRows, int secondRows,
                   int firstGuardRows, int secondGuardRows, int available,
                   long firstVersion, long secondVersion) { }

    static final class OperatorGateway {
        private final Connection connection;
        OperatorGateway(Connection connection) { this.connection = connection; }

        int available() throws SQLException {
            try (Statement s = connection.createStatement();
                 ResultSet r = s.executeQuery("select count(*) from operator_availability where available = true")) {
                r.next(); return r.getInt(1);
            }
        }
        long version(int id) throws SQLException {
            try (PreparedStatement s = connection.prepareStatement("select version from operator_availability where id = ?")) {
                s.setInt(1, id);
                try (ResultSet r = s.executeQuery()) {
                    if (!r.next()) throw new SQLException("Missing operator " + id);
                    return r.getLong(1);
                }
            }
        }
        int disable(int id, long expectedVersion) throws SQLException {
            try (PreparedStatement s = connection.prepareStatement(
                    "update operator_availability set available = false, version = version + 1 " +
                    "where id = ? and available = true and version = ?")) {
                s.setInt(1, id); s.setLong(2, expectedVersion); return s.executeUpdate();
            }
        }
        long guardVersion() throws SQLException {
            try (Statement s = connection.createStatement(); ResultSet r = s.executeQuery("select version from roster_guard where id = 1")) {
                r.next(); return r.getLong(1);
            }
        }
        int advanceGuard(long expected) throws SQLException {
            try (PreparedStatement s = connection.prepareStatement("update roster_guard set version = version + 1 where id = 1 and version = ?")) {
                s.setLong(1, expected); return s.executeUpdate();
            }
        }
    }

    static Decision decide(OperatorGateway gateway, int operatorId, int minimum, boolean sharedGuard) throws SQLException {
        // Caller owns the transaction; business policy stays out of the SQL gateway.
        long guard = sharedGuard ? gateway.guardVersion() : -1;
        int available = gateway.available();
        return new Decision(operatorId, gateway.version(operatorId), available, available > minimum, guard);
    }

    static Outcome execute(String name, int minimum, boolean sharedGuard) throws Exception {
        String url = "jdbc:h2:mem:" + name + ";LOCK_TIMEOUT=1000";
        try (Connection setup = DriverManager.getConnection(url)) {
            try (Statement s = setup.createStatement()) {
                s.execute("create table operator_availability (id int primary key, available boolean not null, version bigint not null)");
                s.execute("create table roster_guard (id int primary key, version bigint not null)");
                s.execute("insert into operator_availability values (1, true, 0), (2, true, 0)");
                s.execute("insert into roster_guard values (1, 0)");
            }
            try (Connection a = DriverManager.getConnection(url); Connection b = DriverManager.getConnection(url)) {
                for (Connection c : new Connection[] {a, b}) {
                    c.setAutoCommit(false);
                    c.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                    if (c.getTransactionIsolation() != Connection.TRANSACTION_READ_COMMITTED) throw new AssertionError("Isolation mismatch");
                }
                OperatorGateway ga = new OperatorGateway(a), gb = new OperatorGateway(b);
                Decision da = decide(ga, 1, minimum, sharedGuard);
                Decision db = decide(gb, 2, minimum, sharedGuard);
                if (da.observedAvailable != 2 || db.observedAvailable != 2) throw new AssertionError("Required overlap missing");
                int aGuard = sharedGuard ? ga.advanceGuard(da.expectedGuardVersion) : -1;
                int aRows = da.allowed && (!sharedGuard || aGuard == 1) ? ga.disable(da.operatorId, da.expectedVersion) : 0;
                if (aRows == 1) a.commit(); else a.rollback();
                int bGuard = sharedGuard ? gb.advanceGuard(db.expectedGuardVersion) : -1;
                int bRows = db.allowed && (!sharedGuard || bGuard == 1) ? gb.disable(db.operatorId, db.expectedVersion) : 0;
                if (bRows == 1) b.commit(); else b.rollback();
                OperatorGateway observer = new OperatorGateway(setup);
                return new Outcome(minimum, sharedGuard, aRows, bRows, aGuard, bGuard,
                        observer.available(), observer.version(1), observer.version(2));
            }
        }
    }

    static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
    public static void main(String[] args) throws Exception {
        System.out.println("java=" + System.getProperty("java.runtime.version"));
        try (Connection c = DriverManager.getConnection("jdbc:h2:mem:metadata")) {
            System.out.println("database=" + c.getMetaData().getDatabaseProductVersion());
        }
        Outcome required = execute("required_one", 1, false);
        require(required.firstRows == 1 && required.secondRows == 1, "Both disjoint row-version writes must succeed");
        require(required.available == 0 && required.available < required.minimum, "Cross-row invariant should be exposed as violated");
        require(required.firstVersion == 1 && required.secondVersion == 1, "Both row versions must advance");
        Outcome independent = execute("zero_required", 0, false);
        require(independent.firstRows == 1 && independent.secondRows == 1, "Both independent changes should succeed");
        require(independent.available == 0 && independent.available >= independent.minimum, "Zero-active contract permits the outcome");
        Outcome coordinated = execute("coordinated", 1, true);
        require(coordinated.firstRows == 1 && coordinated.secondRows == 0, "Shared guard should reject the stale second decision");
        require(coordinated.firstGuardRows == 1 && coordinated.secondGuardRows == 0, "Guard CAS must conflict");
        require(coordinated.available == 1 && coordinated.firstVersion == 1 && coordinated.secondVersion == 0, "Rejected writer must preserve state");
        System.out.println(required);
        System.out.println(independent);
        System.out.println(coordinated);
        System.out.println("PASS: 8 assertions; three deterministic two-transaction controls; no ORM or PostgreSQL executed");
    }
}
