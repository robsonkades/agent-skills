import fs from 'node:fs';
import { spawnSync } from 'node:child_process';
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const h2 = 'C:/Users/robso/.m2/repository/com/h2database/h2/2.2.224/h2-2.2.224.jar';
if (!fs.existsSync(h2)) throw new Error('Prerequisite H2 2.2.224 jar missing; do not install automatically.');
fs.mkdirSync('classes', { recursive: true });
const env = { ...process.env };
for (const key of ['JDK_JAVA_OPTIONS', 'JAVA_TOOL_OPTIONS', '_JAVA_OPTIONS']) delete env[key];
for (const [bin, args] of [
    ['javac', ['--release', '17', '-cp', h2, '-d', 'classes', 'CaseProbe.java']],
    ['java', ['-cp', 'classes;' + h2, 'CaseProbe']]
]) {
    const r = spawnSync(jdk + '/' + bin + '.exe', args, { cwd: process.cwd(), env, encoding: 'utf8', windowsHide: true, timeout: 20000 });
    process.stdout.write(r.stdout ?? '');
    process.stderr.write(r.stderr ?? '');
    if (r.error) throw r.error;
    if (r.status !== 0) process.exit(r.status ?? 1);
}
