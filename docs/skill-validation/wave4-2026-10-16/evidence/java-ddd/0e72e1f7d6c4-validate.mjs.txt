function command(id, executable,args) {
  const ran=spawnSync(executable,args,{cwd:repo,encoding:'utf8',windowsHide:true,timeout:30000});
  const stdout=join(evidence,'logs',id+'.stdout.txt');
  const stderr=join(evidence,'logs',id+'.stderr.txt');
  writeFileSync(stdout,ran.stdout??'');
  writeFileSync(stderr,ran.stderr??'');
  result.commands.push({id,executable,args,cwd:repo,status:ran.status,signal:ran.signal,error:ran.error?String(ran.error):null,stdout,stderr});
  return ran.status;
}

function walk(directory,prefix='') {
  for(const entry of readdirSync(directory,{withFileTypes:true})){
    const relative=prefix+entry.name;

}
