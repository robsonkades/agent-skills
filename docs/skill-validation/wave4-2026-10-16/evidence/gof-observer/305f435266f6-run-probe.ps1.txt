$ErrorActionPreference = 'Stop'
$probeDirectory = $PSScriptRoot
$javaToolDirectory = 'C:/Users/robso/.jdks/temurin-25.0.3/bin'
$classesDirectory = Join-Path $probeDirectory 'classes'
New-Item -ItemType Directory -Path $classesDirectory -Force | Out-Null
& (Join-Path $javaToolDirectory 'java.exe') -version 2>&1 | Tee-Object -FilePath (Join-Path $probeDirectory 'java-version.log')
& (Join-Path $javaToolDirectory 'javac.exe') --release 17 -d $classesDirectory (Join-Path $probeDirectory 'SnapshotBus.java') (Join-Path $probeDirectory 'ObserverProbe.java') 2>&1 | Tee-Object -FilePath (Join-Path $probeDirectory 'compile.log')
$compileExit = $LASTEXITCODE
Write-Output "javac exit code: $compileExit"
if ($compileExit -ne 0) { throw 'Probe compilation failed' }
$stdoutPath = Join-Path $probeDirectory 'probe.stdout.log'
$stderrPath = Join-Path $probeDirectory 'probe.stderr.log'
$probeArguments = @('-cp', ('"' + $classesDirectory + '"'), 'ObserverProbe')
$probeProcess = Start-Process -FilePath (Join-Path $javaToolDirectory 'java.exe') -ArgumentList $probeArguments -PassThru -WindowStyle Hidden -RedirectStandardOutput $stdoutPath -RedirectStandardError $stderrPath
$completed = $probeProcess.WaitForExit(30000)
if (-not $completed) {
    $probeProcess.Kill()
    $probeProcess.WaitForExit()
    throw 'ObserverProbe exceeded 30 seconds'
}
$probeProcess.Refresh()
Get-Content -LiteralPath $stdoutPath
Get-Content -LiteralPath $stderrPath
Write-Output "java exit code: $($probeProcess.ExitCode)"
if ($probeProcess.ExitCode -ne 0) { throw 'ObserverProbe failed' }
