import java.util.List;
import java.util.Objects;
import java.util.concurrent.CopyOnWriteArrayList;

// The field and publish body below are copied verbatim from the reviewed partial example.
// Nested types, constructor and subscribe are harness completions, not package content.
final class SnapshotBus {
    record Event(int sequence) {}

    @FunctionalInterface
    interface Listener {
        void on(Event event);
    }

    @FunctionalInterface
    interface FailureRecorder {
        void record(Listener listener, Event event, RuntimeException failure);
    }

    @FunctionalInterface
    interface Subscription extends AutoCloseable {
        @Override
        void close();
    }

    private final FailureRecorder failures;

    SnapshotBus(FailureRecorder failures) {
        this.failures = Objects.requireNonNull(failures);
    }

    Subscription subscribe(Listener listener) {
        listeners.add(Objects.requireNonNull(listener));
        return () -> listeners.remove(listener);
    }

    private final List<Listener> listeners = new CopyOnWriteArrayList<>();
    
    void publish(Event event) {
        for (Listener listener : listeners) {          // snapshot iteration: safe to modify during
            try {
                listener.on(event);
            } catch (RuntimeException e) {
                failures.record(listener, event, e);    // policy: isolate and record
            }
        }
    }
}
