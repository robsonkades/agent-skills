import java.beans.PropertyChangeSupport;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicReference;

public final class ObserverProbe {
    public static void main(String[] args) throws Exception {
        removedBeforePublication();
        removedAfterSnapshotCapture();
        runtimeFailureIsolated();
        recorderFailurePropagates();
        errorPropagates();
        reentrantPublication();
        propertyChangeEquality();
        System.out.println("PASS: 7 observer contract controls");
    }

    private static SnapshotBus noFailures() {
        return new SnapshotBus((listener, event, failure) -> {
            throw new AssertionError("Unexpected listener failure", failure);
        });
    }

    private static void removedBeforePublication() {
        SnapshotBus bus = noFailures();
        AtomicInteger calls = new AtomicInteger();
        SnapshotBus.Subscription subscription = bus.subscribe(event -> calls.incrementAndGet());
        subscription.close();
        bus.publish(new SnapshotBus.Event(1));
        check(calls.get() == 0, "removed before publication must not receive an event");
        System.out.println("PASS: remove before publish, calls=0");
    }

    private static void removedAfterSnapshotCapture() throws Exception {
        SnapshotBus bus = noFailures();
        CountDownLatch captured = new CountDownLatch(1);
        CountDownLatch release = new CountDownLatch(1);
        AtomicInteger calls = new AtomicInteger();
        AtomicReference<Throwable> workerFailure = new AtomicReference<>();
        bus.subscribe(event -> {
            if (event.sequence() == 1) {
                captured.countDown();
                try {
                    if (!release.await(5, TimeUnit.SECONDS)) {
                        throw new AssertionError("Timed out waiting for release");
                    }
                } catch (InterruptedException interrupted) {
                    Thread.currentThread().interrupt();
                    throw new AssertionError(interrupted);
                }
            }
        });
        SnapshotBus.Subscription subscription = bus.subscribe(event -> calls.incrementAndGet());
        Thread publisher = new Thread(() -> {
            try {
                bus.publish(new SnapshotBus.Event(1));
            } catch (Throwable failure) {
                workerFailure.set(failure);
            }
        }, "observer-snapshot-probe");
        publisher.start();
        try {
            check(captured.await(5, TimeUnit.SECONDS), "publisher did not reach captured snapshot");
            subscription.close();
            check(calls.get() == 0, "second listener must still be pending");
            release.countDown();
            publisher.join(6000);
            check(!publisher.isAlive(), "publisher did not finish");
            check(workerFailure.get() == null, "publisher failed: " + workerFailure.get());
            check(calls.get() == 1, "captured snapshot should retain the removed listener");
            bus.publish(new SnapshotBus.Event(2));
            check(calls.get() == 1, "subsequent snapshot should omit the removed listener");
            System.out.println("PASS: remove after snapshot, captured calls=1; subsequent added calls=0");
        } finally {
            release.countDown();
            publisher.join(6000);
            if (publisher.isAlive()) {
                publisher.interrupt();
                publisher.join(1000);
            }
            check(!publisher.isAlive(), "publisher thread leaked");
        }
    }

    private static void runtimeFailureIsolated() {
        AtomicInteger recorded = new AtomicInteger();
        AtomicInteger later = new AtomicInteger();
        IllegalStateException sentinel = new IllegalStateException("listener sentinel");
        SnapshotBus bus = new SnapshotBus((listener, event, failure) -> {
            check(failure == sentinel, "failure identity was changed");
            recorded.incrementAndGet();
        });
        bus.subscribe(event -> { throw sentinel; });
        bus.subscribe(event -> later.incrementAndGet());
        bus.publish(new SnapshotBus.Event(1));
        check(recorded.get() == 1 && later.get() == 1, "isolation policy not exercised");
        System.out.println("PASS: RuntimeException recorded=1; later listener calls=1");
    }

    private static void recorderFailurePropagates() {
        AtomicInteger later = new AtomicInteger();
        IllegalStateException sentinel = new IllegalStateException("recorder sentinel");
        SnapshotBus bus = new SnapshotBus((listener, event, failure) -> { throw sentinel; });
        bus.subscribe(event -> { throw new IllegalArgumentException("listener failure"); });
        bus.subscribe(event -> later.incrementAndGet());
        try {
            bus.publish(new SnapshotBus.Event(1));
            throw new AssertionError("Expected recorder failure");
        } catch (IllegalStateException failure) {
            check(failure == sentinel, "unexpected exception");
        }
        check(later.get() == 0, "recorder failure must abort this loop");
        System.out.println("PASS: throwing recorder propagated; later listener calls=0");
    }

    private static void errorPropagates() {
        AtomicInteger recorded = new AtomicInteger();
        AtomicInteger later = new AtomicInteger();
        AssertionError sentinel = new AssertionError("listener Error sentinel");
        SnapshotBus bus = new SnapshotBus((listener, event, failure) -> recorded.incrementAndGet());
        bus.subscribe(event -> { throw sentinel; });
        bus.subscribe(event -> later.incrementAndGet());
        try {
            bus.publish(new SnapshotBus.Event(1));
            throw new AssertionError("Expected listener Error");
        } catch (AssertionError failure) {
            check(failure == sentinel, "unexpected Error");
        }
        check(recorded.get() == 0 && later.get() == 0, "Error was treated as RuntimeException");
        System.out.println("PASS: Error propagated; recorded=0; later listener calls=0");
    }

    private static void reentrantPublication() {
        SnapshotBus bus = noFailures();
        List<Integer> observed = new ArrayList<>();
        bus.subscribe(event -> {
            if (event.sequence() == 1) bus.publish(new SnapshotBus.Event(2));
        });
        bus.subscribe(event -> observed.add(event.sequence()));
        bus.publish(new SnapshotBus.Event(1));
        check(observed.equals(List.of(2, 1)), "nested publication did not precede outer delivery");
        System.out.println("PASS: reentrant later-listener sequence=" + observed);
    }

    private static void propertyChangeEquality() {
        PropertyChangeSupport changes = new PropertyChangeSupport(new Object());
        AtomicInteger calls = new AtomicInteger();
        changes.addPropertyChangeListener(event -> calls.incrementAndGet());
        changes.firePropertyChange("value", new String("same"), new String("same"));
        check(calls.get() == 0, "equal non-null values should be suppressed");
        changes.firePropertyChange("value", (Object) null, "changed");
        check(calls.get() == 1, "null to value should fire");
        changes.firePropertyChange("value", (Object) null, (Object) null);
        check(calls.get() == 2, "two null Object values should fire");
        System.out.println("PASS: PropertyChangeSupport equality suppressed=1; null-involving events=2");
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
