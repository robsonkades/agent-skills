import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import { spawnSync } from 'node:child_process';
import { fileURLToPath, pathToFileURL } from 'node:url';

const repo = 'C:/git/agent-skills';
const skill = 'architecture-fitness-functions';
const root = join(repo, 'skills', skill);
const evidence = dirname(fileURLToPath(import.meta.url));
const baseline = JSON.parse(readFileSync(join(evidence, 'baseline.json'), 'utf8').replace(/^\uFEFF/, ''));
mkdirSync(join(evidence, 'logs'), { recursive: true });
const commands = [];
function run(label, command, args, cwd = root) {
  const result = spawnSync(command, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = {
    label, command, args, cwd, exitCode: result.status, signal: result.signal,
    error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '',
  };
  record.log = join(evidence, 'logs', label + '.json');
  writeFileSync(record.log, JSON.stringify(record, null, 2) + '\n');
  commands.push(record);
  assert.equal(record.error, null, label + ': ' + record.error);
  assert.equal(record.signal, null, label);
  assert.equal(record.exitCode, 0, label + '\n' + record.stdout + record.stderr);
  return record;
}

const tests = run('register-tests', process.execPath, ['--test', 'scripts/check-governance-register.test.mjs']);
assert.match(tests.stdout, /tests 25/);
assert.match(tests.stdout, /pass 25/);
assert.match(tests.stdout, /fail 0/);
assert.match(tests.stdout, /skipped 0/);

const { loadPackageFromDirectory } = await import(pathToFileURL(join(repo, 'packages/core/src/application/package-loader.ts')).href);
const { validatePackage } = await import(pathToFileURL(join(repo, 'packages/core/src/application/validate-package.ts')).href);
const { NodeFileSystem } = await import(pathToFileURL(join(repo, 'packages/node/src/file-system.ts')).href);
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), root, { strict: true });
const validated = validatePackage(loaded.pkg, { strict: true, directoryName: skill });
const issues = [...loaded.issues, ...validated.issues];
assert.deepEqual(issues, []);
assert.equal(loaded.pkg.manifest.version, '1.3.3');
assert.equal(loaded.pkg.files.length, 7);
const references = [];
for (const file of baseline.hashes.filter((file) => file.path.endsWith('.md'))) {
  const absolute = join(repo, file.path);
  const body = readFileSync(absolute, 'utf8');
  for (const match of body.matchAll(/\[[^\]]+\]\(([^)]+)\)/g)) {
    if (/^(?:https?:|#)/.test(match[1])) continue;
    const target = resolve(dirname(absolute), match[1].split('#')[0]);
    assert.ok(existsSync(target), absolute + ': ' + match[1]);
    references.push({ from: file.path, target: match[1] });
  }
}
const referrals = loaded.pkg.manifest.suggests;
for (const referral of referrals) {
  assert.ok(existsSync(join(repo, 'skills', referral, 'SKILL.md')), referral);
}
const expectedReferrals = [
  'architecture-characteristics', 'architecture-testing', 'architecture-trade-off-analysis',
  'performance-regression-ci', 'quality-gates', 'slo-and-alerting',
];
assert.deepEqual([...referrals].sort(), expectedReferrals.sort());

run('prettier-check', process.execPath, [
  join(repo, 'node_modules/prettier/bin/prettier.cjs'), '--check',
  ...baseline.hashes.map((file) => join(repo, file.path)),
], repo);
run('diff-check', 'git', ['diff', '--check', '--', 'skills/' + skill], repo);
const status = run('status', 'git', ['status', '--short', '--', 'skills/' + skill], repo);
const diff = run('diff', 'git', ['diff', '--', 'skills/' + skill], repo);
const staged = run('staged-diff', 'git', ['diff', '--cached', '--', 'skills/' + skill], repo);
assert.equal(status.stdout, '');
assert.equal(diff.stdout, '');
assert.equal(staged.stdout, '');
const finalHashes = baseline.hashes.map((file) => ({
  path: file.path,
  sha256: createHash('sha256').update(readFileSync(join(repo, file.path))).digest('hex'),
}));
for (const file of finalHashes) {
  assert.equal(file.sha256, baseline.hashes.find((entry) => entry.path === file.path).sha256, file.path);
}
const results = {
  skill, checkedAt: new Date().toISOString(), node: process.version, platform: process.platform,
  commands, strictValidation: { issues, shippedFiles: loaded.pkg.files.length },
  internalLinks: references, optionalReferrals: referrals, finalHashes,
  changedSinceAssignment: [], noActiveChildren: true,
};
writeFileSync(join(evidence, 'local-results.json'), JSON.stringify(results, null, 2) + '\n');
process.stdout.write(JSON.stringify({
  tests: 25, failures: 0, skipped: 0, commands: commands.length, shippedFiles: loaded.pkg.files.length,
  issues: issues.length, links: references.length, referrals: referrals.length,
  baselineHashMatches: finalHashes.length, changedSinceAssignment: [],
}, null, 2) + '\n');
