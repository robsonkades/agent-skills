package imports;

import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.time.format.ResolverStyle;
import java.util.Locale;
import java.util.Properties;

// Evaluator-only differential control, not an actor fixture or performance benchmark.
public final class UncachedControl {
    public static void main(String[] args) throws Exception {
        Path directory = Path.of(args[0]);
        var properties = new Properties();
        try (var input = Files.newInputStream(directory.resolve("import.properties"))) {
            properties.load(input);
        }
        boolean overrides = Boolean.parseBoolean(properties.getProperty("allowLocaleOverride"));
        Locale defaultLocale = Locale.forLanguageTag(properties.getProperty("defaultLocale"));
        for (String row : Files.readAllLines(directory.resolve("sample-inputs.csv")).subList(1, 3)) {
            String[] fields = row.split(",", -1);
            Locale locale = overrides ? Locale.forLanguageTag(fields[0]) : defaultLocale;
            try {
                System.out.println(fields[0] + "," + LocalDate.parse(fields[2],
                        DateTimeFormatter.ofPattern(fields[1], locale)
                                .withResolverStyle(ResolverStyle.STRICT)));
            } catch (DateTimeParseException invalid) {
                System.out.println(fields[0] + ",REJECTED");
            }
        }
    }
}
