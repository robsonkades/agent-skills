package example.application;

import java.io.IOException;
import java.io.InputStream;
import java.util.Arrays;
import java.util.zip.CRC32;

public final class ExportPipeline {
    public long process(ExportBody body, int reads) throws IOException {
        if (reads < 1 || reads > 2) throw new IllegalArgumentException("reads");
        byte[] first = null;
        for (int pass = 0; pass < reads; pass++) {
            try (InputStream input = body.open()) {
                byte[] bytes = input.readAllBytes();
                if (first == null) first = bytes;
                else if (!Arrays.equals(first, bytes)) throw new IllegalStateException("content changed");
            }
        }
        CRC32 checksum = new CRC32();
        checksum.update(first);
        return checksum.getValue();
    }
}
