package example.adapter;

import example.application.ExportBody;
import java.io.ByteArrayInputStream;
import org.springframework.core.io.InputStreamResource;

public final class ExportFactory {
    private ExportFactory() {}

    public static ExportBody create(byte[] bytes) {
        if (bytes.length > 4096) throw new IllegalArgumentException("export limit");
        byte[] snapshot = bytes.clone();
        return new SpringExportBody(new InputStreamResource(() -> new ByteArrayInputStream(snapshot)));
    }
}
