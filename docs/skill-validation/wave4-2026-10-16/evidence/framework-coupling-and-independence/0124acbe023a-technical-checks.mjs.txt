import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join, dirname, delimiter } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';

const root = dirname(fileURLToPath(import.meta.url));
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const jar = 'C:/Users/robso/.m2/repository/org/springframework/spring-core/6.2.10/spring-core-6.2.10.jar';
const expectedJarHash = '28122D6E4A48EB8B6A0E4E087A55E96E46FB4D5E7D5D2C74DC8051401D8101DA';
assert.equal(createHash('sha256').update(readFileSync(jar)).digest('hex').toUpperCase(), expectedJarHash);
const app = join(root, 'outputs/application');
const original = join(root, 'outputs/original');
const replayable = join(root, 'outputs/replayable');
const logs = join(root, 'logs');
for (const path of [app, original, replayable, logs]) mkdirSync(path, { recursive: true });
const commands = [];
function run(id, executable, args, expectedExit = 0) {
    const result = spawnSync(join(jdk, executable + '.exe'), args, {
        cwd: root, encoding: 'utf8', timeout: 30000, windowsHide: true
    });
    const stdoutPath = join(logs, id + '.stdout.txt');
    const stderrPath = join(logs, id + '.stderr.txt');
    writeFileSync(stdoutPath, result.stdout ?? '');
    writeFileSync(stderrPath, result.stderr ?? '');
    commands.push({ id, executable: join(jdk, executable + '.exe'), args, cwd: root,
        expectedExit, actualExit: result.status, signal: result.signal,
        error: result.error?.message ?? null, stdoutPath, stderrPath });
    writeFileSync(join(root, 'commands.json'), JSON.stringify(commands, null, 2) + '\n');
    assert.equal(result.error, undefined, id + ': process error');
    assert.equal(result.status, expectedExit, id + ': exit mismatch');
    return { stdout: result.stdout, stderr: result.stderr };
}
const src = join(root, 'actor/common/src');
const appSources = ['example/application/ExportBody.java', 'example/application/ExportPipeline.java'].map(p => join(src, p));
const adapter = join(src, 'example/adapter/SpringExportBody.java');
const runner = join(src, 'example/Runner.java');
const control = join(root, 'controls/ResourceControls.java');
const caseA = join(root, 'actor/case-a/consumer.properties');
const caseB = join(root, 'actor/case-b/consumer.properties');
const javaVersion = run('01-java-version', 'java', ['-version']);
const javacVersion = run('02-javac-version', 'javac', ['-version']);
run('03-application-without-spring', 'javac', ['--release', '17', '-d', app, ...appSources]);
run('04-original-with-spring', 'javac', ['--release', '17', '-cp', [app, jar].join(delimiter), '-d', original,
    adapter, join(src, 'example/adapter/ExportFactory.java'), runner, control]);
const originalCp = [app, original, jar].join(delimiter);
const singleRead = run('05-original-single-read', 'java', ['-cp', originalCp, 'example.Runner', caseA]);
assert.match(singleRead.stdout, /PASS reads=1 checksum=/);
const twoReads = run('06-original-two-reads', 'java', ['-cp', originalCp, 'example.Runner', caseB], 1);
assert.match(twoReads.stderr, /IllegalStateException/);
assert.match(twoReads.stderr, /already been read/);
const originalControls = run('07-original-controls', 'java', ['-cp', originalCp, 'ResourceControls', '1']);
assert.match(originalControls.stdout, /PASS lifecycle, forwarding, direct resource use, failure cleanup, snapshot; opens=1 closes=1/);
run('08-replayable-with-spring', 'javac', ['--release', '17', '-cp', [app, jar].join(delimiter), '-d', replayable,
    adapter, join(root, 'controls/replayable/example/adapter/ExportFactory.java'), runner, control]);
const replayableCp = [app, replayable, jar].join(delimiter);
const fixedTwo = run('09-replayable-two-reads', 'java', ['-cp', replayableCp, 'example.Runner', caseB]);
assert.match(fixedTwo.stdout, /PASS reads=2 checksum=/);
const fixedOne = run('10-replayable-single-read', 'java', ['-cp', replayableCp, 'example.Runner', caseA]);
assert.match(fixedOne.stdout, /PASS reads=1 checksum=/);
const fixedControls = run('11-replayable-controls', 'java', ['-cp', replayableCp, 'ResourceControls', '2']);
assert.match(fixedControls.stdout, /PASS lifecycle, forwarding, direct resource use, failure cleanup, snapshot; opens=1 closes=1/);
const summary = {

        originalControls: originalControls.stdout.trim(),

    },
    conclusions: [
