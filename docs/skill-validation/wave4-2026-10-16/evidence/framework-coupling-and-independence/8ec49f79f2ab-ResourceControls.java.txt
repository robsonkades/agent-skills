import example.adapter.ExportFactory;
import example.adapter.SpringExportBody;
import example.application.ExportBody;
import example.application.ExportPipeline;
import java.io.ByteArrayInputStream;
import java.io.FilterInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;
import java.util.concurrent.atomic.AtomicInteger;
import org.springframework.core.io.ByteArrayResource;
import org.springframework.core.io.InputStreamResource;
import org.springframework.core.io.Resource;

public final class ResourceControls {
    private static final byte[] PAYLOAD = "invoice=17;total=42.00".getBytes(StandardCharsets.UTF_8);

    private static InputStream tracked(byte[] bytes, AtomicInteger opened, AtomicInteger closed) {
        opened.incrementAndGet();
        return new FilterInputStream(new ByteArrayInputStream(bytes)) {
            @Override public void close() throws IOException {
                closed.incrementAndGet();
                super.close();
            }
        };
    }

    public static void main(String[] args) throws Exception {
        AtomicInteger opened = new AtomicInteger();
        AtomicInteger closed = new AtomicInteger();
        Resource once = new InputStreamResource(() -> tracked(PAYLOAD, opened, closed));
        if (!once.isOpen()) throw new AssertionError("InputStreamResource lifecycle marker");
        long checksum = new ExportPipeline().process(new SpringExportBody(once), 1);
        try {
            new ExportPipeline().process(new SpringExportBody(once), 1);
            throw new AssertionError("a forwarding wrapper must not manufacture replayability");
        } catch (IllegalStateException expected) {
            if (opened.get() != 1 || closed.get() != 1) throw new AssertionError("stream ownership");
        }

        Resource replayable = new ByteArrayResource(PAYLOAD.clone());
        if (replayable.isOpen()) throw new AssertionError("replayable descriptor marker");
        ExportBody directAdapterUse = replayable::getInputStream;
        if (new ExportPipeline().process(directAdapterUse, 2) != checksum)
            throw new AssertionError("existing framework resource is adequate");

        AtomicInteger failureClosed = new AtomicInteger();
        ExportBody failingRead = () -> new InputStream() {
            @Override public int read() throws IOException { throw new IOException("controlled read failure"); }
            @Override public void close() { failureClosed.incrementAndGet(); }
        };
        try {
            new ExportPipeline().process(failingRead, 1);
            throw new AssertionError("failure expected");
        } catch (IOException expected) {
            if (failureClosed.get() != 1) throw new AssertionError("failure cleanup");
        }

        byte[] mutable = PAYLOAD.clone();
        ExportBody ownedSnapshot = ExportFactory.create(mutable);
        Arrays.fill(mutable, (byte) 0);
        if (new ExportPipeline().process(ownedSnapshot, Integer.parseInt(args[0])) != checksum)
            throw new AssertionError("caller mutation changed exported snapshot");

        System.out.println("PASS lifecycle, forwarding, direct resource use, failure cleanup, snapshot; opens=" + opened + " closes=" + closed);
    }
}
