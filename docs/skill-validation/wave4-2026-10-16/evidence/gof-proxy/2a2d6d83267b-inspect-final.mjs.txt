import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {spawnSync} from 'node:child_process';
const cwd='C:/git/agent-skills';
const root=cwd+'/skills/gof-proxy';
const baseline='C:/Users/robso/AppData/Local/Temp/skills-wave4-validation-Ceekn9/baseline/gof-proxy';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex').toUpperCase();
const commands=[
 ['status','git',['status','--short','--untracked-files=all','--ignored','--','skills/gof-proxy']],
 ['diffAgainstHead','git',['diff','HEAD','--','skills/gof-proxy']],
 ['staged','git',['diff','--cached','--','skills/gof-proxy']],
 ['unstaged','git',['diff','--','skills/gof-proxy']],
 ['headManifest','git',['show','HEAD:skills/gof-proxy/skill.yaml']],
 ['diffCheck','git',['diff','--check','--','skills/gof-proxy']]
];
const checks=commands.map(([id,command,args])=>{
 const r=spawnSync(command,args,{cwd,encoding:'utf8',windowsHide:true,timeout:15000});
 return {id,command,args,cwd,exitCode:r.status,stdout:r.stdout,stderr:r.stderr,error:r.error?.message??null};
});
const files=[];
for(const e of fs.readdirSync(root,{recursive:true,withFileTypes:true})){
 if(!e.isFile())continue;
 const full=path.join(e.parentPath,e.name);
 const relative=path.relative(root,full).replaceAll('\\','/');
 const current=fs.readFileSync(full),initial=fs.readFileSync(path.join(baseline,relative));
 files.push({path:full.replaceAll('\\','/'),relative,bytes:current.length,sha256:sha(current),baselineSha256:sha(initial),matchesBaseline:current.equals(initial)});
}
const result={checks,files,unchanged:files.every(f=>f.matchesBaseline),newPaths:files.filter(f=>!fs.existsSync(path.join(baseline,f.relative))).map(f=>f.relative)};
console.log(JSON.stringify(result,null,2));
