import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const here=path.dirname(new URL(import.meta.url).pathname.replace(/^\/([A-Za-z]:)/,'$1'));
const jdk='C:/Users/robso/.jdks/temurin-25.0.3/bin';
const root='C:/git/agent-skills/skills/gof-proxy';
const commands=[
 {id:'01-extract',command:process.execPath,args:['extract-examples.mjs',root,here]},
 {id:'02-javac-version',command:jdk+'/javac.exe',args:['-version']},
 {id:'03-java-version',command:jdk+'/java.exe',args:['-version']},
 {id:'04-compile',command:jdk+'/javac.exe',args:['--release','17','-Xlint:all','-Werror','-XDrawDiagnostics','-d','classes','Report.java','ReportSpec.java','ReportEngine.java','LazyReportEngine.java','LazyProbe.java','ProxyProbe.java']},
 {id:'05-lazy',command:jdk+'/java.exe',args:['-cp','classes','LazyProbe']},
 {id:'06-proxy',command:jdk+'/java.exe',args:['-cp','classes','ProxyProbe']}
];
const results=[];
for(const c of commands){
 const r=spawnSync(c.command,c.args,{cwd:here,encoding:'utf8',timeout:15000,windowsHide:true});
 const result={...c,cwd:here,exitCode:r.status,signal:r.signal,error:r.error?.message??null,stdout:r.stdout??'',stderr:r.stderr??''};
 fs.writeFileSync(path.join(here,c.id+'.log'),[c.command,...c.args].join(' ')+'\n'+result.stdout+result.stderr);
 results.push(result);
 console.log(JSON.stringify(result));
 if(r.status!==0||r.error){process.exitCode=1;break;}
}
fs.writeFileSync(path.join(here,'technical-results.json'),JSON.stringify({checks:results,scope:'Extracted Java 17 snippets and synthetic API controls; runtime JDK 25, not a JDK 17 or framework integration run'},null,2)+'\n');
