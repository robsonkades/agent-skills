import fs from 'node:fs';
import path from 'node:path';
const root=process.argv[2];
const out=process.argv[3];
const worked=fs.readFileSync(path.join(root,'references/worked-example.md'),'utf8');
const hazards=fs.readFileSync(path.join(root,'references/proxy-kinds-and-hazards.md'),'utf8');
const fenced=[...worked.matchAll(/```java\r?\n([\s\S]*?)```/g)][0][1];
const cut=fenced.indexOf('public final class LazyReportEngine');
if(cut<0)throw Error('missing lazy source');
fs.writeFileSync(path.join(out,'ReportEngine.java'),fenced.slice(0,cut).trim()+'\n');
fs.writeFileSync(path.join(out,'LazyReportEngine.java'),'import java.util.function.Supplier;\n\n'+fenced.slice(cut).trim()+'\n');
fs.writeFileSync(path.join(out,'Report.java'),'public record Report(int value) {}\n');
fs.writeFileSync(path.join(out,'ReportSpec.java'),'public record ReportSpec(int value) {}\n');
const handlerFence=[...hazards.matchAll(/```java\r?\n([\s\S]*?)```/g)][0][1];
const statement=handlerFence.slice(handlerFence.indexOf('Foo foo ='),handlerFence.indexOf('// CGLIB')).trim();
if(!statement.endsWith('});'))throw Error('unexpected handler');
const template=fs.readFileSync(path.join(out,'ProxyProbe.template.java'),'utf8');
if(!template.includes('/* EXTRACTED_HANDLER */'))throw Error('missing insertion');
fs.writeFileSync(path.join(out,'ProxyProbe.java'),template.replace('/* EXTRACTED_HANDLER */',statement));
console.log(JSON.stringify({status:'extracted',sourceRoot:root,
 sources:['references/worked-example.md:first Java block','references/proxy-kinds-and-hazards.md:first Java block'],
 adaptations:['Split exact public interface/class into separate files; added Supplier import','Added Report/ReportSpec record stubs','Placed exact Foo handler statement in a probe factory; Foo, target and loader supplied by harness'],
 generated:['ReportEngine.java','LazyReportEngine.java','Report.java','ReportSpec.java','ProxyProbe.java']},null,2));
