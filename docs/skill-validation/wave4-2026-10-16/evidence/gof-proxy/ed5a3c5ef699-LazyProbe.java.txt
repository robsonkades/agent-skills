import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

public final class LazyProbe {
    private static int checks;
    private static void check(boolean value,String name) {
        if(!value)throw new AssertionError(name);
        checks++;
        System.out.println("PASS "+name);
    }
    private interface Action { void run() throws Exception; }
    private static Throwable thrown(Action action) {
        try {action.run(); throw new AssertionError("expected failure");}
        catch(Exception|Error failure) {return failure;}
    }
    public static void main(String[] args) throws Exception {
        AtomicInteger built=new AtomicInteger(), rendered=new AtomicInteger();
        CountDownLatch entered=new CountDownLatch(1), release=new CountDownLatch(1);
        LazyReportEngine lazy=new LazyReportEngine(()-> {
            built.incrementAndGet(); entered.countDown();
            try {if(!release.await(5,TimeUnit.SECONDS))throw new IllegalStateException("factory release expired");}
            catch(InterruptedException e) {Thread.currentThread().interrupt(); throw new IllegalStateException(e);}
            int initialized=777;
            return spec->{rendered.incrementAndGet(); return new Report(initialized+spec.value());};
        });
        check(built.get()==0,"construction deferred until first use");
        ExecutorService workers=Executors.newFixedThreadPool(24);
        try {
            List<Future<Report>> reports=new ArrayList<>();
            for(int i=0;i<24;i++)reports.add(workers.submit(()->lazy.render(new ReportSpec(1))));
            check(entered.await(5,TimeUnit.SECONDS),"factory entered with concurrent consumers");
            release.countDown();
            boolean allInitialized=true;
            for(Future<Report> result:reports)allInitialized &= result.get(5,TimeUnit.SECONDS).value()==778;
            check(allInitialized,"all concurrent calls observe initialized target in this run");
            check(built.get()==1,"one factory invocation across 24 consumers");
            check(rendered.get()==24,"all consumers reach target");
        } finally {
            release.countDown(); workers.shutdownNow();
            if(!workers.awaitTermination(5,TimeUnit.SECONDS))throw new AssertionError("workers still active");
        }

        AtomicInteger failedBuilds=new AtomicInteger();
        RuntimeException original=new IllegalStateException("templates missing");
        LazyReportEngine failed=new LazyReportEngine(()->{failedBuilds.incrementAndGet();throw original;});
        check(thrown(()->failed.render(new ReportSpec(0)))==original,"initialization RuntimeException preserves instance");
        check(thrown(()->failed.render(new ReportSpec(0)))==original && failedBuilds.get()==1,
              "initialization RuntimeException cached without retry");
        check(thrown(()->new LazyReportEngine(null)) instanceof NullPointerException,"null supplier rejected");

        AtomicInteger nullBuilds=new AtomicInteger();
        LazyReportEngine nullResult=new LazyReportEngine(()->{nullBuilds.incrementAndGet();return null;});
        Throwable firstNull=thrown(()->nullResult.render(new ReportSpec(0)));
        check(firstNull instanceof NullPointerException,"null factory result rejected");
        check(thrown(()->nullResult.render(new ReportSpec(0)))==firstNull && nullBuilds.get()==1,
              "null-result failure cached");

        AtomicReference<LazyReportEngine> recursive=new AtomicReference<>();
        AtomicInteger recursiveBuilds=new AtomicInteger();
        recursive.set(new LazyReportEngine(()->{
            recursiveBuilds.incrementAndGet();
            recursive.get().render(new ReportSpec(0));
            return spec->new Report(0);
        }));
        Throwable cycle=thrown(()->recursive.get().render(new ReportSpec(0)));
        check(cycle instanceof IllegalStateException && "recursive initialization".equals(cycle.getMessage()),
              "same-thread recursive factory entry rejected");
        check(thrown(()->recursive.get().render(new ReportSpec(0)))==cycle && recursiveBuilds.get()==1,
              "uncaught recursive runtime failure retained");

        AtomicInteger errorBuilds=new AtomicInteger();
        AssertionError error=new AssertionError("synthetic factory Error");
        LazyReportEngine errorCase=new LazyReportEngine(()->{
            if(errorBuilds.incrementAndGet()==1)throw error;
            return spec->new Report(9);
        });
        check(thrown(()->errorCase.render(new ReportSpec(0)))==error,"Error propagated without conversion");
        check(errorCase.render(new ReportSpec(0)).value()==9 && errorBuilds.get()==2,"Error not cached; later attempt allowed");

        AtomicInteger calls=new AtomicInteger(), renderBuilds=new AtomicInteger();
        LazyReportEngine renderer=new LazyReportEngine(()->{
            renderBuilds.incrementAndGet();
            return spec->{if(calls.incrementAndGet()==1)throw new IllegalStateException("render failure");return new Report(12);};
        });
        check(thrown(()->renderer.render(new ReportSpec(0))) instanceof IllegalStateException,"target render failure propagated");
        check(renderer.render(new ReportSpec(0)).value()==12 && renderBuilds.get()==1,"render failure does not recreate target");
        System.out.println("PASS lazy checks="+checks+"; executor terminated=true");
    }
}
