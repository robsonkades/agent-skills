function run(label, command, args) {
  const result = spawnSync(command, args, { cwd: evidence, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = { label, command, args, cwd: evidence, exitCode: result.status, signal: result.signal,
    error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  record.log = join(evidence, 'logs', label + '.json');
  commands.push(record);
  writeFileSync(record.log, JSON.stringify(record, null, 2) + '\n');
  assert.equal(result.status, 0, label + '\n' + record.stdout + record.stderr);
  return record;
}
