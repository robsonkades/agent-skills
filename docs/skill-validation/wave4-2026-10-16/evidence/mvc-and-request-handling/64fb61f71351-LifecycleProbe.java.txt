import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.util.ArrayList;
import java.util.List;
import org.springframework.mock.web.MockHttpServletRequest;
import org.springframework.mock.web.MockHttpServletResponse;
import org.springframework.web.servlet.DispatcherServlet;
import org.springframework.web.servlet.HandlerAdapter;
import org.springframework.web.servlet.HandlerExecutionChain;
import org.springframework.web.servlet.HandlerInterceptor;
import org.springframework.web.servlet.ModelAndView;

public class LifecycleProbe {
    static final ThreadLocal<String> REQUEST_TAG = new ThreadLocal<>();

    static class TagInterceptor implements HandlerInterceptor {
        final boolean acquireBeforeDecision;
        final String outcome;
        final List<String> events;

        TagInterceptor(boolean acquireBeforeDecision, String outcome, List<String> events) {
            this.acquireBeforeDecision = acquireBeforeDecision;
            this.outcome = outcome;
            this.events = events;
        }

        public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
            events.add("tag.pre");
            if (acquireBeforeDecision) REQUEST_TAG.set("request-1");
            if (outcome.equals("throw")) throw new IllegalStateException("decision failed");
            if (outcome.equals("stop")) {
                response.setStatus(503);
                return false;
            }
            if (!acquireBeforeDecision) REQUEST_TAG.set("request-1");
            return true;
        }

        public void postHandle(HttpServletRequest request, HttpServletResponse response,
                               Object handler, ModelAndView result) {
            events.add("tag.post");
        }

        public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                    Object handler, Exception failure) {
            events.add("tag.after:" + (failure == null ? "none" : failure.getMessage()));
            REQUEST_TAG.remove();
        }
    }

    static class ProbeServlet extends DispatcherServlet {
        final HandlerExecutionChain chain;
        final List<String> events;
        final boolean handlerFailure;

        ProbeServlet(HandlerInterceptor interceptor, List<String> events, boolean handlerFailure) {
            this.events = events;
            this.handlerFailure = handlerFailure;
            HandlerInterceptor prior = new HandlerInterceptor() {
                public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
                    events.add("prior.pre");
                    return true;
                }
                public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                            Object handler, Exception failure) {
                    events.add("prior.after");
                }
            };
            chain = new HandlerExecutionChain(new Object(), prior, interceptor);
        }

        protected HandlerExecutionChain getHandler(HttpServletRequest request) { return chain; }

        protected HandlerAdapter getHandlerAdapter(Object handler) {
            return new HandlerAdapter() {
                public boolean supports(Object candidate) { return true; }
                public ModelAndView handle(HttpServletRequest request, HttpServletResponse response,
                                           Object candidate) throws Exception {
                    events.add("handler");
                    if (handlerFailure) throw new IllegalStateException("handler failed");
                    response.getWriter().write("ok");
                    return null;
                }
                public long getLastModified(HttpServletRequest request, Object candidate) { return -1; }
            };
        }

        void dispatch(HttpServletRequest request, HttpServletResponse response) throws Exception {
            doDispatch(request, response);
        }
    }

    static void check(boolean condition, String detail) {
        if (!condition) throw new AssertionError(detail);
    }

    static void exercise(boolean before, String outcome, boolean handlerFailure) throws Exception {
        REQUEST_TAG.remove();
        List<String> events = new ArrayList<>();
        var servlet = new ProbeServlet(new TagInterceptor(before, outcome, events), events, handlerFailure);
        var response = new MockHttpServletResponse();
        Exception caught = null;
        try {
            servlet.dispatch(new MockHttpServletRequest("GET", "/report"), response);
        } catch (Exception failure) {
            caught = failure;
        }
        boolean proceeded = outcome.equals("continue");
        boolean leaked = REQUEST_TAG.get() != null;
        check(events.contains("handler") == proceeded, "handler invocation: " + events);
        check(events.stream().anyMatch(event -> event.startsWith("tag.after:")) == proceeded,
              "own completion callback: " + events);
        check(events.contains("prior.after"), "previous accepted interceptor was not completed");
        check(leaked == (before && !proceeded), "request-tag lifetime: " + events);
        if (outcome.equals("stop")) check(response.getStatus() == 503 && caught == null, "stop response");
        else check((caught != null) == (outcome.equals("throw") || handlerFailure), "failure outcome");
        if (handlerFailure) check(events.contains("tag.after:handler failed"), "handler failure callback");
        System.out.println("before=" + before + " outcome=" + outcome + " handlerFailure=" + handlerFailure
                + " requestTagRetained=" + leaked + " events=" + events);
        REQUEST_TAG.remove();
    }

    public static void main(String[] args) throws Exception {
        for (boolean before : new boolean[] {true, false}) {
            exercise(before, "continue", false);
            exercise(before, "stop", false);
            exercise(before, "throw", false);
            exercise(before, "continue", true);
        }
        System.out.println("PASS 8 real DispatcherServlet/HandlerExecutionChain callback cases; mock Servlet request/response, selected handler only");
    }
}
