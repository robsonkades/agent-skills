import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const repo = 'C:/git/agent-skills';
const skill = 'mvc-and-request-handling';
const evidence = dirname(fileURLToPath(import.meta.url));
const wave = dirname(evidence);
const commands = [];
function git(args) {
  const result = spawnSync('git', args, { cwd: repo, encoding: 'utf8', timeout: 30000, windowsHide: true });
  commands.push({ command: 'git', args, cwd: repo, exitCode: result.status, stdout: result.stdout, stderr: result.stderr });
  assert.equal(result.status, 0);
  return result.stdout;
}
const status = git(['status', '--short', '--', 'skills/' + skill]);
const unstaged = git(['diff', '--', 'skills/' + skill]);
const staged = git(['diff', '--cached', '--', 'skills/' + skill]);
assert.equal(status + unstaged + staged, '');
const committedManifest = git(['show', 'HEAD:skills/' + skill + '/skill.yaml']);
const assignmentManifest = readFileSync(join(repo, 'skills', skill, 'skill.yaml'), 'utf8');
const version = (body) => body.match(/^version:\s*(\S+)/m)[1];
assert.equal(version(committedManifest), '1.3.5');
assert.equal(version(assignmentManifest), '1.3.5');
const files = ['SKILL.md', 'skill.yaml', 'references/application-controller.md', 'references/page-vs-front-controller.md'];
const hashes = files.map((file) => {
  const current = readFileSync(join(repo, 'skills', skill, file));
  assert.deepEqual(current, readFileSync(join(wave, 'baseline', skill, file)));
  return { path: 'skills/' + skill + '/' + file, sha256: createHash('sha256').update(current).digest('hex'), bytes: current.length };
});
const baseline = { skill, capturedAt: new Date().toISOString(), committedVersion: version(committedManifest), assignmentVersion: version(assignmentManifest), status, unstaged, staged, hashes, coordinatorSnapshotMatches: true, commands };
writeFileSync(join(evidence, 'baseline.json'), JSON.stringify(baseline, null, 2) + '\n');
process.stdout.write(JSON.stringify(baseline, null, 2) + '\n');
