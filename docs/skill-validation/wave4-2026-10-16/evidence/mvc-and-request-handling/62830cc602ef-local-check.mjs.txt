import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { dirname, join, resolve } from 'node:path';

import { fileURLToPath, pathToFileURL } from 'node:url';

const repo = 'C:/git/agent-skills';
const skill = 'mvc-and-request-handling';
const root = join(repo, 'skills', skill);
const evidence = dirname(fileURLToPath(import.meta.url));

const jars = investigation.cachedJars.map((jar) => jar.path);
const commands = [];
function run(label, command, args, cwd = root, expected = 0) {
  const result = spawnSync(command, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = { label, command, args, cwd, exitCode: result.status, expectedExitCode: expected,
    signal: result.signal, error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  record.log = join(evidence, 'logs', label + '.json');
  writeFileSync(record.log, JSON.stringify(record, null, 2) + '\n');
  commands.push(record);
  assert.equal(record.error, null, label + ': ' + record.error);
  assert.equal(record.signal, null, label);
  assert.equal(record.exitCode, expected, label + '\n' + record.stdout + record.stderr);
  return record;
}

  mkdirSync(classes, { recursive: true });

}

const links = [];
const textReferrals = new Set();
for (const file of baseline.hashes.filter((file) => file.path.endsWith('.md'))) {
  const absolute = join(repo, file.path);
  const body = readFileSync(absolute, 'utf8');
  for (const match of body.matchAll(/\[[^\]]+\]\(([^)]+)\)/g)) {
    if (/^(?:https?:|#)/.test(match[1])) continue;
    assert.ok(existsSync(resolve(dirname(absolute), match[1].split('#')[0])), match[1]);
    links.push({ from: file.path, target: match[1] });
  }
  for (const match of body.matchAll(/\x60([a-z]+(?:-[a-z]+)+)\x60/g)) {
    if (existsSync(join(repo, 'skills', match[1], 'SKILL.md'))) textReferrals.add(match[1]);
  }
}
