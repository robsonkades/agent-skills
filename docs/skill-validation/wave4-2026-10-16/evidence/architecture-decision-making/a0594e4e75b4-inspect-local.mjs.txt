import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, readdirSync, mkdirSync, existsSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
import path from 'node:path';
import { parseManifest } from 'file:///C:/git/agent-skills/packages/core/src/domain/manifest.ts';
import { parseSkillDocument, documentName, documentDescription } from 'file:///C:/git/agent-skills/packages/core/src/domain/skill-document.ts';

const repo = 'C:/git/agent-skills';
const owned = 'skills/architecture-decision-making';
const out = 'C:/Users/robso/AppData/Local/Temp/skills-wave4-validation-Ceekn9/architecture-decision-making';
mkdirSync(out, { recursive: true });
const commands = [];
function run(command, args) {
  const result = spawnSync(command, args, { cwd: repo, encoding: 'utf8', windowsHide: true });
  const log = { command, args, cwd: repo, exitCode: result.status, stdout: result.stdout, stderr: result.stderr };
  commands.push(log);
  assert.equal(result.status, 0, JSON.stringify(log));
  return result.stdout;
}
function walk(dir) {
  return readdirSync(path.join(repo, dir), { withFileTypes: true }).flatMap(e => e.isDirectory() ? walk(`${dir}/${e.name}`) : [`${dir}/${e.name}`]);
}
const status = run('git', ['status', '--short', '--untracked-files=all', '--', owned]);
const unstagedDiff = run('git', ['diff', '--', owned]);
const stagedDiff = run('git', ['diff', '--cached', '--', owned]);
const committedManifest = run('git', ['show', `HEAD:${owned}/skill.yaml`]);
const files = walk(owned).sort().map(file => {
  const bytes = readFileSync(path.join(repo, file));
  return { path: file, bytes: bytes.length, sha256: createHash('sha256').update(bytes).digest('hex'), content: bytes.toString('utf8') };
});
const manifestText = files.find(f => f.path === `${owned}/skill.yaml`).content;
const parsed = parseManifest(manifestText, { strict: true });
assert.deepEqual(parsed.issues, []);
const manifest = parsed.manifest;
const doc = parseSkillDocument(files.find(f => f.path === `${owned}/SKILL.md`).content);
assert.equal(documentName(doc), manifest.name);
assert.equal(documentDescription(doc), manifest.description);
assert.equal(String(manifest.version), '2.1.4');
assert.equal(String(parseManifest(committedManifest).manifest.version), '2.1.4');
assert.equal(status, '');
assert.equal(unstagedDiff, '');
assert.equal(stagedDiff, '');
const waveBaseline = JSON.parse(readFileSync(path.join(repo, 'docs/skill-validation/wave4-2026-10-16/baseline.json'), 'utf8'));
const initial = waveBaseline.packages.find(p => p.name === manifest.name);
assert.ok(initial);
assert.equal(initial.initialVersion, '2.1.4');
assert.equal(initial.committedVersion, '2.1.4');
const baselineHashes = [];
for (const file of files) {
  const prior = initial.files.find(f => f.path === file.path);
  assert.ok(prior, file.path);
  assert.equal(file.sha256, prior.sha256, file.path);
  baselineHashes.push({ path: file.path, unchanged: true, sha256: file.sha256, bytes: file.bytes });
}
assert.equal(initial.files.length, files.length);
const localLinks = [];
for (const file of files.filter(f => f.path.endsWith('.md'))) {
  for (const match of file.content.matchAll(/\]\(([^)]+)\)/g)) {
    if (/^[a-z][a-z0-9+.-]*:/i.test(match[1]) || match[1].startsWith('#')) continue;
    const target = path.resolve(repo, path.dirname(file.path), match[1].split('#')[0]);
    assert.ok(existsSync(target), `${file.path}: ${match[1]}`);
    localLinks.push({ source: file.path, target: match[1], exists: true });
  }
}
const mentions = ['architecture-coupling-and-quanta', 'architecture-characteristics', 'architecture-fitness-functions', 'architecture-trade-off-analysis', 'engineering-communication', 'technical-debt-decisions'];
const dependencyNames = manifest.dependencies.map(d => d.name);
for (const name of mentions) assert.ok([...dependencyNames, ...manifest.suggests].includes(name), name);
run(process.execPath, ['node_modules/prettier/bin/prettier.cjs', '--check', ...files.map(f => f.path)]);
run('git', ['diff', '--check', '--', owned]);
const result = { capturedAt: new Date().toISOString(), node: process.version, scope: owned, status, unstagedDiff, stagedDiff, versions: { committed: '2.1.4', assignment: '2.1.4', final: '2.1.4' }, fileCount: files.length, filesUnchangedFromWaveBaseline: baselineHashes, localLinks, metadata: { strictIssues: parsed.issues, nameMatches: true, descriptionMatches: true, referralDeclarationCount: mentions.length, catalogWideResolutionChecked: false }, commands, boundaries: 'Read-only checks scoped to the assigned package. No agent execution, ADR checker implementation, compilation, runtime or performance test.' };
writeFileSync(path.join(out, 'assignment-baseline.json'), JSON.stringify({ capturedAt: result.capturedAt, skill: manifest.name, status, unstagedDiff, stagedDiff, committedManifest, files }, null, 2) + '\n');
writeFileSync(path.join(out, 'local-validation.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify({ ...result, commands: result.commands.map(c => ({ ...c, stdout: c.command === 'git' && c.args[0] === 'show' ? '[manifest retained in assignment-baseline.json]' : c.stdout })) }, null, 2));
