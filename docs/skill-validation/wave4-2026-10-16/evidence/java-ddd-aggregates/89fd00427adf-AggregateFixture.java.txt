import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.sql.Connection;
import java.sql.SQLException;
import java.util.Properties;
import javax.sql.DataSource;

public final class AggregateFixture {
    public enum ReadPath { COMPAT, COMBINED }

    public static ReadPath readPath(Path configuration) throws IOException {
        var properties = new Properties();
        try (var input = Files.newInputStream(configuration)) {
            properties.load(input);
        }
        return switch (properties.getProperty("orders.reader")) {
            case "compat" -> ReadPath.COMPAT;
            case "combined" -> ReadPath.COMBINED;
            default -> throw new IllegalArgumentException("Unsupported orders.reader");
        };
    }

    public record OrderState(long version, long units) {
        public void accept(int quantity) {
            if (quantity < 1 || units > 10L - quantity) {
                throw new IllegalArgumentException("Order limit exceeded");
            }
        }
    }

    private static long scalar(Connection connection, String sql, String id)
            throws SQLException {
        try (var statement = connection.prepareStatement(sql)) {
            statement.setString(1, id);
            try (var result = statement.executeQuery()) {
                if (!result.next()) throw new SQLException("Order not found");
                return result.getLong(1);
            }
        }
    }

    public static OrderState load(Connection connection, String id, ReadPath path)
            throws SQLException {
        if (path == ReadPath.COMPAT) {
            long units = scalar(connection,
                    "SELECT COALESCE(SUM(quantity), 0) FROM order_lines WHERE order_id = ?", id);
            long version = scalar(connection, "SELECT version FROM orders WHERE id = ?", id);
            return new OrderState(version, units);
        }
        try (var statement = connection.prepareStatement(
                "SELECT o.version, COALESCE(SUM(l.quantity), 0) "
                + "FROM orders o LEFT JOIN order_lines l ON l.order_id = o.id "
                + "WHERE o.id = ? GROUP BY o.version")) {
            statement.setString(1, id);
            try (var result = statement.executeQuery()) {
                if (!result.next()) throw new SQLException("Order not found");
                return new OrderState(result.getLong(1), result.getLong(2));
            }
        }
    }

    public static void add(DataSource source, Path configuration, String orderId,
            String lineId, int quantity) throws IOException, SQLException {
        var path = readPath(configuration);
        try (var connection = source.getConnection()) {
            connection.setAutoCommit(false);
            connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
            try {
                var state = load(connection, orderId, path);
                state.accept(quantity);
                try (var claim = connection.prepareStatement(
                        "UPDATE orders SET version = version + 1 WHERE id = ? AND version = ?")) {
                    claim.setString(1, orderId);
                    claim.setLong(2, state.version());
                    if (claim.executeUpdate() != 1) {
                        throw new SQLException("Order write conflict");
                    }
                }
                try (var append = connection.prepareStatement(
                        "INSERT INTO order_lines(order_id, id, quantity) VALUES (?, ?, ?)")) {
                    append.setString(1, orderId);
                    append.setString(2, lineId);
                    append.setInt(3, quantity);
                    append.executeUpdate();
                }
                connection.commit();
            } catch (SQLException | RuntimeException failure) {
                try {
                    connection.rollback();
                } catch (SQLException rollbackFailure) {
                    failure.addSuppressed(rollbackFailure);
                }
                throw failure;
            }
        }
    }
}
