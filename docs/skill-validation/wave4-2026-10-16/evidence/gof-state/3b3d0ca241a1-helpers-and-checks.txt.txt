static int checks;
static void check(boolean good, String name) { checks++; if (!good) throw new AssertionError(name); }
static <T extends Throwable> T expect(Class<T> type, Runnable action) {
    try { action.run(); } catch (Throwable failure) {
        check(type.isInstance(failure), "wrong exception " + failure);
        return type.cast(failure);
    }
    throw new AssertionError("missing " + type.getSimpleName());
}
record PaymentReference(String value) { PaymentReference { Objects.requireNonNull(value); } }
record TrackingId(String value) { TrackingId { Objects.requireNonNull(value); } }
enum Reason { USER, EXPIRED }
sealed interface OrderEvent permits Pay, Ship, Cancel, RefundSettled {}
record Pay(Instant at, PaymentReference reference) implements OrderEvent {}
record Ship(TrackingId tracking, Instant at) implements OrderEvent {}
record Cancel(Instant at, Reason reason) implements OrderEvent {}
record RefundSettled(Instant at) implements OrderEvent {}
static final class IllegalTransition extends RuntimeException {
    IllegalTransition(OrderState state, OrderEvent event) { super(state.getClass().getSimpleName() + ":" + event.getClass().getSimpleName()); }
}
static final class UnknownPersistedState extends RuntimeException {
    UnknownPersistedState(String state) { super(state); }
}
record Row(Map<String, Object> values) {
    String string(String name) { return (String) values.get(name); }
    Instant instant(String name) { return (Instant) values.get(name); }
}
static Reason reasonFromCode(String code) {
    return switch (Objects.requireNonNull(code)) {
        case "USER_REQUEST" -> Reason.USER;
        case "EXPIRED" -> Reason.EXPIRED;
        default -> throw new IllegalArgumentException("unknown reason");
    };
}
public static void main(String[] args) {
    var now = Instant.parse("2026-10-16T12:00:00Z");
    var ref = new PaymentReference("payment");
    var tracking = new TrackingId("tracking");
    List<OrderState> states = List.of(new Draft(), new Paid(now, ref), new Shipped(tracking, now), new Refunding(now, ref, Reason.USER), new Cancelled(now, Reason.USER));
    List<OrderEvent> events = List.of(new Pay(now, ref), new Ship(tracking, now), new Cancel(now, Reason.USER), new RefundSettled(now));
    check(states.stream().map(Object::getClass).collect(java.util.stream.Collectors.toSet()).equals(Set.of(OrderState.class.getPermittedSubclasses())), "state inventory");
    check(events.stream().map(Object::getClass).collect(java.util.stream.Collectors.toSet()).equals(Set.of(OrderEvent.class.getPermittedSubclasses())), "event inventory");
    Class<?>[][] expected = {
        {Paid.class, null, Cancelled.class, null},
        {null, Shipped.class, Refunding.class, null},
        {null, null, null, null},
        {null, null, null, Cancelled.class},
        {null, null, null, null}
    };
    for (int i = 0; i < states.size(); i++) {
        for (int j = 0; j < events.size(); j++) {
            var state = states.get(i); var event = events.get(j);
            if (expected[i][j] == null) {
                expect(IllegalTransition.class, () -> transition(state, event));
                expect(IllegalTransition.class, () -> referenceTransition(state, event));
            } else {
                var actual = transition(state, event);
                check(actual.getClass() == expected[i][j], "transition table");
                check(actual.equals(referenceTransition(state, event)), "both documented transition functions agree");
            }
        }
    }
    var pending = (Refunding) transition(states.get(1), events.get(2));
    check(pending.reference() == ref && pending.reason() == Reason.USER, "refund preserves payment and reason");
    var cancelled = (Cancelled) transition(pending, events.get(3));
    check(cancelled.reason() == Reason.USER && cancelled.at().equals(now), "settlement preserves reason and completion time");
    expect(NullPointerException.class, () -> transition(null, events.get(0)));
    expect(NullPointerException.class, () -> transition(states.get(0), null));
    expect(NullPointerException.class, () -> new Paid(null, ref));
    expect(NullPointerException.class, () -> new Shipped(null, now));
    expect(NullPointerException.class, () -> new Refunding(now, ref, null));
    expect(NullPointerException.class, () -> new Cancelled(null, Reason.USER));

    var data = new HashMap<String, Object>();
    data.put("paid_at", now); data.put("payment_ref", "payment"); data.put("tracking", "tracking");
    data.put("shipped_at", now); data.put("cancelled_at", now); data.put("cancel_reason", "USER_REQUEST");
    var codes = List.of("DRAFT", "PAID", "SHIPPED", "REFUNDING", "CANCELLED");
    for (int i = 0; i < codes.size(); i++) {
        data.put("status", codes.get(i)); var row = new Row(data);
        check(fromRow(row).equals(states.get(i)), "worked mapper " + codes.get(i));
        check(fromRow(codes.get(i), row).equals(states.get(i)), "reference mapper " + codes.get(i));
    }
    data.put("status", "NEW_CODE");
    expect(UnknownPersistedState.class, () -> fromRow(new Row(data)));
    expect(UnknownPersistedState.class, () -> fromRow("NEW_CODE", new Row(data)));
    data.put("status", "PAID"); data.remove("paid_at");
    expect(NullPointerException.class, () -> fromRow(new Row(data)));
    data.put("status", "REFUNDING"); data.put("paid_at", now); data.put("cancel_reason", "unknown");
    expect(IllegalArgumentException.class, () -> fromRow(new Row(data)));

    OrderState originalDraft = new Draft();
    var latest = new AtomicReference<OrderState>(originalDraft);
    int attempts = 0;
    OrderState current, next;
    do {
        current = latest.get();
        next = transition(current, events.get(2));
        if (attempts++ == 0) latest.set(new Paid(now, ref)); // controlled intervening commit
    } while (!latest.compareAndSet(current, next));
    check(attempts == 2 && latest.get() instanceof Refunding, "latest-state contract re-evaluates cancellation");
    var revisionBound = new AtomicReference<OrderState>(originalDraft);
    OrderState requestedResult = transition(originalDraft, events.get(2));
    var paymentCommit = new Paid(now, ref);
    revisionBound.set(paymentCommit);
    check(!revisionBound.compareAndSet(originalDraft, requestedResult), "original precondition conflicts");
    check(revisionBound.get() == paymentCommit, "bound command cannot silently become refund");
    System.out.println("PASS " + checks + " controls: complete 5x4 table in both functions, null and payload guards, five row codes, unknown/corrupt rows, CAS re-evaluation versus bound precondition");
}
