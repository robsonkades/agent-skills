import java.nio.file.*;
import java.util.*;

public final class ConnectionApp {
    static final class OwnedConnection implements AutoCloseable {
        final String id;
        int closeCount;
        OwnedConnection(String id) { this.id = id; }
        boolean isClosed() { return closeCount != 0; }
        public void close() {
            if (isClosed()) throw new IllegalStateException("duplicate close: " + id);
            closeCount++;
        }
    }
    static final class Machine {
        enum Phase { CONNECTING, CONNECTED, CLOSED }
        private Phase phase = Phase.CONNECTING;
        private final long attempt;
        private OwnedConnection active;
        Machine(long attempt) { this.attempt = attempt; }
        synchronized void connected(long eventAttempt, OwnedConnection result) {
            if (phase != Phase.CONNECTING || attempt != eventAttempt) {
                if (result != active && !result.isClosed()) result.close();
                return;
            }
            active = result;
            phase = Phase.CONNECTED;
        }
        synchronized boolean serves(OwnedConnection connection) {
            return phase == Phase.CONNECTED && active == connection && !connection.isClosed();
        }
        synchronized void close() {
            phase = Phase.CLOSED;
            if (active != null) active.close();
        }
    }
    public static void main(String[] args) throws Exception {
        var config = new Properties();
        try (var reader = Files.newBufferedReader(Path.of(args[0]))) { config.load(reader); }
        var machine = new Machine(7);
        var initial = new OwnedConnection("initial");
        machine.connected(7, initial);
        var callbackResult = switch (config.getProperty("callback.result")) {
            case "same" -> initial;
            case "new" -> new OwnedConnection("extra");
            default -> throw new IllegalArgumentException("unsupported callback result");
        };
        machine.connected(7, callbackResult);
        System.out.println("servingInitial=" + machine.serves(initial));
        System.out.println("beforeStop: initialClosed=" + initial.closeCount + ", callbackClosed=" + callbackResult.closeCount);
        machine.close();
        System.out.println("afterStop: initialClosed=" + initial.closeCount + ", callbackClosed=" + callbackResult.closeCount);
    }
}
