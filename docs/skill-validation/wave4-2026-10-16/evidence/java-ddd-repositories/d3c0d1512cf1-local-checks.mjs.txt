import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, existsSync, readdirSync, statSync } from 'node:fs';
import { resolve, join, dirname } from 'node:path';
import { pathToFileURL, fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';

const root = dirname(fileURLToPath(import.meta.url));
const repo = 'C:/git/agent-skills';
const skill = join(repo, 'skills/java-ddd-repositories');
const snapshot = 'C:/Users/robso/AppData/Local/Temp/skills-wave4-validation-Ceekn9/baseline/java-ddd-repositories';
const { parseManifest } = await import(pathToFileURL(join(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(join(repo, 'packages/core/src/domain/skill-document.ts')));
const { manifest, issues } = parseManifest(readFileSync(join(skill, 'skill.yaml'), 'utf8'), { strict: true });
assert.deepEqual(issues, []);
const doc = parseSkillDocument(readFileSync(join(skill, 'SKILL.md'), 'utf8'));
assert.equal(documentName(doc), manifest.name);
assert.equal(documentName(doc), 'java-ddd-repositories');
assert.equal(documentDescription(doc), manifest.description);
assert.equal(manifest.version, '1.0.1');
for (const entry of manifest.files) assert.ok(existsSync(join(skill, entry)), entry);
const baseline = JSON.parse(readFileSync(join(root, 'baseline-hashes.json'), 'utf8'));
assert.equal(baseline.length, 5);
const normalized = value => value.replaceAll('\r\n', '\n');
const finalFiles = [];
const localLinks = [];
const oldURL = 'https://jakarta.ee/specifications/persistence/3.2/jakarta-persistence-spec-3.2';
for (const before of baseline) {
  const current = readFileSync(before.Path);
  const baselineBytes = readFileSync(join(snapshot, before.RelativePath));
  assert.equal(createHash('sha256').update(baselineBytes).digest('hex').toUpperCase(), before.SHA256);
  const relative = before.RelativePath.replaceAll('\\', '/');
  let expected = normalized(baselineBytes.toString('utf8'));
  if (relative === 'skill.yaml') expected = expected.replace('version: 1.0.0', 'version: 1.0.1');
  if (relative === 'references/reconstitution-and-concurrency.md') expected = expected.replace(oldURL + ')', oldURL + '.html)');
  assert.equal(normalized(current.toString('utf8')), expected, relative);
  finalFiles.push({ path: before.Path, relativePath: relative, bytes: current.length,
    sha256: createHash('sha256').update(current).digest('hex').toUpperCase() });
  if (!relative.endsWith('.md')) continue;
  for (const match of current.toString('utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
    const href = match[1];
    if (/^[a-z][a-z\d+.-]*:/i.test(href)) continue;
    const target = resolve(dirname(before.Path), href.split('#')[0]);
    assert.ok(existsSync(target), href);
    localLinks.push({ source: relative, href, target });
  }
}
for (const ref of readdirSync(join(skill, 'references'))) assert.ok(doc.body.includes('references/' + ref), ref);
assert.equal(readFileSync(join(root, 'actor/case-a/transaction.properties'), 'utf8')
  .replace('isolation=REPEATABLE_READ', 'isolation=READ_COMMITTED'),
  readFileSync(join(root, 'actor/case-b/transaction.properties'), 'utf8'));
for (const fixture of JSON.parse(readFileSync(join(root, 'cases-inputs.json'), 'utf8')).sharedInputs)
  assert.ok(existsSync(join(root, fixture)));

 ['status','--short','--untracked-files=all','--','skills/java-ddd-repositories'],

 ['show','HEAD:skills/java-ddd-repositories/skill.yaml']

assert.match(commands.at(-1).stdout, /version: 1\.0\.0/);
assert.equal(commands[0].stdout, '');
assert.equal(commands[3].stdout, '');
assert.equal(commands[5].stdout, '');
const sourceFiles = [
 'C:/git/ddd-example/domain/src/main/java/com/robsonkades/admin/catalogo/domain/category/CategoryGateway.java',
 'C:/git/ddd-example/infrastructure/src/main/java/com/robsonkades/admin/catalogo/infrastructure/category/CategoryMySQLGateway.java',

const result = { scope:'Owned package only; no global validator or shared writes.',
 nodeVersion:process.version, strictManifestIssues:issues.length, descriptionMatches:true,
 finalVersion:manifest.version, baselineVersion:'1.0.0', packageFiles:baseline.length, finalFiles,
