import java.util.Optional;
import javax.persistence.EntityManager;

final class BudgetJpaGateway implements BudgetGateway {
    private final EntityManager entityManager;

    BudgetJpaGateway(final EntityManager entityManager) {
        this.entityManager = entityManager;
    }

    public Optional<Budget> findById(final BudgetId id) {
        final var root = entityManager.find(BudgetJpaEntity.class, id.value());
        if (root == null) {
            return Optional.empty();
        }
        final long observedVersion = root.getVersion();
        final var amounts = entityManager.createQuery(
                "select a.amount from AllocationJpaEntity a where a.budgetId = :id order by a.id",
                Long.class).setParameter("id", id.value()).getResultList();
        final var verification = entityManager.find(BudgetJpaEntity.class, id.value());
        if (verification == null || verification.getVersion() != observedVersion) {
            throw new InconsistentLoad(id);
        }
        return Optional.of(Budget.with(root.getId(), root.getSpendingLimit(),
                observedVersion, amounts, root.getCreatedAt(), root.getUpdatedAt()));
    }
}
