import assert from 'node:assert/strict';
import { readFileSync, readdirSync, statSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
const root=dirname(fileURLToPath(import.meta.url));
const handoffPath=join(root,'..','java-ddd-repositories.json');
const raw=readFileSync(handoffPath);
const handoff=JSON.parse(raw.toString('utf8'));
assert.equal(handoff.skill,'java-ddd-repositories');
assert.equal(handoff.status,'READY_FOR_GLOBAL_VALIDATION');
assert.equal(handoff.allWritesStopped,true);
assert.deepEqual(handoff.activeBackgroundProcesses,[]);
assert.equal(handoff.versions.final,'1.0.1');
for (const file of [...handoff.evidence.artifacts,...handoff.baseline.finalFiles]) {
 const bytes=readFileSync(file.path);
 assert.equal(bytes.length,file.bytes,file.path);
 assert.equal(createHash('sha256').update(bytes).digest('hex').toUpperCase(),file.sha256,file.path);
}
const status=execFileSync('git',['status','--short','--untracked-files=all','--','skills/java-ddd-repositories'],{cwd:'C:/git/agent-skills',encoding:'utf8'});
assert.equal(status,' M skills/java-ddd-repositories/references/reconstitution-and-concurrency.md\n M skills/java-ddd-repositories/skill.yaml\n');
const diffCheck=execFileSync('git',['diff','--check','--','skills/java-ddd-repositories'],{cwd:'C:/git/agent-skills',encoding:'utf8'});
assert.equal(diffCheck,'');
console.log(JSON.stringify({handoffPath,handoffSHA256:createHash('sha256').update(raw).digest('hex').toUpperCase(),
 status:handoff.status,allWritesStopped:true,verifiedArtifacts:handoff.evidence.artifacts.length,
 verifiedOwnedFiles:handoff.baseline.finalFiles.length,gitStatus:status,diffCheck},null,2));
