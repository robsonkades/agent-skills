import assert from 'node:assert/strict';
import { writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';
const root = dirname(fileURLToPath(import.meta.url));
const oldURL = 'https://jakarta.ee/specifications/persistence/3.2/jakarta-persistence-spec-3.2';
const results = await Promise.allSettled([oldURL, oldURL + '.html'].map(async url => {
  const response = await fetch(url, { signal: AbortSignal.timeout(15000), redirect: 'follow' });
  const body = await response.text();
  return { requestURL: url, finalURL: response.url, status: response.status,
    contentType: response.headers.get('content-type'), serverDate: response.headers.get('date'),
    bodyBytes: Buffer.byteLength(body), bodySHA256: createHash('sha256').update(body).digest('hex'),
    title: body.match(/<title[^>]*>([\s\S]*?)<\/title>/i)?.[1] ?? null,
    containsOptimisticLockingHeading: body.includes('Optimistic Locking'),
    containsForceIncrement: body.includes('OPTIMISTIC_FORCE_INCREMENT') };
}));
const evidence = results.map((item, index) => item.status === 'fulfilled' ? item.value :
  { requestURL: index ? oldURL + '.html' : oldURL, failure: String(item.reason), cause: String(item.reason?.cause ?? '') });
writeFileSync(join(root, 'link-http-results.json'), JSON.stringify(evidence, null, 2) + '\n');
console.log(JSON.stringify(evidence, null, 2));
assert.equal(evidence[0].status, 404);
assert.equal(evidence[1].status, 200);
assert.equal(evidence[1].containsOptimisticLockingHeading, true);
assert.equal(evidence[1].containsForceIncrement, true);
