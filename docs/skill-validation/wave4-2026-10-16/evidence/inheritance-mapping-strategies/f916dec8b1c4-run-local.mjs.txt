import { writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
const root = dirname(fileURLToPath(import.meta.url));
const repository = 'C:/git/agent-skills';
const commands = [];
function run(id, executable, args) {
  const result = spawnSync(executable, args, { cwd: repository, encoding: 'utf8', timeout: 30000, windowsHide: true });
  writeFileSync(resolve(root, `${id}.stdout.txt`), result.stdout ?? '');
  writeFileSync(resolve(root, `${id}.stderr.txt`), result.stderr ?? '');
  commands.push({ id, executable, args, cwd: repository, status: result.status, signal: result.signal, error: result.error?.message, stdout: `${id}.stdout.txt`, stderr: `${id}.stderr.txt` });
}
run('local-validation', process.execPath, [resolve(root, 'local-validation.mjs')]);
run('prettier', process.execPath, [resolve(repository, 'node_modules/prettier/bin/prettier.cjs'), '--check', 'skills/inheritance-mapping-strategies/SKILL.md', 'skills/inheritance-mapping-strategies/skill.yaml', 'skills/inheritance-mapping-strategies/references/strategy-comparison.md', 'skills/inheritance-mapping-strategies/references/schema-evolution.md']);
run('diff-check', 'git', ['diff', '--check', '--', 'skills/inheritance-mapping-strategies']);
run('scoped-status', 'git', ['status', '--short', '--untracked-files=all', '--', 'skills/inheritance-mapping-strategies']);
run('scoped-diff', 'git', ['diff', '--', 'skills/inheritance-mapping-strategies']);
run('staged-diff', 'git', ['diff', '--cached', '--', 'skills/inheritance-mapping-strategies']);
run('case-preparation', process.execPath, [resolve(root, 'prepare-cases.mjs')]);
writeFileSync(resolve(root, 'commands.json'), JSON.stringify(commands, null, 2) + '\n');
console.log(JSON.stringify(commands.map(({ id, status, error }) => ({ id, status, error })), null, 2));
if (commands.some(command => command.status !== 0)) process.exitCode = 1;
