import java.io.FilterInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.nio.ByteBuffer;
import java.nio.channels.Channels;
import java.nio.channels.ClosedChannelException;
import java.nio.channels.IllegalBlockingModeException;
import java.nio.channels.Pipe;
import java.nio.channels.ReadableByteChannel;
import java.util.Arrays;
import java.util.function.Function;

public final class ChannelContractProbe {
    private static int checks;

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
        checks++;
    }

    @FunctionalInterface
    interface CheckedAction { void run() throws Exception; }

    private static <T extends Throwable> T expect(Class<T> expected, CheckedAction action)
            throws Exception {
        try { action.run(); }
        catch (Throwable actual) {
            if (!expected.isInstance(actual)) throw new AssertionError(actual);
            checks++;
            return expected.cast(actual);
        }
        throw new AssertionError("Expected " + expected.getName());
    }

    // Review fixture: synchronous in-memory bytes and injected I/O failures, not a vendor client.
    static final class TrackingChannel implements ReadableByteChannel {
        private final byte[] bytes;
        private final IOException failure;
        private int offset;
        private int closeCount;
        private boolean open = true;

        TrackingChannel(byte[] bytes) { this(bytes, null); }
        TrackingChannel(byte[] bytes, IOException failure) {
            this.bytes = bytes;
            this.failure = failure;
        }

        @Override public synchronized int read(ByteBuffer target) throws IOException {
            if (!open) throw new ClosedChannelException();
            if (failure != null) throw failure;
            if (!target.hasRemaining()) return 0;
            if (offset == bytes.length) return -1;
            int count = Math.min(target.remaining(), bytes.length - offset);
            target.put(bytes, offset, count);
            offset += count;
            return count;
        }

        @Override public synchronized boolean isOpen() { return open; }
        @Override public synchronized void close() {
            if (open) { closeCount++; open = false; }
        }
    }

    static InputStream borrowedView(ReadableByteChannel channel) {
        return new FilterInputStream(Channels.newInputStream(channel)) {
            // This fixture's accepted borrowed-view contract permits close to be a no-op.
            @Override public void close() { }
        };
    }

    public static void main(String[] args) throws Exception {
        Function<ReadableByteChannel, InputStream> simplestAdapter = Channels::newInputStream;

        TrackingChannel owned = new TrackingChannel(new byte[] {1, 2, 3, 4});
        try (InputStream stream = simplestAdapter.apply(owned)) {
            check(Arrays.equals(stream.readNBytes(2), new byte[] {1, 2}), "prefix changed");
            check(owned.offset == 2, "eager source consumption");
            check(!stream.markSupported(), "unexpected repeatability support");
            expect(IOException.class, stream::reset);
        }
        check(!owned.isOpen() && owned.closeCount == 1, "owned resource not released");
        expect(ClosedChannelException.class, () -> owned.read(ByteBuffer.allocate(1)));

        TrackingChannel borrowedDirect = new TrackingChannel(new byte[] {1, 2, 3, 4});
        try (InputStream stream = simplestAdapter.apply(borrowedDirect)) {
            check(stream.read() == 1, "borrowed direct data mismatch");
        }
        // Refuting control: same signatures do not satisfy the borrowed lifecycle.
        expect(ClosedChannelException.class, () -> borrowedDirect.read(ByteBuffer.allocate(1)));

        TrackingChannel borrowed = new TrackingChannel(new byte[] {1, 2, 3, 4});
        try {
            try (InputStream stream = borrowedView(borrowed)) {
                check(Arrays.equals(stream.readNBytes(2), new byte[] {1, 2}), "borrowed prefix");
            }
            check(borrowed.isOpen() && borrowed.closeCount == 0, "borrowed resource closed");
            ByteBuffer remainder = ByteBuffer.allocate(2);
            check(borrowed.read(remainder) == 2
                    && Arrays.equals(remainder.array(), new byte[] {3, 4}), "remainder lost");
        } finally { borrowed.close(); }
        check(borrowed.closeCount == 1, "owner cleanup omitted");

        IOException failure = new IOException("synthetic read failure");
        TrackingChannel ownedFailure = new TrackingChannel(new byte[0], failure);
        IOException observed = expect(IOException.class, () -> {
            try (InputStream stream = simplestAdapter.apply(ownedFailure)) { stream.read(); }
        });
        check(observed == failure, "observed JDK25 failure changed");
        check(ownedFailure.closeCount == 1, "owned failure leaked resource");

        TrackingChannel borrowedFailure = new TrackingChannel(new byte[0], failure);
        try {
            IOException borrowedObserved = expect(IOException.class, () -> {
                try (InputStream stream = borrowedView(borrowedFailure)) { stream.read(); }
            });
            check(borrowedObserved == failure && borrowedFailure.isOpen(),
                    "borrowed failure changed cause/ownership");
        } finally { borrowedFailure.close(); }
        check(borrowedFailure.closeCount == 1, "borrowed owner cleanup omitted");

        expect(NullPointerException.class,
                () -> simplestAdapter.apply(null));

        Pipe pipe = Pipe.open();
        try (Pipe.SourceChannel source = pipe.source(); Pipe.SinkChannel sink = pipe.sink()) {
            check(sink.isOpen(), "pipe sink unavailable");
            source.configureBlocking(false);
            try (InputStream stream = simplestAdapter.apply(source)) {
                expect(IllegalBlockingModeException.class, stream::read);
            }
        }
        System.out.println("runtime=" + Runtime.version() + "; compiler target=17; preview=false");
        System.out.println("PASS: " + checks + " channel adaptation checks");
        System.out.println("No concurrency stress or real provider interaction performed.");
    }
}
