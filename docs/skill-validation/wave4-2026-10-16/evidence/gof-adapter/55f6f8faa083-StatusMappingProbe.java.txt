public final class StatusMappingProbe {
    private AuthorisationStatus status(String raw) {
        if (raw == null) throw new UnknownGatewayStatus("missing status");
        return switch (raw) {
            case "authorised" -> AuthorisationStatus.AUTHORISED;
            case "pending" -> AuthorisationStatus.PENDING;
            case "failed" -> AuthorisationStatus.FAILED;
            default -> throw new UnknownGatewayStatus("unrecognized status");
        };
    }

    public static void main(String[] args) {
        StatusMappingProbe probe = new StatusMappingProbe();
        if (probe.status("authorised") != AuthorisationStatus.AUTHORISED
                || probe.status("pending") != AuthorisationStatus.PENDING
                || probe.status("failed") != AuthorisationStatus.FAILED) {
            throw new AssertionError("Known status translation changed");
        }
        String[] invalid = {null, "", "new_status", "private-token\r\nforged-log"};
        for (String raw : invalid) {
            try { probe.status(raw); throw new AssertionError("Unknown status accepted"); }
            catch (UnknownGatewayStatus expected) {
                String message = raw == null ? "missing status" : "unrecognized status";
                if (!message.equals(expected.getMessage())) {
                    throw new AssertionError("Unexpected or raw diagnostic");
                }
            }
        }
        System.out.println("PASS: 3 known mappings and 4 null/unknown-status controls");
    }
}
enum AuthorisationStatus { AUTHORISED, PENDING, FAILED }
final class UnknownGatewayStatus extends RuntimeException {
    private static final long serialVersionUID = 1L;
    UnknownGatewayStatus(String message) { super(message); }
}
