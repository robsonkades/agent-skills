import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";
import { fileURLToPath } from "node:url";

const outputDir = path.dirname(fileURLToPath(import.meta.url));
const sourcePath = "C:/git/agent-skills/skills/gof-adapter/references/worked-example.md";
const source = fs.readFileSync(sourcePath, "utf8");
const matches = [...source.matchAll(/    private AuthorisationStatus status\(String raw\) \{[\s\S]*?\n    \}/g)];
if (matches.length !== 1) throw Error("Expected exactly one shipped status method");
const method = matches[0][0];
const fixture = `public final class StatusMappingProbe {
${method}

    public static void main(String[] args) {
        StatusMappingProbe probe = new StatusMappingProbe();
        if (probe.status("authorised") != AuthorisationStatus.AUTHORISED
                || probe.status("pending") != AuthorisationStatus.PENDING
                || probe.status("failed") != AuthorisationStatus.FAILED) {
            throw new AssertionError("Known status translation changed");
        }
        String[] invalid = {null, "", "new_status", "private-token\\r\\nforged-log"};
        for (String raw : invalid) {
            try { probe.status(raw); throw new AssertionError("Unknown status accepted"); }
            catch (UnknownGatewayStatus expected) {
                String message = raw == null ? "missing status" : "unrecognized status";
                if (!message.equals(expected.getMessage())) {
                    throw new AssertionError("Unexpected or raw diagnostic");
                }
            }
        }
        System.out.println("PASS: 3 known mappings and 4 null/unknown-status controls");
    }
}
enum AuthorisationStatus { AUTHORISED, PENDING, FAILED }
final class UnknownGatewayStatus extends RuntimeException {
    private static final long serialVersionUID = 1L;
    UnknownGatewayStatus(String message) { super(message); }
}
`;
fs.writeFileSync(path.join(outputDir, "StatusMappingProbe.java"), fixture);
fs.writeFileSync(path.join(outputDir, "extraction-manifest.json"), JSON.stringify({
    sourcePath,
    sourceSha256: crypto.createHash("sha256").update(fs.readFileSync(sourcePath)).digest("hex"),
    extractedMethodUnchanged: true,
    methodSha256: crypto.createHash("sha256").update(method).digest("hex"),
    additions: "Class wrapper, a three-value enum, a minimal exception and assertions only. No SDK/payment body extracted or simulated as integration."
}, null, 2));
console.log("Extracted unchanged status method and wrote isolated compilation wrapper.");
