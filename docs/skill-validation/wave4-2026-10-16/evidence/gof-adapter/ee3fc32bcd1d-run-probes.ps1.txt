$ErrorActionPreference = 'Stop'
$jdkBin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin'
$results = [System.Collections.Generic.List[object]]::new()
function Run-Check {
    param([string]$Id, [string]$Tool, [string[]]$ToolArgs)
    $output = & (Join-Path $jdkBin $Tool) @ToolArgs 2>&1
    $actualExit = $LASTEXITCODE
    $logPath = Join-Path $PSScriptRoot ($Id + '.log')
    $output | Set-Content -LiteralPath $logPath -Encoding utf8
    $output | ForEach-Object { Write-Output $_ }
    $results.Add([pscustomobject]@{id=$Id;executable=(Join-Path $jdkBin $Tool);arguments=$ToolArgs;cwd=$PSScriptRoot;expectedExit=0;actualExit=$actualExit;log=$logPath})
    if ($actualExit -ne 0) { throw "Unexpected failure in $Id with exit $actualExit" }
}
Push-Location $PSScriptRoot
try {
    New-Item -ItemType Directory -Path 'classes' -Force | Out-Null
    Run-Check -Id 'java-version' -Tool 'java.exe' -ToolArgs @('--version')
    Run-Check -Id 'javac-version' -Tool 'javac.exe' -ToolArgs @('--version')
    Run-Check -Id 'compile' -Tool 'javac.exe' -ToolArgs @('--release','17','-Xlint:all','-Werror','-XDrawDiagnostics','-d','classes','ChannelContractProbe.java','StatusMappingProbe.java')
    Run-Check -Id 'channel-contract' -Tool 'java.exe' -ToolArgs @('-cp','classes','ChannelContractProbe')
    Run-Check -Id 'status-mapping' -Tool 'java.exe' -ToolArgs @('-cp','classes','StatusMappingProbe')
} finally {
    $results | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $PSScriptRoot 'technical-check-results.json') -Encoding utf8
    Pop-Location
}
exit 0
