import java.util.ArrayList;
import java.util.List;
import java.util.function.Supplier;

public final class FamilyContractChecks {
    static final class Pdf {}
    static final class Html {}
    record Part<F>(Object owner) {}
    static <F> boolean sameOwner(Part<F> first, Part<F> second) {
        return first.owner() == second.owner();
    }
    static <F> void requireSameOwner(Part<F> first, Part<F> second) {
        if (!sameOwner(first, second)) throw new IllegalArgumentException("Different owner");
    }
    static final class Resource implements AutoCloseable {
        final List<String> events;
        Resource(List<String> events) { this.events = events; events.add("acquire"); }
        public void close() { events.add("close"); }
    }
    static Resource failingOpen(List<String> events, boolean cleansUp) {
        Resource acquired = new Resource(events);
        try {
            throw new IllegalStateException("second acquisition failed");
        } catch (RuntimeException failure) {
            if (cleansUp) acquired.close();
            throw failure;
        }
    }
    static int checks;
    static void check(boolean value, String name) {
        if (!value) throw new AssertionError(name);
        checks++;
        System.out.println("PASS " + name);
    }
    public static void main(String[] args) {
        App app = new App();
        check(app.export(new Reports.Report("Title"), Reports.Format.PDF).equals("PDF:default:Title"),
                "coherent-pdf");
        check(app.export(new Reports.Report("Title"), Reports.Format.HTML).equals("HTML:default:Title"),
                "coherent-html");
        try {
            app.export(new Reports.Report("Title"), Reports.Format.TEXT);
            throw new AssertionError("unsupported key accepted");
        } catch (IllegalArgumentException expected) { check(true, "unsupported-key-rejected"); }
        Reports.Family pdf = Reports.create(Reports.Format.PDF);
        Reports.Family html = Reports.create(Reports.Format.HTML);
        try {
            new Reports.Family(null, pdf.paginator(), pdf.styles());
            throw new AssertionError("null accepted");
        } catch (NullPointerException expected) { check(true, "null-rejected"); }

        var mixed = new Reports.Family(pdf.renderer(), html.paginator(), pdf.styles());
        String result = mixed.export(new Reports.Report("Title"));
        check(result.equals("PDF:default:Title<br>"), "mixed-record-accepted-marker-visible");
        System.out.println("OBSERVATION mixed-output=" + result);

        Object owner = new Object();
        Part<Pdf> first = new Part<>(owner);
        Part<Pdf> matched = new Part<>(owner);
        Part<Pdf> separate = new Part<>(new Object());
        check(sameOwner(first, matched), "same-type-same-owner");
        check(!sameOwner(first, separate), "same-type-different-owner-call-compiles");
        try {
            requireSameOwner(first, separate);
            throw new AssertionError("different owner accepted");
        } catch (IllegalArgumentException expected) { check(true, "explicit-owner-rejection"); }

        Supplier<Object> shared = () -> owner;
        Supplier<Object> fresh = Object::new;
        check(shared.get() == shared.get(), "supplier-can-reuse-product");
        check(fresh.get() != fresh.get(), "supplier-can-create-product");

        for (boolean cleansUp : new boolean[] {false, true}) {
            List<String> events = new ArrayList<>();
            try (Resource resource = failingOpen(events, cleansUp)) {
                throw new AssertionError("open unexpectedly succeeded");
            } catch (IllegalStateException expected) {
                check(events.equals(cleansUp ? List.of("acquire", "close") : List.of("acquire")),
                        cleansUp ? "provider-cleans-failed-acquisition" : "caller-cannot-close-unreturned-resource");
            }
        }
        List<String> exportFailure = new ArrayList<>();
        try (Resource resource = new Resource(exportFailure)) {
            throw new IllegalStateException("export");
        } catch (IllegalStateException expected) {
            check(exportFailure.equals(List.of("acquire", "close")), "returned-owner-closed-on-export-failure");
        }
        System.out.println("SUMMARY checks=" + checks);
    }
}
