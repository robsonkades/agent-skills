import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, existsSync, mkdirSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { loadPackageFromDirectory } from 'file:///C:/git/agent-skills/packages/core/src/application/package-loader.ts';
import { validatePackage } from 'file:///C:/git/agent-skills/packages/core/src/application/validate-package.ts';
import { NodeFileSystem } from 'file:///C:/git/agent-skills/packages/node/src/file-system.ts';
const root = import.meta.dirname;
const repository = 'C:/git/agent-skills';
const skill = 'gof-abstract-factory';
const skillRoot = resolve(repository, 'skills', skill);
const javaBin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const files = ['SKILL.md','skill.yaml','references/decision-and-alternatives.md','references/worked-example.md'];
const result = { skill, node: process.version, classification: 'Structural checks and isolated Java contract fixtures; no agent evaluation', processes: [] };
const read = path => readFileSync(path, 'utf8').replace(/^\uFEFF/, '');
function save() { writeFileSync(resolve(root, 'validation-results.json'), JSON.stringify(result, null, 2)+'\n'); }
function run(label, command, args, expectedStatus = 0, cwd = repository) {
    const output = spawnSync(command, args, {cwd, windowsHide: true, encoding:'utf8', timeout:30000});
    const stdoutPath = resolve(root, label+'.stdout.log');
    const stderrPath = resolve(root, label+'.stderr.log');
    writeFileSync(stdoutPath, output.stdout ?? '');
    writeFileSync(stderrPath, output.stderr ?? '');
    result.processes.push({label,command,args,cwd,status:output.status,expectedStatus,
        signal:output.signal,error:output.error?.message??null,stdoutPath,stderrPath});
    save();
    assert.equal(output.status, expectedStatus, label+': '+output.stdout+'\n'+output.stderr);
    assert.equal(output.signal,null);
    assert.equal(output.error,undefined);
    return output;
}
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), skillRoot, {strict:true});
const issues = [...loaded.issues, ...validatePackage(loaded.pkg,{directoryName:skill}).issues];
assert.deepEqual(issues, []);
assert.equal(loaded.pkg.manifest.version,'1.2.4');
assert.equal(loaded.pkg.files.length,4);
result.package = { version:loaded.pkg.manifest.version, fileCount:loaded.pkg.files.length, issues };
result.unchangedFiles = [];
for (const file of files) {
    const bytes = readFileSync(resolve(skillRoot,file));
    assert(bytes.equals(readFileSync(resolve(root,'baseline',skill,file))), 'Changed: '+file);
    result.unchangedFiles.push({path:'skills/'+skill+'/'+file,bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex')});
}
const routed = files.filter(file=>file.startsWith('references/'));
for (const ref of routed) assert(read(resolve(skillRoot,'SKILL.md')).includes(ref));
let links = 0;
for (const file of files.filter(file=>file.endsWith('.md'))) {
    for (const match of read(resolve(skillRoot,file)).matchAll(/\]\(([^)]+)\)/g)) {
        const target=match[1].split('#')[0];
        if (!target || /^[a-z]+:/i.test(target)) continue;
        assert(existsSync(resolve(dirname(resolve(skillRoot,file)), target)), 'Missing link: '+target);
        links++;
    }
}
result.references = {routed,localLinks:links};
const a = resolve(root,'cases/trusted-assembly'), b = resolve(root,'cases/public-sdk');
for(const file of ['input.md','README.md','src/Reports.java']) {
    assert(readFileSync(resolve(a,file)).equals(readFileSync(resolve(b,file))), 'Unexpected fixture difference '+file);
}
assert.notEqual(read(resolve(a,'api-contract.md')),read(resolve(b,'api-contract.md')));
const inputs = JSON.parse(read(resolve(root,'case-inputs.json')));
const criteria = JSON.parse(read(resolve(root,'case-criteria.json')));
assert.equal(inputs.length,2); assert.equal(criteria.length,2);
for(const item of inputs) for(const file of item.files) assert(existsSync(resolve(item.fixtureRoot,file)));
result.cases = {count:2,identicalFiles:['input.md','README.md','src/Reports.java'],
    soleDifference:'Supported consumer/enforcement contract in api-contract.md',criteriaSeparate:true};
run('java-version',resolve(javaBin,'java.exe'),['-version']);
run('javac-version',resolve(javaBin,'javac.exe'),['-version']);
const classes=resolve(root,'classes');
mkdirSync(classes,{recursive:true});
run('java-compile',resolve(javaBin,'javac.exe'),['--release','17','-d',classes,
    resolve(a,'src/Reports.java'),resolve(root,'technical/FamilyContractChecks.java')]);
const checkOutput=run('java-contract-checks',resolve(javaBin,'java.exe'),['-cp',classes,'FamilyContractChecks'],0,root);
assert.equal((checkOutput.stdout.match(/^PASS /gm)??[]).length,13);
assert(checkOutput.stdout.includes('SUMMARY checks=13'));
const rejected=run('java-type-mismatch',resolve(javaBin,'javac.exe'),['--release','17','-cp',classes,'-d',classes,
    resolve(root,'technical/TypeMismatch.java')],1);
assert(rejected.stderr.includes('incompatible') || rejected.stderr.includes('cannot be applied'));
assert(!existsSync(resolve(classes,'TypeMismatch.class')));
result.java = {release:17,preview:false,assertions:13,negativeCompile:'Expected exit 1 for different generic family types',
    source:'Reviewer-created isolated contract fixtures; shipped sketches unmodified, not represented as complete compiled examples',
    limits:'Runtime is the installed JDK 25; no Java17 VM, Spring, PDF/HTML encoder, real database, concurrency schedule or agent evaluation.'};
run('prettier',process.execPath,['node_modules/prettier/bin/prettier.cjs','--check',...files.map(file=>'skills/'+skill+'/'+file)]);
run('diff-check','git',['diff','--check','--','skills/'+skill]);
for (const [label,args] of [
    ['final-owned-diff',['diff','--','skills/'+skill]],
    ['final-owned-status',['status','--short','--untracked-files=all','--','skills/'+skill]],
    ['final-staged-diff',['diff','--cached','--','skills/'+skill]]
]) assert.equal(run(label,'git',args).stdout,'');
result.allProcessesExited=true;
save();
console.log(JSON.stringify(result,null,2));
