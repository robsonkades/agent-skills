package application.stock;
import java.time.Instant;
import java.util.Properties;
import domain.order.Order;
import domain.stock.StockEligibility;
import domain.stock.StockEligibility.Assessment;
import domain.stock.StockEligibility.Result;
import domain.stock.StockEligibility.Snapshot;

public final class AssessStockUseCase {
    public interface StockGateway { Snapshot snapshot(String tenant, String sku); }
    public interface Orders { void saveWithVersionCheck(Order order); }
    private final StockEligibility rule;
    private final StockGateway stock;
    private final Orders orders;
    private final String operation;
    public AssessStockUseCase(StockEligibility rule, StockGateway stock, Orders orders, Properties runtime) {
        this.rule = rule; this.stock = stock; this.orders = orders;
        this.operation = runtime.getProperty("stock.operation");
        if (!operation.equals("PREVIEW") && !operation.equals("SUBMIT")) throw new IllegalArgumentException("mode");
    }
    public Assessment execute(Order order, Instant decisionAt) {
        var request = order.request();
        var facts = stock.snapshot(request.tenant(), request.sku());
        var assessment = rule.assess(request, facts, decisionAt);
        if (operation.equals("SUBMIT") && assessment.result() == Result.ELIGIBLE) {
            order.submit(assessment);
            orders.saveWithVersionCheck(order);
        }
        return assessment;
    }
}
