package domain.stock;
import java.time.Duration;
import java.time.Instant;
import java.util.Objects;

public final class StockEligibility {
    public record Request(String tenant, String orderId, String sku, int quantity) {
        public Request {
            Objects.requireNonNull(tenant); Objects.requireNonNull(orderId); Objects.requireNonNull(sku);
            if (quantity <= 0) throw new IllegalArgumentException("quantity");
        }
    }
    public record Snapshot(String tenant, String sku, int available, Instant observedAt, String sourceVersion) { }
    public enum Result { ELIGIBLE, INELIGIBLE, NEEDS_EVIDENCE }
    public record Assessment(Request request, Result result, String sourceVersion, Instant decisionAt) { }
    public Assessment assess(Request request, Snapshot facts, Instant decisionAt) {
        if (facts == null || facts.observedAt() == null || facts.sourceVersion() == null
                || !request.tenant().equals(facts.tenant()) || !request.sku().equals(facts.sku())) {
            return new Assessment(request, Result.NEEDS_EVIDENCE, null, decisionAt);
        }
        var age = Duration.between(facts.observedAt(), decisionAt);
        if (age.isNegative() || age.compareTo(Duration.ofSeconds(30)) > 0 || facts.available() < 0) {
            return new Assessment(request, Result.NEEDS_EVIDENCE, facts.sourceVersion(), decisionAt);
        }
        var result = facts.available() >= request.quantity() ? Result.ELIGIBLE : Result.INELIGIBLE;
        return new Assessment(request, result, facts.sourceVersion(), decisionAt);
    }
}
