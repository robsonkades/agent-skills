import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
const here = dirname(fileURLToPath(import.meta.url));
const sources = JSON.parse(readFileSync(resolve(here, 'source-evidence.json'), 'utf8'));
const extraction = [];
for (const source of sources) {
  assert.equal(source.status, 'CAPTURED');
  const output = resolve(here, 'sources', source.id + '.txt');
  if (source.id === 'evans-ddd-reference') {
    const executable = 'C:/Users/robso/AppData/Local/Microsoft/WinGet/Packages/oschwartz10612.Poppler_Microsoft.Winget.Source_8wekyb3d8bbwe/poppler-25.07.0/Library/bin/pdftotext.exe';
    const args = ['-layout', source.evidence, output];
    const run = spawnSync(executable, args, { cwd: here, encoding: 'utf8', windowsHide: true, timeout: 12000 });
    assert.equal(run.status, 0);
    writeFileSync(resolve(here, 'sources', source.id + '.normalized.txt'),
      readFileSync(output, 'utf8').replace(/\r/g, '').replace(/\t/g, ' ').replace(/\u00a0/g, ' '));
    extraction.push({ source: source.id, command: [executable, ...args].join(' '), cwd: here,
      exitCode: run.status, stdout: run.stdout, stderr: run.stderr, output });
  } else {
    const plain = readFileSync(source.evidence, 'utf8').replace(/<script\b[\s\S]*?<\/script>/gi, '')
      .replace(/<style\b[\s\S]*?<\/style>/gi, '').replace(/<!--[\s\S]*?-->/g, '')
      .replace(/<\/(p|div|section|h[1-6]|li|tr|pre)>/gi, '\n').replace(/<[^>]+>/g, ' ')
      .replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&quot;/g, '"').replace(/&#39;/g, "'")
      .split(/\r?\n/).map(s => s.replace(/\s+/g, ' ').trim()).filter(Boolean).join('\n');
    writeFileSync(output, plain + '\n');
    extraction.push({ source: source.id, method: 'Readable text from captured HTML; original retained', output });
  }
}
writeFileSync(resolve(here, 'source-text-results.json'), JSON.stringify(extraction, null, 2) + '\n');
console.log(JSON.stringify(extraction, null, 2));
