package domain.order;
import domain.stock.StockEligibility.Assessment;
import domain.stock.StockEligibility.Request;
import domain.stock.StockEligibility.Result;

public final class Order {
    private final Request request;
    private boolean submitted;
    public Order(Request request) { this.request = java.util.Objects.requireNonNull(request); }
    public Request request() { return request; }
    public void submit(Assessment assessment) {
        if (submitted) throw new IllegalStateException("already submitted");
        if (!request.equals(assessment.request()) || assessment.result() != Result.ELIGIBLE) {
            throw new IllegalArgumentException("assessment does not authorize these order inputs");
        }
        submitted = true;
    }
}
