import { mkdirSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
const here = dirname(fileURLToPath(import.meta.url));
mkdirSync(resolve(here, 'sources'), { recursive: true });
const sources = [
  ['evans-ddd-reference', 'https://www.domainlanguage.com/wp-content/uploads/2016/05/DDD_Reference_2015-03.pdf', 'Eric Evans, Domain-Driven Design Reference, March 2015; Services and Factories', '.pdf'],
  ['fowler-domain-model', 'https://martinfowler.com/eaaCatalog/domainModel.html', 'Martin Fowler, Domain Model catalog entry', '.html'],
  ['fowler-anemic-model', 'https://martinfowler.com/bliki/AnemicDomainModel.html', 'Martin Fowler, Anemic Domain Model, 25 November 2003', '.html'],
  ['aws-idempotent', 'https://aws.amazon.com/builders-library/making-retries-safe-with-idempotent-APIs/', 'AWS Builders Library, Making retries safe with idempotent APIs; request identity, lost responses and changed intent', '.html'],
  ['postgres17-isolation', 'https://www.postgresql.org/docs/17/transaction-iso.html', 'PostgreSQL 17 section 13.2 Transaction Isolation', '.html'],
  ['spring-data-specifications', 'https://docs.spring.io/spring-data/jpa/reference/3.5/jpa/specifications.html', 'Spring Data JPA 3.5 Specifications reference requested; verify resolved version before interpretation', '.html']
];
const results = await Promise.all(sources.map(async ([id, url, section, extension]) => {
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(18000) });
    if (!response.ok) throw new Error('HTTP ' + response.status);
    const bytes = Buffer.from(await response.arrayBuffer());
    const path = resolve(here, 'sources', id + extension);
    writeFileSync(path, bytes);
    return { id, url, resolvedUrl: response.url, section, consulted: '2026-10-16', primary: true,
      status: 'CAPTURED', httpStatus: response.status, evidence: path, bytes: bytes.length,
      sha256: createHash('sha256').update(bytes).digest('hex') };
  } catch (error) {
    return { id, url, section, consulted: '2026-10-16', primary: true, status: 'UNAVAILABLE', error: error.message };
  }
}));
writeFileSync(resolve(here, 'source-evidence.json'), JSON.stringify(results, null, 2) + '\n');
console.log(JSON.stringify(results, null, 2));
