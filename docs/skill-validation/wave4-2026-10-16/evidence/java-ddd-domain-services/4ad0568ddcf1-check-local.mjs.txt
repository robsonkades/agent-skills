import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync, readdirSync, mkdirSync, existsSync } from 'node:fs';
import { dirname, resolve, relative } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { spawnSync } from 'node:child_process';
const here = dirname(fileURLToPath(import.meta.url));
const repo = 'C:/git/agent-skills';
const name = 'java-ddd-domain-services';
const owned = 'skills/' + name;
const skill = resolve(repo, owned);
const coordinatorBaseline = resolve(here, '../baseline', name);
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const gitChecks = [];
function git(args) {
  const r = spawnSync('git', args, { cwd: repo, encoding: 'utf8', windowsHide: true });
  gitChecks.push({ command: 'git ' + args.join(' '), cwd: repo, exitCode: r.status, stdout: r.stdout, stderr: r.stderr });
  assert.equal(r.status, 0);
  return r.stdout;
}
for (const args of [['status', '--short', '--', owned], ['diff', '--', owned], ['diff', '--cached', '--', owned], ['diff', '--check', '--', owned]]) {
  assert.equal(git(args).trim(), '');
}
const committed = git(['show', 'HEAD:' + owned + '/skill.yaml']);
const manifestText = readFileSync(resolve(skill, 'skill.yaml'), 'utf8');
assert.equal(committed.match(/^version:\s*(\S+)/m)[1], '1.0.0');
assert.equal(manifestText.match(/^version:\s*(\S+)/m)[1], '1.0.0');
const pathsFor = root => readdirSync(root, { recursive: true, withFileTypes: true }).filter(e => e.isFile())
  .map(e => relative(root, resolve(e.parentPath, e.name)).replaceAll('\\', '/')).sort();
const paths = pathsFor(skill);
assert.deepEqual(paths, pathsFor(coordinatorBaseline));
const files = paths.map(path => {
  const bytes = readFileSync(resolve(skill, path));
  const hash = sha(bytes);
  assert.equal(hash, sha(readFileSync(resolve(coordinatorBaseline, path))));
  return { path, sha256: hash };
});
const baselinePath = resolve(here, 'baseline.json');
if (!existsSync(baselinePath)) {
  for (const path of paths) {
    const destination = resolve(here, 'baseline', path);
    mkdirSync(dirname(destination), { recursive: true });
    writeFileSync(destination, readFileSync(resolve(skill, path)));
  }
  writeFileSync(baselinePath, JSON.stringify({ skill: name, committedVersion: '1.0.0', assignmentVersion: '1.0.0',
    coordinatorBaseline, preExistingChanges: [], files }, null, 2) + '\n');
}
assert.deepEqual(JSON.parse(readFileSync(baselinePath, 'utf8')).files, files);
const { parseManifest } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/skill-document.ts')));
const { manifest, issues } = parseManifest(manifestText, { strict: true });
const doc = parseSkillDocument(readFileSync(resolve(skill, 'SKILL.md'), 'utf8'));
assert.deepEqual(issues, []);
assert.equal(documentName(doc), manifest.name);
assert.equal(documentDescription(doc), manifest.description);
for (const entry of manifest.files) assert.ok(existsSync(resolve(skill, entry)), entry);
let links = 0;
for (const path of paths.filter(p => p.endsWith('.md'))) {
  for (const m of readFileSync(resolve(skill, path), 'utf8').matchAll(/\]\(([^\s)]+)\)/g)) {
    const target = m[1].replace(/^<|>$/g, '');
    if (/^(https?:|#)/.test(target)) continue;
    assert.ok(existsSync(resolve(skill, dirname(path), target)), target);
    links++;
  }
}
const referrals = ['domain-logic-organization', 'gof-strategy', 'java-ddd-aggregates', 'java-ddd-domain-events',
  'java-ddd-repositories', 'java-ddd-testing', 'java-ddd-use-cases', 'java-ddd-value-objects', 'query-objects-and-specifications'];
for (const target of referrals) {
  assert.ok(manifest.suggests.includes(target), target);
  assert.ok(existsSync(resolve(repo, 'skills', target, 'SKILL.md')), target);
}
const result = { status: 'PASS', skill: name, version: manifest.version, packageFiles: paths.length, strictManifestIssues: 0,
  metadataAgreement: true, relativeLinks: links, directReferrals: referrals.length,
  assignmentAndCoordinatorBaselineByteIdentical: true, files, gitChecks };
writeFileSync(resolve(here, 'structure-results.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify({ ...result, gitChecks: gitChecks.map(c => ({ command: c.command, exitCode: c.exitCode })) }, null, 2));
