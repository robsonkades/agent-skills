import assert from 'node:assert/strict';
import { mkdirSync, writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
const here = dirname(fileURLToPath(import.meta.url));
const executable = 'C:/Users/robso/.jdks/temurin-25.0.3/bin/javac.exe';
const classes = resolve(here, 'case-classes');
mkdirSync(classes, { recursive: true });
const sourceRoot = resolve(here, 'cases/case-a/src');
const env = { ...process.env };
for (const key of ['JAVA_TOOL_OPTIONS', '_JAVA_OPTIONS', 'JDK_JAVA_OPTIONS']) delete env[key];
const records = [];
for (const args of [['-version'], ['--release', '17', '-Xlint:all', '-d', classes,
  resolve(sourceRoot, 'domain/stock/StockEligibility.java'), resolve(sourceRoot, 'domain/order/Order.java'),
  resolve(sourceRoot, 'application/stock/AssessStockUseCase.java')]]) {
  const run = spawnSync(executable, args, { cwd: here, encoding: 'utf8', env, windowsHide: true, timeout: 12000 });
  records.push({ command: [executable, ...args].join(' '), cwd: here, exitCode: run.status, stdout: run.stdout, stderr: run.stderr });
  assert.equal(run.status, 0, JSON.stringify(records.at(-1)));
}
const result = { status: 'PASS', records, actorRuns: 0, runtimeExecutions: 0,
  scope: 'Compilation of the shared synthetic case sources only. Case B sources are byte-identical. Not a test of shipped design sketches or actual persistence/provider behavior.' };
writeFileSync(resolve(here, 'case-source-results.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify(result, null, 2));
