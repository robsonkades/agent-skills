import { mkdirSync, writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
const root = dirname(fileURLToPath(import.meta.url));
const repository = 'C:/git/agent-skills';
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const commands = [];
function run(id, executable, args, cwd = repository) {
  const result = spawnSync(executable, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
  writeFileSync(resolve(root, `${id}.stdout.txt`), result.stdout ?? '');
  writeFileSync(resolve(root, `${id}.stderr.txt`), result.stderr ?? '');
  commands.push({ id, executable, args, cwd, status: result.status, signal: result.signal, error: result.error?.message, stdout: `${id}.stdout.txt`, stderr: `${id}.stderr.txt` });
  return result.status === 0;
}
run('local-validation', process.execPath, [resolve(root, 'local-validation.mjs')]);
run('prettier', process.execPath, [resolve(repository, 'node_modules/prettier/bin/prettier.cjs'), '--check', 'skills/domain-logic-organization/SKILL.md', 'skills/domain-logic-organization/skill.yaml', 'skills/domain-logic-organization/references/domain-model.md', 'skills/domain-logic-organization/references/transaction-script-and-table-module.md']);
run('diff-check', 'git', ['diff', '--check', '--', 'skills/domain-logic-organization']);
run('scoped-status', 'git', ['status', '--short', '--untracked-files=all', '--', 'skills/domain-logic-organization']);
run('scoped-diff', 'git', ['diff', '--', 'skills/domain-logic-organization']);
run('staged-diff', 'git', ['diff', '--cached', '--', 'skills/domain-logic-organization']);
run('java-version', resolve(jdk, 'java.exe'), ['-version']);
run('javac-version', resolve(jdk, 'javac.exe'), ['-version']);
if (run('case-preparation', process.execPath, [resolve(root, 'prepare-cases.mjs')])) {
  const output = resolve(root, 'case-compiled');
  mkdirSync(output, { recursive: true });
  run('case-compile', resolve(jdk, 'javac.exe'), ['--release', '17', '-d', output, ...['Bootstrap.java', 'StockGateway.java', 'ReservationService.java'].map(name => resolve(root, 'cases/case-a', name))]);
}
writeFileSync(resolve(root, 'commands.json'), JSON.stringify(commands, null, 2) + '\n');
console.log(JSON.stringify(commands.map(({ id, status, error }) => ({ id, status, error })), null, 2));
if (commands.some(command => command.status !== 0)) process.exitCode = 1;
