package warehouse;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import javax.sql.DataSource;

public final class Bootstrap {
    public static ReservationService service(DataSource database) throws IOException {
        try (var input = Bootstrap.class.getResourceAsStream("read-stock.sql")) {
            if (input == null) throw new IOException("missing read-stock.sql");
            String readSql = new String(input.readAllBytes(), StandardCharsets.UTF_8);
            return new ReservationService(database, new StockGateway(readSql));
        }
    }
}
