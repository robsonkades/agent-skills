  'StockGateway.java': `package warehouse;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.UUID;

public final class StockGateway {
    private final String readSql;

    public StockGateway(String readSql) { this.readSql = readSql; }

    public int available(Connection connection, String sku) throws SQLException {
        try (var statement = connection.prepareStatement(readSql)) {
            statement.setString(1, sku);
            try (var result = statement.executeQuery()) {
                if (!result.next()) throw new IllegalArgumentException("unknown sku");
                return result.getInt("available");
            }
        }
    }

    public void setAvailable(Connection connection, String sku, int value) throws SQLException {
        try (var statement = connection.prepareStatement(
                "UPDATE stock SET available = ? WHERE sku = ?")) {
            statement.setInt(1, value);
            statement.setString(2, sku);
            if (statement.executeUpdate() != 1) throw new SQLException("stock row missing");
        }
    }

    public void insertReservation(Connection connection, UUID requestId, String sku, int quantity)
            throws SQLException {
        try (var statement = connection.prepareStatement(
                "INSERT INTO reservation (request_id, sku, quantity) VALUES (?, ?, ?)")) {
            statement.setObject(1, requestId);
            statement.setString(2, sku);
            statement.setInt(3, quantity);
            statement.executeUpdate();
        }
    }
}
`,
  'ReservationService.java': `package warehouse;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.Objects;
import java.util.UUID;
import javax.sql.DataSource;

public final class ReservationService {
    private final DataSource database;
    private final StockGateway stock;

    public ReservationService(DataSource database, StockGateway stock) {
        this.database = database;
        this.stock = stock;
    }

    public UUID reserve(UUID requestId, String sku, int quantity) throws SQLException {
        Objects.requireNonNull(requestId);
        Objects.requireNonNull(sku);
        if (quantity <= 0) throw new IllegalArgumentException("quantity must be positive");
        try (var connection = database.getConnection()) {
            connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
            connection.setAutoCommit(false);
            try {
                int available = stock.available(connection, sku);
                if (available < quantity) throw new IllegalStateException("insufficient stock");
                stock.setAvailable(connection, sku, available - quantity);
                stock.insertReservation(connection, requestId, sku, quantity);
                connection.commit();
                return requestId;
            } catch (SQLException | RuntimeException failure) {
                try { connection.rollback(); }
                catch (SQLException rollbackFailure) { failure.addSuppressed(rollbackFailure); }
                throw failure;
            }
        }
    }
}
`,
