import java.io.IOException;
import java.util.Objects;

final class ReportPublisher {
    private final Store store;

    ReportPublisher(Store store) {
        this.store = Objects.requireNonNull(store);
    }

    synchronized boolean publish(String key, Report report) throws IOException {
        if (store.exists(key)) return false;
        store.write(key, report.render());
        return true;
    }
}
