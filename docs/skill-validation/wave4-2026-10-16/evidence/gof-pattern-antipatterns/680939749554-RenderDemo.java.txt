import java.util.List;
public class RenderDemo {
  interface Node { String label(); String accept(Visitor visitor); }
  record Text(String label) implements Node { public String accept(Visitor v) { return v.text(this); } }
record Heading(String label) implements Node { public String accept(Visitor v) { return v.heading(this); } }
record Badge(String label) implements Node { public String accept(Visitor v) { return v.badge(this); } }
  
  interface Visitor {
    String text(Text node);
    String heading(Heading node);
    String unknown(Node node);
    String badge(Badge node);
  }
  static class Plain implements Visitor {
    public String text(Text node) { return node.label(); }
    public String heading(Heading node) { return node.label(); }
    public String unknown(Node node) { return node.label(); }
  }
  static class Upper implements Visitor {
    public String text(Text node) { return node.label().toUpperCase(java.util.Locale.ROOT); }
    public String heading(Heading node) { return node.label().toUpperCase(java.util.Locale.ROOT); }
    public String unknown(Node node) { return node.label().toUpperCase(java.util.Locale.ROOT); }
  }
  static String plain(Node node) { return node.label(); }
  static String upper(Node node) { return node.label().toUpperCase(java.util.Locale.ROOT); }
  static int assertions;
  static void eq(String actual, String expected) {
    assertions++;
    if (!actual.equals(expected)) throw new AssertionError("expected=" + expected + " actual=" + actual);
  }
  public static void main(String[] args) {
    List<Node> nodes = List.of(new Text("daily"), new Heading("status"), new Badge("staging"));
    for (Node node : nodes) {
      eq(plain(node), node.label());
      eq(upper(node), node.label().toUpperCase(java.util.Locale.ROOT));
      
      eq(node.accept(new Plain()), node.label());
      eq(node.accept(new Upper()), node.label().toUpperCase(java.util.Locale.ROOT));
    }
    System.out.println("nodes=" + nodes.size() + " assertions=" + assertions);
  }
}
