import java.util.List;
public class RenderDemo {
  interface Node { String label();  }
  record Text(String label) implements Node {  }
record Heading(String label) implements Node {  }
record Badge(String label) implements Node {  }
  
  static String plain(Node node) { return node.label(); }
  static String upper(Node node) { return node.label().toUpperCase(java.util.Locale.ROOT); }
  static int assertions;
  static void eq(String actual, String expected) {
    assertions++;
    if (!actual.equals(expected)) throw new AssertionError("expected=" + expected + " actual=" + actual);
  }
  public static void main(String[] args) {
    List<Node> nodes = List.of(new Text("daily"), new Heading("status"), new Badge("staging"));
    for (Node node : nodes) {
      eq(plain(node), node.label());
      eq(upper(node), node.label().toUpperCase(java.util.Locale.ROOT));
      
    }
    System.out.println("nodes=" + nodes.size() + " assertions=" + assertions);
  }
}
