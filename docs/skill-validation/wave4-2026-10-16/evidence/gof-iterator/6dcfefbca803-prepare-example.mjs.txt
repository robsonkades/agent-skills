import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';

const root = dirname(fileURLToPath(import.meta.url));
const source = 'C:/git/agent-skills/skills/gof-iterator/references/worked-example.md';
const markdown = readFileSync(source, 'utf8');
const blocks = [...markdown.matchAll(/\x60\x60\x60java\r?\n([\s\S]*?)\r?\n\x60\x60\x60/g)].map(m => m[1]);
if (!blocks[0]?.includes('public interface AuditPageFetcher') || !blocks[1]?.includes('final class AuditSpliterator')) throw new Error('Unexpected source fences');
const split = blocks[0].indexOf('public record AuditPage');
const imports = 'import java.util.*;\nimport java.util.function.*;\nimport java.util.concurrent.CancellationException;\n';
const target = join(root, 'actor/common/src');
mkdirSync(target, { recursive: true });
const extracted = {
    'AuditPageFetcher.java': imports + blocks[0].slice(0, split).trim() + '\n',
    'AuditPage.java': imports + blocks[0].slice(split).trim() + '\n',
    'AuditSpliterator.java': imports + blocks[1] + '\n'
};
for (const [name, text] of Object.entries(extracted)) writeFileSync(join(target, name), text);
writeFileSync(join(root, 'extraction.json'), JSON.stringify({source, sourceSHA256:createHash('sha256').update(readFileSync(source)).digest('hex'), scope:'First two Java fences only; imports added and public top-level types split into files; no logic edits.', files:Object.keys(extracted)}, null, 2) + '\n');
console.log('Extracted unchanged AuditPageFetcher, AuditPage and AuditSpliterator; imports/top-level file boundaries only.');
