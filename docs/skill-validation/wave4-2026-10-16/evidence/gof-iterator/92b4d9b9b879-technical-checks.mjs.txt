import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { mkdirSync, readdirSync, writeFileSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = dirname(fileURLToPath(import.meta.url));
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const logs = join(root, 'logs-attempt2'), classes = join(root, 'classes-attempt2');
for (const p of [logs, classes]) mkdirSync(p, { recursive: true });
const commands = [];
function run(id, executable, args, expectedExit = 0) {
    const result = spawnSync(join(jdk, executable + '.exe'), args, { cwd: root, encoding: 'utf8', timeout: 30000, windowsHide: true });
    const stdoutPath = join(logs, id + '.stdout.txt'), stderrPath = join(logs, id + '.stderr.txt');
    writeFileSync(stdoutPath, result.stdout ?? ''); writeFileSync(stderrPath, result.stderr ?? '');
    commands.push({ id, executable: join(jdk, executable + '.exe'), args, cwd: root, expectedExit, actualExit: result.status, signal: result.signal, error: result.error?.message ?? null, stdoutPath, stderrPath });
    writeFileSync(join(root, 'commands-attempt2.json'), JSON.stringify(commands, null, 2) + '\n');
    assert.equal(result.error, undefined, id + ': process failure');
    assert.equal(result.status, expectedExit, id + ': unexpected exit');
    return { stdout: result.stdout, stderr: result.stderr };
}
const javaVersion = run('01-java-version', 'java', ['-version']);
const javacVersion = run('02-javac-version', 'javac', ['-version']);
const dirs = [join(root, 'actor/common/src'), join(root, 'controls')];
const sources = dirs.flatMap(dir => readdirSync(dir).filter(p => p.endsWith('.java')).map(p => join(dir, p)));
run('03-compile-release17', 'javac', ['--release', '17', '-d', classes, ...sources]);
const checks = run('04-traversal-checks', 'java', ['-cp', classes, 'TraversalChecks']);
assert.match(checks.stdout, /PASS 28 focused checks/);
const a = join(root, 'actor/case-a/consumer.properties'), b = join(root, 'actor/case-b/consumer.properties');
const originalPrefix = run('05-original-prefix', 'java', ['-cp', classes, 'CaseProbe', a, 'original']);
const originalFull = run('06-original-complete', 'java', ['-cp', classes, 'CaseProbe', b, 'original'], 1);
assert.match(originalFull.stderr, /expected INCOMPLETE/);
const fixedFull = run('07-control-complete', 'java', ['-cp', classes, 'CaseProbe', b, 'complete']);
const wrongPrefix = run('08-overbroad-control-prefix', 'java', ['-cp', classes, 'CaseProbe', a, 'complete'], 1);
assert.match(wrongPrefix.stderr, /expected COMPLETE/);
const summary = {
    javaVersion: javaVersion.stderr.trim(), javacVersion: javacVersion.stdout.trim(),
    sourceAndApiTarget: 17, actualRuntime: 25, executedChildCommands: commands.length,
    expectedNegativeExits: 2, unexpectedFailures: [],
    sourceCompilation: 'Unchanged extracted first two Java fences; imports and top-level files adapted only.',
    focusedChecks: checks.stdout.trim(),
    cases: { originalPrefix: originalPrefix.stdout.trim(), originalComplete: originalFull.stdout.trim(), completeControl: fixedFull.stdout.trim(), overbroadPrefixControl: wrongPrefix.stdout.trim() },
    limits: ['No agent evaluation or measured incremental gain.', 'Controlled provider and deadline state; no HTTP, monotonic-time/transport or server-cursor integration.', 'onClose notification observed; no real resource acquisition/release tested.', 'Sequential/parallel agreement is on an unsplittable source and does not prove decomposition or speedup.', '--release17 compilation used JDK25; no JDK17 VM execution.', 'ArrayList non-detection is a build-specific counterexample, not a future behavior guarantee.']
};
writeFileSync(join(root, 'technical-results.json'), JSON.stringify(summary, null, 2) + '\n');
console.log(JSON.stringify(summary, null, 2));
