import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.StreamSupport;

public final class ExportJob {
    public record Receipt(String status, List<Integer> recordedIds) {}

    public static Receipt run(FixtureFetcher fetcher, AtomicInteger closes) {
        List<Integer> recorded = new ArrayList<>();
        var source = new AuditSpliterator(fetcher, Cursor.START, new Deadline(), 2);
        try (var events = StreamSupport.stream(source, false).onClose(closes::incrementAndGet)) {
            events.limit(2).forEach(event -> recorded.add(event.id()));
        }
        return new Receipt("COMPLETE", List.copyOf(recorded));
    }
}
