import assert from 'node:assert/strict';
import {readFileSync, readdirSync, existsSync} from 'node:fs';
import path from 'node:path';
import {loadPackageFromDirectory} from './packages/core/src/application/package-loader.ts';
import {validatePackage} from './packages/core/src/application/validate-package.ts';
import {NodeFileSystem} from './packages/node/src/file-system.ts';
import {ClaudeCodeAdapter} from './packages/adapter-claude/src/index.ts';
import {CodexAdapter} from './packages/adapter-codex/src/index.ts';
const root='skills/orm-structural-mapping';
const loaded=await loadPackageFromDirectory(new NodeFileSystem(),root,{strict:true});
const report=validatePackage(loaded.pkg,{directoryName:'orm-structural-mapping',adapters:[new ClaudeCodeAdapter(),new CodexAdapter()]});
const issues=[...loaded.issues,...report.issues];
assert.equal(issues.length,0,JSON.stringify(issues));
const entry=readFileSync(path.join(root,'SKILL.md'),'utf8');
const refs=readdirSync(path.join(root,'references'));
for(const ref of refs) assert.ok(entry.includes('references/'+ref),'unrouted '+ref);
let links=0;
for(const file of ['SKILL.md',...refs.map(f=>'references/'+f)]) {
 const body=readFileSync(path.join(root,file),'utf8');
 for(const m of body.matchAll(/\]\(([^)]+)\)/g)) {
  const href=m[1];
  if(/^(?:<)?[a-z]+:\/\//i.test(href)) continue;
  const [target,anchor]=href.split('#');
  const resolved=target?path.resolve(root,path.dirname(file),target):path.resolve(root,file);
  assert.ok(existsSync(resolved),file+': '+href);
  if(anchor) {
   const targetText=readFileSync(resolved,'utf8');
   const anchors=[...targetText.matchAll(/^#+ (.+)$/gm)].map(m=>m[1].toLowerCase().replace(/[^a-z0-9\s-]/g,'').trim().replace(/\s+/g,'-'));
   assert.ok(anchors.includes(anchor),file+': '+href);
  }
  links++;
 }
}
console.log(JSON.stringify({skill:'orm-structural-mapping',version:loaded.pkg.manifest.version,issues,referenceFilesRouted:refs.length,localLinksAndAnchorsChecked:links},null,2));
