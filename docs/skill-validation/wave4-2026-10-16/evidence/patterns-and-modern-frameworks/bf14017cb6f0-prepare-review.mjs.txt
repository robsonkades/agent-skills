const dir = `${root}/${skill}`;
const hash = b => crypto.createHash('sha256').update(b).digest('hex');
const commands = [];
function cmd(exe, args) {
  const p = spawnSync(exe, args, { cwd: repo, encoding: 'utf8', timeout: 20000 });
  const result = { exe, args, cwd: repo, exitCode: p.status, stdout: p.stdout, stderr: p.stderr, error: p.error?.message };
  commands.push(result);
  assert.equal(p.status, 0, JSON.stringify(result));
  return p.stdout;
}
function walk(p, prefix = '') {
  return fs.readdirSync(p, { withFileTypes: true }).flatMap(e => e.isDirectory() ? walk(path.join(p, e.name), prefix + e.name + '/') : [prefix + e.name]).sort();
}
const files = walk(`${repo}/skills/${skill}`).map(name => {
  const bytes = fs.readFileSync(`${repo}/skills/${skill}/${name}`);
  const supplied = fs.readFileSync(`${root}/baseline/${skill}/${name}`);
  assert.ok(bytes.equals(supplied), `Assignment baseline differs: ${name}`);
  return { name, sha256: hash(bytes), bytes: bytes.length, content: bytes.toString('utf8') };
});
assert.deepEqual(files.map(x => x.name), walk(`${root}/baseline/${skill}`));
const status = cmd('git', ['status', '--short', '--untracked-files=all', '--', `skills/${skill}`]);
const unstaged = cmd('git', ['diff', '--', `skills/${skill}`]);
const staged = cmd('git', ['diff', '--cached', '--', `skills/${skill}`]);
const committedManifest = cmd('git', ['show', `HEAD:skills/${skill}/skill.yaml`]);
assert.equal(status.trim(), '');
assert.equal(unstaged, '');
assert.equal(staged, '');
assert.ok(committedManifest.includes('version: 1.2.4'));
fs.mkdirSync(`${dir}/sources`, { recursive: true });
fs.writeFileSync(`${dir}/assignment-baseline.json`, JSON.stringify({ skill, capturedAt: new Date().toISOString(), status, unstaged, staged, committedManifest, assignmentVersion: '1.2.4', matchedCoordinatorBaseline: true, files, commands }, null, 2));
