package app;
import org.springframework.transaction.annotation.Transactional;
public class CheckoutFlow {
    private final CatalogCommands commands;
    public CheckoutFlow(CatalogCommands commands) { this.commands = commands; }
    @Transactional
    public void submit(String id, boolean failAtEnd) {
        commands.change(id, false);
        if (failAtEnd) throw new IllegalStateException("checkout rejected");
    }
}
