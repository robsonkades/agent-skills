package app;
import org.springframework.aop.support.AopUtils;
import org.springframework.cache.Cache;
import org.springframework.cache.CacheManager;
import org.springframework.cache.concurrent.ConcurrentMapCacheManager;
import org.springframework.cache.transaction.TransactionAwareCacheDecorator;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.transaction.support.TransactionTemplate;

public class IntegrationChecks {
    static int checks;
    static void check(boolean value, String label) {
        if (!value) throw new AssertionError(label);
        checks++; System.out.println("PASS " + label);
    }
    static void rejects(Runnable run, String label) {
        try { run.run(); } catch (IllegalStateException expected) { check(true, label); return; }
        throw new AssertionError(label);
    }
    public static void main(String[] args) {
        boolean aware = Boolean.parseBoolean(args[0]);
        try (var context = new AnnotationConfigApplicationContext(CacheConfiguration.class)) {
            Cache raw = context.getBean(ConcurrentMapCacheManager.class).getCache("catalog");
            Cache configured = context.getBean(CacheManager.class).getCache("catalog");
            var flow = context.getBean(CheckoutFlow.class);
            var commands = context.getBean(CatalogCommands.class);
            var tx = context.getBean(SandboxTransactionManager.class);
            check(AopUtils.isAopProxy(flow), "outer transaction bean is proxied");
            check(AopUtils.isAopProxy(commands), "cache command bean is proxied");
            check((configured instanceof TransactionAwareCacheDecorator) == aware, "effective primary manager matches fixture");
            raw.put("sku-7", "old");
            flow.submit("sku-7", false);
            check(raw.get("sku-7") == null && tx.commits == 1, "successful outer transaction invalidates");
            raw.put("sku-7", "old");
            rejects(() -> flow.submit("sku-7", true), "outer failure propagated");
            check(tx.rollbacks == 1, "outer rollback executed");
            check((raw.get("sku-7") != null) == aware, "rollback cache retention depends on manager");
            raw.put("sku-7", "old");
            rejects(() -> commands.change("sku-7", true), "method failure propagated");
            check(raw.get("sku-7") != null, "method failure skips default eviction");
            commands.change("sku-7", false);
            check(raw.get("sku-7") == null, "without transaction eviction is immediate");
            if (aware) {
                var template = new TransactionTemplate(tx);
                raw.put("sku-7", "old");
                template.executeWithoutResult(status -> {
                    configured.put("sku-7", "new");
                    check("old".equals(raw.get("sku-7", String.class)), "put deferred while synchronization active");
                    status.setRollbackOnly();
                });
                check("old".equals(raw.get("sku-7", String.class)), "put omitted on rollback");
                template.executeWithoutResult(status -> {
                    configured.clear();
                    check(raw.get("sku-7") != null, "clear deferred while synchronization active");
                    status.setRollbackOnly();
                });
                check(raw.get("sku-7") != null, "clear omitted on rollback");
                template.executeWithoutResult(status -> {
                    configured.evictIfPresent("sku-7");
                    check(raw.get("sku-7") == null, "evictIfPresent immediate");
                    configured.putIfAbsent("inserted", "v");
                    check(raw.get("inserted") != null, "putIfAbsent immediate");
                    status.setRollbackOnly();
                });
                check(raw.get("sku-7") == null && raw.get("inserted") != null, "immediate operations not restored by rollback");
                template.executeWithoutResult(status -> { configured.invalidate(); status.setRollbackOnly(); });
                check(raw.get("inserted") == null, "invalidate immediate despite rollback");
            }
        }
        System.out.println("controls=" + checks + " PASS");
    }
}
