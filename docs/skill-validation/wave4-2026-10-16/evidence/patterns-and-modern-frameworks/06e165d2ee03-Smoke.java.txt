package app;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.cache.concurrent.ConcurrentMapCacheManager;
public class Smoke {
    public static void main(String[] args) {
        try (var context = new AnnotationConfigApplicationContext(CacheConfiguration.class)) {
            var cache = context.getBean(ConcurrentMapCacheManager.class).getCache("catalog");
            cache.put("sku-7", "old projection");
            context.getBean(CheckoutFlow.class).submit("sku-7", false);
            if (cache.get("sku-7") != null) throw new AssertionError("commit did not invalidate");
            System.out.println("smoke=1 PASS");
        }
    }
}
