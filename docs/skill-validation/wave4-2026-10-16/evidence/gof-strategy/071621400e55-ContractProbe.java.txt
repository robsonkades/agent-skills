import java.io.Reader;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.Properties;

class ContractProbe {
    private static int checks;
    private static void equal(Object expected, Object actual, String label) {
        if (!expected.equals(actual)) throw new AssertionError(label + ": expected " + expected + ", got " + actual);
        checks++;
    }
    public static void main(String[] args) throws Exception {
        Properties properties = new Properties();
        try (Reader reader = Files.newBufferedReader(Path.of(args[0]))) {
            properties.load(reader);
        }
        ExportService service = Wiring.service(properties);
        equal("alpha\nbeta\n", service.export("TEXT", List.of("alpha", "beta")), "first TEXT");
        equal("GAMMA\n", service.export("UPPER", List.of("gamma")), "UPPER contract");
        equal("delta\n", service.export("TEXT", List.of("delta")), "independent second TEXT");
        equal("", service.export("TEXT", List.of()), "empty TEXT after prior jobs");
        equal("EPSILON\nZETA\n", service.export("UPPER", List.of("epsilon", "zeta")), "repeated UPPER");
        try {
            service.export("TXT", List.of("forbidden fallback"));
            throw new AssertionError("unknown key was accepted");
        } catch (IllegalArgumentException expected) {
            checks++;
        }
        equal("eta\n", service.export("TEXT", List.of("eta")), "after unknown-key failure");
        System.out.println("PASS " + checks + " caller-contract checks");
    }
}
