import java.util.Locale;
import java.util.Map;
import java.util.Properties;
import java.util.function.Supplier;
import java.util.function.UnaryOperator;

final class Wiring {
    static ExportService service(Properties properties) {
        boolean cache = Boolean.parseBoolean(properties.getProperty("encoding.cache.instances"));
        Supplier<Encoder> text = FactoryPolicy.build(
                () -> new LineEncoder(UnaryOperator.identity()), cache);
        Supplier<Encoder> upper = () -> new LineEncoder(row -> row.toUpperCase(Locale.ROOT));
        return new ExportService(Map.of("TEXT", text, "UPPER", upper));
    }
}
