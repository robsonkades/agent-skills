package com.acme.orders.domain;

import com.acme.orders.integration.LocalNotifier;
import java.util.Objects;

public final class OrderService {
    private final LocalNotifier notifier;

    public OrderService(LocalNotifier notifier) {
        this.notifier = Objects.requireNonNull(notifier);
    }

    public String confirm(String orderId) {
        if (orderId == null || orderId.isBlank()) {
            throw new IllegalArgumentException("orderId");
        }
        notifier.send("confirmed:" + orderId);
        return orderId;
    }
}
