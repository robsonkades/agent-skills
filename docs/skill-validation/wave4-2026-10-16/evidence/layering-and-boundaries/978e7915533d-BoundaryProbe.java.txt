import com.acme.orders.Bootstrap;
import com.acme.orders.domain.OrderService;
import com.acme.orders.integration.LocalNotifier;
import com.acme.orders.web.OrderController;
import java.time.Clock;
import java.time.Instant;
import java.time.ZoneOffset;
import java.util.List;

public final class BoundaryProbe {
    public static void main(String[] args) {
        if (Instant.now() == null || Clock.systemUTC().instant() == null) {
            throw new AssertionError("JDK wall clocks returned null");
        }
        System.out.println("PASS: JDK wall-clock access without framework classes");
        Instant deadline = Instant.parse("2026-10-16T12:00:00Z");
        check(!expired(deadline, Clock.fixed(deadline.minusNanos(1), ZoneOffset.UTC)),
                "before deadline");
        check(expired(deadline, Clock.fixed(deadline, ZoneOffset.UTC)), "at deadline");
        check(expired(deadline, Clock.fixed(deadline.plusNanos(1), ZoneOffset.UTC)),
                "after deadline");
        System.out.println("PASS: fixed clock before/at/after deadline without DI framework");

        LocalNotifier notifier = new LocalNotifier();
        OrderController controller = new OrderController(new OrderService(notifier));
        check(controller.confirm("order-17").equals("order-17"), "confirmation return");
        check(notifier.messages().equals(List.of("confirmed:order-17")), "notification");
        try {
            controller.confirm(" ");
            throw new AssertionError("invalid ID accepted");
        } catch (IllegalArgumentException expected) {
            check(notifier.messages().size() == 1, "invalid ID produced side effect");
        }
        check(Bootstrap.controller().confirm("boot-1").equals("boot-1"), "composition");
        System.out.println("PASS: confirmation, invalid-ID control and composition");
    }

    private static boolean expired(Instant deadline, Clock clock) {
        return !Instant.now(clock).isBefore(deadline);
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
