import java.util.ArrayList;
import java.util.List;

public final class PluginExample extends Runner {
    public final List<String> events = new ArrayList<>();

    @Override protected void loaded(String input) {
        events.add("loaded:" + input);
        if (input.equals("outer")) {
            String nested = run("inner");
            events.add("nested:" + nested);
        }
    }

    @Override protected String publish() {
        String value = super.publish();
        events.add("published:" + value);
        return value;
    }
}
