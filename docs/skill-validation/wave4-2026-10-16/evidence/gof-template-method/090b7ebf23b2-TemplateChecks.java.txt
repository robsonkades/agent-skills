import java.time.*;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

public class TemplateChecks {
    private static int checks;
    private static final LocalDate DATE = LocalDate.of(2026, 10, 16);
    private static void check(String name, boolean condition) {
        if (!condition) throw new AssertionError(name);
        checks++;
        System.out.println("PASS " + name);
    }
    static class NoSuppression extends RuntimeException {
        NoSuppression() { super("primary", null, false, true); }
    }
    static class Scenario implements SettlementSteps, AuditLog {
        final List<String> trace = new ArrayList<>();
        String failAt;
        Throwable primary;
        RuntimeException reporting;
        void step(String name) {
            trace.add(name);
            if (name.equals(failAt)) {
                if (primary instanceof Error error) throw error;
                throw (RuntimeException) primary;
            }
        }
        SettlementRun runner() { return new SettlementRun(this, this, Clock.fixed(Instant.EPOCH, ZoneOffset.UTC)); }
        public Batch load(RunContext context) { step("load"); return new Batch(context.date().toString()); }
        public ValidatedBatch validate(Batch batch, RunContext context) { step("validate"); return new ValidatedBatch(batch.data()); }
        public Results settle(ValidatedBatch batch, RunContext context) { step("settle"); return new Results(batch.data()); }
        public void record(Results results, RunContext context) { step("record"); }
        public void runStarted(RunContext context) { step("start"); }
        public void runFinished(RunContext context, RunReport report) { step("finish"); }
        public void runFailed(RunContext context, RuntimeException failure) {
            trace.add("failure");
            if (reporting != null) throw reporting;
        }
        Throwable failure() {
            try { runner().run(DATE, new RunId("logical-1"), new Deadline(10)); }
            catch (Throwable result) { return result; }
            return null;
        }
    }
    public static void main(String[] args) {
        var success = new Scenario();
        var report = success.runner().run(DATE, new RunId("logical-1"), new Deadline(10));
        check("successful fixed sequence and returned context", success.trace.equals(List.of("start", "load", "validate", "settle", "record", "finish")) && report.data().equals(DATE.toString()));
        var order = List.of("start", "load", "validate", "settle", "record");
        for (String stage : List.of("load", "validate", "settle", "record")) {
            var scenario = new Scenario();
            scenario.failAt = stage;
            scenario.primary = new IllegalStateException(stage);
            check(stage + " preserves primary identity", scenario.failure() == scenario.primary);
            var expected = new ArrayList<>(order.subList(0, order.indexOf(stage) + 1));
            expected.add("failure");
            check(stage + " skips success-only successors", scenario.trace.equals(expected));
        }
        var start = new Scenario(); start.failAt = "start"; start.primary = new IllegalStateException("start");
        check("start-audit failure has no failure-audit attempt", start.failure() == start.primary && start.trace.equals(List.of("start")));
        var finish = new Scenario(); finish.failAt = "finish"; finish.primary = new IllegalStateException("finish");
        check("finish-audit failure can follow recording", finish.failure() == finish.primary && finish.trace.equals(List.of("start", "load", "validate", "settle", "record", "finish", "failure")));
        var secondary = new Scenario(); secondary.failAt = "settle"; secondary.primary = new IllegalStateException("settle"); secondary.reporting = new IllegalArgumentException("audit");
        check("distinct audit failure is suppressed", secondary.failure() == secondary.primary && secondary.primary.getSuppressed().length == 1 && secondary.primary.getSuppressed()[0] == secondary.reporting);
        var same = new Scenario(); same.failAt = "record"; same.primary = new IllegalStateException("same"); same.reporting = (RuntimeException) same.primary;
        check("same failure is not self-suppressed", same.failure() == same.primary && same.primary.getSuppressed().length == 0);
        var disabled = new Scenario(); disabled.failAt = "settle"; disabled.primary = new NoSuppression(); disabled.reporting = new IllegalStateException("audit");
        check("disabled suppression does not retain secondary", disabled.failure() == disabled.primary && disabled.primary.getSuppressed().length == 0);
        var error = new Scenario(); error.failAt = "validate"; error.primary = new AssertionError("error");
        check("Error bypasses runtime-failure audit", error.failure() == error.primary && error.trace.equals(List.of("start", "load", "validate")));
        var missing = new Scenario();
        try { missing.runner().run(DATE, null, new Deadline(10)); throw new AssertionError("null run ID"); }
        catch (NullPointerException expected) { check("null logical ID rejected before audit", missing.trace.isEmpty()); }
        var input = new CompletableFuture<String>();
        var effects = new AtomicInteger();
        var continuation = input.thenApply(value -> { effects.incrementAndGet(); return value.length(); });
        check("dependent success effect waits for incomplete input", !continuation.isDone() && effects.get() == 0);
        input.complete("data");
        check("dependent success effect follows completion", continuation.join() == 4 && effects.get() == 1);
        var failed = new CompletableFuture<String>();
        var skipped = failed.thenRun(effects::incrementAndGet);
        failed.completeExceptionally(new IllegalStateException("async failure"));
        check("exceptional completion skips success effect", skipped.isCompletedExceptionally() && effects.get() == 1);
        Thread caller = Thread.currentThread();
        var inline = CompletableFuture.completedFuture("done").thenApply(v -> Thread.currentThread());
        check("completed CompletableFuture continuation observed inline", inline.join() == caller);
        System.out.println("checks=" + checks);
    }
}
