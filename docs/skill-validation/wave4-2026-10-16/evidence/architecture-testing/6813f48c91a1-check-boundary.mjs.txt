  const result = spawnSync(executable, args, { cwd, encoding: 'utf8', env, timeout: 30000 });
  const record = { name, executable, args, cwd, expectedExitCode: expected, exitCode: result.status, stdout: result.stdout ?? '', stderr: result.stderr ?? '', error: result.error?.message ?? null };

  writeFileSync(join(fixture, `${name}.json`), JSON.stringify(record, null, 2));
  assert.equal(result.error, undefined, name);
  assert.equal(result.status, expected, `${name}: ${result.stdout}\n${result.stderr}`);
  return record;
}
