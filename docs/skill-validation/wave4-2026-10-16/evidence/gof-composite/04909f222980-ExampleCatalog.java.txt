import java.util.List;

public final class ExampleCatalog {
    public static Node create() {
        Leaf sharedManual = new Leaf("manual", 6);
        Leaf independentManual = new Leaf("manual", 6);
        Branch local = new Branch("local", List.of(sharedManual));
        Branch partner = new Branch("partner", List.of(sharedManual, independentManual));
        return new Branch("catalog", List.of(local, partner));
    }
}
