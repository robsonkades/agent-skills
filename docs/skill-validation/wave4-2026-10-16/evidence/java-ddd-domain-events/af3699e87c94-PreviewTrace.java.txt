import java.time.Instant;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Queue;
import java.util.UUID;

public final class PreviewTrace {
    record QuoteAccepted(UUID eventId, UUID quoteId, Instant occurredOn) {}
    record QuoteRow(UUID quoteId, long totalCents) {}
    record Preview(UUID eventId, UUID quoteId, Instant occurredOn, long totalCents) {}

    static final class Quote {
        private final UUID id;
        private long totalCents;
        private boolean accepted;
        private final List<QuoteAccepted> pending = new ArrayList<>();
        Quote(UUID id, long totalCents) {
            if (totalCents < 0) throw new IllegalArgumentException("negative total");
            this.id = id;
            this.totalCents = totalCents;
        }
        void accept(UUID eventId, Instant time) {
            if (accepted) throw new IllegalStateException("already accepted");
            var event = new QuoteAccepted(eventId, id, time);
            accepted = true;
            pending.add(event);
        }
        void amendTotal(long value) {
            if (!accepted || value < 0) throw new IllegalStateException("invalid amendment");
            totalCents = value;
        }
        QuoteRow row() { return new QuoteRow(id, totalCents); }
        List<QuoteAccepted> pullEvents() {
            var batch = List.copyOf(pending);
            pending.clear();
            return batch;
        }
    }

    static final class MemoryRepository {
        private final Map<UUID, QuoteRow> rows = new HashMap<>();
        void save(QuoteRow row) { rows.put(row.quoteId(), row); }
        QuoteRow find(UUID id) { return rows.get(id); }
    }

    static final class PreviewConsumer {
        private final MemoryRepository repository;
        PreviewConsumer(MemoryRepository repository) { this.repository = repository; }
        Preview on(QuoteAccepted event) {
            var row = repository.find(event.quoteId());
            return new Preview(event.eventId(), event.quoteId(), event.occurredOn(), row.totalCents());
        }
    }

    public static void main(String[] args) {
        var quoteId = new UUID(0, 17);
        var eventId = new UUID(0, 41);
        var occurredOn = Instant.parse("2026-10-16T10:00:00Z");
        var repository = new MemoryRepository();
        Queue<QuoteAccepted> localQueue = new ArrayDeque<>();
        var quote = new Quote(quoteId, 10000);
        quote.accept(eventId, occurredOn);
        repository.save(quote.row());
        localQueue.addAll(quote.pullEvents());
        System.out.println("after acceptance row=" + repository.find(quoteId));
        quote.amendTotal(12500);
        repository.save(quote.row());
        System.out.println("after amendment row=" + repository.find(quoteId));
        var preview = new PreviewConsumer(repository).on(localQueue.remove());
        System.out.println("preview=" + preview);
    }
}
