import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import { fileURLToPath } from 'node:url';
import path from 'node:path';
const dir = path.dirname(fileURLToPath(import.meta.url));
const inputs = JSON.parse(await readFile(path.join(dir, 'case-inputs.json'), 'utf8'));
const criteria = JSON.parse(await readFile(path.join(dir, 'case-criteria.json'), 'utf8'));
let checks = 0;
const check = (value, message) => { assert.ok(value, message); checks++; console.log('PASS ' + message); };
check(inputs.length === 2, 'two separate case inputs');
check(criteria.cases.length === 2, 'two separately stored evaluator criteria');
check(inputs[0].task === inputs[1].task, 'same actor task');
check(inputs[0].files['PreviewTrace.java'] === inputs[1].files['PreviewTrace.java'], 'same ordinary code fixture');
check(inputs[0].files['README.md'] === inputs[1].files['README.md'], 'same runtime and fixture limitations');
const lines = inputs.map(c => c.files['preview-contract.md'].split('\n'));
check(lines[0].length === lines[1].length, 'contract sizes comparable');
const differences = lines[0].map((line, i) => line === lines[1][i] ? null : i).filter(i => i !== null);
check(differences.length === 1, 'exactly one decisive contract sentence differs');
for (const c of inputs) {
  check(!('expected' in c) && !('criteria' in c), c.id + ': no evaluator answers in task schema');
  check((await readFile(path.join(c.artifactRoot, 'PreviewTrace.java'), 'utf8')) === c.files['PreviewTrace.java'], c.id + ': retained code matches input');
}
console.log('PASS ' + checks + ' input-integrity checks; no actor run.');
