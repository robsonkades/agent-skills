package com.example.domain.order;
import com.example.domain.order.event.OrderSubmitted;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.Set;
import java.util.UUID;

// Fixture supplies the omitted base classes and collection owner.
// Only submit() is extracted unchanged from the skill's partial snippet.
public final class Order {
    private final OrderID id;
    private final List<String> lines;
    private OrderStatus status;
    private final List<OrderSubmitted> pending = new ArrayList<>();

    private Order(OrderID id, OrderStatus status, List<String> lines) {
        this.id = id;
        this.status = status;
        this.lines = List.copyOf(lines);
    }
    public static Order draft(OrderID id, List<String> lines) {
        return new Order(id, OrderStatus.DRAFT, lines);
    }
    public static Order restore(OrderID id, OrderStatus status, List<String> lines) {
        return new Order(id, status, lines);
    }
    public OrderID getId() { return id; }
    public OrderStatus status() { return status; }
    private void registerEvent(OrderSubmitted event) { pending.add(event); }
    public List<OrderSubmitted> pendingEvents() { return List.copyOf(pending); }
    public void acknowledge(Set<UUID> eventIds) {
        pending.removeIf(event -> eventIds.contains(event.eventId()));
    }
    // Deliberate fixture business contract: reopen permits another submission.
    public void reopen() {
        if (status != OrderStatus.SUBMITTED) throw new DomainException("not submitted");
        status = OrderStatus.DRAFT;
    }

    public void submit(final UUID eventId, final Instant occurredOn) {
        if (status != OrderStatus.DRAFT || lines.isEmpty()) {
            throw new DomainException("only a non-empty draft can be submitted");
        }
        final var submitted = new OrderSubmitted(eventId, getId(), occurredOn);
        status = OrderStatus.SUBMITTED;
        registerEvent(submitted);
    }
}
