import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

public final class ContinuationControls {
    static int checks;
    static void check(boolean ok, String label) {
        checks++;
        if (!ok) throw new AssertionError(label);
    }
    record Request(String id) {}
    record Response(String body) {}
    interface Next { Response handle(Request request); }
    interface Stage { Response handle(Request request, Next next); }
    static Next compose(List<Stage> stages, Next terminal) {
        Next next = terminal;
        for (int i = stages.size() - 1; i >= 0; i--) {
            Stage stage = stages.get(i);
            Next remainder = next;
            next = request -> stage.handle(request, remainder);
        }
        return next;
    }
    static final class Lease implements AutoCloseable {
        final String name;
        final List<String> trace;
        final RuntimeException closeFailure;
        boolean closed;
        Lease(String name, List<String> trace) { this(name, trace, null); }
        Lease(String name, List<String> trace, RuntimeException closeFailure) {
            this.name = name; this.trace = trace; this.closeFailure = closeFailure;
            trace.add(name + ".enter");
        }
        String read() {
            if (closed) throw new IllegalStateException("read after close");
            trace.add(name + ".read");
            return "ok";
        }
        public void close() {
            check(!closed, "close exactly once");
            closed = true;
            trace.add(name + ".cleanup");
            if (closeFailure != null) throw closeFailure;
        }
    }
    static Stage stage(String name, List<String> trace, String action) {
        return (request, next) -> {
            try (var lease = new Lease(name, trace)) {
                if (action.equals("reject")) return new Response("rejected");
                if (action.equals("throw")) throw new IllegalArgumentException("stage failure");
                return next.handle(request);
            }
        };
    }
    static final class QueueExecutor implements Executor {
        final Queue<Runnable> tasks = new ArrayDeque<>();
        public void execute(Runnable task) { tasks.add(task); }
        void drain() { while (!tasks.isEmpty()) tasks.remove().run(); }
    }
    static void asyncCancellation(boolean closeOnResult) {
        var executor = new QueueExecutor();
        var trace = new ArrayList<String>();
        var lease = new Lease("lease", trace);
        var result = new CompletableFuture<String>();
        var workDone = new CompletableFuture<Void>();
        var workFailure = new AtomicReference<Throwable>();
        if (closeOnResult) result.whenComplete((value, failure) -> lease.close());
        else workDone.whenComplete((value, failure) -> lease.close());
        executor.execute(() -> {
            try { result.complete(lease.read()); }
            catch (Throwable failure) { workFailure.set(failure); result.completeExceptionally(failure); }
            finally { workDone.complete(null); }
        });
        check(result.cancel(true), "result cancellation accepted");
        check(!workDone.isDone(), "result cancellation does not finish queued computation");
        executor.drain();
        check(workDone.isDone() && lease.closed, "work and cleanup finish");
        if (closeOnResult) {
            check(workFailure.get() instanceof IllegalStateException, "negative control exposes early release");
            check(trace.equals(List.of("lease.enter", "lease.cleanup")), "cancel callback closed before read");
        } else {
            check(workFailure.get() == null, "work-owned completion keeps resource alive");
            check(trace.equals(List.of("lease.enter", "lease.read", "lease.cleanup")), "read precedes work completion cleanup");
        }
    }
    public static void main(String[] args) {
        var request = new Request("r1");
        var trace = new ArrayList<String>();
        var calls = new AtomicInteger();
        Next terminal = r -> { calls.incrementAndGet(); trace.add("terminal"); return new Response("ok"); };
        var chain = compose(List.of(stage("A", trace, "forward"), stage("B", trace, "forward")), terminal);
        check(chain.handle(request).body().equals("ok"), "result preserved");
        check(calls.get() == 1 && trace.equals(List.of("A.enter", "B.enter", "terminal", "B.cleanup", "A.cleanup")), "nested success order");

        trace.clear(); calls.set(0);
        var rejecting = compose(List.of(stage("A", trace, "forward"), stage("B", trace, "reject")), terminal);
        check(rejecting.handle(request).body().equals("rejected"), "rejection preserved");
        check(calls.get() == 0 && trace.equals(List.of("A.enter", "B.enter", "B.cleanup", "A.cleanup")), "zero forwarding cleanup");
        trace.clear();
        var throwing = compose(List.of(stage("A", trace, "forward"), stage("B", trace, "throw")), terminal);
        try { throwing.handle(request); throw new AssertionError("missing stage failure"); }
        catch (IllegalArgumentException expected) { check(expected.getMessage().equals("stage failure"), "stage failure preserved"); }
        check(calls.get() == 0 && trace.equals(List.of("A.enter", "B.enter", "B.cleanup", "A.cleanup")), "throw before next cleanup");

        trace.clear(); calls.set(0);
        var primary = new IllegalStateException("terminal failure");
        var closeFailure = new IllegalArgumentException("close failure");
        Stage closeThrows = (r, next) -> {
            try (var lease = new Lease("B", trace, closeFailure)) { return next.handle(r); }
        };
        var failureChain = compose(List.of(stage("A", trace, "forward"), closeThrows), r -> {
            calls.incrementAndGet(); trace.add("terminal"); throw primary;
        });
        try { failureChain.handle(request); throw new AssertionError("missing terminal failure"); }
        catch (IllegalStateException expected) {
            check(expected == primary, "primary exception retained by try-with-resources");
            check(Arrays.equals(expected.getSuppressed(), new Throwable[] { closeFailure }), "cleanup failure suppressed");
        }
        check(calls.get() == 1 && trace.equals(List.of("A.enter", "B.enter", "terminal", "B.cleanup", "A.cleanup")), "failure cleanup order");

        trace.clear(); calls.set(0);
        Stage duplicate = (r, next) -> { next.handle(r); return next.handle(r); };
        compose(List.of(duplicate), terminal).handle(request);
        check(calls.get() == 2, "negative control: a generic continuation does not enforce one call");

        asyncCancellation(true);
        asyncCancellation(false);
        System.out.println("PASS " + checks + " continuation controls: zero/one/two forwarding, success/failure cleanup, suppressed close error, cancellation versus actual work completion");
    }
}
