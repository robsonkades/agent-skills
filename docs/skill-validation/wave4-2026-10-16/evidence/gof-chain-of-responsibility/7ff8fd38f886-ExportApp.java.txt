import java.nio.file.*;
import java.util.*;
import java.util.concurrent.*;

public final class ExportApp {
    interface Next { CompletionStage<String> handle(Session session); }
    static final class Session implements AutoCloseable {
        private final List<String> trace;
        private boolean closed;
        Session(List<String> trace) { this.trace = trace; trace.add("open"); }
        String read() {
            if (closed) throw new IllegalStateException("session closed");
            trace.add("read");
            return "report";
        }
        public void close() {
            if (closed) throw new IllegalStateException("double close");
            closed = true;
            trace.add("close");
        }
    }
    static final class SessionStage {
        private final List<String> trace;
        SessionStage(List<String> trace) { this.trace = trace; }
        CompletionStage<String> handle(Next next) {
            try (var session = new Session(trace)) {
                return next.handle(session);
            }
        }
    }
    // Deterministic adapter for the host's single-thread task queue.
    static final class TaskQueue implements Executor {
        private final Queue<Runnable> tasks = new ArrayDeque<>();
        public void execute(Runnable task) { tasks.add(task); }
        void drain() { while (!tasks.isEmpty()) tasks.remove().run(); }
    }
    static Executor executor(Properties config, TaskQueue taskQueue) {
        return switch (config.getProperty("execution.mode")) {
            case "inline" -> Runnable::run;
            case "queued" -> taskQueue;
            default -> throw new IllegalArgumentException("unsupported execution mode");
        };
    }
    public static void main(String[] args) throws Exception {
        var config = new Properties();
        try (var reader = Files.newBufferedReader(Path.of(args[0]))) { config.load(reader); }
        var queue = new TaskQueue();
        var worker = executor(config, queue);
        var trace = new ArrayList<String>();
        Next render = session -> CompletableFuture.supplyAsync(session::read, worker).minimalCompletionStage();
        var response = new SessionStage(trace).handle(render);
        queue.drain();
        try { System.out.println(response.toCompletableFuture().join()); }
        catch (CompletionException failure) {
            System.out.println("FAILED: " + failure.getCause().getClass().getSimpleName() + ": " + failure.getCause().getMessage());
        }
        System.out.println(trace);
    }
}
