import java.util.*;
import java.util.function.Function;

public final class RuleControls {
    static int checks;
    static void check(boolean ok, String message) {
        checks++;
        if (!ok) throw new AssertionError(message);
    }
    static <T extends Throwable> T expect(Class<T> type, Runnable action) {
        try { action.run(); } catch (Throwable failure) {
            check(type.isInstance(failure), "wrong failure: " + failure);
            return type.cast(failure);
        }
        throw new AssertionError("expected " + type.getSimpleName());
    }
    static AuthorisationRule rule(String name, List<String> trace, Function<Payment, Optional<Decision>> body) {
        return new AuthorisationRule() {
            public String name() { return name; }
            public Optional<Decision> apply(Payment payment) { trace.add(name); return body.apply(payment); }
        };
    }
    public static void main(String[] args) {
        var request = new Payment("p1", "product");
        var decline = new Decision(Outcome.DECLINE, "denied", List.of());
        var approve = new Decision(Outcome.APPROVE, "", List.of());
        var trace = new ArrayList<String>();
        var ordered = new ArrayList<AuthorisationRule>(List.of(
            rule("abstain", trace, p -> Optional.empty()),
            rule("deny", trace, p -> Optional.of(decline)),
            rule("allow", trace, p -> Optional.of(approve))));
        var chain = new AuthorisationRules(ordered);
        ordered.clear();
        check(chain.decide(request) == decline, "present rejection must win");
        check(trace.equals(List.of("abstain", "deny")), "no invocation after first match; frozen membership");
        trace.clear();
        var streamRules = List.of(
            rule("deny", trace, p -> Optional.of(decline)),
            rule("allow", trace, p -> Optional.of(approve)));
        var first = streamRules.stream().flatMap(h -> h.apply(request).stream()).findFirst();
        check(first.orElseThrow() == decline && trace.equals(List.of("deny")), "sequential stream first decision");

        trace.clear();
        var absent = new AuthorisationRules(List.of(rule("empty", trace, p -> Optional.empty())));
        var unhandled = expect(NoAuthorisationRule.class, () -> absent.decide(request));
        check(unhandled.paymentId.equals("p1") && unhandled.productCode.equals("product"), "unhandled diagnostics fixture");
        check(trace.equals(List.of("empty")), "one abstention followed by unhandled");
        expect(NoAuthorisationRule.class, () -> new AuthorisationRules(List.of()).decide(request));

        trace.clear();
        var primary = new IllegalStateException("handler failure");
        var failing = new AuthorisationRules(List.of(
            rule("failure", trace, p -> { throw primary; }),
            rule("allow", trace, p -> Optional.of(approve))));
        check(expect(IllegalStateException.class, () -> failing.decide(request)) == primary, "preserve handler failure");
        check(trace.equals(List.of("failure")), "failure must not become abstention");

        trace.clear();
        var invalid = new AuthorisationRules(List.of(
            rule("null", trace, p -> null),
            rule("allow", trace, p -> Optional.of(approve))));
        expect(NullPointerException.class, () -> invalid.decide(request));
        check(trace.equals(List.of("null")), "null Optional is contract failure");
        trace.clear();
        expect(NullPointerException.class, () -> chain.decide(null));
        check(trace.isEmpty(), "null input rejected before handler");
        expect(NullPointerException.class, () -> new AuthorisationRules(null));
        expect(NullPointerException.class, () -> new AuthorisationRules(Arrays.asList((AuthorisationRule) null)));

        var event = new MutableEvent("before");
        var events = new ArrayList<DomainEvent>(List.of(event));
        var result = new Decision(Outcome.REVIEW, "review", events);
        events.clear();
        check(result.events().size() == 1, "record freezes list membership");
        expect(UnsupportedOperationException.class, () -> result.events().clear());
        event.value = "after";
        check(((MutableEvent) result.events().get(0)).value.equals("after"), "copy does not deep-freeze event payload");

        System.out.println("PASS " + checks + " rule controls: first denial, abstention, unhandled, failure propagation, null contract, membership ownership, shallow payload limit");
    }
}
record Payment(String id, String productCode) {}
enum Outcome { APPROVE, REVIEW, DECLINE }
interface DomainEvent {}
final class MutableEvent implements DomainEvent {
    String value;
    MutableEvent(String value) { this.value = value; }
}
final class NoAuthorisationRule extends RuntimeException {
    final String paymentId;
    final String productCode;
    NoAuthorisationRule(String paymentId, String productCode) {
        super(paymentId + ":" + productCode);
        this.paymentId = paymentId;
        this.productCode = productCode;
    }
}
