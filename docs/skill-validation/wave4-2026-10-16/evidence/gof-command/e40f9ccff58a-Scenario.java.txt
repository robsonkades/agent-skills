import java.io.Reader;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Objects;
import java.util.Properties;

public final class Scenario {
    private static void check(boolean condition, String label) {
        if (!condition) throw new AssertionError(label);
    }
    public static void main(String[] args) throws Exception {
        Properties contract = new Properties();
        try (Reader in = Files.newBufferedReader(Path.of(args[0]))) { contract.load(in); }
        String selection = contract.getProperty("target.selection");
        check(selection.equals("creation") || selection.equals("execution"), "known target contract");
        Editor.Document a = new Editor.Document("draft-A");
        Editor.Document b = new Editor.Document("draft-B");
        Editor ui = new Editor(a);
        Runnable queued = ui.queueSave();
        a.label = "revised-A";
        b.label = "current-B";
        ui.active = b;
        queued.run();
        Editor.Document expected = selection.equals("creation") ? a : b;
        Editor.Document other = selection.equals("creation") ? b : a;
        check(expected.saves == 1, "chosen document saved exactly once");
        check(other.saves == 0, "other document left unchanged");
        check(Objects.equals(expected.lastSavedLabel, expected.label), "save uses live state at execution");
        System.out.println("PASS target=" + selection + "; chosen receiver, untouched other receiver, live label");
    }
}
