import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

public class ReceiverTimingProbe {
    private static int checks;
    private static final List<String> calls = new ArrayList<>();

    static final class Order {
        final String id;
        String state;
        Order(String id, String state) { this.id = id; this.state = state; }
        void cancel() { calls.add(id + ":" + state); }
    }

    static final class UI {
        Order order;
        UI(Order order) { this.order = order; }
        Runnable bindNow() {
            Runnable selectedNow = this.order::cancel;
            return selectedNow;
        }
        Runnable selectWhenRun() {
            Runnable selectedWhenRun = () -> this.order.cancel();
            return selectedWhenRun;
        }
        Runnable bindLocal() {
            Order selected = this.order;
            return () -> selected.cancel();
        }
    }

    static void equal(Object expected, Object actual, String label) {
        if (!Objects.equals(expected, actual))
            throw new AssertionError(label + ": expected=" + expected + ", actual=" + actual);
        checks++;
    }
    static void nullFailure(Runnable action, String label) {
        try { action.run(); }
        catch (NullPointerException expected) { checks++; return; }
        throw new AssertionError(label + ": missing NullPointerException");
    }
    static String scenario(boolean boundNow) {
        calls.clear();
        Order a = new Order("A", "draft");
        Order b = new Order("B", "draft");
        UI ui = new UI(a);
        Runnable action = boundNow ? ui.bindNow() : ui.selectWhenRun();
        a.state = "revised";
        b.state = "current";
        ui.order = b;
        action.run();
        return calls.get(0);
    }
    public static void main(String[] args) {
        Order a = new Order("A", "draft");
        Order b = new Order("B", "current");
        UI ui = new UI(a);
        Runnable now = ui.bindNow();
        Runnable later = ui.selectWhenRun();
        Runnable local = ui.bindLocal();
        equal(0, calls.size(), "creation executes no effect");
        a.state = "revised";
        ui.order = b;
        now.run();
        equal(List.of("A:revised"), List.copyOf(calls), "binding preserves identity but observes later state");
        later.run();
        equal(List.of("A:revised", "B:current"), List.copyOf(calls), "field selection happens on invocation");
        local.run();
        equal("A:revised", calls.get(2), "effectively final local keeps its receiver");
        ui.order = null;
        now.run();
        equal("A:revised", calls.get(3), "bound target survives field replacement with null");
        nullFailure(later, "later field dereference sees null");
        equal(4, calls.size(), "failed dereference adds no effect");
        now.run();
        equal(5, calls.size(), "repeating callback repeats invocation");
        UI empty = new UI(null);
        nullFailure(() -> empty.bindNow(), "bound null fails at creation");
        Runnable initiallyEmpty = empty.selectWhenRun();
        equal(5, calls.size(), "lambda creation with null field does not invoke body");
        empty.order = b;
        initiallyEmpty.run();
        equal("B:current", calls.get(5), "later field population is visible under confinement");
        empty.order = null;
        nullFailure(initiallyEmpty, "null fails when lambda body runs");
        equal("A:revised", scenario(true), "creation-target contract accepts bound callback");
        equal("B:current", scenario(false), "execution-target contract accepts field callback");
        equal(false, "A:revised".equals(scenario(false)), "field callback rejected by creation-target contract");
        equal(false, "B:current".equals(scenario(true)), "bound callback rejected by execution-target contract");
        System.out.println("PASS " + checks + " assertions; two valid contracts and two rejected cross-contract substitutions.");
    }
}
