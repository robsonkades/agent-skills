import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, mkdirSync, readdirSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
const root=dirname(fileURLToPath(import.meta.url));
const source='C:/git/agent-skills/skills/service-layer-design/references/anaemic-and-god-services.md';
const body=readFileSync(source,'utf8');
const extracted=[...body.matchAll(/```bash\r?\n([\s\S]*?)```/g)][0][1];
assert.equal(readFileSync(join(root,'history-example.sh'),'utf8').trimEnd(),extracted.trimEnd());
const posix=p=>p.replaceAll('\\','/').replace(/^([A-Za-z]):/,(_,drive)=>'/'+drive.toLowerCase());
const bash='C:/Program Files/Git/bin/bash.exe';
const commands=[];
function run(id,args,expectedExit,env={}) {
 const result=spawnSync(bash,args,{cwd:root,encoding:'utf8',env:{...process.env,...env},timeout:10000});
 const stdoutPath=join(root,id+'.stdout.txt'),stderrPath=join(root,id+'.stderr.txt');
 writeFileSync(stdoutPath,result.stdout??'');writeFileSync(stderrPath,result.stderr??'');
 const row={id,executable:bash,args,cwd:root,expectedExit,actualExit:result.status,signal:result.signal,error:result.error?String(result.error):null,stdoutPath,stderrPath};
 commands.push(row);writeFileSync(join(root,'history-commands.json'),JSON.stringify(commands,null,2)+'\n');
 assert.equal(result.status,expectedExit,id);
 return result;
}
run('01-syntax',['--noprofile','--norc','-n',posix(join(root,'history-example.sh'))],0);
const cases=[
 {id:'02-numeric-binary',stdout:'commit abc123 2026-10-16 update\n12\t3\tsrc/main/OrderService.java\n-\t-\tsrc/binary/OrderService.java\n2\t1\tsrc/other/OrderService.java\n',stderr:'',status:0},
 {id:'03-empty',stdout:'',stderr:'warning: controlled empty selection\n',status:0},
 {id:'04-failed-producer',stdout:'commit def456 2026-10-16 partial\n999\t0\tsrc/main/OrderService.java\n',stderr:'fatal: controlled producer failure\n',status:23}
];
for (const item of cases) {
 const dir=join(root,item.id);mkdirSync(dir,{recursive:true});
 writeFileSync(join(dir,'producer.out'),item.stdout);writeFileSync(join(dir,'producer.err'),item.stderr);
 const result=run(item.id,['--noprofile','--norc',posix(join(root,'history-wrapper.sh'))],item.status,{
 TMPDIR:posix(dir),SERVICE_TEST_ARGUMENTS:posix(join(dir,'argv.bin')),SERVICE_TEST_STDOUT:posix(join(dir,'producer.out')),
 SERVICE_TEST_STDERR:posix(join(dir,'producer.err')),SERVICE_TEST_STATUS:String(item.status),
 SERVICE_TEST_SCRIPT:posix(join(root,'history-example.sh'))
 });
 assert.deepEqual(readFileSync(join(dir,'argv.bin'),'utf8').split('\0').slice(0,-1),
 ['log','--format=commit %h %ad %s','--date=short','--numstat','--',':(glob)src/**/OrderService.java']);
 const captures=readdirSync(dir).filter(name=>name.startsWith('service-history.'));
 assert.equal(captures.length,1);
 assert.equal(readFileSync(join(dir,captures[0],'history.out'),'utf8'),item.stdout);
 assert.equal(readFileSync(join(dir,captures[0],'history.err'),'utf8'),item.stderr);
 if(item.id==='02-numeric-binary') assert.match(result.stdout,/Numeric additions: 14 across 2 rows; binary rows: 1/);
 if(item.id==='03-empty') { assert.match(result.stdout,/No matching history/); assert.doesNotMatch(result.stdout,/Numeric additions/); }
 if(item.status!==0) { assert.match(result.stderr,/Git failed \(status 23\); partial stdout is not evidence/); assert.doesNotMatch(result.stdout,/Numeric additions/); }
}
const summary={source,sourceSHA256:createHash('sha256').update(body).digest('hex'),extraction:'Bash fence unchanged except final trailing whitespace normalization.',executedCommands:commands.length,expectedProducerFailure:23,passed:['syntax','numeric plus binary rows','empty output is missing evidence','failed producer rejects partial stdout','literal quoted Git pathspec','raw stdout/stderr retained for every case'],limits:['Controlled git shell function, not real history/classification.','No Git state mutated.','No actual Java/provider/database or agent evaluation.']};
writeFileSync(join(root,'history-results.json'),JSON.stringify(summary,null,2)+'\n');
console.log(JSON.stringify(summary,null,2));
