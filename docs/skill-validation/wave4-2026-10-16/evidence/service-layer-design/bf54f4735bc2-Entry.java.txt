// Partial source; Endpoint and Entry are not advised, outer uses the same manager.
class Entry {
    private final SubmissionEndpoint endpoint;
    private final TransactionTemplate outer;
    private final Consumer<Receipt> replies;
    public void handle(SubmitCommand command, boolean outerTransaction) {
        Runnable work = () -> replies.accept(endpoint.submit(command));
        if (outerTransaction) {
            outer.executeWithoutResult(status -> work.run());
        } else {
            work.run();
        }
    }
}
