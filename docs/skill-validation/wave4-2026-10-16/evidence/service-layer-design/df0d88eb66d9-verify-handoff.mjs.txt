import assert from 'node:assert/strict';
import { readFileSync, statSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';

const root = dirname(fileURLToPath(import.meta.url));
const handoffPath = resolve(root, '../service-layer-design.json');
const raw = readFileSync(handoffPath);
const handoff = JSON.parse(raw);
assert.equal(handoff.skill, 'service-layer-design');
assert.equal(handoff.status, 'NO_CHANGES_NEEDED');
assert.equal(handoff.versions.final, '1.2.4');
assert.equal(handoff.allWritesStopped, true);
assert.deepEqual(handoff.activeWritingProcesses, []);
function checkFile(item) {
  const data = readFileSync(item.path);
  assert.equal(statSync(item.path).size, item.bytes, item.path);
  assert.equal(createHash('sha256').update(data).digest('hex').toUpperCase(), item.sha256.toUpperCase(), item.path);
}
handoff.artifacts.forEach(checkFile);
handoff.baseline.files.forEach(checkFile);
const checks = [
  ['status', '--porcelain=v1', '--untracked-files=all', '--', 'skills/service-layer-design'],
  ['diff', '--check', '--', 'skills/service-layer-design'],
  ['diff', '--numstat', '--', 'skills/service-layer-design'],
  ['diff', '--cached', '--numstat', '--', 'skills/service-layer-design'],
  ['ls-files', '--others', '--ignored', '--exclude-standard', '--', 'skills/service-layer-design']
].map(args => {
  const result = spawnSync('git', args, { cwd: 'C:/git/agent-skills', encoding: 'utf8', windowsHide: true });
  assert.equal(result.status, 0, JSON.stringify({ args, stderr: result.stderr }));
  assert.equal(result.stdout, '', JSON.stringify({ args, stdout: result.stdout }));
  return { executable: 'git', args, cwd: 'C:/git/agent-skills', exitCode: result.status, stdout: result.stdout, stderr: result.stderr };
});
console.log(JSON.stringify({
  handoff: handoffPath,
  sha256: createHash('sha256').update(raw).digest('hex').toUpperCase(),
  artifactCount: handoff.artifacts.length, ownedFiles: handoff.baseline.files.length,
  checks, allWritesStopped: handoff.allWritesStopped
}, null, 2));
