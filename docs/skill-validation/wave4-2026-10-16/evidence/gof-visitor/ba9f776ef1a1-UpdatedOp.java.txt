public final class UpdatedOp implements Model.Visitor<Integer> {
    @Override public Integer visitText(Model.Text text) { return text.value().length(); }
    @Override public Integer visitCode(Model.Code code) { return code.value().length(); }
    public static void main(String[] args) {
        Model.Visitor<Integer> visitor = new UpdatedOp();
        int oldValue = new Model.Text("four").accept(visitor);
        int newValue = new Model.Code("example").accept(visitor);
        if (oldValue != 4 || newValue != 7) throw new AssertionError("explicit handling failed");
        System.out.println("EXPLICIT_HANDLING: OLD_PATH=" + oldValue + "; NEW_PATH=" + newValue);
    }
}
