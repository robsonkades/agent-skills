public final class App {
    public static void main(String[] args) {
        Delivery delivery = Assembly.completion();
        int[] views = {0};
        CaptureCoordinator coordinator = new CaptureCoordinator(
                Runnable::run, delivery, () -> views[0]++);
        coordinator.request();
        delivery.drain();
        System.out.println("phase=" + coordinator.phase() + " views=" + views[0]);
    }
}
