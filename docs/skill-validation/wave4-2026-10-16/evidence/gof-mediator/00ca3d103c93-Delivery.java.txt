import java.util.ArrayDeque;
import java.util.Deque;
import java.util.concurrent.Executor;
import java.util.concurrent.RejectedExecutionException;

public interface Delivery extends Executor {
    void drain();

    final class Inline implements Delivery {
        public void execute(Runnable task) { task.run(); }
        public void drain() {}
    }
    final class Queued implements Delivery {
        private final Deque<Runnable> pending = new ArrayDeque<>();
        private boolean draining;
        public void execute(Runnable task) {
            if (pending.size() >= 8) throw new RejectedExecutionException("completion capacity");
            pending.addLast(task);
        }
        public void drain() {
            if (draining) return;
            draining = true;
            try {
                while (!pending.isEmpty()) pending.removeFirst().run();
            } finally {
                draining = false;
            }
        }
    }
}
