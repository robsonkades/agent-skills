import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.Executor;
import java.util.concurrent.ExecutorCompletionService;
import java.util.concurrent.Future;
import java.util.concurrent.TimeUnit;

public final class ProtocolStateCheck {
    static int checks;
    static void check(boolean value, String label) {
        if (!value) throw new AssertionError(label);
        checks++;
        System.out.println("PASS " + label);
    }
    static final class ManualExecutor implements Executor {
        final List<Runnable> tasks = new ArrayList<>();
        public void execute(Runnable task) { tasks.add(task); }
    }
    public static void main(String[] args) throws Exception {
        FulfilmentState initial = FulfilmentState.initial();
        check(!initial.readyToPack(), "initial-not-ready");
        check(!initial.withStockReserved().readyToPack(), "stock-alone-not-ready");
        check(!initial.withPaymentSettled().readyToPack(), "payment-alone-not-ready");
        var ready = initial.withStockReserved().withPaymentSettled();
        check(ready.readyToPack(), "stock-then-payment-ready");
        check(initial.withPaymentSettled().withStockReserved().readyToPack(), "payment-then-stock-ready");
        var claimed = ready.withPackingRequested();
        check(!claimed.readyToPack(), "claimed-not-ready");
        check(!claimed.withStockReserved().withPaymentSettled().readyToPack(), "duplicates-do-not-reclaim");
        check(!new FulfilmentState(true, true, true, false).readyToPack(), "cancelled-not-ready");
        check(!initial.stockReserved() && !initial.paymentSettled(), "initial-state-unchanged");
        ManualExecutor executor = new ManualExecutor();
        ExecutorCompletionService<String> completion = new ExecutorCompletionService<>(executor);
        completion.submit(() -> "first");
        completion.submit(() -> "second");
        executor.tasks.get(1).run();
        executor.tasks.get(0).run();
        Future<String> firstCompleted = completion.poll();
        Future<String> secondCompleted = completion.poll();
        check(firstCompleted != null && firstCompleted.get(1, TimeUnit.SECONDS).equals("second"),
                "completed-second-submission-observed-first");
        check(secondCompleted != null && secondCompleted.get(1, TimeUnit.SECONDS).equals("first"),
                "completed-first-submission-observed-second");
        check(completion.poll() == null, "completion-queue-drained");
        System.out.println("SUMMARY checks=" + checks);
    }
}
