import java.util.concurrent.Executor;

public final class CaptureCoordinator {
    public enum Phase { IDLE, RUNNING, DONE, UNKNOWN }
    @FunctionalInterface public interface Driver { void capture(Runnable captured); }
    @FunctionalInterface public interface View { void captured(); }

    private final Driver driver;
    private final Executor completion;
    private final View view;
    private Phase phase = Phase.IDLE;

    public CaptureCoordinator(Driver driver, Executor completion, View view) {
        this.driver = driver;
        this.completion = completion;
        this.view = view;
    }
    public Phase phase() { return phase; }
    public void request() {
        if (phase != Phase.IDLE) return;
        try {
            driver.capture(() -> completion.execute(this::captured));
            phase = Phase.RUNNING;
        } catch (RuntimeException failure) {
            phase = Phase.UNKNOWN;
            throw failure;
        }
    }
    private void captured() {
        phase = Phase.DONE;
        view.captured();
    }
}
