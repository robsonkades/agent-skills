import { writeFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
const urls = [
  'https://jakarta.ee/specifications/servlet/6.0/apidocs/jakarta.servlet/jakarta/servlet/filter',
  'https://jakarta.ee/specifications/servlet/6.0/apidocs/jakarta.servlet/jakarta/servlet/Filter.html',
];
const results = await Promise.all(urls.map(async (url) => {
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(15000) });
    const body = await response.text();
    return { url, status: response.status, finalUrl: response.url, title: body.match(/<title>(.*?)<\/title>/s)?.[1] ?? null };
  } catch (error) {
    return { url, error: error.message };
  }
}));
writeFileSync(fileURLToPath(new URL('servlet-link-results.json', import.meta.url)),
  JSON.stringify({ checkedAt: new Date().toISOString(), results }, null, 2) + '\n');
process.stdout.write(JSON.stringify(results, null, 2) + '\n');
