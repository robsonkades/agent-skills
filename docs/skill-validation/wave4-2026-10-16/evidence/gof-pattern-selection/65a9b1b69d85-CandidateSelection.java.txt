import java.util.List;

public final class CandidateSelection {
    @FunctionalInterface
    public interface Rule {
        boolean permits(String customer);
    }

    public static boolean permits(List<Rule> rules, String customer) {
        for (Rule rule : rules) {
            if (rule.permits(customer)) {
                return true;
            }
        }
        return false;
    }
}
