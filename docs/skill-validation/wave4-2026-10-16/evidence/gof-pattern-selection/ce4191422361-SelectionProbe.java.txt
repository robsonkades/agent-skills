import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public final class SelectionProbe {
    private record Example(boolean[] decisions, boolean any, boolean all,
                           List<Integer> successfulAnyTrace,
                           List<Integer> successfulAllTrace) {}

    private static boolean allPermit(List<CandidateSelection.Rule> rules, String customer) {
        if (rules.isEmpty()) return false;
        for (CandidateSelection.Rule rule : rules) {
            if (!rule.permits(customer)) return false;
        }
        return true;
    }

    public static void main(String[] args) {
        if (args.length != 2 || !List.of("first", "all").contains(args[0])
                || !List.of("any", "all").contains(args[1])) {
            throw new IllegalArgumentException("SelectionProbe first|all any|all");
        }
        List<Example> examples = List.of(
                new Example(new boolean[] {true, false}, true, false, List.of(0), List.of()),
                new Example(new boolean[] {false, true}, true, false, List.of(0, 1), List.of()),
                new Example(new boolean[] {false, false}, false, false, List.of(), List.of()),
                new Example(new boolean[] {true, true}, true, true, List.of(0), List.of(0, 1)),
                new Example(new boolean[] {}, false, false, List.of(), List.of()));
        int passed = 0;
        for (int index = 0; index < examples.size(); index++) {
            Example example = examples.get(index);
            List<Integer> trace = new ArrayList<>();
            List<CandidateSelection.Rule> rules = new ArrayList<>();
            for (int i = 0; i < example.decisions().length; i++) {
                final int ruleIndex = i;
                rules.add(customer -> {
                    trace.add(ruleIndex);
                    return example.decisions()[ruleIndex];
                });
            }
            boolean actual = args[0].equals("first")
                    ? CandidateSelection.permits(rules, "sample-customer")
                    : allPermit(rules, "sample-customer");
            boolean expected = args[1].equals("any") ? example.any() : example.all();
            List<Integer> successTrace = args[1].equals("any")
                    ? example.successfulAnyTrace() : example.successfulAllTrace();
            // A denied request may fail fast; only successful execution has a required trace.
            boolean ok = actual == expected && (!actual || trace.equals(successTrace));
            if (ok) passed++;
            System.out.printf("case=%d decisions=%s actual=%s expected=%s trace=%s result=%s%n",
                    index + 1, Arrays.toString(example.decisions()), actual, expected,
                    trace, ok ? "PASS" : "FAIL");
        }
        System.out.printf("implementation=%s contract=%s passed=%d total=%d%n",
                args[0], args[1], passed, examples.size());
        if (passed != examples.size()) System.exit(1);
    }
}
