import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave4-validation-Ceekn9';
const report='C:/git/agent-skills/docs/skill-validation/wave4-2026-10-16';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const queue=JSON.parse(fs.readFileSync(report+'/queue.json','utf8'));
assert.equal(queue.tasks.length,68);
assert.ok(queue.tasks.every(t=>t.diffReviewed&&['implemented','validated'].includes(t.status)));
const handoffs=queue.tasks.map(t=>{
 const raw='handoffs-raw/'+t.name+'.json.txt',b=fs.readFileSync(report+'/'+raw);
 assert.equal(sha(b),t.capturedHandoffSha);
 assert.deepEqual(JSON.parse(b),JSON.parse(fs.readFileSync(report+'/handoffs/'+t.name+'.json','utf8')));
 return {skill:t.name,owner:t.owner,raw,sha256:sha(b),diff:'diffs/'+t.name+'.patch',reviewedDiffSha256:t.reviewedDiffSha256};
});
const helpers=[];
fs.mkdirSync(report+'/helpers',{recursive:true});
for(const name of fs.readdirSync(root).filter(n=>/\.(?:mjs|cjs|ps1)$/.test(n)).sort()){
 const b=fs.readFileSync(root+'/'+name),target='helpers/'+name+'.txt';
 fs.writeFileSync(report+'/'+target,b);helpers.push({name,source:root+'/'+name,target,bytes:b.length,sha256:sha(b)});
}
fs.writeFileSync(report+'/review-provenance.json',JSON.stringify({createdAt:new Date().toISOString(),policy:'Original stopped handoffs and coordinator helper sources preserved by bytes/hash. Helpers are historical tooling, not an instruction to replay baseline, actors or mutations.',handoffs,helpers},null,2)+'\n');
console.log(JSON.stringify({handoffs:handoffs.length,helpers:helpers.length}));
