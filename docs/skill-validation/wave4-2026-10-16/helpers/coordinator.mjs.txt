import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import {spawnSync} from 'node:child_process';
const repo='C:/git/agent-skills',root='C:/Users/robso/AppData/Local/Temp/skills-wave4-validation-Ceekn9',report=repo+'/docs/skill-validation/wave4-2026-10-16';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const queue=JSON.parse(fs.readFileSync(report+'/queue.json','utf8'));
const item=name=>{const t=queue.tasks.find(x=>x.name===name);assert.ok(t,name);return t;};
const diff=name=>{const r=spawnSync('git',['diff','--no-index','--no-ext-diff','--',path.join(root,'baseline',name),path.join(repo,'skills',name)],{cwd:repo,encoding:'utf8',maxBuffer:16*1024*1024});assert.ok(r.status===0||r.status===1,r.stderr);return r.stdout;};
for(const t of queue.tasks){
 const p=root+'/'+t.name+'.json';if(!fs.existsSync(p))continue;
 const b=fs.readFileSync(p),hash=sha(b);if(hash===t.ignoredHandoffSha||hash===t.capturedHandoffSha)continue;
 const h=JSON.parse(b);if(h.allWritesStopped!==true)continue;
 assert.equal(h.skill,t.name);assert.ok(['READY_FOR_GLOBAL_VALIDATION','NO_CHANGES_NEEDED','BLOCKED'].includes(h.status));
 fs.writeFileSync(report+'/handoffs/'+t.name+'.json',b);
 fs.mkdirSync(report+'/handoffs-raw',{recursive:true});fs.writeFileSync(report+'/handoffs-raw/'+t.name+'.json.txt',b);
 Object.assign(t,{status:h.status==='BLOCKED'?'blocked':'implemented',reviewResult:h.status,finalVersion:fs.readFileSync(repo+'/skills/'+t.name+'/skill.yaml','utf8').match(/^version:\s*(.+)$/m)[1],capturedHandoffSha:hash,diffReviewed:false,validation:'local complete; coordinator diff review and global checks pending'});
}
for(const arg of process.argv.slice(2)){
 const [action,value]=arg.split('=');
 if(action==='assign')for(const pair of value.split(',')){const [owner,name]=pair.split(':'),t=item(name);assert.equal(t.status,'pending');assert.ok(!queue.tasks.some(x=>x.owner===owner&&x.status==='running'));Object.assign(t,{owner,status:'running',validation:'local pending; global deferred'});}
 else if(action==='reopen'){const t=item(value),b=fs.readFileSync(root+'/'+t.name+'.json');fs.mkdirSync(report+'/handoff-revisions',{recursive:true});fs.writeFileSync(report+'/handoff-revisions/'+t.name+'-'+sha(b).slice(0,12)+'.json.txt',b);Object.assign(t,{status:'running',diffReviewed:false,ignoredHandoffSha:sha(b),validation:'coordinator finding under review; global deferred'});}
 else if(action==='diff')for(const name of value.split(',')){const d=diff(name);fs.writeFileSync(report+'/diffs/'+name+'.patch',d);console.log('SKILL '+name+'\n'+(d||'(no changes)'));}
 else if(action==='review')for(const name of value.split(',')){const t=item(name);assert.equal(t.status,'implemented');const d=diff(name);fs.writeFileSync(report+'/diffs/'+name+'.patch',d);Object.assign(t,{diffReviewed:true,reviewedDiffSha256:sha(d),validation:'local checks and coordinator diff review passed; global deferred'});}
 else throw Error('Unknown action '+action);
}
queue.updatedAt=new Date().toISOString();fs.writeFileSync(report+'/queue.json',JSON.stringify(queue,null,2)+'\n');
fs.writeFileSync(report+'/tasks.md','# Onda 4 — fila de revisão\n\nEscopo: 68 skills. Conclusão local aguarda validação global.\n\n| Skill | Responsável | Estado | Validação |\n| --- | --- | --- | --- |\n'+queue.tasks.map(t=>`| ${t.name} | ${t.owner??'—'} | ${t.status} | ${t.validation} |`).join('\n')+'\n');
if(!process.argv.some(x=>x.startsWith('diff=')))console.log(JSON.stringify({counts:queue.tasks.reduce((a,t)=>(a[t.status]=(a[t.status]??0)+1,a),{}),ready:queue.tasks.filter(t=>t.status==='implemented'&&!t.diffReviewed).map(t=>t.name),running:queue.tasks.filter(t=>t.status==='running').map(t=>t.owner+':'+t.name),next:queue.tasks.filter(t=>t.status==='pending').slice(0,20).map(t=>t.name)},null,2));
