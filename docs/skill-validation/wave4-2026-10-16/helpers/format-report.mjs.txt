import fs from 'node:fs';
import {spawnSync} from 'node:child_process';
import assert from 'node:assert/strict';
const repo='C:/git/agent-skills',report=repo+'/docs/skill-validation/wave4-2026-10-16';
const label=process.argv[2];assert.match(label??'',/^[a-z0-9-]+$/);
const mode=process.argv.includes('--check')?'--check':'--write';
const log=report+'/format-'+label+'.log.txt';assert.ok(!fs.existsSync(log));
const args=['node_modules/prettier/bin/prettier.cjs',mode,'docs/skill-validation/wave4-2026-10-16/**/*.{json,md,yaml,yml}'];
const r=spawnSync(process.execPath,args,{cwd:repo,encoding:'utf8',windowsHide:true,maxBuffer:8*1024*1024});
fs.writeFileSync(log,r.stdout+'\n'+r.stderr);
fs.writeFileSync(report+'/format-'+label+'-result.json',JSON.stringify({at:new Date().toISOString(),command:process.execPath,args,exit:r.status,log:log.split('/').at(-1),scope:'Wave4 report only; raw .txt evidence and .patch diffs untouched'},null,2)+'\n');
console.log(JSON.stringify({mode,exit:r.status,log}));
console.log((r.stdout+'\n'+r.stderr).split(/\r?\n/).slice(-6).join('\n'));
process.exitCode=r.status??1;
