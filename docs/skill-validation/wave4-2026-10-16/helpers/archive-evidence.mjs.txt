 const files=new Set(),seen=new Set(),sourcePaths=new Set();
 function identifySources(v,sourceContext=false){
  if(typeof v==='string'&&sourceContext&&/^C:[/\\]/.test(v))sourcePaths.add(v.replaceAll('\\','/').replace(/\/$/,''));
  else if(Array.isArray(v))v.forEach(x=>identifySources(x,sourceContext));
  else if(v&&typeof v==='object'){
   const primary=sourceContext||Object.entries(v).some(([k,x])=>/^(?:url|sourceUrl|uri)$/i.test(k)&&typeof x==='string'&&/^https?:/.test(x));
   for(const [k,x] of Object.entries(v))identifySources(x,primary||/^(?:sources|primarySources|verifiedSources|sourceEvidence|sourceDownloads|sourceSnapshots)$/i.test(k));
  }
 }
 identifySources(h);
 function add(p){p=p.replaceAll('\\','/').replace(/\/$/,'');if(!p.startsWith(temp)||p===root||p===temp.slice(0,-1)||/\/(?:gradle-user-home|\.gradle|daemon|caches)\//.test(p)||seen.has(p)||!fs.existsSync(p))return;seen.add(p);const s=fs.lstatSync(p);if(s.isSymbolicLink())return;if(s.isDirectory()){for(const e of fs.readdirSync(p,{withFileTypes:true}))if(!e.isSymbolicLink()&&!skipDirs.has(e.name))add(p+'/'+e.name);}else if(s.isFile())files.add(p);}
 function visit(v){if(typeof v==='string'&&/^C:[/\\]/.test(v))add(v);else if(Array.isArray(v))v.forEach(visit);else if(v&&typeof v==='object')Object.values(v).forEach(visit);}
 visit(h);
 const entries=[];
 for(const f of [...files].sort().filter(p=>p!==root+'/'+skill+'.json')){
  const size=fs.statSync(f).size,ext=path.extname(f).toLowerCase();
  if(size>8*1024*1024){entries.push({path:f,bytes:size,archived:false,reason:'Large artifact retained at original location; not read or vendored by archiver.'});continue;}
  const b=fs.readFileSync(f),hash=sha(b),start=b.subarray(0,6000).toString('utf8');
  const sourceMarked=[...sourcePaths].some(p=>f===p||f.startsWith(p+'/'));
  const bulkSourceRecord=ext==='.json'&&(/\[wordlim:/.test(b.toString('utf8'))||(/source-evidence\.json$/i.test(f)&&size>16000&&/excerpt/.test(b.toString('utf8'))));
  const thirdParty=sourceMarked||bulkSourceRecord||/<!doctype\s+html|<html(?:\s|>)/i.test(start)||(/\.(java|js|mjs|sh|ps1|cpp|hpp)$/i.test(f)&&/copyright|licensed under|SPDX-License-Identifier/i.test(start))||(/\/(?:sources|source-pages|source-snapshots|research)\//i.test(f)&&/\.(?:md|txt|html|java|adoc)$/i.test(f))||(/\.txt$/i.test(f)&&fs.existsSync(f.replace(/\.txt$/i,'.pdf')));
  if(!allowed.has(ext)||size>1048576||thirdParty){entries.push({path:f,bytes:size,sha256:hash,archived:false,reason:thirdParty?'Third-party source/documentation remains referenced, not vendored.':!allowed.has(ext)?'Binary or non-text artifact remains referenced.':'Large text remains referenced.'});continue;}

  entries.push({path:f,bytes:b.length,sha256:hash,archived:true,target});
 }
 records.push({skill,embeddedEvidence:{rawHandoff:rawPath,sha256:sha(raw),classification:'Original handoff contains authored findings, command records and source references. Its preservation is not a new execution.'},entries});
}
