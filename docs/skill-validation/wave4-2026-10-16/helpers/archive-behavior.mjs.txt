import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const report='C:/git/agent-skills/docs/skill-validation/wave4-2026-10-16';
const manifest=JSON.parse(fs.readFileSync(report+'/behavior-manifest.frozen.json.txt','utf8'));
const judgments=JSON.parse(fs.readFileSync(report+'/behavior-results.json','utf8'));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const artifacts=[],preservation=[];
function archive(from,to){const b=fs.readFileSync(from);assert.ok(b.length<4*1024*1024,'Unexpected large actor artifact');fs.mkdirSync(path.dirname(report+'/'+to),{recursive:true});fs.writeFileSync(report+'/'+to,b);artifacts.push({path:to,source:from,bytes:b.length,sha256:sha(b),encoding:'raw bytes, unmodified; remove .txt suffix to restore original extension'});}
function walk(base,sub=''){return fs.readdirSync(base+'/'+sub,{withFileTypes:true}).flatMap(e=>{const p=sub?sub+'/'+e.name:e.name;assert.ok(!e.isSymbolicLink());return e.isDirectory()?walk(base,p):[p];});}
for(const r of manifest.runs){
 const result=judgments.runs.find(x=>x.runId===r.runId);assert.equal(result.allWritesStopped,true);assert.equal(result.criteriaSha256,r.criteriaSha256);
 const missing=[];for(const f of r.inputs){const actual=fs.readFileSync(r.actorDirectory+'/'+f.path),snapshot=fs.readFileSync(report+'/'+r.inputSnapshot+'/'+f.path+'.txt');assert.equal(sha(actual),f.sha256);assert.equal(sha(snapshot),f.sha256);artifacts.push({path:r.inputSnapshot+'/'+f.path+'.txt',source:r.actorDirectory+'/'+f.path,bytes:snapshot.length,sha256:f.sha256,encoding:'raw bytes, unmodified; remove .txt suffix to restore original extension'});}
 const all=walk(r.actorDirectory),inputSet=new Set(r.inputs.map(x=>x.path));
 const outside=all.filter(p=>!inputSet.has(p)&&!p.startsWith('output/'));assert.deepEqual(outside,[]);
 const outputs=all.filter(p=>p.startsWith('output/'));assert.ok(outputs.includes('output/decision.md'));assert.ok(outputs.includes('output/execution.json'));
 for(const p of outputs)archive(r.actorDirectory+'/'+p,'behavior-outputs/'+r.runId+'/'+p.slice(7)+'.txt');
 preservation.push({runId:r.runId,passed:true,inputFilesUnchanged:r.inputs.length,outputs:outputs.length,unexpectedFiles:outside,missing});
}
fs.writeFileSync(report+'/behavior-artifact-manifest.json',JSON.stringify({createdAt:new Date().toISOString(),policy:'All supplied actor inputs and outputs are byte-preserved, including tiny project-authored jars/classes; no external dependency binary is included. No new actor execution by this archiver.',artifacts,preservation},null,2)+'\n');
console.log(JSON.stringify({runs:preservation.length,files:artifacts.length,bytes:artifacts.reduce((n,a)=>n+a.bytes,0),inputPreservation:true},null,2));
