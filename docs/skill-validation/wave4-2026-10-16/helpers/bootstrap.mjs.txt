import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';
import YAML from 'file:///C:/git/agent-skills/node_modules/yaml/dist/index.js';

const git=(args)=>execFileSync('git',args,{cwd:repo,encoding:'utf8',maxBuffer:64*1024*1024});
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const names=fs.readFileSync(repo+'/docs/skills-evolution-inventory.md','utf8').split(/\r?\n/).filter(l=>/^\| \[/.test(l)&&/\| 4\s*\|$/.test(l)).map(l=>l.match(/\[([^\]]+)\]/)[1]).sort();

const allFiles=allPaths.filter(p=>fs.existsSync(repo+'/'+p)).map(p=>{const b=fs.readFileSync(repo+'/'+p);return {path:p,sha256:sha(b),bytes:b.length};});
const packages=names.map(name=>{assert.ok(fs.existsSync(repo+'/skills/'+name+'/SKILL.md'));const manifest=YAML.parse(fs.readFileSync(repo+'/skills/'+name+'/skill.yaml','utf8'));const committed=YAML.parse(git(['show','HEAD:skills/'+name+'/skill.yaml']));return {name,initialVersion:manifest.version,committedVersion:committed.version,status:git(['status','--short','--','skills/'+name]),files:allFiles.filter(f=>f.path.startsWith('skills/'+name+'/'))};});
const baseline={capturedAt:new Date().toISOString(),head:git(['rev-parse','HEAD']).trim(),wave:4,repo,temporaryRoot:root,registryBefore:fs.readFileSync(repo+'/registry/skills.yaml','utf8'),gitStatus:git(['status','--short']),gitDiffStat:git(['diff','--stat']),gitCachedDiffStat:git(['diff','--cached','--stat']),allFiles,packages};

for(const p of packages)fs.cpSync(repo+'/skills/'+p.name,root+'/baseline/'+p.name,{recursive:true,errorOnExist:true,force:false});
const queue={wave:4,scope:'Exactly the 68 inventory rows marked wave 4',createdAt:baseline.capturedAt,tasks:packages.map(p=>({name:p.name,owner:null,status:'pending',initialVersion:p.initialVersion,committedVersion:p.committedVersion,preexistingChanges:p.status,validation:'not started',diffReviewed:false}))};

console.log(JSON.stringify({root,report,skills:packages.length,filesPreserved:allFiles.length,head:baseline.head,preexistingSelectedChanges:packages.filter(p=>p.status).map(p=>({name:p.name,status:p.status,initialVersion:p.initialVersion,committedVersion:p.committedVersion})),firstBatch:names.slice(0,20)},null,2));
