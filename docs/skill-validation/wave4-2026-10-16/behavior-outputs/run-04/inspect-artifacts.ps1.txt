$ErrorActionPreference = 'Stop'
$runRoot = Split-Path -Parent $PSScriptRoot
$javaRoot = 'C:/Users/robso/.jdks/temurin-25.0.3/bin'
$runner = Join-Path $PSScriptRoot 'run-command.ps1'
& $runner -Executable "$javaRoot/java.exe" -CommandArguments @('-version') -Label 'java-version'
& $runner -Executable "$javaRoot/javac.exe" -CommandArguments @('-version') -Label 'javac-version'
& $runner -Executable "$javaRoot/jar.exe" -CommandArguments @('--version') -Label 'jar-version'
& $runner -Executable "$javaRoot/javap.exe" -CommandArguments @('-version') -Label 'javap-version'
foreach ($artifact in @(
    'input/published/text-core-1.0.0.jar',
    'input/published/text-core-1.0.1.jar',
    'input/published/report-app-1.0.0.jar',
    'input/bundles/report-1.0.0.jar'
)) {
    & $runner -Executable "$javaRoot/jar.exe" -CommandArguments @('--list', '--file', $artifact) -Label "list-$artifact"
}
foreach ($version in @('1.0.0', '1.0.1')) {
    & $runner -Executable "$javaRoot/javap.exe" -CommandArguments @('-classpath', "input/published/text-core-$version.jar", '-public', '-s', 'com.example.text.LineEndings') -Label "api-text-core-$version"
    & $runner -Executable "$javaRoot/javap.exe" -CommandArguments @('-classpath', "input/published/text-core-$version.jar", '-c', '-verbose', 'com.example.text.LineEndings') -Label "bytecode-text-core-$version"
}
& $runner -Executable "$javaRoot/javap.exe" -CommandArguments @('-classpath', 'input/published/report-app-1.0.0.jar', '-c', '-verbose', 'com.example.app.ReportApp') -Label 'bytecode-report-app-1.0.0'
