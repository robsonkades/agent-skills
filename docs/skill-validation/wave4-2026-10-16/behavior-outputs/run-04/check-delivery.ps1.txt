function Run-Check([string]$Executable, [string[]]$CommandArguments, [string]$Label, [int]$ExpectedCode = 0) {
    $resultJson = & $runner -Executable $Executable -CommandArguments $CommandArguments -Label $Label
    $resultJson
    $result = $resultJson | ConvertFrom-Json
    if ($result.timedOut -or $result.exitCode -ne $ExpectedCode) {
        throw "$Label failed: exit $($result.exitCode); expected $ExpectedCode"
    }
}
