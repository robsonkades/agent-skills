$ErrorActionPreference = 'Stop'
$runRoot = Split-Path -Parent $PSScriptRoot
$logPath = Join-Path $PSScriptRoot 'execution.json'
$document = Get-Content -LiteralPath $logPath -Raw | ConvertFrom-Json
$expectedFailureLabels = @(
    'external-old-rejects-corrected-assertion',
    'external-candidate-rejects-old-assertion',
    'old-library-contract-negative-control'
)
foreach ($entry in $document.executions) {
    $expectedCode = if ($entry.label -in $expectedFailureLabels) { 1 } else { 0 }
    if ($entry.timedOut -or $entry.exitCode -ne $expectedCode) {
        throw "Unexpected command result: $($entry.label)"
    }
}
function Get-Execution([string]$Label) {
    $matches = @($document.executions | Where-Object { $_.label -eq $Label })
    if ($matches.Count -ne 1) { throw "Expected exactly one execution for $Label" }
    return $matches[0]
}
if ((Get-Execution 'api-text-core-1.0.0').stdout -cne (Get-Execution 'api-text-core-1.0.1').stdout) { throw 'Public APIs differ' }
foreach ($label in @('external-old-baseline', 'external-rollback-old-selection-new-process')) {
    if ((Get-Execution $label).stdout.Trim() -cne 'A<CR>B') { throw "Unexpected stdout: $label" }
}
foreach ($label in @('external-candidate-same-app-binary', 'unchanged-app-with-rebuilt-candidate')) {
    if ((Get-Execution $label).stdout.Trim() -cne 'A<LF>B') { throw "Unexpected stdout: $label" }
}
foreach ($label in @('published-candidate-normalization-contract', 'rebuilt-candidate-normalization-contract')) {
    if ((Get-Execution $label).stdout -notmatch 'cases=9, failures=0') { throw "Incomplete regression run: $label" }
}
$loadedClasses = @(Get-Content -LiteralPath (Join-Path $PSScriptRoot 'classload-candidate.log') | Where-Object { $_ -match 'com\.example\.(app\.ReportApp|text\.LineEndings) source:' })
if ($loadedClasses.Count -ne 2) { throw 'Expected exactly two application class-load records' }
if (-not ($loadedClasses -match 'com\.example\.app\.ReportApp source: .*input/published/report-app-1\.0\.0\.jar')) { throw 'Wrong application source' }
if (-not ($loadedClasses -match 'com\.example\.text\.LineEndings source: .*input/published/text-core-1\.0\.1\.jar')) { throw 'Wrong library source' }
$rebuiltHash = (Get-FileHash -LiteralPath (Join-Path $PSScriptRoot 'classes/library/com/example/text/LineEndings.class') -Algorithm SHA256).Hash.ToLowerInvariant()
$publishedBytecode = (Get-Execution 'bytecode-text-core-1.0.1').stdout
if ($publishedBytecode -notmatch "SHA-256 checksum $rebuiltHash") { throw 'Rebuilt class differs from published candidate' }
$before = @(Get-Content -LiteralPath (Join-Path $PSScriptRoot 'input-hashes-before.json') -Raw | ConvertFrom-Json)
$after = @(Get-ChildItem -LiteralPath (Join-Path $runRoot 'input') -Recurse -File | Get-FileHash -Algorithm SHA256 | Select-Object Path, Hash)
if (@(Compare-Object ($before | ForEach-Object { "$($_.Path):$($_.Hash)" }) ($after | ForEach-Object { "$($_.Path):$($_.Hash)" })).Count -ne 0) { throw 'Inputs changed' }
$proposal = Get-Content -LiteralPath (Join-Path $PSScriptRoot 'delivery.proposed.json') -Raw | ConvertFrom-Json
if ($proposal.applicationVersion -ne '1.0.0' -or $proposal.libraryVersion -ne '1.0.1' -or $proposal.layout -ne 'external') { throw 'Unexpected proposal' }
if (-not (Test-Path -LiteralPath (Join-Path $PSScriptRoot 'decision.md'))) { throw 'Decision missing' }
