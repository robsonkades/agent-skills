$execution.versionsUsed = [ordered]@{
    javaRuntime = 'OpenJDK Runtime Environment Temurin-25.0.3+9 (build 25.0.3+9-LTS)'
    javaVm = 'OpenJDK 64-Bit Server VM Temurin-25.0.3+9'

$execution.summary = [ordered]@{
    processCount = $execution.executions.Count
    expectedAssertionFailures = @($execution.executions | Where-Object { $_.expectedExitCode -eq 1 }).Count
    unexpectedResults = @($execution.executions | Where-Object { -not $_.matchedExpectedExit }).Count
    timeouts = @($execution.executions | Where-Object { $_.timedOut }).Count

    allWritesStopped = $true

    deploymentPerformed = $false

if ($execution.summary.processCount -ne 30 -or $execution.summary.unexpectedResults -ne 0 -or $execution.summary.timeouts -ne 0) {

}

}

        throw "Empty required output: $required"
    }

        throw "stderr mismatch for $($run.id)"
    }
}
