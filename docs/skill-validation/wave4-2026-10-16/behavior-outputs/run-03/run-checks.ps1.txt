function Save-Execution {
    $execution | ConvertTo-Json -Depth 20 | Set-Content -LiteralPath $executionPath -Encoding utf8
}

function Invoke-Checked {
    param(
        [string] $Id,
        [string] $Executable,
        [string[]] $ArgumentList,
        [int] $ExpectedExit = 0,
        [string] $WorkingDirectory = $reviewRoot
    )
    if (@($execution.executions | Where-Object { $_.id -eq $Id }).Count -ne 0) {
        throw "Execution already recorded: $Id"
    }
    $startInfo = [System.Diagnostics.ProcessStartInfo]::new()
    $startInfo.FileName = $Executable
    $startInfo.WorkingDirectory = $WorkingDirectory
    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true
    foreach ($argument in $ArgumentList) {
        [void]$startInfo.ArgumentList.Add($argument)
    }
    $process = [System.Diagnostics.Process]::new()
    $process.StartInfo = $startInfo
    $watch = [System.Diagnostics.Stopwatch]::StartNew()
    $startedAt = [DateTime]::UtcNow.ToString('o')
    try {
        [void]$process.Start()
        $stdoutTask = $process.StandardOutput.ReadToEndAsync()
        $stderrTask = $process.StandardError.ReadToEndAsync()
        $timedOut = -not $process.WaitForExit(30000)
        if ($timedOut) {
            $process.Kill($true)
            $process.WaitForExit()
        }
        $stdout = $stdoutTask.GetAwaiter().GetResult()
        $stderr = $stderrTask.GetAwaiter().GetResult()
        $exitCode = $process.ExitCode
        $watch.Stop()
        $stdoutPath = Join-Path $reviewLogDirectory "$Id.stdout.log"
        $stderrPath = Join-Path $reviewLogDirectory "$Id.stderr.log"
        [System.IO.File]::WriteAllText($stdoutPath, $stdout)
        [System.IO.File]::WriteAllText($stderrPath, $stderr)
        $execution.executions += [ordered]@{
            id = $Id
            command = $Executable
            arguments = @($ArgumentList)
            cwd = $WorkingDirectory
            startedAtUtc = $startedAt
            elapsedMilliseconds = $watch.ElapsedMilliseconds
            timeoutMilliseconds = 30000
            timedOut = $timedOut
            exitCode = $exitCode
            expectedExitCode = $ExpectedExit
            matchedExpectedExit = (-not $timedOut -and $exitCode -eq $ExpectedExit)
            stdout = $stdout
            stderr = $stderr
            stdoutFile = $stdoutPath
            stderrFile = $stderrPath
        }
        Save-Execution
        Write-Output "$Id : exit=$exitCode expected=$ExpectedExit timeout=$timedOut"
        if ($timedOut -or $exitCode -ne $ExpectedExit) {
            throw "Unexpected command result for $Id; inspect execution.json and logs."
        }
    } finally {
        if (-not $process.HasExited) {
            $process.Kill($true)
            $process.WaitForExit()
        }
        $process.Dispose()
    }
}

function Get-ArtifactHashes {
    $artifacts = @(
        'input/published/text-core-1.0.0.jar',
        'input/published/text-core-1.0.1.jar',
        'input/published/report-app-1.0.0.jar',
        'input/bundles/report-1.0.0.jar'
    )
    $hashes = [ordered]@{}
    foreach ($artifact in $artifacts) {
        $hashes[$artifact] = (Get-FileHash -LiteralPath (Join-Path $reviewRoot $artifact) -Algorithm SHA256).Hash
    }
    return $hashes
}

$java = Join-Path $reviewJdk 'java.exe'
$javac = Join-Path $reviewJdk 'javac.exe'
$jar = Join-Path $reviewJdk 'jar.exe'
$javap = Join-Path $reviewJdk 'javap.exe'

switch ($Stage) {
    'inspect' {
        $execution.artifactHashes.before = Get-ArtifactHashes
        Save-Execution
        Invoke-Checked 'java-version' $java @('-version')
        Invoke-Checked 'javac-version' $javac @('-version')
        Invoke-Checked 'jar-version' $jar @('--version')
        Invoke-Checked 'javap-version' $javap @('-version')
        Invoke-Checked 'list-text-core-100' $jar @('--list', '--file', 'input/published/text-core-1.0.0.jar')
        Invoke-Checked 'list-text-core-101' $jar @('--list', '--file', 'input/published/text-core-1.0.1.jar')
        Invoke-Checked 'list-app-100' $jar @('--list', '--file', 'input/published/report-app-1.0.0.jar')
        Invoke-Checked 'list-bundle-100' $jar @('--list', '--file', 'input/bundles/report-1.0.0.jar')
        Invoke-Checked 'bytecode-text-core-100' $javap @('-classpath', 'input/published/text-core-1.0.0.jar', '-public', '-s', '-c', '-verbose', 'com.example.text.LineEndings')
        Invoke-Checked 'bytecode-text-core-101' $javap @('-classpath', 'input/published/text-core-1.0.1.jar', '-public', '-s', '-c', '-verbose', 'com.example.text.LineEndings')
        Invoke-Checked 'bytecode-app-100' $javap @('-classpath', 'input/published/report-app-1.0.0.jar', '-public', '-s', '-c', '-verbose', 'com.example.app.ReportApp')
        Invoke-Checked 'bytecode-bundled-library-100' $javap @('-classpath', 'input/bundles/report-1.0.0.jar', '-public', '-s', '-c', '-verbose', 'com.example.text.LineEndings')
    }
    'compatibility' {
        Invoke-Checked 'old-bundle-legacy-pass' $java @('-jar', 'input/bundles/report-1.0.0.jar', 'CR')
        Invoke-Checked 'old-bundle-correction-fails' $java @('-jar', 'input/bundles/report-1.0.0.jar', 'LF') 1
        Invoke-Checked 'old-bundle-with-external-candidate-still-fails' $java @('-cp', 'input/published/text-core-1.0.1.jar', '-jar', 'input/bundles/report-1.0.0.jar', 'LF') 1
        Invoke-Checked 'old-app-old-library-pass' $java @('-cp', 'input/published/report-app-1.0.0.jar;input/published/text-core-1.0.0.jar', 'com.example.app.ReportApp', 'CR')
        Invoke-Checked 'old-app-candidate-library-pass' $java @('-cp', 'input/published/report-app-1.0.0.jar;input/published/text-core-1.0.1.jar', 'com.example.app.ReportApp', 'LF')
        [void][System.IO.Directory]::CreateDirectory((Join-Path $reviewOutput 'rebuilt-consumer'))
        Invoke-Checked 'compile-consumer-against-candidate' $javac @('--release', '17', '-cp', 'input/published/text-core-1.0.1.jar', '-d', 'output/rebuilt-consumer', 'input/consumer/com/example/app/ReportApp.java')
        Invoke-Checked 'rebuilt-consumer-candidate-pass' $java @('-cp', 'output/rebuilt-consumer;input/published/text-core-1.0.1.jar', 'com.example.app.ReportApp', 'LF')
        [void][System.IO.Directory]::CreateDirectory((Join-Path $reviewOutput 'contract-classes'))
        Invoke-Checked 'compile-contract-probe' $javac @('--release', '17', '-cp', 'input/published/text-core-1.0.1.jar', '-d', 'output/contract-classes', 'output/check-contract/CheckLineEndings.java')
        Invoke-Checked 'contract-old-library-legacy' $java @('-cp', 'output/contract-classes;input/published/text-core-1.0.0.jar', 'CheckLineEndings', 'legacy')
        Invoke-Checked 'contract-candidate-library-corrected' $java @('-cp', 'output/contract-classes;input/published/text-core-1.0.1.jar', 'CheckLineEndings', 'corrected')
    }
    'bundle' {
        $consumerDirectory = Join-Path $reviewOutput 'bundle-consumer'
        $libraryDirectory = Join-Path $reviewOutput 'bundle-library'
        [void][System.IO.Directory]::CreateDirectory($consumerDirectory)
        [void][System.IO.Directory]::CreateDirectory($libraryDirectory)
        Invoke-Checked 'extract-original-consumer' $jar @('--extract', '--file', (Join-Path $reviewRoot 'input/bundles/report-1.0.0.jar'), 'com/example/app/ReportApp.class') 0 $consumerDirectory
        Invoke-Checked 'extract-candidate-library' $jar @('--extract', '--file', (Join-Path $reviewRoot 'input/published/text-core-1.0.1.jar'), 'com/example/text/LineEndings.class') 0 $libraryDirectory
        Invoke-Checked 'create-candidate-bundle' $jar @('--create', '--file', 'output/report-review-text-core-1.0.1.jar', '--main-class', 'com.example.app.ReportApp', '-C', 'output/bundle-consumer', '.', '-C', 'output/bundle-library', '.')
        Invoke-Checked 'list-candidate-bundle' $jar @('--list', '--file', 'output/report-review-text-core-1.0.1.jar')
        Invoke-Checked 'candidate-bundle-correction-pass' $java @('-jar', 'output/report-review-text-core-1.0.1.jar', 'LF')
        Invoke-Checked 'candidate-bundle-old-assertion-fails' $java @('-jar', 'output/report-review-text-core-1.0.1.jar', 'CR') 1
        Invoke-Checked 'contract-candidate-bundle-corrected' $java @('-cp', 'output/contract-classes;output/report-review-text-core-1.0.1.jar', 'CheckLineEndings', 'corrected')
    }
    'verify-retention' {
        Invoke-Checked 'rollback-original-bundle-pass' $java @('-jar', 'input/bundles/report-1.0.0.jar', 'CR')
        $execution.artifactHashes.after = Get-ArtifactHashes
        $same = $true
        foreach ($artifact in $execution.artifactHashes.before.Keys) {
            if ($execution.artifactHashes.before[$artifact] -ne $execution.artifactHashes.after[$artifact]) {
                $same = $false
            }
        }
        $execution.artifactHashes.inputsUnchanged = $same
        $execution.artifactHashes.localCandidateBundle = (Get-FileHash -LiteralPath (Join-Path $reviewOutput 'report-review-text-core-1.0.1.jar') -Algorithm SHA256).Hash
        Save-Execution
        if (-not $same) {
            throw 'Input artifact hashes changed.'
        }
        Write-Output 'Input JAR SHA-256 hashes remain unchanged.'
    }
}
