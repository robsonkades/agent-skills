function Save-Executions {
    $record = [ordered]@{
        taskRoot = $taskRoot
        status = 'in-progress'
        timeoutSecondsPerChild = 30
        environment = [ordered]@{ os = [System.Runtime.InteropServices.RuntimeInformation]::OSDescription; javaToolsDirectory = $jdkRoot; compileTarget = 17; runtime = 'pending version command' }
        commands = @($script:executions)
        limitations = @('Synthetic ReportApp asserts lone-CR normalization; no full service or live deployment was tested.', 'All generated files are confined to output; input artifacts are read-only.')
    }
    $record | ConvertTo-Json -Depth 12 | Set-Content -LiteralPath $executionPath -Encoding utf8
}

function Invoke-RecordedProcess {
    param([string]$Id, [string]$Executable, [string[]]$Arguments, [string]$WorkingDirectory = $taskRoot, [int[]]$AllowedExitCodes = @(0))
    $start = [System.Diagnostics.ProcessStartInfo]::new()
    $start.FileName = $Executable
    $start.WorkingDirectory = $WorkingDirectory
    $start.UseShellExecute = $false
    $start.CreateNoWindow = $true
    $start.RedirectStandardOutput = $true
    $start.RedirectStandardError = $true
    foreach ($argument in $Arguments) { $start.ArgumentList.Add($argument) }
    $process = [System.Diagnostics.Process]::new()
    $process.StartInfo = $start
    $startedAt = [DateTimeOffset]::UtcNow
    if (-not $process.Start()) { throw "Could not start $Id" }
    $stdoutTask = $process.StandardOutput.ReadToEndAsync()
    $stderrTask = $process.StandardError.ReadToEndAsync()
    $timedOut = -not $process.WaitForExit(30000)
    if ($timedOut) { $process.Kill($true); $process.WaitForExit() }
    $stdout = $stdoutTask.GetAwaiter().GetResult()
    $stderr = $stderrTask.GetAwaiter().GetResult()
    $exitCode = $process.ExitCode
    $stdoutLog = Join-Path $outputRoot "logs/$Id.stdout.log"
    $stderrLog = Join-Path $outputRoot "logs/$Id.stderr.log"
    [System.IO.File]::WriteAllText($stdoutLog, $stdout)
    [System.IO.File]::WriteAllText($stderrLog, $stderr)
    $script:executions.Add([ordered]@{
        id = $Id; executable = $Executable; arguments = @($Arguments); cwd = $WorkingDirectory
        startedAtUtc = $startedAt.ToString('o'); durationMs = [int]([DateTimeOffset]::UtcNow - $startedAt).TotalMilliseconds
        timeoutSeconds = 30; timedOut = $timedOut; exitCode = $exitCode; allowedExitCodes = @($AllowedExitCodes)
        stdout = $stdout; stderr = $stderr; stdoutLog = $stdoutLog; stderrLog = $stderrLog
    })
    Save-Executions
    $process.Dispose()
    [pscustomobject]@{ id = $Id; exitCode = $exitCode; timedOut = $timedOut; stdout = $stdout; stderr = $stderr } | ConvertTo-Json -Compress
    if ($timedOut -or $AllowedExitCodes -notcontains $exitCode) { throw "Unexpected result for $Id (exit=$exitCode, timeout=$timedOut)" }
}
