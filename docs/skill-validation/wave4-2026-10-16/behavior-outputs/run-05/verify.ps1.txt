function Run-Check {
    param([string]$Id, [string]$Executable, [string[]]$Arguments, [int]$ExpectedExit = 0)
    $start = [DateTime]::UtcNow
    $startInfo = [System.Diagnostics.ProcessStartInfo]::new()
    $startInfo.FileName = $Executable
    $startInfo.WorkingDirectory = $runRoot
    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true
    foreach ($argValue in $Arguments) { $startInfo.ArgumentList.Add($argValue) }
    $child = [System.Diagnostics.Process]::new()
    $child.StartInfo = $startInfo
    $timedOut = $false
    try {
        $child.Start() | Out-Null
        $outTask = $child.StandardOutput.ReadToEndAsync()
        $errTask = $child.StandardError.ReadToEndAsync()
        if (-not $child.WaitForExit(30000)) {
            $timedOut = $true
            $child.Kill($true)
            $child.WaitForExit()
        }
        $stdout = $outTask.GetAwaiter().GetResult()
        $stderr = $errTask.GetAwaiter().GetResult()
        $exitCode = $child.ExitCode
        $entry = [ordered]@{
            id = $Id; executable = $Executable; arguments = $Arguments; cwd = $runRoot
            startedUtc = $start.ToString('o'); durationMilliseconds = ([DateTime]::UtcNow - $start).TotalMilliseconds
            exitCode = $exitCode; expectedExitCode = $ExpectedExit; timedOut = $timedOut
            stdout = $stdout; stderr = $stderr
            expectationMet = (-not $timedOut -and $exitCode -eq $ExpectedExit)
        }
        $runs.Add($entry)
        [System.IO.File]::WriteAllText((Join-Path $outputRoot "logs/$Id.stdout.txt"), $stdout, $utf8)
        [System.IO.File]::WriteAllText((Join-Path $outputRoot "logs/$Id.stderr.txt"), $stderr, $utf8)
        Save-Record
        Write-Output ("{0}: exit={1}, expected={2}, timeout={3}" -f $Id, $exitCode, $ExpectedExit, $timedOut)
        if ($timedOut) { throw "Timed out: $Id" }
    } finally {
        if (-not $child.HasExited) { $child.Kill($true); $child.WaitForExit() }
        $child.Dispose()
    }
}
