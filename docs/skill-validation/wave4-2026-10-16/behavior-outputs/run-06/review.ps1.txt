        'input/published/report-app-1.0.0.jar',

    )
    foreach ($relativePath in $paths) {
        $fullPath = Join-Path $reviewRoot $relativePath
        [ordered]@{
            path = $relativePath
            sizeBytes = (Get-Item -LiteralPath $fullPath).Length
            sha256 = (Get-FileHash -Algorithm SHA256 -LiteralPath $fullPath).Hash.ToLowerInvariant()
        }
    }
}

function Invoke-Recorded {
    param(
        [string]$Id,
        [string]$Tool,
        [string[]]$ToolArguments,
        [int]$ExpectedExitCode = 0,
        [string]$Purpose
    )
    $executable = Join-Path $reviewJdk ($Tool + '.exe')
    $start = [Diagnostics.ProcessStartInfo]::new()
    $start.FileName = $executable
    $start.WorkingDirectory = $reviewRoot
    $start.UseShellExecute = $false
    $start.CreateNoWindow = $true
    $start.RedirectStandardOutput = $true
    $start.RedirectStandardError = $true
    foreach ($argument in $ToolArguments) { $start.ArgumentList.Add($argument) }
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $start
    $watch = [Diagnostics.Stopwatch]::StartNew()
    $started = [DateTimeOffset]::UtcNow.ToString('o')
    [void]$process.Start()
    $stdoutTask = $process.StandardOutput.ReadToEndAsync()
    $stderrTask = $process.StandardError.ReadToEndAsync()
    $timedOut = -not $process.WaitForExit(30000)
    if ($timedOut) {
        $process.Kill($true)
        $process.WaitForExit()
    }
    $stdout = $stdoutTask.GetAwaiter().GetResult()
    $stderr = $stderrTask.GetAwaiter().GetResult()
    $exitCode = $process.ExitCode
    $watch.Stop()
    $process.Dispose()
    $stdoutLog = 'logs/' + $Id + '.stdout.txt'
    $stderrLog = 'logs/' + $Id + '.stderr.txt'
    [IO.File]::WriteAllText((Join-Path $reviewOutput $stdoutLog), $stdout, [Text.UTF8Encoding]::new($false))
    [IO.File]::WriteAllText((Join-Path $reviewOutput $stderrLog), $stderr, [Text.UTF8Encoding]::new($false))
    $matches = -not $timedOut -and $exitCode -eq $ExpectedExitCode
    $reviewRuns.Add([ordered]@{
        id = $Id
        purpose = $Purpose
        command = $executable
        arguments = @($ToolArguments)
        cwd = $reviewRoot
        startedAtUtc = $started
        durationMs = $watch.ElapsedMilliseconds
        timeoutSeconds = 30
        timedOut = $timedOut
        exitCode = $exitCode
        expectedExitCode = $ExpectedExitCode
        matchedExpectedExitCode = $matches
        stdout = $stdout
        stderr = $stderr
        stdoutLog = $stdoutLog
        stderrLog = $stderrLog
    })
    if (-not $matches) { $reviewUnexpected.Add($Id) }
    Write-ExecutionRecord
    Write-Output ($Id + ': exit=' + $exitCode + ' expected=' + $ExpectedExitCode + ' timeout=' + $timedOut)
}
