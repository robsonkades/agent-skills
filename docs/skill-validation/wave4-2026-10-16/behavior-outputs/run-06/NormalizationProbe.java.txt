import com.example.text.LineEndings;

public final class NormalizationProbe {
    public static void main(String[] args) {
        String[][] cases = {
            {"empty", "", ""},
            {"plain", "AB", "AB"},
            {"LF", "A\nB", "A\nB"},
            {"CRLF", "A\r\nB", "A\nB"},
            {"CR", "A\rB", "A\nB"},
            {"mixed", "\rA\r\nB\nC\r", "\nA\nB\nC\n"}
        };
        int failures = 0;
        for (String[] test : cases) {
            String actual = LineEndings.normalize(test[1]);
            boolean matches = actual.equals(test[2]);
            System.out.println(test[0] + ": " + (matches ? "PASS" : "FAIL")
                + " actual=" + visible(actual) + " expected=" + visible(test[2]));
            if (!matches) failures++;
            if (!LineEndings.normalize(actual).equals(actual)) {
                failures++;
                System.out.println(test[0] + ": FAIL idempotence");
            }
        }
        System.out.println("provider=" + LineEndings.class.getProtectionDomain()
            .getCodeSource().getLocation());
        System.out.println("failures=" + failures);
        if (failures != 0) throw new AssertionError("Normalization failures: " + failures);
    }

    private static String visible(String value) {
        return value.replace("\r", "<CR>").replace("\n", "<LF>");
    }
}
