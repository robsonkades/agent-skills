$script:Execution.versions.PowerShell = $PSVersionTable.PSVersion.ToString()

Save-Execution
Invoke-Recorded -Id 'archive-manifests-class-identity' -Executable $hostExe -CommandArguments @('-NoProfile', '-File', (Join-Path $PSScriptRoot 'inspect-archives.ps1'))
$archiveResult = $script:Execution.executions[-1].stdout | ConvertFrom-Json -AsHashtable
$script:Execution.archiveChecks = $archiveResult
$unchanged = @($script:Execution.artifactHashes | ForEach-Object {
    $current = Get-FileHash -LiteralPath (Join-Path $script:TaskRoot $_.path) -Algorithm SHA256
    [ordered]@{ path = $_.path; unchanged = ($current.Hash -eq $_.sha256); sha256 = $current.Hash }
})
if ($unchanged.unchanged -contains $false) { throw 'An input artifact changed.' }
$script:Execution.inputArtifactRetention = $unchanged
