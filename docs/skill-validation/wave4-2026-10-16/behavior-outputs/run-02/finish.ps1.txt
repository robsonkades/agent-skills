. (Join-Path $PSScriptRoot 'runner.ps1')
$allowedUnexpected = @('javap-version')
$unexpected = @($script:Execution.executions | Where-Object { -not $_.matchedExpectedExitCode -and $_.id -notin $allowedUnexpected })
if ($unexpected.Count -ne 0) { throw 'An unresolved validation failure remains.' }
if (@($script:Execution.executions | Where-Object timedOut).Count -ne 0) { throw 'A subprocess timed out.' }
if (-not (Test-Path -LiteralPath (Join-Path $PSScriptRoot 'decision.md'))) { throw 'Missing decision.md' }
if (-not ($script:Execution.archiveChecks.unchangedConsumer -and $script:Execution.archiveChecks.publishedCandidateInBundle -and $script:Execution.archiveChecks.oneLibraryClass)) {
    throw 'Missing artifact identity validation.'
}
$script:Execution.summary = [ordered]@{
    validationSubprocesses = $script:Execution.executions.Count
    matchedExpectedExitCodes = @($script:Execution.executions | Where-Object matchedExpectedExitCode).Count
    correctedToolOptionFailure = 'javap-version: unknown --version option; corrected -version invocation passed.'
    intentionalNegativeControls = @('deployed-old-bundle-fix-absent', 'published-new-jar-does-not-update-bundle')
    allInputJarHashesUnchanged = -not ($script:Execution.inputArtifactRetention.unchanged -contains $false)
    childrenExited = $true
    deploymentPerformed = $false
}
$script:Execution.completedUtc = [DateTime]::UtcNow.ToString('o')
$script:Execution.allWritesStopped = $true
Save-Execution
$verified = Get-Content -LiteralPath $script:ExecutionPath -Raw | ConvertFrom-Json
if (-not $verified.allWritesStopped) { throw 'Invalid completion record.' }
$verified.summary | ConvertTo-Json -Depth 8
Write-Output 'allWritesStopped=true; all started validation processes exited.'
