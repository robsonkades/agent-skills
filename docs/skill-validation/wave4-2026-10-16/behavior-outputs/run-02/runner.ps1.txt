$ErrorActionPreference = 'Stop'
$script:TaskRoot = [IO.Path]::GetFullPath((Join-Path $PSScriptRoot '..'))
$script:JavaBin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin'
$script:ExecutionPath = Join-Path $PSScriptRoot 'execution.json'
$script:LogRoot = Join-Path $PSScriptRoot 'logs'
[IO.Directory]::CreateDirectory($script:LogRoot) | Out-Null

if (Test-Path -LiteralPath $script:ExecutionPath) {
    $script:Execution = Get-Content -LiteralPath $script:ExecutionPath -Raw | ConvertFrom-Json -AsHashtable
} else {
    $script:Execution = [ordered]@{
        taskRoot = $script:TaskRoot
        scope = 'Read only input/ and skill/; all writes in output/; no deployment, network, install or real configuration changes.'
        processTimeoutSeconds = 30
        versions = [ordered]@{ compilationTarget = 17; skill = 'component-and-release-boundaries 1.2.4' }
        limitations = @()
        executions = @()
        localActions = @()
        allWritesStopped = $false
    }
}

function Save-Execution {
    $script:Execution | ConvertTo-Json -Depth 30 | Set-Content -LiteralPath $script:ExecutionPath -Encoding utf8
}

function Add-LocalAction {
    param([string]$Description)
    $script:Execution.localActions += [ordered]@{ atUtc = [DateTime]::UtcNow.ToString('o'); description = $Description }
    Save-Execution
}

function Invoke-Recorded {
    param(
        [string]$Id,
        [string]$Executable,
        [string[]]$CommandArguments,
        [int]$ExpectedExitCode = 0,
        [string]$WorkingDirectory = $script:TaskRoot
    )
    if ($script:Execution.executions.id -contains $Id) { throw "Duplicate execution id: $Id" }
    $start = [Diagnostics.ProcessStartInfo]::new()
    $start.FileName = $Executable
    $start.WorkingDirectory = $WorkingDirectory
    $start.UseShellExecute = $false
    $start.CreateNoWindow = $true
    $start.RedirectStandardOutput = $true
    $start.RedirectStandardError = $true
    foreach ($argument in $CommandArguments) { $start.ArgumentList.Add($argument) }
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $start
    $began = [DateTime]::UtcNow
    $process.Start() | Out-Null
    $stdoutTask = $process.StandardOutput.ReadToEndAsync()
    $stderrTask = $process.StandardError.ReadToEndAsync()
    $timedOut = -not $process.WaitForExit(30000)
    if ($timedOut) { $process.Kill($true); $process.WaitForExit() }
    $stdout = $stdoutTask.GetAwaiter().GetResult()
    $stderr = $stderrTask.GetAwaiter().GetResult()
    $exitCode = $process.ExitCode
    $process.Dispose()
    $record = [ordered]@{
        id = $Id
        executable = $Executable
        arguments = @($CommandArguments)
        cwd = $WorkingDirectory
        startedUtc = $began.ToString('o')
        elapsedMilliseconds = [int]([DateTime]::UtcNow - $began).TotalMilliseconds
        timeoutSeconds = 30
        timedOut = $timedOut
        exitCode = $exitCode
        expectedExitCode = $ExpectedExitCode
        matchedExpectedExitCode = ($exitCode -eq $ExpectedExitCode -and -not $timedOut)
        stdout = $stdout
        stderr = $stderr
    }
    $stdout | Set-Content -LiteralPath (Join-Path $script:LogRoot "$Id.stdout.txt") -Encoding utf8 -NoNewline
    $stderr | Set-Content -LiteralPath (Join-Path $script:LogRoot "$Id.stderr.txt") -Encoding utf8 -NoNewline
    $script:Execution.executions += $record
    Save-Execution
    $record | ConvertTo-Json -Depth 10 -Compress | Write-Output
    if (-not $record.matchedExpectedExitCode) { throw "Unexpected result for $Id (exit $exitCode, timeout $timedOut)" }
}

function Invoke-JavaTool {
    param([string]$Id, [string]$Tool, [string[]]$CommandArguments, [int]$ExpectedExitCode = 0, [string]$WorkingDirectory = $script:TaskRoot)
    Invoke-Recorded -Id $Id -Executable (Join-Path $script:JavaBin "$Tool.exe") -CommandArguments $CommandArguments -ExpectedExitCode $ExpectedExitCode -WorkingDirectory $WorkingDirectory
}
