import fs from 'node:fs';
import assert from 'node:assert/strict';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave10-c71a8e23';
const read=n=>JSON.parse(fs.readFileSync(root+'/'+n+'.json','utf8'));
const expected=read('calibration-expected'),g=read('review/calibration-grades'),ar=read('arithmetic-result');
const comparisons=[];
assert.equal(g.samples.length,12);
for(const s of g.samples)for(const stage of ['1','2'])for(const key of ['C1','C2','C3','C4']){
 const v=s.stages[stage].criteria[key];assert.equal(v.pass,expected[s.id],s.id+'/'+stage+'/'+key);assert.ok(v.reason&&v.evidence.length);
 comparisons.push({id:s.id,stage,criterion:key,expected:expected[s.id],actual:v.pass,agreed:true});
}
assert.equal(ar.checks.length,16);assert.ok(ar.checks.every(c=>c.pass));
const result={at:new Date().toISOString(),calibrationSamples:12,validAlternatives:8,invalidControls:4,criterionAgreement:comparisons.length,comparisons,arithmetic:ar,limit:'Short authored controls verify these judgments; no claim of general evaluator reliability or agent gain.'};
fs.writeFileSync(root+'/preflight.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify({calibrationAgreement:comparisons.length,arithmetic:ar.checks.length}));
