import fs from 'node:fs';
import path from 'node:path';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave10-c71a8e23';
const actors=JSON.parse(fs.readFileSync(root+'/actors.json','utf8'));
const records=[];
for(const a of actors)for(const stage of[1,2]){
 const p=a.dir+'/output/access'+stage+'.json',x=JSON.parse(fs.readFileSync(p,'utf8'));
 const reads=x.paths_read??x.caminhos_lidos,writes=x.files_written??x.arquivos_escritos;
 if(!Array.isArray(reads)||!Array.isArray(writes))throw Error('Unknown schema '+p);
 const resolve=p=>path.resolve(a.dir,p).replaceAll('\\','/');
 const outside=[...reads.filter(p=>!resolve(p).startsWith(a.dir+'/')),...writes.filter(p=>!resolve(p).startsWith(a.dir+'/output/'))];
 const reportedOutside=x.out_of_perimeter_reads??x.reads_outside_perimeter??x.outside_perimeter_reads??x.leituras_fora_do_perimetro;
 if(!Array.isArray(reportedOutside))throw Error('Missing disclosure '+p);
 records.push({id:a.id,stage,condition:a.condition,reads,writes,outside,reportedOutside,failures:x.failures??x.falhas??[],skillReads:reads.filter(p=>resolve(p).startsWith(a.dir+'/input/skill/')).map(p=>resolve(p).slice(a.dir.length+1))});
}
const result={at:new Date().toISOString(),basis:'Actor-declared access files only. Paths checked mechanically; this is not a complete tool trace, independent observation of reads, enforced isolation or proof of absence of leakage.',records,declaredOutside:records.flatMap(r=>[...r.outside,...r.reportedOutside]),skillCoverage:actors.map(a=>({id:a.id,condition:a.condition,reads:[...new Set(records.filter(r=>r.id===a.id).flatMap(r=>r.skillReads))]}))};
fs.writeFileSync(root+'/access-audit.json',JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify({logs:records.length,outside:result.declaredOutside,failures:records.filter(r=>r.failures.length).map(r=>({id:r.id,stage:r.stage,failures:r.failures})),skillCoverage:result.skillCoverage.filter(a=>a.condition!=='none')}));
