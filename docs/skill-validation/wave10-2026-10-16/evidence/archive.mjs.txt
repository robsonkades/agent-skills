import fs from 'node:fs';
import crypto from 'node:crypto';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave10-c71a8e23',report='C:/git/agent-skills/docs/skill-validation/wave10-2026-10-16';
const walk=d=>fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(d+'/'+e.name):[d+'/'+e.name]);
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const artifacts=[];
const copy=(src,dst)=>{fs.mkdirSync(report+'/'+dst.slice(0,dst.lastIndexOf('/')),{recursive:true});fs.copyFileSync(src,report+'/'+dst);artifacts.push({source:src,path:dst,sha256:sha(src)});};
for(const folder of['runs','snapshot','stage-seals','packets','review'])for(const p of walk(root+'/'+folder))copy(p,'raw/'+p.slice(root.length+1)+'.txt');
for(const p of fs.readdirSync(root).filter(p=>p.endsWith('.mjs')||p.endsWith('.json'))){if(p==='actors.json'||p==='cases.json'||p==='criteria.json'||p==='calibration.json')continue;copy(root+'/'+p,'evidence/'+p+'.txt');}
for(const p of['result-summary','integrity-final','access-audit','preflight','research'])fs.copyFileSync(root+'/'+p+'.json',report+'/'+p+'.json');
fs.writeFileSync(report+'/artifact-manifest.json',JSON.stringify({at:new Date().toISOString(),artifacts},null,2)+'\n');
console.log(JSON.stringify({rawArtifacts:artifacts.length,allHashesMatch:artifacts.every(a=>sha(report+'/'+a.path)===a.sha256)}));
