import fs from 'node:fs';
import crypto from 'node:crypto';

 for(let j=0;j<3;j++){

  if(condition!=='none')fs.cpSync(condition==='reference'?root+'/snapshot':repo+'/skills/architecture-trade-off-analysis',dir+'/input/skill',{recursive:true});
  const resources=condition==='none'?'Não há pacote adicional de instruções neste diretório.':'Use o pacote adicional em input/skill: leia SKILL.md e siga suas rotas de referência quando úteis. Todos os arquivos desse pacote estão disponíveis, inclusive referências distribuídas.';
  const task=`# Trabalho\n\n${cases[c].task}\n\nLeia os documentos em input/project antes de responder. ${resources}\n\nEscreva output/answer1.md em português; procure manter até350 palavras, sem transformar esse limite em motivo para omitir uma decisão necessária. Use dados locais como premissas de um caso autoral; não os apresente como medição que você executou. Não é necessário nem autorizado consultar internet, outros projetos, skills instaladas, diretórios irmãos, resultados ou critérios de avaliação. Ferramentas locais para leitura e cálculo estão disponíveis. Não altere input. Sua propriedade de escrita é somente output deste diretório. Você não está sozinho no filesystem: preserve trabalho alheio.\n\nRegistre em output/access1.json uma lista literal dos caminhos lidos, comandos executados e arquivos escritos nesta fase, e eventuais falhas ou leituras fora do perímetro; isso é um registro declaratório, não sandbox. Finalize esta fase após gravar a resposta; uma atualização posterior poderá chegar. Não procure atualizações em outros arquivos. Não se autoavalie nem consulte critérios.\n`;
  fs.writeFileSync(dir+'/TASK.md',task);
  const prompt=`Execute somente a fase1 da tarefa em ${dir}/TASK.md. Seu diretório de trabalho é ${dir}. Leia apenas TASK.md, input/** e seus próprios output/**; pode usar ferramentas locais para calcular. Não consulte outros diretórios, a versão do repositório, skills instaladas, internet, critérios ou outros agentes. Sua escrita pertence apenas a ${dir}/output/**; não está sozinho, preserve edições alheias. Produza answer1.md e access1.json conforme a tarefa e pare. Não delegue. Não conhece nem deve procurar resultado esperado. Na resposta final, informe somente arquivos gerados e limitações de execução.`;
  actors.push({id,case:c,repetition:rep,condition,version:condition==='none'?null:condition==='reference'?'1.2.4':'1.2.5',dir,prompt,stage2Prompt:`Leia somente ${dir}/input/UPDATE.md recém-fornecido e seus documentos/resposta anteriores, mantendo o mesmo perímetro. Escreva output/answer2.md com a recomendação atualizada e output/access2.json com as leituras/comandos/escritas desta fase. Não altere answer1.md nem input. Não há necessidade de comparar seu desempenho nem consultar critérios. Preserve o trabalho alheio; não delegue. Termine após esses dois arquivos.`});
 }
}
fs.writeFileSync(root+'/actors.json',JSON.stringify(actors,null,2)+'\n');
console.log(JSON.stringify({actors:actors.length,phases:actors.length*2,conditions:[...new Set(actors.map(a=>a.condition))],noMandatoryDependencies:true}));
