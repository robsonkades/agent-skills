import java.nio.file.Path;
import java.time.Duration;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import jdk.jfr.Recording;
import jdk.jfr.consumer.RecordingFile;

public class PinningBoundaryProbe {
    private static final CountDownLatch ENTERED = new CountDownLatch(1);
    private static final CountDownLatch RELEASE = new CountDownLatch(1);

    private static void waitForRelease() {
        ENTERED.countDown();
        try {
            RELEASE.await();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    private static class Initializing {
        static final int VALUE;
        static {
            waitForRelease();
            VALUE = 1;
        }
    }

    public static void main(String[] args) throws Exception {
        String mode = args[0];
        boolean expectedPin = Boolean.parseBoolean(args[1]);
        Path target = Path.of(args[2]);
        Thread task = null;
        try (Recording recording = new Recording()) {
            recording.enable("jdk.VirtualThreadPinned").withThreshold(Duration.ZERO).withStackTrace();
            recording.start();
            task = Thread.ofVirtual().name("pinning-boundary-probe").start(() -> {
                if (mode.equals("classinit")) {
                    if (Initializing.VALUE != 1) throw new AssertionError();
                } else if (mode.equals("monitor")) {
                    synchronized (PinningBoundaryProbe.class) {
                        waitForRelease();
                    }
                } else {
                    throw new IllegalArgumentException(mode);
                }
            });
            if (!ENTERED.await(3, TimeUnit.SECONDS)) throw new AssertionError("Task did not enter");
            long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(3);
            while (task.getState() != Thread.State.WAITING && System.nanoTime() < deadline) {
                Thread.sleep(1);
            }
            if (task.getState() != Thread.State.WAITING) throw new AssertionError("No waiting state");
            RELEASE.countDown();
            task.join(3000);
            if (task.isAlive()) throw new AssertionError("Task failed to finish");
            recording.stop();
            recording.dump(target);
        } finally {
            RELEASE.countDown();
            if (task != null) {
                task.interrupt();
                task.join(3000);
            }
        }
        var pins = RecordingFile.readAllEvents(target).stream()
                .filter(event -> event.getEventType().getName().equals("jdk.VirtualThreadPinned"))
                .filter(event -> event.getThread() != null
                        && event.getThread().getJavaName().equals("pinning-boundary-probe"))
                .toList();
        System.out.printf("runtime=%s mode=%s pinEvents=%d expectedPin=%s%n",
                Runtime.version(), mode, pins.size(), expectedPin);
        pins.forEach(System.out::println);
        if ((!pins.isEmpty()) != expectedPin) throw new AssertionError("Unexpected pin outcome");
    }
}
