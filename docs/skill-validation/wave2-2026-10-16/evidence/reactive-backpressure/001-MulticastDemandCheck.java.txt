import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Objects;

import org.reactivestreams.Subscription;
import reactor.core.Exceptions;
import reactor.core.publisher.BaseSubscriber;
import reactor.core.publisher.Flux;
import reactor.core.publisher.Sinks;

public final class MulticastDemandCheck {
    private static int assertions;

    static final class Probe extends BaseSubscriber<Integer> {
        final List<Integer> received = new ArrayList<Integer>();
        Throwable failure;
        boolean completed;

        @Override
        protected void hookOnSubscribe(Subscription subscription) {
            // Deliberately no default/unbounded demand.
        }

        @Override
        protected void hookOnNext(Integer value) {
            received.add(value);
        }

        @Override
        protected void hookOnError(Throwable error) {
            failure = error;
        }

        @Override
        protected void hookOnComplete() {
            completed = true;
        }
    }

    private static void equal(Object expected, Object actual, String label) {
        assertions++;
        if (!Objects.equals(expected, actual)) {
            throw new AssertionError(label + ": expected=" + expected + ", actual=" + actual);
        }
    }

    private static Sinks.Many<Integer> sink(boolean allOrNothing) {
        return allOrNothing
            ? Sinks.many().multicast().<Integer>directAllOrNothing()
            : Sinks.many().multicast().<Integer>directBestEffort();
    }

    private static void directDemand(boolean allOrNothing) {
        String mode = allOrNothing ? "directAllOrNothing" : "directBestEffort";
        Sinks.Many<Integer> source = sink(allOrNothing);
        equal(Sinks.EmitResult.FAIL_ZERO_SUBSCRIBER, source.tryEmitNext(0), mode + " no subscriber");

        Probe fast = new Probe();
        Probe slow = new Probe();
        source.asFlux().subscribe(fast);
        source.asFlux().subscribe(slow);
        equal(Sinks.EmitResult.FAIL_OVERFLOW, source.tryEmitNext(0), mode + " zero demand");
        equal(Collections.emptyList(), fast.received, mode + " zero demand fast");
        equal(Collections.emptyList(), slow.received, mode + " zero demand slow");
        equal(null, fast.failure, mode + " try failure is not terminal fast");
        equal(null, slow.failure, mode + " try failure is not terminal slow");

        fast.request(2);
        Sinks.EmitResult first = source.tryEmitNext(1);
        equal(allOrNothing ? Sinks.EmitResult.FAIL_OVERFLOW : Sinks.EmitResult.OK,
              first, mode + " asymmetric result");
        equal(allOrNothing ? Collections.emptyList() : Arrays.asList(1),
              fast.received, mode + " asymmetric fast");
        equal(Collections.emptyList(), slow.received, mode + " asymmetric slow");

        slow.request(1);
        equal(Collections.emptyList(), slow.received, mode + " no replay on later demand");
        equal(Sinks.EmitResult.OK, source.tryEmitNext(2), mode + " restored demand");
        equal(allOrNothing ? Arrays.asList(2) : Arrays.asList(1, 2),
              fast.received, mode + " restored fast");
        equal(Arrays.asList(2), slow.received, mode + " restored slow");
        equal(null, fast.failure, mode + " no terminal fast");
        equal(null, slow.failure, mode + " no terminal slow");

        fast.request(1);
        slow.cancel();
        equal(Sinks.EmitResult.OK, source.tryEmitNext(3), mode + " cancelled slow");
        equal(allOrNothing ? Arrays.asList(2, 3) : Arrays.asList(1, 2, 3),
              fast.received, mode + " remaining fast");
        equal(Arrays.asList(2), slow.received, mode + " cancelled receives no more");
        fast.cancel();
        equal(Sinks.EmitResult.FAIL_ZERO_SUBSCRIBER, source.tryEmitNext(4), mode + " all cancelled");
        equal(Sinks.EmitResult.OK, source.tryEmitComplete(), mode + " explicit completion");
        equal(Sinks.EmitResult.FAIL_TERMINATED, source.tryEmitNext(5), mode + " after terminal");

        System.out.println(mode + ": first=" + first + "; fast=" + fast.received
            + "; slow=" + slow.received + "; zero/no-subscriber/cancel/terminal controls passed");
    }

    private static void failFastOverflow(boolean allOrNothing) {
        String mode = allOrNothing ? "directAllOrNothing" : "directBestEffort";
        Sinks.Many<Integer> source = sink(allOrNothing);
        Probe first = new Probe();
        Probe second = new Probe();
        source.asFlux().subscribe(first);
        source.asFlux().subscribe(second);
        source.emitNext(7, Sinks.EmitFailureHandler.FAIL_FAST);
        equal(true, Exceptions.isOverflow(first.failure), mode + " emit overflow first");
        equal(true, Exceptions.isOverflow(second.failure), mode + " emit overflow second");
        equal(Collections.emptyList(), first.received, mode + " emit receives no value first");
        equal(Collections.emptyList(), second.received, mode + " emit receives no value second");
        equal(Sinks.EmitResult.FAIL_TERMINATED, source.tryEmitNext(8), mode + " emit terminal");
        System.out.println(mode + ": emitNext FAIL_FAST overflow terminated both subscribers");
    }

    public static void main(String[] args) {
        equal("3.7.5", Flux.class.getPackage().getImplementationVersion(), "Reactor artifact version");
        directDemand(false);
        directDemand(true);
        failFastOverflow(false);
        failFastOverflow(true);
        System.out.println("PASS assertions=" + assertions
            + "; compiler target=Java 8; runtime=" + System.getProperty("java.runtime.version"));
    }
}
