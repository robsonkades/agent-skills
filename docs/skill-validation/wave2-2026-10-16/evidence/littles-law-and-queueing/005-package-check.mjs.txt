import assert from 'node:assert/strict';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { pathToFileURL } from 'node:url';
import { createHash } from 'node:crypto';

const repo = 'C:/git/agent-skills';
const root = resolve(repo, 'skills/littles-law-and-queueing');
const { loadPackageFromDirectory } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/package-loader.ts')).href);
const { validatePackage } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/validate-package.ts')).href);
const { NodeFileSystem } = await import(pathToFileURL(resolve(repo, 'packages/node/src/file-system.ts')).href);
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), root, { strict: true });
const report = validatePackage(loaded.pkg, { directoryName: 'littles-law-and-queueing', strict: true });
const issues = [...loaded.issues, ...report.issues];
assert.equal(issues.length, 0, JSON.stringify(issues));
const files = ['SKILL.md', 'skill.yaml', 'references/sizing-worksheet.md', 'references/utilisation-curve.md'];
assert.equal(loaded.pkg.files.length, files.length);
const links = [];
const hashes = {};
for (const file of files) {
  const absolute = resolve(root, file);
  const bytes = readFileSync(absolute);
  hashes[file] = createHash('sha256').update(bytes).digest('hex').toUpperCase();
  if (!file.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
    const target = match[1];
    if (/^<?[a-z]+:/i.test(target)) continue;
    const [path, anchor] = target.split('#');
    assert.ok(existsSync(resolve(dirname(absolute), path)), `${file}: ${target}`);
    assert.equal(anchor, undefined, 'Anchor validation needed if an internal anchor is added');
    links.push({ file, target });
  }
}
assert.equal(hashes['references/utilisation-curve.md'], 'D295C1A66EFF7CD9D5744A6C32BBE49D09995C98B7870ED807178EBE4D37F0FE');
assert.equal(loaded.pkg.manifest.version, '1.5.6');
console.log(JSON.stringify({ runtime: process.version, name: loaded.pkg.manifest.name, version: loaded.pkg.manifest.version, packageFiles: loaded.pkg.files.length, issues, localMarkdownLinks: links, hashes, unchangedReferenceVerified: true }, null, 2));
