import assert from 'node:assert/strict';

const window = { start: 0, end: 10 };
const trace = [[0, 2], [1, 0], [9, 2], [10, 0]];
const meanRows = rows => rows.reduce((sum, [, n]) => sum + n, 0) / rows.length;
const meanTime = rows => rows.slice(0, -1).reduce(
  (sum, [time, n], i) => sum + n * (rows[i + 1][0] - time), 0,
) / (window.end - window.start);
const results = [];
const check = (name, actual, expected) => {
  assert.ok(Math.abs(actual - expected) < 1e-12, `${name}: ${actual} != ${expected}`);
  results.push({ name, actual, expected, passed: true });
};

check('complete post-change trace time average', meanTime(trace), 0.4);
const residenceIntervals = [[-1, 1], [-1, 1], [9, 10], [9, 10]];
const clippedArea = residenceIntervals.reduce((sum, [start, end]) => sum +
  Math.max(0, Math.min(end, window.end) - Math.max(start, window.start)), 0);
check('independent clipped residence integral', clippedArea / 10, 0.4);
check('negative control: arithmetic row mean is not time mean', meanRows(trace), 1);
assert.notEqual(meanRows(trace), meanTime(trace));
const redundant = [[0, 2], [1, 0], [5, 0], [9, 2], [10, 0]];
check('redundant unchanged observation leaves time average invariant', meanTime(redundant), 0.4);
check('redundant unchanged observation changes arithmetic row mean', meanRows(redundant), 0.8);

const unseenBurst = [[0, 2], [1, 0], [4, 3], [5, 0], [9, 2], [10, 0]];
for (const [time, observed] of trace) {
  const candidates = unseenBurst.filter(([t]) => t <= time);
  assert.equal(candidates.at(-1)[1], observed);
}
check('same sparse observations admit a different true time average', meanTime(unseenBurst), 0.7);
const fullResidencePerWindow = residenceIntervals.reduce((sum, [start, end]) => sum + end - start, 0) / 10;
check('preserved edge-effect control: complete departure residences differ from clipped area', fullResidencePerWindow, 0.6);
assert.notEqual(fullResidencePerWindow, clippedArea / 10);
console.log(JSON.stringify({ runtime: process.version, classification: 'deterministic numerical controls; no workload or agent evaluation', checks: results.length, results }, null, 2));
