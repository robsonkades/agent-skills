import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const directory = fileURLToPath(new URL('.', import.meta.url));
const java = 'C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe';
const controls = [
  { id: 'zgc-obsolete-mode', flags: ['-XX:+UseZGC', '-XX:+ZGenerational'], exit: 0, patterns: [/Ignoring option ZGenerational; support was removed in 24\.0/] },
  { id: 'shenandoah-default', flags: ['-XX:+UseShenandoahGC'], exit: 0, patterns: [/Shenandoah GC mode: Snapshot-At-The-Beginning \(SATB\)/, /ShenandoahGCMode\s+= satb/] },
  { id: 'shenandoah-generational', flags: ['-XX:+UseShenandoahGC', '-XX:ShenandoahGCMode=generational'], exit: 0, patterns: [/Shenandoah GC mode: Generational/, /ShenandoahGCMode\s+= generational/] },
  { id: 'shenandoah-generational-compact', flags: ['-XX:+UseShenandoahGC', '-XX:ShenandoahGCMode=generational', '-XX:ShenandoahGCHeuristics=compact'], exit: 0, patterns: [/Ignoring -XX:ShenandoahGCHeuristics=compact/, /only supports adaptive heuristics/, /Heuristics: Adaptive/] },
  { id: 'shenandoah-invalid-buffer', flags: ['-XX:+UseShenandoahGC', '-XX:ShenandoahMaxSATBBufferSize=1024'], exit: 1, patterns: [/Unrecognized VM option 'ShenandoahMaxSATBBufferSize=1024'/] },
  { id: 'zgc-fixed-heap-uncommit', flags: ['-XX:+UseZGC', '-Xms64m', '-XX:+ZUncommit'], exit: 0, patterns: [/Uncommit: Disabled \(Xms equals Xmx\)/] }
];
const results = [];
for (const control of controls) {
  const args = ['-Xms16m', '-Xmx64m', ...control.flags, '-Xlog:gc+init=info', '-XX:+PrintFlagsFinal', '-version'];
  const result = spawnSync(java, args, { cwd: directory, encoding: 'utf8', windowsHide: true, timeout: 20000, maxBuffer: 2 * 1024 * 1024 });
  writeFileSync(resolve(directory, control.id + '.stdout.log'), result.stdout ?? '');
  writeFileSync(resolve(directory, control.id + '.stderr.log'), result.stderr ?? '');
  const output = (result.stdout ?? '') + (result.stderr ?? '');
  const patterns = control.patterns.map(pattern => ({ pattern: pattern.source, matched: pattern.test(output) }));
  results.push({ id: control.id, executable: java, arguments: args, status: result.status, signal: result.signal, error: result.error?.message ?? null, expectedExit: control.exit, patterns, stdout: control.id + '.stdout.log', stderr: control.id + '.stderr.log' });
}
writeFileSync(resolve(directory, 'launch-results.json'), JSON.stringify({ runtimeScope: 'Temurin 25.0.3+9 Windows x64 owned launch only; no workload performance measurement', results }, null, 2) + '\n');
console.log(JSON.stringify(results.map(({ id, status, expectedExit, patterns, error }) => ({ id, status, expectedExit, patterns, error })), null, 2));
for (const result of results) {
  assert.equal(result.error, null, result.id);
  assert.equal(result.status, result.expectedExit, result.id);
  assert.ok(result.patterns.every(pattern => pattern.matched), result.id);
}
