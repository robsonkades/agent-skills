$ErrorActionPreference = 'Stop'
$validationDirectory = $PSScriptRoot
$sources = @(
  @{ Name = 'jep-521.html'; Url = 'https://openjdk.org/jeps/521' },
  @{ Name = 'jep-535.html'; Url = 'https://openjdk.org/jeps/535' },
  @{ Name = 'jep-523.html'; Url = 'https://openjdk.org/jeps/523' },
  @{ Name = 'jep-535-issue.json'; Url = 'https://bugs.openjdk.org/rest/api/2/issue/JDK-8379682?fields=summary,status,fixVersions' },
  @{ Name = 'oracle-jdk25-zgc.html'; Url = 'https://docs.oracle.com/en/java/javase/25/gctuning/z-garbage-collector1.html' },
  @{ Name = 'z_globals.hpp'; Url = 'https://raw.githubusercontent.com/openjdk/jdk/jdk-25-ga/src/hotspot/share/gc/z/z_globals.hpp' },
  @{ Name = 'zPageAllocator.cpp'; Url = 'https://raw.githubusercontent.com/openjdk/jdk/jdk-25-ga/src/hotspot/share/gc/z/zPageAllocator.cpp' },
  @{ Name = 'shenandoahForwarding.inline.hpp'; Url = 'https://raw.githubusercontent.com/openjdk/jdk/jdk-25-ga/src/hotspot/share/gc/shenandoah/shenandoahForwarding.inline.hpp' },
  @{ Name = 'shenandoahGeneration.cpp'; Url = 'https://raw.githubusercontent.com/openjdk/jdk/jdk-25-ga/src/hotspot/share/gc/shenandoah/shenandoahGeneration.cpp' },
  @{ Name = 'shenandoah_globals.hpp'; Url = 'https://raw.githubusercontent.com/openjdk/jdk/jdk-25-ga/src/hotspot/share/gc/shenandoah/shenandoah_globals.hpp' }
)
$results = foreach ($source in $sources) {
  $destination = Join-Path $validationDirectory $source.Name
  try {
    $response = Invoke-WebRequest -Uri $source.Url -OutFile $destination -PassThru -TimeoutSec 15
    [ordered]@{ url = $source.Url; file = $destination; status = [int]$response.StatusCode; bytes = (Get-Item -LiteralPath $destination).Length; sha256 = (Get-FileHash -LiteralPath $destination -Algorithm SHA256).Hash; error = $null }
  } catch {
    [ordered]@{ url = $source.Url; file = $destination; status = $null; bytes = $null; sha256 = $null; error = $_.Exception.Message }
  }
}
$results | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $validationDirectory 'source-fetch-results.json') -Encoding utf8
$results | ConvertTo-Json -Depth 5
