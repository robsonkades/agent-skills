import java.time.Duration;
import java.util.concurrent.*;

public final class BulkheadChecks {
    private static final long WAIT = TimeUnit.MINUTES.toNanos(1);
    private static int passed;

    public static void main(String[] args) throws Exception {
        if (args.length != 0) {
            if (args[0].equals("unsafe-timeout")) unsafeTimeout();
            else if (args[0].equals("fresh-gate")) callerScope(false);
            else throw new IllegalArgumentException(args[0]);
            throw new AssertionError("negative control was not detected");
        }
        basicGate();
        inputAndInterrupt();
        observer("success");
        observer("failure");
        observer("timeout");
        observer("cancel");
        callerScope(true);
        System.out.println("PASS " + passed + " positive cases");
    }

    private static ConcurrencyGate.Lease acquire(ConcurrencyGate gate) throws Exception {
        ConcurrencyGate.Lease lease = gate.tryAcquire(Duration.ZERO);
        if (lease == null) throw new AssertionError("expected free permit");
        return lease;
    }

    private static void held(ConcurrencyGate gate) throws Exception {
        ConcurrencyGate.Lease unexpected = gate.tryAcquire(Duration.ZERO);
        if (unexpected != null) {
            unexpected.close();
            throw new AssertionError("resource admitted while initial operation open");
        }
    }

    private static void restored(ConcurrencyGate gate) throws Exception {
        try (ConcurrencyGate.Lease lease = acquire(gate)) { held(gate); }
    }

    private static void basicGate() throws Exception {
        ConcurrencyGate gate = new ConcurrencyGate(1, true);
        ConcurrencyGate.Lease lease = acquire(gate);
        held(gate);
        lease.close();
        lease.close();
        restored(gate);
        passed++;
    }

    private static void inputAndInterrupt() throws Exception {
        ConcurrencyGate gate = new ConcurrencyGate(1, true);
        try {
            gate.tryAcquire(Duration.ofNanos(-1));
            throw new AssertionError("negative duration accepted");
        } catch (IllegalArgumentException expected) { }
        try (ConcurrencyGate.Lease lease = gate.tryAcquire(Duration.ofSeconds(Long.MAX_VALUE))) {
            if (lease == null) throw new AssertionError("overflow budget lost free permit");
            Thread.currentThread().interrupt();
            try {
                gate.tryAcquire(Duration.ZERO);
                throw new AssertionError("interruption ignored");
            } catch (InterruptedException expected) { }
            held(gate);
        }
        restored(gate);
        passed++;
    }

    private static void observer(String mode) throws Exception {
        ConcurrencyGate gate = new ConcurrencyGate(1, true);
        CompletableFuture<String> actualDone = new CompletableFuture<>();
        CompletableFuture<String> observed = ObserverPattern.observe(
                actualDone, acquire(gate), mode.equals("timeout") ? 0 : WAIT);
        try {
            held(gate);
            if (mode.equals("timeout")) {
                try {
                    observed.get(5, TimeUnit.SECONDS);
                    throw new AssertionError("timeout observer succeeded");
                } catch (ExecutionException expected) {
                    if (!(expected.getCause() instanceof TimeoutException)) throw expected;
                }
            } else if (mode.equals("cancel")) {
                if (!observed.cancel(true)) throw new AssertionError("observer not cancelled");
            }
            if (actualDone.isDone()) throw new AssertionError("observer completed lifecycle");
            held(gate);
            if (mode.equals("failure")) actualDone.completeExceptionally(new IllegalStateException("provider failed"));
            else actualDone.complete("done");
            if (mode.equals("success") && !observed.get(5, TimeUnit.SECONDS).equals("done")) {
                throw new AssertionError("wrong result");
            }
            if (mode.equals("failure")) {
                try {
                    observed.get(5, TimeUnit.SECONDS);
                    throw new AssertionError("provider failure hidden");
                } catch (ExecutionException expected) {
                    if (!(expected.getCause() instanceof IllegalStateException)) throw expected;
                }
            }
            restored(gate);
        } finally {
            actualDone.complete("cleanup");
        }
        passed++;
    }

    private static void unsafeTimeout() throws Exception {
        ConcurrencyGate gate = new ConcurrencyGate(1, true);
        ConcurrencyGate.Lease lease = acquire(gate);
        CompletableFuture<String> source = new CompletableFuture<>();
        CompletableFuture<String> afterRelease = source.whenComplete((value, failure) -> lease.close());
        source.orTimeout(0, TimeUnit.NANOSECONDS);
        try { afterRelease.get(5, TimeUnit.SECONDS); }
        catch (ExecutionException expected) {
            if (!(expected.getCause() instanceof TimeoutException)) throw expected;
        }
        // The controlled resource is deliberately still open; no real completion was signalled.
        held(gate);
    }

    private static void callerScope(boolean shared) throws Exception {
        ConcurrencyGate stable = new ConcurrencyGate(1, true);
        Client first = new Client(stable);
        Client second = new Client(shared ? stable : new ConcurrencyGate(1, true));
        try (ConcurrencyGate.Lease lease = acquire(first.gate)) { held(second.gate); }
        restored(stable);
        passed++;
    }

    private static final class Client {
        final ConcurrencyGate gate;
        Client(ConcurrencyGate gate) { this.gate = gate; }
    }
}
