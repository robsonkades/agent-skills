-- Reproducible adversarial permission fixture, authored but NOT executed in this review.
-- Run only as an administrator in a newly provisioned disposable PostgreSQL 17 instance,
-- in a database named sdo_wave2_validation, with psql -X --set=ON_ERROR_STOP=1.
-- Supply that instance's explicit connection parameters; never use production defaults.
-- Roles are NOLOGIN and exercised by SET ROLE: this checks permission semantics only,
-- not direct authentication, deployment secrets, pools or live-session revocation.
-- All fixture roles/objects and mutations are in one transaction and rolled back.

\set ON_ERROR_STOP on
DO $guard$
BEGIN
    IF current_setting('server_version_num')::integer / 10000 <> 17 THEN
        RAISE EXCEPTION 'Expected PostgreSQL 17';
    END IF;
    IF current_database() <> 'sdo_wave2_validation' THEN
        RAISE EXCEPTION 'Expected the disposable sdo_wave2_validation database';
    END IF;
END;
$guard$;

SELECT version(), current_database(), session_user, current_user;
BEGIN;
CREATE ROLE sdo_wave2_owner NOLOGIN;
CREATE ROLE sdo_wave2_report NOLOGIN;
CREATE SCHEMA sdo_wave2 AUTHORIZATION sdo_wave2_owner;
SET LOCAL ROLE sdo_wave2_owner;
CREATE TABLE sdo_wave2.order_line (
    order_id integer PRIMARY KEY,
    quantity integer NOT NULL CHECK (quantity > 0)
);
INSERT INTO sdo_wave2.order_line VALUES (42, 3);
CREATE VIEW sdo_wave2.billing_order_v1 AS
    SELECT order_id, quantity FROM sdo_wave2.order_line;
GRANT USAGE ON SCHEMA sdo_wave2 TO sdo_wave2_report;
GRANT SELECT ON sdo_wave2.billing_order_v1 TO sdo_wave2_report;
GRANT UPDATE(quantity) ON sdo_wave2.order_line TO sdo_wave2_report;
RESET ROLE;
SET LOCAL ROLE sdo_wave2_report;
SAVEPOINT before_hostile_mutation;

DO $hostile$
DECLARE
    changed integer;
    observed integer;
BEGIN
    IF current_user <> 'sdo_wave2_report'
       OR NOT has_column_privilege('sdo_wave2.order_line', 'quantity', 'UPDATE')
       OR has_column_privilege('sdo_wave2.order_line', 'quantity', 'SELECT')
       OR has_column_privilege('sdo_wave2.order_line', 'order_id', 'SELECT') THEN
        RAISE EXCEPTION 'Unexpected hostile-fixture effective privileges';
    END IF;
    BEGIN
        UPDATE sdo_wave2.order_line SET quantity = quantity WHERE order_id = 42;
        RAISE EXCEPTION 'Read-dependent probe unexpectedly succeeded';
    EXCEPTION WHEN insufficient_privilege THEN
        RAISE NOTICE 'Expected negative-control outcome: read-dependent UPDATE returns 42501';
    END;
    UPDATE sdo_wave2.order_line SET quantity = 7;
    GET DIAGNOSTICS changed = ROW_COUNT;
    SELECT quantity INTO STRICT observed
      FROM sdo_wave2.billing_order_v1 WHERE order_id = 42;
    IF changed <> 1 OR observed <> 7 THEN
        RAISE EXCEPTION 'Literal probe did not expose the unauthorized mutation';
    END IF;
    RAISE NOTICE 'Expected hostile outcome: literal UPDATE changed one synthetic row to 7';
END;
$hostile$;

ROLLBACK TO SAVEPOINT before_hostile_mutation;
RESET ROLE;
SET LOCAL ROLE sdo_wave2_owner;
REVOKE UPDATE(quantity) ON sdo_wave2.order_line FROM sdo_wave2_report;
RESET ROLE;
SET LOCAL ROLE sdo_wave2_report;

DO $corrected$
DECLARE
    observed integer;
BEGIN
    IF current_user <> 'sdo_wave2_report'
       OR has_column_privilege('sdo_wave2.order_line', 'quantity', 'UPDATE') THEN
        RAISE EXCEPTION 'Unexpected corrected-fixture effective privileges';
    END IF;
    BEGIN
        UPDATE sdo_wave2.order_line SET quantity = 7;
        RAISE EXCEPTION 'Literal UPDATE unexpectedly succeeded after revocation';
    EXCEPTION WHEN insufficient_privilege THEN
        RAISE NOTICE 'Expected corrected outcome: literal UPDATE returns 42501';
    END;
    BEGIN
        PERFORM quantity FROM sdo_wave2.order_line WHERE order_id = 42;
        RAISE EXCEPTION 'Private base-table SELECT unexpectedly succeeded';
    EXCEPTION WHEN insufficient_privilege THEN
        RAISE NOTICE 'Expected private-read outcome: base-table SELECT returns 42501';
    END;
    SELECT quantity INTO STRICT observed
      FROM sdo_wave2.billing_order_v1 WHERE order_id = 42;
    IF observed <> 3 THEN
        RAISE EXCEPTION 'Supported view did not preserve the original synthetic row';
    END IF;
    RAISE NOTICE 'Expected allowed outcome: supported view returns original quantity 3';
END;
$corrected$;

RESET ROLE;
ROLLBACK;
\echo Fixture transaction rolled back; compare all expected notices and exit status.
