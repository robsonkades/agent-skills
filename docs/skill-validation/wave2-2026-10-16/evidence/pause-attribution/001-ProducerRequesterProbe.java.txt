import java.nio.file.Path;
import java.time.Duration;
import java.util.List;
import jdk.jfr.Recording;
import jdk.jfr.consumer.RecordedEvent;
import jdk.jfr.consumer.RecordedThread;
import jdk.jfr.consumer.RecordingFile;

public final class ProducerRequesterProbe {
    private static String identity(RecordedThread thread) {
        if (thread == null) return "null";
        return "javaName=" + thread.getJavaName() + ", osName=" + thread.getOSName()
                + ", javaId=" + thread.getJavaThreadId() + ", osId=" + thread.getOSThreadId();
    }

    private static void requireVmProducer(RecordedEvent event) {
        if (event.getThread() == null || !"VM Thread".equals(event.getThread().getOSName())) {
            throw new AssertionError("Unexpected producer: " + identity(event.getThread()));
        }
    }

    public static void main(String[] args) throws Exception {
        Thread.currentThread().setName("requester-control");
        Path recordingPath = Path.of("producer-requester.jfr");
        try (Recording recording = new Recording()) {
            for (String event : List.of("jdk.SafepointBegin", "jdk.SafepointEnd",
                    "jdk.SafepointStateSynchronization", "jdk.ExecuteVMOperation")) {
                recording.enable(event).withThreshold(Duration.ZERO);
            }
            recording.start();
            System.gc();
            recording.stop();
            recording.dump(recordingPath);
        }

        List<RecordedEvent> events = RecordingFile.readAllEvents(recordingPath);
        RecordedEvent operation = events.stream()
                .filter(event -> event.getEventType().getName().equals("jdk.ExecuteVMOperation"))
                .filter(event -> event.getString("operation").equals("G1CollectFull"))
                .findFirst().orElseThrow(() -> new AssertionError("No G1CollectFull event"));
        requireVmProducer(operation);
        RecordedThread caller = operation.getThread("caller");
        if (caller == null || !"requester-control".equals(caller.getJavaName())) {
            throw new AssertionError("Unexpected requester: " + identity(caller));
        }
        long id = operation.getLong("safepointId");
        if (id == 0) throw new AssertionError("No safepoint id");
        System.out.println("runtime=" + Runtime.version());
        System.out.println("operation=G1CollectFull, safepointId=" + id);
        System.out.println("operation.eventThread={" + identity(operation.getThread()) + "}");
        System.out.println("operation.caller={" + identity(caller) + "}");
        for (String eventType : List.of("jdk.SafepointBegin",
                "jdk.SafepointStateSynchronization", "jdk.SafepointEnd")) {
            RecordedEvent cycle = events.stream()
                    .filter(event -> event.getEventType().getName().equals(eventType))
                    .filter(event -> event.getLong("safepointId") == id)
                    .findFirst().orElseThrow(() -> new AssertionError("No matching " + eventType));
            requireVmProducer(cycle);
            System.out.println(eventType + ".eventThread={" + identity(cycle.getThread())
                    + "}, durationNanos=" + cycle.getDuration().toNanos());
        }
        System.out.println("PASS: VM producer differs from requester. No delayed-thread claim tested.");
    }
}
