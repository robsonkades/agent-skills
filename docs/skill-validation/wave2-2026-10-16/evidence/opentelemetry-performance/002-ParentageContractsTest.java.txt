import io.opentelemetry.api.trace.Span;
import io.opentelemetry.api.trace.StatusCode;
import io.opentelemetry.api.trace.Tracer;
import io.opentelemetry.context.Context;
import io.opentelemetry.context.Scope;
import io.opentelemetry.sdk.trace.SdkTracerProvider;
import io.opentelemetry.sdk.trace.data.SpanData;
import io.opentelemetry.sdk.trace.export.SimpleSpanProcessor;
import io.opentelemetry.sdk.testing.exporter.InMemorySpanExporter;
import java.util.ArrayDeque;
import java.util.List;
import java.util.concurrent.*;

public class ParentageContractsTest {
    static final InMemorySpanExporter exported = InMemorySpanExporter.create();
    static final SdkTracerProvider provider = SdkTracerProvider.builder()
        .addSpanProcessor(SimpleSpanProcessor.create(exported)).build();
    static final Tracer tracer = provider.get("parentage-contract-control");
    static int checks;

    static final class QueueExecutor implements Executor {
        final ArrayDeque<Runnable> queue = new ArrayDeque<>();
        public void execute(Runnable task) { queue.add(task); }
        void drain() { while (!queue.isEmpty()) queue.remove().run(); }
    }
    static Span root(String name) { return tracer.spanBuilder(name).setNoParent().startSpan(); }
    static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
    static void same(Object expected, Object actual, String what) {
        if (!expected.equals(actual)) throw new AssertionError(what + ": expected=" + expected + ", actual=" + actual);
    }
    static DeferredRecipe.Price work(String name, boolean fail, int value) {
        Span child = tracer.spanBuilder(name).startSpan();
        try (Scope scope = child.makeCurrent()) {
            if (fail) throw new IllegalStateException("planned-action-failure");
            return new DeferredRecipe.Price(value + 1);
        } catch (RuntimeException error) {
            child.setStatus(StatusCode.ERROR);
            child.recordException(error);
            throw error;
        } finally {
            child.end();
        }
    }
    static SpanData child(String name) {
        List<SpanData> items = exported.getFinishedSpanItems().stream()
            .filter(s -> s.getName().equals(name)).toList();
        same(1, items.size(), "one ended callback span");
        return items.get(0);
    }
    static void verifyParent(Span expected, SpanData child) {
        same(expected.getSpanContext().getSpanId(), child.getParentSpanId(), "callback parent");
        same(expected.getSpanContext().getTraceId(), child.getTraceId(), "callback trace");
    }
    static void outcome(CompletableFuture<DeferredRecipe.Price> result, boolean failure) throws Exception {
        try {
            same(8, result.get(2, TimeUnit.SECONDS).amount, "result");
            require(!failure, "expected action failure");
        } catch (ExecutionException error) {
            require(failure && error.getCause() instanceof IllegalStateException, "unexpected action failure");
        }
    }
    static void one(boolean complete, boolean inline, boolean failure, boolean submissionOnly) throws Exception {
        exported.reset();
        Span request = root("request-A"), completer = root("provider-B"), worker = root("worker-C");
        Context a = Context.root().with(request), b = Context.root().with(completer), c = Context.root().with(worker);
        QueueExecutor queue = new QueueExecutor();
        Executor executor = inline ? Runnable::run : queue;
        CompletableFuture<Integer> source = complete ? CompletableFuture.completedFuture(7) : new CompletableFuture<>();
        CompletableFuture<DeferredRecipe.Price> result;
        try {
            try (Scope scope = a.makeCurrent()) {
                if (submissionOnly) {
                    result = source.thenApplyAsync(value -> work("child", failure, value), Context.taskWrapping(executor));
                } else {
                    result = new DeferredRecipe(value -> work("child", failure, value)).attach(source, executor);
                }
                same(a, Context.current(), "registration context restored");
            }
            request.end(); // Capturing parent Context does not keep its span open.
            try (Scope scope = b.makeCurrent()) {
                if (!complete) source.complete(7);
                same(b, Context.current(), "completer context restored");
            }
            try (Scope scope = c.makeCurrent()) {
                queue.drain();
                same(c, Context.current(), "worker context restored");
            }
            outcome(result, failure);
            verifyParent(submissionOnly && !complete ? completer : request, child("child"));
            same(failure ? StatusCode.ERROR : StatusCode.UNSET, child("child").getStatus().getStatusCode(), "child status");
            same(Context.root(), Context.current(), "outer context restored");
            checks++;
        } finally { request.end(); completer.end(); worker.end(); }
    }
    static void interleaved() throws Exception {
        exported.reset();
        Span a = root("A"), b = root("B"), worker = root("worker");
        QueueExecutor queue = new QueueExecutor();
        CompletableFuture<Integer> inputA = new CompletableFuture<>(), inputB = new CompletableFuture<>();
        CompletableFuture<DeferredRecipe.Price> resultA, resultB;
        try {
            try (Scope scope = a.makeCurrent()) {
                resultA = new DeferredRecipe(value -> work("child-A", false, value)).attach(inputA, queue);
            }
            try (Scope scope = b.makeCurrent()) {
                resultB = new DeferredRecipe(value -> work("child-B", false, value)).attach(inputB, queue);
            }
            try (Scope scope = worker.makeCurrent()) {
                inputB.complete(7); inputA.complete(7); queue.drain();
                same(worker.getSpanContext(), Span.current().getSpanContext(), "shared worker restored");
            }
            outcome(resultA, false); outcome(resultB, false);
            verifyParent(a, child("child-A")); verifyParent(b, child("child-B"));
            checks++;
        } finally { a.end(); b.end(); worker.end(); }
    }
    static void rejected(boolean complete) throws Exception {
        exported.reset();
        Span a = root("rejected-A");
        CompletableFuture<Integer> source = complete ? CompletableFuture.completedFuture(7) : new CompletableFuture<>();
        CompletableFuture<DeferredRecipe.Price> result;
        try {
            try (Scope scope = a.makeCurrent()) {
                result = new DeferredRecipe(value -> work("rejected-child", false, value))
                    .attach(source, task -> { throw new RejectedExecutionException("planned-rejection"); });
            }
            if (!complete) source.complete(7);
            try { result.get(2, TimeUnit.SECONDS); throw new AssertionError("rejection lost"); }
            catch (ExecutionException error) {
                require(error.getCause() instanceof RejectedExecutionException, "wrong rejection cause");
            }
            require(exported.getFinishedSpanItems().isEmpty(), "rejected action ran");
            same(Context.root(), Context.current(), "rejection leaked context");
            checks++;
        } finally { a.end(); }
    }
    public static void main(String[] args) throws Exception {
        try {
            for (boolean complete : new boolean[]{true, false})
                for (boolean inline : new boolean[]{true, false})
                    for (boolean failure : new boolean[]{false, true})
                        one(complete, inline, failure, false);
            one(true, false, false, true);
            one(false, false, false, true);
            interleaved();
            rejected(true); rejected(false);
            System.out.println("PASS " + checks + " contract cases/controls; no agent, collector or performance claim.");
        } finally { provider.close(); }
    }
}
