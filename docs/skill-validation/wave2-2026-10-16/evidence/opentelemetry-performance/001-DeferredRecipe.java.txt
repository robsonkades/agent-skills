import io.opentelemetry.context.Context;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.Executor;
import java.util.function.Function;

final class DeferredRecipe {
    static final class Price {
        final int amount;
        Price(int amount) { this.amount = amount; }
    }
    private final Function<Integer, Price> work;
    DeferredRecipe(Function<Integer, Price> work) { this.work = work; }
    private Price reprice(Integer value) { return work.apply(value); }
    CompletableFuture<Price> attach(CompletableFuture<Integer> source, Executor executor) {
        Context origin = Context.current();
        CompletableFuture<Price> result =
            source.thenApplyAsync(origin.wrapFunction(this::reprice), executor);
        return result;
    }
}
