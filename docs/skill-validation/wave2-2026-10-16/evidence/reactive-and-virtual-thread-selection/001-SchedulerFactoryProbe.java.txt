import java.time.Duration;
import reactor.core.publisher.Mono;
import reactor.core.scheduler.Scheduler;
import reactor.core.scheduler.Schedulers;

public class SchedulerFactoryProbe {
    private static final String PROPERTY = "reactor.schedulers.defaultBoundedElasticOnVirtualThreads";

    private static boolean sample(Scheduler scheduler) {
        return Boolean.TRUE.equals(Mono.fromCallable(() -> Thread.currentThread().isVirtual())
                .subscribeOn(scheduler).block(Duration.ofSeconds(3)));
    }

    public static void main(String[] args) {
        boolean expectedShared = Boolean.parseBoolean(args[0]);
        Scheduler owned = null;
        try {
            boolean shared = sample(Schedulers.boundedElastic());
            owned = Schedulers.newBoundedElastic(2, 4, "owned-jdbc-probe");
            boolean separate = sample(owned);
            System.setProperty(PROPERTY, Boolean.toString(!expectedShared));
            Schedulers.shutdownNow();
            boolean recreatedShared = sample(Schedulers.boundedElastic());
            System.out.printf("{\"runtime\":\"%s\",\"reactor\":\"%s\",\"startupProperty\":%s,"
                            + "\"sharedVirtual\":%s,\"newBoundedElasticVirtual\":%s,"
                            + "\"propertyAfterInitialization\":%s,\"recreatedSharedVirtual\":%s}%n",
                    Runtime.version(), Mono.class.getPackage().getImplementationVersion(),
                    expectedShared, shared, separate, !expectedShared, recreatedShared);
            if (shared != expectedShared || separate || recreatedShared != expectedShared) {
                throw new AssertionError("Scheduler selection differed from the checked contract");
            }
        } finally {
            if (owned != null) owned.dispose();
            Schedulers.shutdownNow();
        }
    }
}
