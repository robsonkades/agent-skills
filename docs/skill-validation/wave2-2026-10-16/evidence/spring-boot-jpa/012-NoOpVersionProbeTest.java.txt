package example.persistence;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertTrue;

import jakarta.persistence.EntityManager;
import jakarta.persistence.EntityManagerFactory;
import jakarta.persistence.LockModeType;
import jakarta.persistence.OptimisticLockException;
import jakarta.persistence.RollbackException;
import org.hibernate.SessionFactory;
import org.hibernate.stat.Statistics;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.context.TestConstructor;

@SpringBootTest(properties = {
        "spring.datasource.url=jdbc:h2:mem:noop-version-probe;DB_CLOSE_DELAY=-1",
        "spring.jpa.hibernate.ddl-auto=create-drop",
        "spring.jpa.properties.hibernate.generate_statistics=true",
        "logging.level.org.hibernate.stat=OFF"
})
@TestConstructor(autowireMode = TestConstructor.AutowireMode.ALL)
class NoOpVersionProbeTest {
    private final InventoryRepository inventory;
    private final EntityManagerFactory entityManagers;
    private final Statistics statistics;

    NoOpVersionProbeTest(InventoryRepository inventory, EntityManagerFactory entityManagers) {
        this.inventory = inventory;
        this.entityManagers = entityManagers;
        this.statistics = entityManagers.unwrap(SessionFactory.class).getStatistics();
    }

    @BeforeEach
    void seed() {
        inventory.deleteAll();
        inventory.saveAndFlush(new Inventory("A", "Unchanged", 20));
        statistics.clear();
    }

    @Test
    void sameValueNormalLoadDoesNotValidateVersionAtCommit() {
        try (EntityManager reader = entityManagers.createEntityManager();
             EntityManager writer = entityManagers.createEntityManager()) {
            reader.getTransaction().begin();
            writer.getTransaction().begin();
            try {
                Inventory observed = reader.find(Inventory.class, "A");
                observed.changeNote(null);
                writer.find(Inventory.class, "A").changeNote("Concurrent writer");
                writer.getTransaction().commit();
                reader.flush();
                reader.getTransaction().commit();
            } finally {
                rollbackIfActive(reader);
                rollbackIfActive(writer);
            }
        }
        assertEquals("Concurrent writer", inventory.findById("A").orElseThrow().getNote());
        assertEquals(1L, statistics.getEntityUpdateCount());
    }

    @Test
    void sameValueOptimisticLoadValidatesVersionAtCompletion() {
        try (EntityManager reader = entityManagers.createEntityManager();
             EntityManager writer = entityManagers.createEntityManager()) {
            reader.getTransaction().begin();
            writer.getTransaction().begin();
            try {
                Inventory observed = reader.find(Inventory.class, "A", LockModeType.OPTIMISTIC);
                observed.changeNote(null);
                writer.find(Inventory.class, "A").changeNote("Concurrent writer");
                writer.getTransaction().commit();
                RollbackException failure = assertThrows(RollbackException.class,
                        reader.getTransaction()::commit);
                assertTrue(hasOptimisticCause(failure));
            } finally {
                rollbackIfActive(reader);
                rollbackIfActive(writer);
            }
        }
        assertEquals("Concurrent writer", inventory.findById("A").orElseThrow().getNote());
        assertEquals(1L, statistics.getEntityUpdateCount());
    }

    @Test
    void changedValueNormalLoadChecksVersionOnWrite() {
        try (EntityManager reader = entityManagers.createEntityManager();
             EntityManager writer = entityManagers.createEntityManager()) {
            reader.getTransaction().begin();
            writer.getTransaction().begin();
            try {
                Inventory observed = reader.find(Inventory.class, "A");
                observed.changeNote("Losing update");
                writer.find(Inventory.class, "A").changeNote("Concurrent writer");
                writer.getTransaction().commit();
                assertThrows(OptimisticLockException.class, reader::flush);
            } finally {
                rollbackIfActive(reader);
                rollbackIfActive(writer);
            }
        }
        assertEquals("Concurrent writer", inventory.findById("A").orElseThrow().getNote());
    }

    @Test
    void optimisticValidationWithoutWriterDoesNotForceIncrement() {
        Long before = inventory.findById("A").orElseThrow().getVersion();
        try (EntityManager reader = entityManagers.createEntityManager()) {
            reader.getTransaction().begin();
            try {
                reader.find(Inventory.class, "A", LockModeType.OPTIMISTIC).changeNote(null);
                reader.getTransaction().commit();
            } finally {
                rollbackIfActive(reader);
            }
        }
        assertEquals(before, inventory.findById("A").orElseThrow().getVersion());
        assertEquals(0L, statistics.getEntityUpdateCount());
    }

    private static void rollbackIfActive(EntityManager manager) {
        if (manager.getTransaction().isActive()) manager.getTransaction().rollback();
    }

    private static boolean hasOptimisticCause(Throwable failure) {
        for (Throwable current = failure; current != null; current = current.getCause()) {
            if (current instanceof OptimisticLockException) return true;
        }
        return false;
    }
}
