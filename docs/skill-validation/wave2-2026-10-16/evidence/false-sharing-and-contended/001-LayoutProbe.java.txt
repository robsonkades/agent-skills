import java.lang.management.ManagementFactory;
import java.lang.reflect.Field;
import com.sun.management.HotSpotDiagnosticMXBean;
import jdk.internal.vm.annotation.Contended;
import sun.misc.Unsafe;

// Review-only offset check. No throughput, coherence or absolute-address claim.
public final class LayoutProbe {
    static final class Plain {
        volatile long produced;
        volatile long consumed;
    }

    static final class Distinct {
        @Contended("producer") volatile long produced;
        @Contended("consumer") volatile long consumed;
    }

    static final class Together {
        @Contended("shared") volatile long produced;
        @Contended("shared") volatile long consumed;
    }

    @Contended
    static final class Whole {
        volatile long produced;
        volatile long consumed;
    }

    static final class StaticAnnotated {
        @Contended("producer") static volatile long produced;
        @Contended("consumer") static volatile long consumed;
    }

    static final class StaticPlain {
        static volatile long produced;
        static volatile long consumed;
    }

    public static void main(String[] args) throws Exception {
        Field singleton = Unsafe.class.getDeclaredField("theUnsafe");
        singleton.setAccessible(true);
        Unsafe unsafe = (Unsafe) singleton.get(null);
        HotSpotDiagnosticMXBean bean = ManagementFactory.getPlatformMXBean(HotSpotDiagnosticMXBean.class);
        System.out.println("runtime=" + System.getProperty("java.runtime.version")
                + " arch=" + System.getProperty("os.arch"));
        for (String option : new String[]{"EnableContended", "RestrictContended", "ContendedPaddingWidth",
                "UseCompactObjectHeaders", "ObjectAlignmentInBytes"}) {
            System.out.println(option + "=" + bean.getVMOption(option).getValue());
        }
        long plain = gap(unsafe, Plain.class, false);
        long distinct = gap(unsafe, Distinct.class, false);
        long together = gap(unsafe, Together.class, false);
        long whole = gap(unsafe, Whole.class, false);
        long statics = gap(unsafe, StaticAnnotated.class, true);
        long staticPlain = gap(unsafe, StaticPlain.class, true);
        check(together == plain, "same-tag fields unexpectedly separated in this build");
        check(whole == plain, "class annotation unexpectedly separated its declared fields");
        check(statics == staticPlain, "static annotation unexpectedly changed the field gap");
        if (args[0].equals("enabled")) {
            check(distinct > plain, "distinct groups were not separated");
        } else if (args[0].equals("ignored")) {
            check(distinct == plain, "distinct groups unexpectedly separated");
        } else {
            throw new IllegalArgumentException("expected enabled or ignored");
        }
        check(Distinct.class.getDeclaredField("produced").getDeclaredAnnotations().length == 1,
                "annotation metadata missing");
        System.out.println("PASS: " + args[0] + " layout controls; runtime export for Contended was not supplied");
    }

    private static long gap(Unsafe unsafe, Class<?> type, boolean isStatic) throws Exception {
        Field first = type.getDeclaredField("produced");
        Field second = type.getDeclaredField("consumed");
        long firstOffset = isStatic ? unsafe.staticFieldOffset(first) : unsafe.objectFieldOffset(first);
        long secondOffset = isStatic ? unsafe.staticFieldOffset(second) : unsafe.objectFieldOffset(second);
        long gap = Math.abs(secondOffset - firstOffset);
        System.out.println(type.getSimpleName() + " produced=" + firstOffset + " consumed=" + secondOffset + " gap=" + gap);
        return gap;
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
