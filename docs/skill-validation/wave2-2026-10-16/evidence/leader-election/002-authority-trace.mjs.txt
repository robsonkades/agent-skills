import assert from 'node:assert/strict';
import { writeFileSync } from 'node:fs';

// Synthetic trace checks, not a live etcd client or a distributed-system proof.
// The sink atomically accepts a term >= its highest accepted term.
function apply(highestTerm, commandTerm) {
  const accepted = commandTerm >= highestTerm;
  return { accepted, highestTerm: accepted ? commandTerm : highestTerm };
}

function mayActivate(participation) {
  return participation.current && participation.acquisition === 'SUCCEEDED';
}

const pending = { current: true, acquisition: 'PENDING', candidateCreationRevision: 42 };
const acquired = { ...pending, acquisition: 'SUCCEEDED' };
const invalidated = { ...acquired, current: false };

const legacyPendingActivation = apply(41, pending.candidateCreationRevision);
const legacyCurrentLeaderWrite = apply(legacyPendingActivation.highestTerm, 41);
const guardedPendingHighest = mayActivate(pending)
  ? apply(41, pending.candidateCreationRevision).highestTerm : 41;
const guardedCurrentLeaderWrite = apply(guardedPendingHighest, 41);
const acquiredActivation = mayActivate(acquired)
  ? apply(41, acquired.candidateCreationRevision) : null;

assert.equal(legacyPendingActivation.accepted, true);
assert.equal(legacyCurrentLeaderWrite.accepted, false);
assert.equal(mayActivate(pending), false);
assert.equal(guardedCurrentLeaderWrite.accepted, true);
assert.equal(acquiredActivation?.accepted, true);
assert.equal(mayActivate(invalidated), false);

// A command admitted under old term 41 is delivered after term 42 activates.
// A later store read returned header revision 100 because unrelated keys changed.
const unboundOldWrite = apply(42, 100);
const successorAfterUnboundOldWrite = apply(unboundOldWrite.highestTerm, 42);
const boundOldWrite = apply(42, 41);
const successorAfterBoundOldWrite = apply(boundOldWrite.highestTerm, 42);
assert.equal(unboundOldWrite.accepted, true);
assert.equal(successorAfterUnboundOldWrite.accepted, false);
assert.equal(boundOldWrite.accepted, false);
assert.equal(successorAfterBoundOldWrite.accepted, true);

const result = {
  classification: 'Executed synthetic counterexample traces; no live store or agent evaluation',
  assumptions: {
    sink: 'atomic highest-seen-term acceptance, equal terms allowed',
    term41: 'previously acquired by A',
    term42: 'candidate revision of B; activation allowed only after acquisition',
    header100: 'store read revision unrelated to a new application grant'
  },
  pendingCandidate: { legacyPendingActivation, legacyCurrentLeaderWrite, guardedCurrentLeaderWrite },
  acquiredCandidate: acquiredActivation,
  invalidatedAcquisitionAccepted: mayActivate(invalidated),
  delayedOldCommand: { unboundOldWrite, successorAfterUnboundOldWrite, boundOldWrite, successorAfterBoundOldWrite },
  assertionsPassed: 10
};
writeFileSync('authority-trace-results.json', JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify(result, null, 2));
