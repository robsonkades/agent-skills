import org.openjdk.jol.info.ClassLayout;

public class PrintableProbe {
    static final class Sample {
        long value = 42L;
    }
    record Pair(int x, int y) {}

    public static void main(String[] args) {
        boolean sourceLaunch = args[0].equals("source");
        int checks = 0;
        for (Object obj : new Object[] { new Sample(), new Pair(3, 5) }) {
            ClassLayout layout = ClassLayout.parseInstance(obj);
            long bytes = layout.instanceSize();
            String offsets = layout.toPrintable(null);
            if (!offsets.contains("Instance size: " + bytes + " bytes")) throw new AssertionError("size");
            checks++;
            String field = obj instanceof Sample ? "value" : "x";
            if (!offsets.contains("." + field)) throw new AssertionError("field metadata");
            checks++;
            if (!offsets.contains("N/A")) throw new AssertionError("values must be unavailable");
            checks++;
            boolean expectedFailure = false;
            try {
                String actual = layout.toPrintable();
                if (sourceLaunch) throw new AssertionError("expected loader failure");
                if (!actual.contains("." + field)) throw new AssertionError("normal field metadata");
                String fieldLine = actual.lines().filter(line -> line.contains("." + field)).findFirst().orElseThrow();
                String expectedValue = obj instanceof Sample ? "42" : "3";
                if (!fieldLine.stripTrailing().endsWith(expectedValue)) throw new AssertionError("missing actual value");
            } catch (IllegalArgumentException exception) {
                if (!sourceLaunch || !exception.getMessage().contains("Class is not found")) throw exception;
                expectedFailure = true;
            }
            if (sourceLaunch != expectedFailure) throw new AssertionError("loader outcome");
            checks++;
            if (layout.instanceSize() != bytes) throw new AssertionError("layout changed");
            checks++;
            System.out.println(obj.getClass().getName() + " size=" + bytes + " sourceLaunch=" + sourceLaunch);
        }
        System.out.println("PASS checks=" + checks);
    }
}
