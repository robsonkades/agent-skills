import java.nio.ByteBuffer;
import java.nio.channels.ClosedByInterruptException;
import java.nio.channels.ClosedChannelException;
import java.nio.channels.FileChannel;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.util.concurrent.atomic.AtomicReference;

public final class ChannelOwnershipProbe {
    private record Observation(int read, boolean closedByInterrupt, boolean interruptSet) {}

    public static void main(String[] args) throws Exception {
        Path data = Path.of(args[0]).resolve("payload.bin");
        Files.write(data, new byte[] {42, 43, 44});
        for (boolean virtual : new boolean[] {false, true}) {
            String mode = virtual ? "virtual" : "platform";
            try (FileChannel shared = open(data)) {
                Observation result = read(shared, false, virtual);
                require(result.read() == 1 && !result.closedByInterrupt(), "normal read failed");
                require(!result.interruptSet() && shared.isOpen(), "normal read changed status/channel");
                require(shared.read(ByteBuffer.allocate(1), 0) == 1, "second reader cannot read control");
                System.out.println("PASS " + mode + ": no-interrupt shared-channel control");
            }
            try (FileChannel shared = open(data)) {
                Observation result = read(shared, true, virtual);
                require(result.closedByInterrupt() && result.interruptSet(), "interrupt contract not observed");
                require(!shared.isOpen(), "borrowed interruptible channel unexpectedly remains open");
                try {
                    shared.read(ByteBuffer.allocate(1), 0);
                    throw new AssertionError("second reader unexpectedly used the invalidated channel");
                } catch (ClosedChannelException expected) {
                    System.out.println("PASS " + mode + ": interrupted borrower invalidates shared channel");
                }
            }
            try (FileChannel retained = open(data); FileChannel taskOwned = open(data)) {
                Observation result = read(taskOwned, true, virtual);
                require(result.closedByInterrupt() && result.interruptSet(), "owned interrupt contract not observed");
                require(!taskOwned.isOpen(), "task-owned channel did not close");
                require(retained.isOpen() && retained.read(ByteBuffer.allocate(1), 0) == 1,
                        "independent channel was invalidated");
                System.out.println("PASS " + mode + ": separate task-owned channel preserves other reader");
            }
        }
        System.out.println("PASS 6 cases; initial-interrupt contract only, no blocking-latency claim");
    }

    private static FileChannel open(Path data) throws Exception {
        return FileChannel.open(data, StandardOpenOption.READ);
    }

    private static Observation read(FileChannel channel, boolean interrupt, boolean virtual) throws Exception {
        AtomicReference<Observation> outcome = new AtomicReference<>();
        AtomicReference<Throwable> failure = new AtomicReference<>();
        Runnable action = () -> {
            try {
                if (interrupt) Thread.currentThread().interrupt();
                int bytes = channel.read(ByteBuffer.allocate(1), 0);
                outcome.set(new Observation(bytes, false, Thread.currentThread().isInterrupted()));
            } catch (ClosedByInterruptException expected) {
                outcome.set(new Observation(-1, true, Thread.currentThread().isInterrupted()));
            } catch (Throwable unexpected) {
                failure.set(unexpected);
            }
        };
        Thread worker = virtual ? Thread.ofVirtual().unstarted(action)
                : Thread.ofPlatform().daemon().unstarted(action);
        worker.start();
        worker.join(5_000);
        if (worker.isAlive()) throw new AssertionError("read exceeded bounded join; child process watchdog applies");
        if (failure.get() != null) throw new AssertionError("worker failed", failure.get());
        require(outcome.get() != null, "worker produced no observation");
        return outcome.get();
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
