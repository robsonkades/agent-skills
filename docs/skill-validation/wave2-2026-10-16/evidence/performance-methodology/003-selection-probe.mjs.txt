import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';

// Synthetic finite populations, not application timings or a production sampler.
// Sequence thinning preserves each latency stratum exactly in this fixture only.
const configPath = process.argv[2];
if (!configPath) throw new Error('Usage: node selection-probe.mjs <config.json> [--require-preserved-p99]');
const config = JSON.parse(readFileSync(configPath, 'utf8'));
if (!['sequence', 'duration'].includes(config.treatmentSelection)) {
  throw new Error('Unknown treatmentSelection');
}
const rows = ['A', 'B'].flatMap(arm => Array.from({ length: 1000 }, (_, sequence) => ({
  arm,
  sequence,
  durationMs: sequence < 100 ? (arm === 'A' ? 400 : 200) : 10,
  successful: true,
})));
const selected = rows.filter(row =>
  row.arm === 'A' || config.treatmentSelection === 'sequence'
    ? row.sequence % 10 !== 0
    : row.durationMs < 100
);
const nearestRank = (values, p) => [...values].sort((a, b) => a - b)[Math.ceil(values.length * p) - 1];
const summarize = population => Object.fromEntries(['A', 'B'].map(arm => {
  const armRows = population.filter(row => row.arm === arm);
  return [arm, {
    count: armRows.length,
    successful: armRows.filter(row => row.successful).length,
    p99Ms: nearestRank(armRows.map(row => row.durationMs), 0.99),
  }];
}));
const full = summarize(rows);
const retained = summarize(selected);
const report = {
  selection: config.treatmentSelection,
  quantileEstimator: 'empirical nearest rank, ceil(n * .99)',
  full,
  retained,
  sameRetainedCount: retained.A.count === retained.B.count,
  capturedP99Reduction: 1 - retained.B.p99Ms / retained.A.p99Ms,
  fullPopulationP99Reduction: 1 - full.B.p99Ms / full.A.p99Ms,
};
assert.equal(full.A.count, 1000);
assert.equal(full.B.count, 1000);
assert.equal(retained.A.count, 900);
assert.equal(retained.B.count, 900);
console.log(JSON.stringify(report, null, 2));
if (process.argv.includes('--require-preserved-p99')) {
  assert.equal(retained.A.p99Ms, full.A.p99Ms, 'control selection changed finite-population p99');
  assert.equal(retained.B.p99Ms, full.B.p99Ms, 'treatment selection changed finite-population p99 despite balanced counts');
}
