import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';

const temp = dirname(fileURLToPath(import.meta.url));
const repo = resolve(process.argv[2] ?? 'C:/git/agent-skills');
const blocks = file => [...readFileSync(resolve(repo, 'skills/timeouts-and-deadlines', file), 'utf8')
  .matchAll(/```java\r?\n([\s\S]*?)```/g)].map(match => match[1]);
const helper = blocks('references/deadline-propagation.md').find(code => code.includes('public final class Deadline'));
const waiter = blocks('references/java-timeout-surface.md').find(code => code.includes('shared.copy()'));
assert.ok(helper && waiter, 'Expected actual shipped examples');
mkdirSync(resolve(temp, 'classes'), { recursive: true });
writeFileSync(resolve(temp, 'Deadline.java'), helper);
writeFileSync(resolve(temp, 'Waiter.java'), `import java.time.Duration;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;
final class Waiter {
  static <T> CompletableFuture<T> attach(CompletableFuture<T> shared, Duration budget) {
${waiter}
  }
}
`);
const checks = [];
for (const [command, args] of [
  ['javac', ['--release', '11', '-d', resolve(temp, 'classes'), resolve(temp, 'Deadline.java'), resolve(temp, 'Waiter.java'), resolve(temp, 'Probe.java')]],
  ['java', ['-cp', resolve(temp, 'classes'), 'Probe']]
]) {
  const result = spawnSync(command, args, { cwd: temp, encoding: 'utf8', timeout: 15000 });
  checks.push({ command, args, cwd: temp, status: result.status, signal: result.signal,
    stdout: result.stdout, stderr: result.stderr, error: result.error?.message });
  writeFileSync(resolve(temp, 'technical-results.json'), JSON.stringify(checks, null, 2));
  console.log(JSON.stringify(checks.at(-1), null, 2));
  assert.equal(result.status, 0, `${command} failed`);
}
