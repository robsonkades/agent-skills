import java.nio.file.Path;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.Map;
import jdk.jfr.DataAmount;
import jdk.jfr.Enabled;
import jdk.jfr.Event;
import jdk.jfr.EventType;
import jdk.jfr.FlightRecorder;
import jdk.jfr.Name;
import jdk.jfr.Recording;
import jdk.jfr.StackTrace;
import jdk.jfr.Timespan;
import jdk.jfr.ValueDescriptor;
import jdk.jfr.consumer.RecordedEvent;
import jdk.jfr.consumer.RecordingFile;

public final class SameNameEvidenceCheck {
    static final String EVENT_NAME = "com.example.Measurement";

    @Name(EVENT_NAME)
    @Enabled(false)
    @StackTrace(false)
    public static final class ByteMeasurement extends Event {
        @DataAmount(DataAmount.BYTES)
        public long value;
    }

    @Name(EVENT_NAME)
    @Enabled(false)
    @StackTrace(false)
    public static final class TimeMeasurement extends Event {
        @Timespan(Timespan.NANOSECONDS)
        public long value;
    }

    public static void main(String[] args) throws Exception {
        Thread deadline = new Thread(() -> {
            try {
                Thread.sleep(30_000);
                System.err.println("Validation exceeded its 30-second process deadline");
                Runtime.getRuntime().halt(124);
            } catch (InterruptedException finished) {
                Thread.currentThread().interrupt();
            }
        });
        deadline.setDaemon(true);
        deadline.start();
        try {
            Path directory = Path.of(args[0]);
            FlightRecorder.register(ByteMeasurement.class);
            FlightRecorder.register(TimeMeasurement.class);
            long byteId = EventType.getEventType(ByteMeasurement.class).getId();
            long timeId = EventType.getEventType(TimeMeasurement.class).getId();
            if (byteId == timeId) throw new AssertionError("Event IDs must distinguish types");
            printCatalogue();
            for (String mode : new String[] {"name", "class", "id"}) {
                Path recordingPath = directory.resolve(mode + ".jfr");
                record(mode, byteId, recordingPath);
                inspect(mode, recordingPath);
            }
            System.out.println("PASS: name enables both schemas; class/id isolate the requested type; mixed-unit aggregate rejected");
        } finally {
            deadline.interrupt();
        }
    }

    private static void printCatalogue() {
        FlightRecorder.getFlightRecorder().getEventTypes().stream()
            .filter(type -> type.getName().equals(EVENT_NAME))
            .sorted(Comparator.comparing(EventType::getName))
            .forEach(type -> {
                System.out.println(type.getName() + " id=" + type.getId()
                    + " enabled=" + type.isEnabled());
                type.getSettingDescriptors().forEach(setting ->
                    System.out.println("  setting " + setting.getName()
                        + " default=" + setting.getDefaultValue()));
                type.getFields().forEach(field ->
                    System.out.println("  field " + field.getName()
                        + " type=" + field.getTypeName()
                        + " content=" + field.getContentType()));
            });
    }

    private static void record(String mode, long byteId, Path destination) throws Exception {
        try (Recording recording = new Recording()) {
            switch (mode) {
                case "name": recording.enable(EVENT_NAME).withoutStackTrace(); break;
                case "class": recording.enable(ByteMeasurement.class).withoutStackTrace(); break;
                case "id": recording.enable(Long.toString(byteId)).withoutStackTrace(); break;
                default: throw new IllegalArgumentException(mode);
            }
            recording.start();
            ByteMeasurement bytes = new ByteMeasurement();
            bytes.value = 1024L;
            bytes.commit();
            TimeMeasurement time = new TimeMeasurement();
            time.value = 7L;
            time.commit();
            recording.stop();
            recording.dump(destination);
        }
    }

    private static void inspect(String mode, Path recordingPath) throws Exception {
        Map<Long, String> types = new LinkedHashMap<>();
        long matches = 0;
        long byteEvents = 0;
        long incompatible = 0;
        long naiveSum = 0;
        long validBytes = 0;
        try (RecordingFile file = new RecordingFile(recordingPath)) {
            while (file.hasMoreEvents()) {
                RecordedEvent event = file.readEvent();
                EventType type = event.getEventType();
                if (!type.getName().equals(EVENT_NAME)) continue;
                matches++;
                ValueDescriptor field = type.getField("value");
                types.put(type.getId(), field.getTypeName() + "/" + field.getContentType());
                naiveSum += event.getLong("value");
                DataAmount unit = field.getAnnotation(DataAmount.class);
                boolean bytesSchema = field.getTypeName().equals("long")
                    && unit != null && unit.value().equals(DataAmount.BYTES);
                if (!bytesSchema) {
                    incompatible++;
                    continue;
                }
                byteEvents++;
                validBytes += event.getLong("value");
            }
        }
        boolean mixed = mode.equals("name");
        if (matches != (mixed ? 2 : 1) || byteEvents != 1 || incompatible != (mixed ? 1 : 0)
                || types.size() != (mixed ? 2 : 1) || validBytes != 1024
                || naiveSum != (mixed ? 1031 : 1024)) {
            throw new AssertionError("Unexpected capture for " + mode + ": " + types
                + ", matches=" + matches + ", byteEvents=" + byteEvents
                + ", incompatible=" + incompatible + ", naiveSum=" + naiveSum);
        }
        String status = incompatible > 0 ? "REJECTED" : "ACCEPTED";
        System.out.println("mode=" + mode + " types=" + types + " count=" + matches
            + " byteEvents=" + byteEvents + " incompatible=" + incompatible
            + " naiveSum=" + naiveSum + " strictAggregate=" + status);
    }
}
