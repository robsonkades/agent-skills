import com.tngtech.archunit.core.importer.ClassFileImporter;
import com.tngtech.archunit.lang.ArchRule;
import java.nio.file.Files;
import java.nio.file.Path;
import javax.tools.ToolProvider;

public final class SqlBoundaryProbe {
    public static void main(String[] args) throws Exception {
        Path root = Path.of(args[0]);
        Path source = root.resolve("src/main/java/example/clean/application/SqlLeak.java");
        Path output = root.resolve("target/classes");
        Files.createDirectories(source.getParent());
        Files.createDirectories(output);
        Files.writeString(source, "package example.clean.application;"
                + " public class SqlLeak { public java.sql.Connection connection; }");
        int compiled = ToolProvider.getSystemJavaCompiler().run(null, null, null,
                "--release", "25", "-d", output.toString(), source.toString());
        if (compiled != 0) throw new AssertionError("Negative control did not compile");
        var field = Class.forName("example.clean.BoundaryTest").getDeclaredField("APPLICATION");
        field.setAccessible(true);
        var existingRule = (ArchRule) field.get(null);
        var imported = new ClassFileImporter().importPath(output);
        try {
            existingRule.check(imported);
        } catch (AssertionError expected) {
            if (!expected.getMessage().contains("java.sql.Connection")
                    || !expected.getMessage().contains("SqlLeak")) {
                throw new AssertionError("Wrong reason for rule failure", expected);
            }
            System.out.println("PASS: compiled java.sql.Connection signature rejected by existing APPLICATION rule.");
            System.out.println(expected.getMessage());
            return;
        }
        throw new AssertionError("Existing rule accepted a java.sql.Connection signature");
    }
}
