import java.lang.reflect.Field;
import java.util.*;

/** Isolated verifier experiment, not a proposed production implementation. */
public final class RoutingAgreementControl {
    private static final int V = 1024;
    private static final int K = 10_000;

    private static HashRing ring(String... nodes) {
        HashRing result = new HashRing(V);
        for (String node : nodes) result.add(node, 1);
        return result;
    }

    private static Object points(HashRing ring) throws ReflectiveOperationException {
        Field field = HashRing.class.getDeclaredField("ring");
        field.setAccessible(true);
        return field.get(ring);
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    public static void main(String[] args) throws ReflectiveOperationException {
        HashRing baseline = ring("n1", "n2", "n3", "n4");
        HashRing candidate = ring("n4", "n3", "n2", "n1");
        HashRing joined = ring("n1", "n2", "n3", "n4", "n5");
        require(points(baseline).equals(points(candidate)), "node-position agreement failed");

        int primaryDifferences = 0;
        int replicaDifferences = 0;
        int rawMoves = 0;
        int prefixedMoves = 0;
        String witness = null;
        for (int i = 0; i < K; i++) {
            String key = "user:" + i;
            String expectedOwner = baseline.owner(key);
            List<String> expectedReplicas = baseline.owners(key, 3);
            require(candidate.owner(key).equals(expectedOwner), "unchanged serializer disagrees");
            require(candidate.owners(key, 3).equals(expectedReplicas), "unchanged replicas disagree");
            require(expectedReplicas.get(0).equals(expectedOwner), "replica order lost primary");
            require(new HashSet<>(expectedReplicas).size() == 3, "replicas are not distinct");

            // Mutation is in the caller's key serializer. Ring points and hash function stay fixed.
            String prefixed = "cache:" + key;
            String changedOwner = candidate.owner(prefixed);
            if (!expectedOwner.equals(changedOwner)) {
                primaryDifferences++;
                if (witness == null) witness = key + ": " + expectedOwner + " -> " + changedOwner
                        + "; replicas " + expectedReplicas + " -> " + candidate.owners(prefixed, 3);
            }
            if (!expectedReplicas.equals(candidate.owners(prefixed, 3))) replicaDifferences++;

            if (!expectedOwner.equals(joined.owner(key))) {
                rawMoves++;
                require(joined.owner(key).equals("n5"), "raw-key join breaks monotonicity");
            }
            if (!changedOwner.equals(joined.owner(prefixed))) {
                prefixedMoves++;
                require(joined.owner(prefixed).equals("n5"), "prefixed-key join breaks monotonicity");
            }
        }
        require(primaryDifferences > 0, "negative control did not exercise primary disagreement");
        require(replicaDifferences > 0, "negative control did not exercise replica disagreement");
        require(Math.abs(rawMoves / (double) K - 0.2) <= 0.02, "raw fixture outside declared movement tolerance");
        require(Math.abs(prefixedMoves / (double) K - 0.2) <= 0.02, "prefixed fixture outside declared movement tolerance");

        System.out.println("PASS unchanged node-point maps and raw-key primary/ordered-replica agreement: " + K + " keys");
        System.out.println("PASS raw and prefixed clients both preserve join monotonicity");
        System.out.println("Join movement: raw=" + rawMoves + "/" + K + ", prefixed=" + prefixedMoves + "/" + K);
        System.out.println("DETECTED serializer control: primary mismatches=" + primaryDifferences + "/" + K
                + ", ordered-replica mismatches=" + replicaDifferences + "/" + K);
        System.out.println("Witness " + witness);
    }
}
