import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';

// A sequential counterexample, not a CDC connector, production sink, or atomicity proof.
const limited = JSON.parse(readFileSync(new URL('./capture-primary-only.json', import.meta.url), 'utf8'));
const complete = JSON.parse(readFileSync(new URL('./capture-with-routing.json', import.meta.url), 'utf8'));
const clone = value => JSON.parse(JSON.stringify(value));
const owner = row => {
  if (!row || (row.tenant_id !== 'A' && row.tenant_id !== 'B')) {
    throw new Error('MISSING_OLD_PLACEMENT');
  }
  return row.tenant_id;
};

class TargetModel {
  constructor(snapshot) {
    this.stores = { A: new Map(), B: new Map() };
    this.checkpoint = snapshot.position;
    this.write(owner(snapshot.row), snapshot.row.id, snapshot.position, snapshot.row);
  }
  write(shard, id, position, row) {
    const previous = this.stores[shard].get(id);
    if (!previous || position > previous.position) {
      this.stores[shard].set(id, {position, row: row ? clone(row) : null});
    }
  }
  apply(event) {
    // Resolve all required destinations before changing this in-memory model.
    // Real deployments still need their actual transaction/recovery protocol.
    const oldOwner = owner(event.before);
    const newOwner = event.after ? owner(event.after) : undefined;
    if (oldOwner !== newOwner) this.write(oldOwner, event.key.id, event.position, null);
    if (newOwner !== undefined) {
      this.write(newOwner, event.key.id, event.position, event.after);
    }
    this.checkpoint = Math.max(this.checkpoint, event.position);
  }
  visible() {
    return Object.fromEntries(Object.entries(this.stores).map(([shard, rows]) => [
      shard, [...rows.values()].filter(value => value.row !== null).map(value => value.row)
    ]));
  }
}

const blocked = new TargetModel(limited.sourceSnapshot[0]);
const beforeFailure = blocked.visible();
assert.throws(() => blocked.apply(limited.events[0]), /MISSING_OLD_PLACEMENT/);
assert.deepEqual(blocked.visible(), beforeFailure);
assert.equal(blocked.checkpoint, 10);
console.log('primary-only: move BLOCKED; checkpoint=10; A retains its last known row');

// Same deleted id can have belonged to either owner. The event key alone does not distinguish them.
const oldA = {id:7,tenant_id:'A',amount:50};
const oldB = {id:7,tenant_id:'B',amount:50};
const deleteProjection = old => ({key:{id:old.id},op:'d',before:{id:old.id},after:null});
assert.deepEqual(deleteProjection(oldA), deleteProjection(oldB));
assert.notEqual(owner(oldA), owner(oldB));
console.log('primary-only delete: two different prior owners yield identical routing input');

// Negative control: using only the new owner performs an upsert while leaving the old row visible.
const afterOnly = new TargetModel(complete.sourceSnapshot[0]);
const move = complete.events[0];
afterOnly.write(owner(move.after), move.key.id, move.position, move.after);
assert.equal(afterOnly.visible().A.length, 1);
assert.equal(afterOnly.visible().B.length, 1);
console.log('after-only negative control: ghost copy exists on A and new copy exists on B');

const routed = new TargetModel(complete.sourceSnapshot[0]);
routed.apply(complete.events[0]);
assert.deepEqual(routed.visible(), {A:[], B:[{id:7,tenant_id:'B',amount:50}]});
routed.apply(complete.events[1]);
assert.deepEqual(routed.visible(), {A:[], B:[]});
routed.apply(complete.events[1]); // duplicate delete
routed.apply(complete.events[0]); // old update after delete
const oldSnapshot = complete.sourceSnapshot[0];
routed.write(owner(oldSnapshot.row), oldSnapshot.row.id, oldSnapshot.position, oldSnapshot.row);
assert.deepEqual(routed.visible(), {A:[], B:[]});
assert.equal(routed.checkpoint,30);
assert.deepEqual(routed.stores.A.get(7),{position:20,row:null});
assert.deepEqual(routed.stores.B.get(7),{position:30,row:null});
console.log('old-routing control: move/delete/replay leave A=[], B=[]; tombstones A=20, B=30');
console.log('PASS: capture identity changes whether the proposed routed application is determined');
