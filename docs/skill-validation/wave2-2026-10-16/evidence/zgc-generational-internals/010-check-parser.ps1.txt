$ErrorActionPreference = 'Stop'
$env:TMPDIR = '/c/Users/robso/AppData/Local/Temp/agent-skills-wave2-zgc-generational-internals-aaa547384a1d48049b0811dd73c11b19/bash-tmp'
$bash = 'C:/Program Files/Git/bin/bash.exe'
$cases = @(
    @{name='actual'; input='zgc.log'; expected=0},
    @{name='empty'; input='empty.log'; expected=0},
    @{name='malformed'; input='malformed.log'; expected=1},
    @{name='unknown-prefix'; input='unknown-prefix.log'; expected=1},
    @{name='missing'; input='missing.log'; expected=1}
)
foreach ($case in $cases) {
    $result = & $bash --noprofile --norc -c 'export PATH=/usr/bin:/bin; exec bash "$@"' -- extract-pauses.sh $case.input 2>&1
    $status = $LASTEXITCODE
    $result | Set-Content -LiteralPath ('parser-' + $case.name + '.log')
    [pscustomobject]@{name=$case.name; exitCode=$status; expected=$case.expected; output=($result -join [Environment]::NewLine)} | ConvertTo-Json -Compress
    if ($status -ne $case.expected) { throw ('Unexpected parser exit for ' + $case.name) }
}
exit 0
