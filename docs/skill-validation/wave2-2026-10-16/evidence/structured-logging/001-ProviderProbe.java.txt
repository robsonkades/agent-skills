import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class ProviderProbe {
  public static void main(String[] args) {
    Logger logger = LoggerFactory.getLogger("probe.service");
    String factory = LoggerFactory.getILoggerFactory().getClass().getName();
    boolean infoEnabled = logger.isInfoEnabled();
    System.out.println("java=" + System.getProperty("java.runtime.version"));
    System.out.println("factory=" + factory);
    System.out.println("logger=" + logger.getClass().getName());
    System.out.println("infoEnabled=" + infoEnabled);
    if (!factory.equals(args[0])) throw new AssertionError("Unexpected factory: " + factory);
    if (infoEnabled != Boolean.parseBoolean(args[1])) {
      throw new AssertionError("Unexpected enabled state: " + infoEnabled);
    }
    logger.atInfo().setMessage("provider-probe-event")
        .addKeyValue("event.name", "probe.provider_selected").log();
    System.out.println("loggingCallReturned=true");
  }
}
