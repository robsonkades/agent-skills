  name: candidate-verification

  args:
    - name: release-id
  metrics:
    - name: verified-orders

      count: 3

      provider:
        prometheus:
          address: http://prometheus.monitoring.svc.cluster.local:9090
          query: candidate_release_verification{release="{{args.release-id}}"}
