import java.util.concurrent.StructuredTaskScope;
import java.util.concurrent.StructureViolationException;

// Java 25 preview only for StructuredTaskScope, not for ScopedValue itself.
public final class StructuredInheritanceControl {
    private static final ScopedValue<String> KEY = ScopedValue.newInstance();

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    public static void main(String[] args) throws Exception {
        ScopedValue.where(KEY, "captured-at-open").call(() -> {
            try (var scope = StructuredTaskScope.open()) {
                var child = scope.fork(() -> KEY.get());
                scope.join();
                check("captured-at-open".equals(child.get()), "structured child lost binding");
            }
            try (var scope = StructuredTaskScope.open()) {
                boolean rejected;
                try {
                    ScopedValue.where(KEY, "rebound").run(() -> scope.fork(() -> KEY.get()));
                    rejected = false;
                } catch (StructureViolationException expected) {
                    rejected = true;
                }
                check(rejected, "rebound fork accepted");
                scope.join();
            }
            check("captured-at-open".equals(KEY.get()), "outer binding not restored");
            return null;
        });
        check(!KEY.isBound(), "binding survived scope");
        System.out.println("PASS: structured inheritance and rebound-fork rejection; runtime="
                + Runtime.version());
    }
}
