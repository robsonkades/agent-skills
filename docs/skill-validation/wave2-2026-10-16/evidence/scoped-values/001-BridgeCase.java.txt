import java.util.List;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.FutureTask;
import java.util.concurrent.TimeUnit;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContext;
import org.springframework.security.core.context.SecurityContextHolder;

// Minimal request-to-virtual-thread bridge. The launch file selects holder strategy.
public final class BridgeCase {
    private static final ScopedValue<String> SUBJECT = ScopedValue.newInstance();
    private static final class WorkFailure extends RuntimeException {}

    private static UsernamePasswordAuthenticationToken auth(String name) {
        return UsernamePasswordAuthenticationToken.authenticated(name, "fixture-only", List.of());
    }

    static Runnable capture(Runnable work) {
        String subject = SUBJECT.get();
        return () -> ScopedValue.where(SUBJECT, subject).run(() -> {
            SecurityContext previous = SecurityContextHolder.getContext();
            previous.setAuthentication(auth(SUBJECT.get()));
            try {
                work.run();
            } finally {
                SecurityContextHolder.setContext(previous);
            }
        });
    }

    static void exercise(boolean exceptional) throws Exception {
        SecurityContext parent = SecurityContextHolder.createEmptyContext();
        parent.setAuthentication(auth("parent"));
        SecurityContextHolder.setContext(parent);
        Runnable work = ScopedValue.where(SUBJECT, "worker").call(() -> capture(() -> {
            SecurityContextHolder.getContext().setAuthentication(auth("worker-after"));
            if (exceptional) throw new WorkFailure();
        }));
        FutureTask<Void> result = new FutureTask<>(work, null);
        Thread child = Thread.ofVirtual().start(result);
        try {
            try {
                result.get(5, TimeUnit.SECONDS);
                if (exceptional) throw new AssertionError("missing work exception");
            } catch (ExecutionException failure) {
                if (!exceptional || !(failure.getCause() instanceof WorkFailure)) throw failure;
            }
        } finally {
            child.interrupt();
            child.join(5000);
            if (child.isAlive()) throw new AssertionError("child still running");
        }
        System.out.println("exceptional=" + exceptional
                + " parent=" + parent.getAuthentication().getName());
        if (!"parent".equals(parent.getAuthentication().getName())) {
            throw new AssertionError("child changed caller authentication");
        }
        SecurityContextHolder.clearContext();
    }

    public static void main(String[] args) throws Exception {
        System.out.println("strategy="
                + SecurityContextHolder.getContextHolderStrategy().getClass().getName());
        exercise(false);
        exercise(true);
    }
}
