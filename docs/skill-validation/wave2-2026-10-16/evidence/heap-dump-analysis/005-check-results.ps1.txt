$ErrorActionPreference = 'Stop'
$validationRoot = $PSScriptRoot
$referenceHash = (Get-FileHash -LiteralPath "$validationRoot/source.hprof" -Algorithm SHA256).Hash
$results = @()
foreach ($condition in @('complete', 'discard', 'discard-keep')) {
    if ((Get-FileHash -LiteralPath "$validationRoot/$condition.hprof" -Algorithm SHA256).Hash -ne $referenceHash) {
        throw "Capture changed between analyses: $condition"
    }
    $histogram = Import-Csv -LiteralPath "$validationRoot/$condition-histogram/pages/Query_Command2.csv"
    $unreachable = Import-Csv -LiteralPath "$validationRoot/$condition-unreachable/pages/Query_Command2.csv"
    $indexedPayload = @($histogram | Where-Object { $_.'Class Name' -eq 'DumpFixture$Payload' })
    $unreachablePayload = @($unreachable | Where-Object { $_.'Class Name' -eq 'DumpFixture$Payload' })
    $payloadArray = @($histogram | Where-Object { $_.'Class Name' -eq 'DumpFixture$Payload[]' })
    $indexedCount = if ($indexedPayload.Count -eq 1) { [int]$indexedPayload[0].Objects } else { 0 }
    $unreachableCount = if ($unreachablePayload.Count -eq 1) { [int]$unreachablePayload[0].Objects } else { 0 }
    $expectedIndexed = if ($condition -eq 'complete') { 128 } else { 0 }
    $expectedUnreachable = if ($condition -eq 'complete') { 0 } else { 128 }
    if ($indexedCount -ne $expectedIndexed -or $unreachableCount -ne $expectedUnreachable) {
        throw "$condition observed indexed=$indexedCount unreachable=$unreachableCount; expected indexed=$expectedIndexed unreachable=$expectedUnreachable"
    }
    if ($payloadArray.Count -ne 1 -or [int]$payloadArray[0].Objects -ne 1) {
        throw "$condition missing retained array positive control"
    }
    $results += [PSCustomObject]@{
        condition=$condition
        indexedPayloads=$indexedCount
        unreachableHistogramPayloads=$unreachableCount
        indexedPayloadArrays=1
        sameCaptureSha256=$referenceHash
        result='PASS'
    }
}
$results | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath "$validationRoot/results.json" -Encoding utf8
$results | Format-Table condition,indexedPayloads,unreachableHistogramPayloads,indexedPayloadArrays,result
