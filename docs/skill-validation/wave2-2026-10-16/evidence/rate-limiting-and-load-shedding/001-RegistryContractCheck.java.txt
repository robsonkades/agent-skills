import java.io.Reader;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Iterator;
import java.util.LinkedHashMap;
import java.util.Locale;
import java.util.Map;
import java.util.Properties;
import java.util.Set;

/** Isolated fixture. The three canonical identities and one process are fixed for the run. */
public final class RegistryContractCheck {
    private static final Set<String> IDENTITIES = Set.of("A", "B", "C");
    private static final long CAPACITY = 10;
    private static final double REFILL_PER_SECOND = 1.0 / 3600.0;

    static final class Registry {
        private final int maximumKeys;
        private final LinkedHashMap<String, TokenBucket> entries =
                new LinkedHashMap<>(4, 0.75f, true);

        Registry(int maximumKeys) {
            if (maximumKeys <= 0) throw new IllegalArgumentException("maximumKeys");
            this.maximumKeys = maximumKeys;
        }

        synchronized TokenBucket.Decision consume(String authenticatedKey, long cost) {
            if (!IDENTITIES.contains(authenticatedKey)) {
                throw new IllegalArgumentException("unknown canonical identity");
            }
            TokenBucket bucket = entries.computeIfAbsent(authenticatedKey,
                    ignored -> new TokenBucket(CAPACITY, REFILL_PER_SECOND));
            while (entries.size() > maximumKeys) {
                Iterator<Map.Entry<String, TokenBucket>> oldest = entries.entrySet().iterator();
                String key = oldest.next().getKey();
                oldest.remove();
                System.out.println("evicted=" + key);
            }
            return bucket.tryConsume(cost);
        }

        synchronized int size() { return entries.size(); }
    }

    public static void main(String[] args) throws Exception {
        Properties properties = new Properties();
        try (Reader reader = Files.newBufferedReader(Path.of(args[0]))) {
            properties.load(reader);
        }
        int maximumKeys = Integer.parseInt(properties.getProperty("maximumKeys"));
        Registry registry = new Registry(maximumKeys);
        long start = System.nanoTime();
        long acceptedByA = 0;
        String[] keys = {"A", "B", "C", "A"};
        long[] costs = {10, 1, 1, 10};
        for (int i = 0; i < keys.length; i++) {
            TokenBucket.Decision decision = registry.consume(keys[i], costs[i]);
            System.out.printf(Locale.ROOT,
                    "event=%d key=%s cost=%d admitted=%s retryAfter=%s registrySize=%d%n",
                    i, keys[i], costs[i], decision.admitted(), decision.retryAfter(), registry.size());
            if (i < 3 && !decision.admitted()) {
                throw new AssertionError("initial legitimate credit unexpectedly unavailable");
            }
            if ("A".equals(keys[i]) && decision.admitted()) acceptedByA += costs[i];
            if (registry.size() > maximumKeys) throw new AssertionError("memory bound");
        }
        double elapsedSeconds = (System.nanoTime() - start) / 1_000_000_000d;
        if (elapsedSeconds >= 60) {
            throw new IllegalStateException("inconclusive fixture timing: " + elapsedSeconds);
        }
        double envelope = CAPACITY + REFILL_PER_SECOND * elapsedSeconds;
        System.out.printf(Locale.ROOT,
                "maximumKeys=%d acceptedByA=%d envelope=%.9f elapsedSeconds=%.9f%n",
                maximumKeys, acceptedByA, envelope, elapsedSeconds);
        if (acceptedByA > envelope) {
            throw new AssertionError("authenticated A exceeded its declared burst/refill envelope");
        }
        System.out.println("PASS per-identity envelope for this workload");
    }
}
