import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import crypto from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const root = path.dirname(fileURLToPath(import.meta.url));
const perl = 'C:/Program Files/Git/usr/bin/perl.exe';
const sourceIndex = JSON.parse(fs.readFileSync(path.join(root, 'source-index.json'), 'utf8'));
for (const source of sourceIndex.sources) {
  const actual = crypto.createHash('sha256').update(fs.readFileSync(path.join(root, source.file))).digest('hex');
  assert.equal(actual, source.sha256, source.file);
}

const calls = [];
function run(args, input) {
  const result = spawnSync(perl, args, { cwd: root, input, encoding: 'utf8' });
  calls.push({ executable: perl, args, status: result.status, stderr: result.stderr, error: result.error?.message });
  assert.equal(result.status, 0, JSON.stringify(calls.at(-1)));
  return result.stdout;
}
const perlVersion = run(['-e', 'print "$^V\n"']).trim();

function write(name, content) {
  fs.writeFileSync(path.join(root, name), content);
}
function diff(left, right, normalize = false) {
  return run(['sources/difffolded.pl', ...(normalize ? ['-n'] : []), left, right]);
}
function render(content, negate = false) {
  return run(['sources/flamegraph.pl', ...(negate ? ['--negate'] : [])], content);
}
function rows(content) {
  return Object.fromEntries(content.trim().split('\n').map(line => {
    const match = /^(.*) (\d+) (\d+)$/.exec(line);
    assert.ok(match, line);
    return [match[1], [Number(match[2]), Number(match[3])]];
  }));
}
function frame(svg, name) {
  const match = new RegExp('<g[^>]*>\\s*<title>' + name + ' \\([^<]+</title>[^]*?</g>').exec(svg);
  return match?.[0];
}

write('baseline.folded', 'root;stable 80\nroot;removed 20\n');
write('candidate.folded', 'root;stable 80\nroot;added 20\n');
const direct = diff('baseline.folded', 'candidate.folded');
const directRows = rows(direct);
assert.deepEqual(directRows['root;stable'], [80, 80]);
assert.deepEqual(directRows['root;added'], [0, 20]);
assert.deepEqual(directRows['root;removed'], [20, 0]);
const totals = [0, 0];
for (const values of Object.values(directRows)) values.forEach((value, index) => totals[index] += value);
assert.deepEqual(totals, [100, 100]);
const afterSvg = render(direct);
assert.match(frame(afterSvg, 'added'), /fill="rgb\(255,0,0\)"/);
assert.equal(frame(afterSvg, 'removed'), undefined);
assert.match(frame(afterSvg, 'root'), /fill="rgb\(255,255,255\)"/);
const reversed = diff('candidate.folded', 'baseline.folded');
const beforeSvg = render(reversed, true);
assert.match(frame(beforeSvg, 'removed'), /fill="rgb\(0,0,255\)"/);
assert.match(frame(beforeSvg, 'removed'), /-20\.00%/);
const wrongSignSvg = render(reversed);
assert.match(frame(wrongSignSvg, 'removed'), /fill="rgb\(255,0,0\)"/);
write('direct.diff.folded', direct);
write('candidate-width.svg', afterSvg);
write('baseline-width.svg', beforeSvg);
write('wrong-sign-control.svg', wrongSignSvg);

write('cost-before.folded', 'request;decode 60\nrequest;encode 40\n');
write('cost-after.folded', 'request;decode 120\nrequest;encode 80\n');
const absolute = rows(diff('cost-before.folded', 'cost-after.folded'));
assert.deepEqual(absolute['request;decode'], [60, 120]);
assert.deepEqual(absolute['request;encode'], [40, 80]);
const normalized = diff('cost-before.folded', 'cost-after.folded', true);
const normalizedRows = rows(normalized);
assert.deepEqual(normalizedRows['request;decode'], [120, 120]);
assert.deepEqual(normalizedRows['request;encode'], [80, 80]);
const normalizedSvg = render(normalized);
assert.match(frame(normalizedSvg, 'decode'), /fill="rgb\(255,255,255\)"/);
assert.match(frame(normalizedSvg, 'encode'), /fill="rgb\(255,255,255\)"/);
write('normalized.diff.folded', normalized);
write('normalized-cost.svg', normalizedSvg);
assert.equal((100 * 0.01) / 1000, 0.001);
assert.equal((200 * 0.01) / 1000, 0.002);
assert.equal((200 * 0.01) / 2000, 0.001);

const report = {
  status: 'PASS',
  scope: 'Pinned classic FlameGraph conversion and synthetic arithmetic; no agent evaluation or production measurement.',
  sourceCommit: sourceIndex.upstreamCommit,
  perlVersion,
  controls: {
    rawComparisonTotals: totals,
    candidateWidthAddedRed: true,
    candidateWidthRemovedAbsent: true,
    unchangedParentSelfColorWhite: true,
    complementaryRemovedBlue: true,
    omittedNegateNegativeControlIsRed: true,
    normalizationErasesUniformDoubling: true,
    fixedWorkCpuPerOperation: [0.001, 0.002],
    doubledWorkCpuPerOperation: [0.001, 0.001]
  },
  calls
};
write('validation-log.json', JSON.stringify(report, null, 2) + '\n');
console.log(JSON.stringify(report, null, 2));
