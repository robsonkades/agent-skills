import java.io.IOException;
import java.io.InputStream;
import java.net.URI;
import java.util.Arrays;
import org.springframework.http.HttpStatus;
import org.springframework.http.client.ClientHttpRequestFactory;
import org.springframework.mock.http.client.MockClientHttpRequest;
import org.springframework.mock.http.client.MockClientHttpResponse;
import org.springframework.web.client.RestClient;

public final class BufferingBoundaryCheck {
    private static final int CAP = 1_024;
    private static final int LARGE = 65_536;

    public static void main(String[] args) {
        for (int status : new int[] {200, 500}) {
            exercise(false, status, LARGE);
            exercise(true, status, LARGE);
        }
        exercise(false, 200, 8);
        exercise(true, 200, 8);
        System.out.println("PASS: six actual RestClient cases; counted source is a transport double, not a socket.");
    }

    private static void exercise(boolean buffering, int status, int size) {
        GeneratedBody body = new GeneratedBody(size);
        int[] requests = {0};
        ClientHttpRequestFactory transport = (uri, method) -> {
            requests[0]++;
            MockClientHttpResponse response =
                    new MockClientHttpResponse(body, HttpStatus.valueOf(status));
            check(response.getHeaders().getContentLength() == -1,
                    "Fixture must have no declared length");
            MockClientHttpRequest request = new MockClientHttpRequest(method, uri) {
                @Override
                protected org.springframework.http.client.ClientHttpResponse executeInternal()
                        throws IOException {
                    check("retained".equals(getHeaders().getFirst("X-Configured")),
                            "Configured header lost");
                    return response;
                }
            };
            return request;
        };
        RestClient client = RestClient.builder()
                .requestFactory(transport)
                .defaultHeader("X-Configured", "retained")
                .bufferContent((uri, method) -> buffering)
                .build();
        boolean rejected = false;
        int result = -1;
        try {
            result = client.get().uri(URI.create("https://peer.example/resource"))
                    .exchange((request, response) -> {
                        byte[] bytes = response.getBody().readNBytes(CAP + 1);
                        if (bytes.length > CAP) {
                            throw new TooLarge();
                        }
                        return bytes.length;
                    });
        } catch (TooLarge expected) {
            rejected = true;
        }
        boolean oversized = size > CAP;
        check(rejected == oversized, "Wrong body classification");
        check(oversized || result == size, "Small response changed");
        int expectedRead = buffering ? size : Math.min(size, CAP + 1);
        check(body.readBytes == expectedRead, "Wrong source-consumption boundary");
        check(body.closed, "Response body was not closed");
        check(requests[0] == 1, "Unexpected outgoing attempts");
        System.out.printf("buffering=%s status=%d size=%d rejected=%s sourceRead=%d closed=%s requests=%d%n",
                buffering, status, size, rejected, body.readBytes, body.closed, requests[0]);
    }

    private static void check(boolean condition, String message) {
        if (!condition) {
            throw new AssertionError(message);
        }
    }

    private static final class TooLarge extends RuntimeException {}

    private static final class GeneratedBody extends InputStream {
        private final int size;
        private int readBytes;
        private boolean closed;

        private GeneratedBody(int size) {
            this.size = size;
        }

        @Override
        public int read() {
            if (readBytes == size) {
                return -1;
            }
            readBytes++;
            return 'x';
        }

        @Override
        public int read(byte[] buffer, int offset, int length) {
            if (length == 0) {
                return 0;
            }
            if (readBytes == size) {
                return -1;
            }
            int count = Math.min(length, size - readBytes);
            Arrays.fill(buffer, offset, offset + count, (byte) 'x');
            readBytes += count;
            return count;
        }

        @Override
        public void close() {
            closed = true;
        }
    }
}
