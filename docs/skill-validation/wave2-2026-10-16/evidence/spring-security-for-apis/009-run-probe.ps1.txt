$ErrorActionPreference = 'Stop'
$probeRoot = 'C:/Users/robso/AppData/Local/Temp/skills-wave2-validation-Bu1zNV/spring-security-for-apis'
$mavenCache = 'C:/Users/robso/.m2/repository'
$jarPaths = @(
  'org/springframework/security/spring-security-core/7.1.0/spring-security-core-7.1.0.jar'
  'org/springframework/security/spring-security-oauth2-core/7.1.0/spring-security-oauth2-core-7.1.0.jar'
  'org/springframework/security/spring-security-oauth2-jose/7.1.0/spring-security-oauth2-jose-7.1.0.jar'
  'org/springframework/security/spring-security-oauth2-resource-server/7.1.0/spring-security-oauth2-resource-server-7.1.0.jar'
  'org/springframework/spring-core/7.0.8/spring-core-7.0.8.jar'
  'org/springframework/spring-web/7.0.8/spring-web-7.0.8.jar'
  'org/springframework/spring-beans/7.0.8/spring-beans-7.0.8.jar'
  'org/springframework/spring-expression/7.0.8/spring-expression-7.0.8.jar'
  'commons-logging/commons-logging/1.3.5/commons-logging-1.3.5.jar'
  'org/jspecify/jspecify/1.0.0/jspecify-1.0.0.jar'
  'com/nimbusds/nimbus-jose-jwt/10.9/nimbus-jose-jwt-10.9.jar'
) | ForEach-Object { Join-Path $mavenCache $_ }
foreach ($jarPath in $jarPaths) {
  if (-not (Test-Path -LiteralPath $jarPath -PathType Leaf)) { throw "Missing cached jar: $jarPath" }
}
$jarPaths | ForEach-Object {
  $jarHash = Get-FileHash -LiteralPath $_ -Algorithm SHA256
  [pscustomobject]@{ path = $_; sha256 = $jarHash.Hash.ToLowerInvariant() }
} | ConvertTo-Json -Depth 3 | Set-Content -LiteralPath (Join-Path $probeRoot 'cached-dependencies.json') -Encoding utf8
$probeClasspath = $jarPaths -join ';'
$probeClasses = Join-Path $probeRoot 'classes'
New-Item -ItemType Directory -Path $probeClasses -Force | Out-Null
Set-Content -LiteralPath (Join-Path $probeRoot 'compile-output.txt') -Value '' -NoNewline -Encoding utf8
& javac --release 17 -cp $probeClasspath -d $probeClasses (Join-Path $probeRoot 'SpringIdentityProbe.java') 2>&1 | Tee-Object -FilePath (Join-Path $probeRoot 'compile-output.txt')
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
& java -cp "$probeClasspath;$probeClasses" SpringIdentityProbe 2>&1 | Tee-Object -FilePath (Join-Path $probeRoot 'probe-output.txt')
exit $LASTEXITCODE
