import java.security.KeyPair;
import java.security.KeyPairGenerator;
import java.security.interfaces.RSAPrivateKey;
import java.security.interfaces.RSAPublicKey;
import java.time.Instant;
import java.util.Date;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.concurrent.atomic.AtomicInteger;

import com.nimbusds.jose.JOSEObjectType;
import com.nimbusds.jose.JWSAlgorithm;
import com.nimbusds.jose.JWSHeader;
import com.nimbusds.jose.crypto.RSASSASigner;
import com.nimbusds.jwt.JWTClaimsSet;
import com.nimbusds.jwt.SignedJWT;
import org.springframework.security.authentication.AbstractAuthenticationToken;
import org.springframework.security.oauth2.core.DelegatingOAuth2TokenValidator;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.security.oauth2.jwt.JwtClaimValidator;
import org.springframework.security.oauth2.jwt.JwtException;
import org.springframework.security.oauth2.jwt.JwtValidators;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationConverter;

/** Isolated documentation probe: real decoder/converter; synthetic account/owner policy. */
public final class SpringIdentityProbe {
    private static final String ISSUER_A = "https://issuer-a.example.test";
    private static final String ISSUER_B = "https://issuer-b.example.test";
    private static final String SUBJECT = "42";
    private static int assertions;

    record PrincipalKey(String issuer, String subject) {
        PrincipalKey {
            if (issuer == null || issuer.isBlank() || subject == null || subject.isBlank()) {
                throw new IllegalArgumentException("Required identity is missing");
            }
        }
    }

    public static void main(String[] args) throws Exception {
        KeyPairGenerator generator = KeyPairGenerator.getInstance("RSA");
        generator.initialize(2048);
        KeyPair keyA = generator.generateKeyPair();
        KeyPair keyB = generator.generateKeyPair();
        NimbusJwtDecoder decoderA = decoder(keyA, ISSUER_A);
        NimbusJwtDecoder decoderB = decoder(keyB, ISSUER_B);
        String tokenA = token(keyA, ISSUER_A, SUBJECT);
        String tokenB = token(keyB, ISSUER_B, SUBJECT);
        Jwt jwtA = decoderA.decode(tokenA);
        Jwt jwtB = decoderB.decode(tokenB);
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        AbstractAuthenticationToken authA = converter.convert(jwtA);
        AbstractAuthenticationToken authB = converter.convert(jwtB);

        check(authA.isAuthenticated() && authB.isAuthenticated(), "both local signed fixtures validate and convert");
        check(SUBJECT.equals(authA.getName()) && authA.getName().equals(authB.getName()),
                "real stock converter gives equal names for different trusted issuers with equal sub");
        check(!jwtA.getClaimAsString("iss").equals(jwtB.getClaimAsString("iss")), "validated issuer claims remain distinct");
        check(hasWriteScope(authA) && hasWriteScope(authB), "existing scope conversion remains available");

        // This deliberately vulnerable owner check is only a synthetic counterexample.
        check(SUBJECT.equals(authB.getName()), "legacy name-only owner key would admit issuer B to issuer A's row");
        PrincipalKey owner = key(jwtA);
        check(!owner.equals(key(jwtB)), "validated issuer-subject keys separate the two callers");
        AtomicInteger state = new AtomicInteger();
        check(writeIfOwner(owner, key(jwtA), state), "original owner remains permitted");
        check(!writeIfOwner(owner, key(jwtB), state), "other trusted issuer cannot mutate this owner's state");
        check(state.get() == 1, "denied attempt leaves protected synthetic state unchanged");

        Map<PrincipalKey, String> linkedAccounts = Map.of(key(jwtA), "account-17", key(jwtB), "account-17");
        check(Objects.equals(linkedAccounts.get(key(jwtA)), linkedAccounts.get(key(jwtB))),
                "an explicit trusted canonical account link can intentionally unify external identities");

        // The map is fixture issuer selection, not Spring's production manager resolver.
        Map<String, NimbusJwtDecoder> singleIssuer = Map.of(ISSUER_A, decoderA);
        Map<String, NimbusJwtDecoder> twoIssuers = Map.of(ISSUER_A, decoderA, ISSUER_B, decoderB);
        check(!singleIssuer.containsKey(ISSUER_B) && twoIssuers.containsKey(ISSUER_B),
                "changing the accepted issuer set changes which local identity namespace must be preserved");
        rejected(decoderA, tokenB, "token signed with another issuer's key is rejected by this decoder");
        rejected(decoderB, token(keyB, ISSUER_A, SUBJECT), "correctly signed wrong-issuer claim is rejected");
        rejected(decoderA, token(keyA, ISSUER_A, null), "explicit required-subject validator rejects missing identity");

        System.out.println("RESULT PASS assertions=" + assertions);
        System.out.println("Spring Security=" + JwtAuthenticationConverter.class.getPackage().getImplementationVersion());
        System.out.println("Java runtime=" + Runtime.version());
        System.out.println("LIMITS Real local RSA signature/issuer/audience/time/required-subject validation and stock converter; synthetic owner/account mapping. No HTTP filters, Boot wiring, repository, cache, remote issuer, TLS, production credentials or agent evaluation.");
    }

    private static NimbusJwtDecoder decoder(KeyPair pair, String issuer) {
        NimbusJwtDecoder decoder = NimbusJwtDecoder.withPublicKey((RSAPublicKey) pair.getPublic()).build();
        JwtClaimValidator<List<String>> audience = new JwtClaimValidator<>("aud",
                values -> values != null && values.contains("orders-api"));
        JwtClaimValidator<String> subject = new JwtClaimValidator<>("sub",
                value -> value != null && !value.isBlank());
        decoder.setJwtValidator(new DelegatingOAuth2TokenValidator<>(
                JwtValidators.createDefaultWithIssuer(issuer), audience, subject));
        return decoder;
    }

    private static String token(KeyPair key, String issuer, String subject) throws Exception {
        Instant now = Instant.now();
        JWTClaimsSet.Builder claims = new JWTClaimsSet.Builder()
                .issuer(issuer).audience("orders-api").issueTime(Date.from(now.minusSeconds(1)))
                .expirationTime(Date.from(now.plusSeconds(300))).claim("scope", "orders.write");
        if (subject != null) claims.subject(subject);
        SignedJWT jwt = new SignedJWT(new JWSHeader.Builder(JWSAlgorithm.RS256)
                .type(JOSEObjectType.JWT).build(), claims.build());
        jwt.sign(new RSASSASigner((RSAPrivateKey) key.getPrivate()));
        return jwt.serialize();
    }

    private static PrincipalKey key(Jwt validated) {
        return new PrincipalKey(validated.getClaimAsString("iss"), validated.getSubject());
    }

    private static boolean hasWriteScope(AbstractAuthenticationToken authentication) {
        return authentication.getAuthorities().stream().anyMatch(value -> value.getAuthority().equals("SCOPE_orders.write"));
    }

    private static boolean writeIfOwner(PrincipalKey owner, PrincipalKey caller, AtomicInteger state) {
        if (!owner.equals(caller)) return false;
        state.incrementAndGet();
        return true;
    }

    private static void rejected(NimbusJwtDecoder decoder, String token, String message) {
        try {
            decoder.decode(token);
            throw new AssertionError(message);
        } catch (JwtException expected) {
            check(true, message);
        }
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
        assertions++;
        System.out.println("PASS " + message);
    }
}
