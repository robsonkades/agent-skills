import { readFile, writeFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const output = path.dirname(fileURLToPath(import.meta.url));
const source = 'C:/git/agent-skills/skills/distributed-aggregation-and-barriers/references/aggregation-correctness.md';
const markdown = await readFile(source, 'utf8');
const blocks = [...markdown.matchAll(/```java\r?\n([\s\S]*?)```/g)].map(match => match[1]);
const matches = blocks.filter(block => block.includes('record SumCount('));
if (matches.length !== 1) throw new Error('Expected exactly one SumCount example');
await writeFile(path.join(output, 'SumCount.java'), 'import java.math.*;\nimport java.util.Optional;\n' + matches[0]);
console.log('Extracted the existing SumCount block verbatim, adding omitted imports only.');
console.log('Scope: local numeric/invocation-shape checks; no distributed engine execution.');
