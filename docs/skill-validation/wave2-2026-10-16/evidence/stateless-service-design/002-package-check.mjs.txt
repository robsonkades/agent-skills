import assert from 'node:assert/strict';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { pathToFileURL } from 'node:url';
import { createHash } from 'node:crypto';

const repo = 'C:/git/agent-skills';
const root = resolve(repo, 'skills/stateless-service-design');
const { loadPackageFromDirectory } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/package-loader.ts')).href);
const { validatePackage } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/validate-package.ts')).href);
const { NodeFileSystem } = await import(pathToFileURL(resolve(repo, 'packages/node/src/file-system.ts')).href);
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), root, { strict: true });
const report = validatePackage(loaded.pkg, { directoryName: 'stateless-service-design', strict: true });
const issues = [...loaded.issues, ...report.issues];
assert.equal(issues.length, 0, JSON.stringify(issues));
const baseline = {
  'SKILL.md': '5C751DD9942F7CEF9AAF17C96849653A7185E0DD010A363D9599BDB26FDB64D0',
  'skill.yaml': '4BDDBFC0DE66397F8CD88CE592C623A39B8E6F659E26FD11E505239DC5A56407',
  'references/state-inventory.md': 'FCAE3D5CBD7370D779959B633C2A6F00091CD3EDC459D5E5067808DBDB9C3668',
  'references/session-placement.md': '887A86505AE95DADFA4DFB89AAAD6739BE7B9A89AB0A824F356D9DE0E9BEE517'
};
assert.equal(loaded.pkg.files.length, Object.keys(baseline).length);
const links = [];
const hashes = {};
for (const [file, expected] of Object.entries(baseline)) {
  const absolute = resolve(root, file);
  const bytes = readFileSync(absolute);
  hashes[file] = createHash('sha256').update(bytes).digest('hex').toUpperCase();
  assert.equal(hashes[file], expected, `Unexpected change: ${file}`);
  if (!file.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
    const target = match[1];
    if (/^<?[a-z]+:/i.test(target)) continue;
    const [path, anchor] = target.split('#');
    assert.ok(existsSync(resolve(dirname(absolute), path)), `${file}: ${target}`);
    assert.equal(anchor, undefined, 'Anchor validation needed if an internal anchor is added');
    links.push({ file, target });
  }
}
const crossSkillResources = ['state-placement.md', 'session-failure-modes.md'].map(name =>
  'skills/session-state-strategies/references/' + name);
for (const resource of crossSkillResources) assert.ok(existsSync(resolve(repo, resource)), resource);
assert.equal(loaded.pkg.manifest.version, '1.2.4');
console.log(JSON.stringify({ runtime: process.version, name: loaded.pkg.manifest.name, version: loaded.pkg.manifest.version, packageFiles: loaded.pkg.files.length, issues, localMarkdownLinks: links, crossSkillResources, hashes, allPackageHashesUnchanged: true }, null, 2));
