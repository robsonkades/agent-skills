import java.util.concurrent.Callable;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.Future;
import java.util.concurrent.ThreadFactory;
import java.util.concurrent.TimeUnit;

// Review-only public-API ownership probe; no GC/unloading assertion.
public final class ContextLoaderProbe {
    private static final class PluginLoader extends ClassLoader {
        PluginLoader(ClassLoader parent) { super(parent); }
    }

    private static final class ExpectedTaskFailure extends RuntimeException { }

    public static void main(String[] args) throws Exception {
        String mode = args[0];
        if (!mode.equals("inherited") && !mode.equals("fixed") && !mode.equals("missing-restore")) {
            throw new IllegalArgumentException(mode);
        }
        Thread submitter = Thread.currentThread();
        ClassLoader original = submitter.getContextClassLoader();
        ClassLoader base = ContextLoaderProbe.class.getClassLoader();
        ClassLoader plugin = new PluginLoader(base);
        ExecutorService pool = null;
        try {
            submitter.setContextClassLoader(base);
            final ThreadFactory defaults = Executors.defaultThreadFactory();
            ThreadFactory factory = mode.equals("inherited") ? defaults : task -> {
                Thread worker = defaults.newThread(task);
                worker.setContextClassLoader(base);
                return worker;
            };
            pool = Executors.newSingleThreadExecutor(factory);
            Future<Thread> first;
            try {
                submitter.setContextClassLoader(plugin);
                first = pool.submit(Thread::currentThread);
            } finally {
                submitter.setContextClassLoader(base);
            }
            Thread worker = first.get(5, TimeUnit.SECONDS);
            ClassLoader expected = mode.equals("inherited") ? plugin : base;
            check(submitter.getContextClassLoader() == base, "submitter was not restored");
            check(worker.getContextClassLoader() == expected, "unexpected worker creation context");
            System.out.println("runtime=" + System.getProperty("java.runtime.version"));
            System.out.println("factory-and-pool-created-under=base; first-worker-created-under=plugin");
            System.out.println("mode=" + mode + "; submitter-restored=true; worker-holds-plugin="
                    + (worker.getContextClassLoader() == plugin));

            pool.submit(scoped(plugin, false, true)).get(5, TimeUnit.SECONDS);
            check(pool.submit(() -> Thread.currentThread().getContextClassLoader()).get(5, TimeUnit.SECONDS) == expected,
                    "context not restored after successful task");
            try {
                pool.submit(scoped(plugin, true, !mode.equals("missing-restore"))).get(5, TimeUnit.SECONDS);
                throw new AssertionError("expected task failure was not reported");
            } catch (ExecutionException failure) {
                check(failure.getCause() instanceof ExpectedTaskFailure, "unexpected task failure");
            }
            check(pool.submit(() -> Thread.currentThread().getContextClassLoader()).get(5, TimeUnit.SECONDS) == expected,
                    "context not restored after failed task");
            check(pool.submit(Thread::currentThread).get(5, TimeUnit.SECONDS) == worker,
                    "worker was replaced rather than reused");
            System.out.println("PASS: first-worker ownership; same-worker restoration after success and failure");
            System.out.println("No collection, class unloading or memory-growth claim was tested.");
        } finally {
            submitter.setContextClassLoader(original);
            if (pool != null) {
                pool.shutdownNow();
                check(pool.awaitTermination(5, TimeUnit.SECONDS), "probe worker did not terminate");
            }
        }
    }

    private static Callable<Void> scoped(ClassLoader desired, boolean fail, boolean restore) {
        return () -> {
            Thread worker = Thread.currentThread();
            ClassLoader previous = worker.getContextClassLoader();
            try {
                worker.setContextClassLoader(desired);
                check(worker.getContextClassLoader() == desired, "task context was not applied");
                if (fail) throw new ExpectedTaskFailure();
                return null;
            } finally {
                if (restore) worker.setContextClassLoader(previous);
            }
        };
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
