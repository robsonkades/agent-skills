import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import com.github.benmanes.caffeine.cache.LoadingCache;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

public final class LoadPathCheck {
    static final String KEY = "product-7";

    static final class Origin {
        final AtomicInteger calls = new AtomicInteger();
        final CountDownLatch entered;
        final CountDownLatch release = new CountDownLatch(1);
        Origin(int expectedConcurrentCalls) { entered = new CountDownLatch(expectedConcurrentCalls); }
        String load(String key) {
            calls.incrementAndGet();
            entered.countDown();
            await(release);
            return "value:" + key;
        }
    }

    static void await(CountDownLatch latch) {
        try {
            if (!latch.await(5, TimeUnit.SECONDS)) throw new AssertionError("latch timeout");
        } catch (InterruptedException failure) {
            Thread.currentThread().interrupt();
            throw new AssertionError(failure);
        }
    }

    static void await(CyclicBarrier barrier) {
        try { barrier.await(5, TimeUnit.SECONDS); }
        catch (InterruptedException failure) {
            Thread.currentThread().interrupt();
            throw new AssertionError(failure);
        } catch (BrokenBarrierException | TimeoutException failure) {
            throw new AssertionError(failure);
        }
    }

    static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    static void concurrent(boolean split, boolean separateCaches, int expected) throws Exception {
        Origin origin = new Origin(expected);
        LoadingCache<String, String> left = Caffeine.newBuilder().maximumSize(100).build(origin::load);
        LoadingCache<String, String> right = separateCaches
                ? Caffeine.newBuilder().maximumSize(100).build(origin::load) : left;
        ExecutorService workers = Executors.newFixedThreadPool(2);
        CyclicBarrier start = new CyclicBarrier(3);
        CyclicBarrier bothMissed = new CyclicBarrier(2);
        try {
            Future<String> first = workers.submit(() -> read(left, origin, split, start, bothMissed));
            Future<String> second = workers.submit(() -> read(right, origin, split, start, bothMissed));
            await(start);
            await(origin.entered);
            origin.release.countDown();
            require(("value:" + KEY).equals(first.get(5, TimeUnit.SECONDS)), "first result");
            require(("value:" + KEY).equals(second.get(5, TimeUnit.SECONDS)), "second result");
            require(origin.calls.get() == expected, "origin attempts: " + origin.calls.get());
            require(("value:" + KEY).equals(left.getIfPresent(KEY)), "left cache populated");
            require(("value:" + KEY).equals(right.getIfPresent(KEY)), "right cache populated");
            System.out.println("PASS " + (split ? "split lookup/load/put" : "LoadingCache.get")
                    + ", cache instances=" + (separateCaches ? 2 : 1)
                    + ", origin attempts=" + origin.calls.get());
        } finally {
            origin.release.countDown();
            workers.shutdownNow();
            require(workers.awaitTermination(5, TimeUnit.SECONDS), "worker termination");
        }
    }

    static String read(LoadingCache<String, String> cache, Origin origin, boolean split,
            CyclicBarrier start, CyclicBarrier bothMissed) {
        await(start);
        if (!split) return cache.get(KEY);
        String value = cache.getIfPresent(KEY);
        require(value == null, "negative control must see two misses");
        await(bothMissed);
        value = origin.load(KEY);
        cache.put(KEY, value);
        return value;
    }

    static void failureAndNull() {
        Cache<String, String> cache = Caffeine.newBuilder().maximumSize(100).build();
        AtomicInteger attempts = new AtomicInteger();
        try {
            cache.get(KEY, key -> {
                attempts.incrementAndGet();
                throw new IllegalStateException("origin unavailable");
            });
            throw new AssertionError("load failure must propagate");
        } catch (IllegalStateException expected) {
            require("origin unavailable".equals(expected.getMessage()), "preserved failure");
        }
        require(cache.getIfPresent(KEY) == null, "failed load must not establish value");
        require("recovered".equals(cache.get(KEY, key -> {
            attempts.incrementAndGet();
            return "recovered";
        })), "recovery value");
        require(attempts.get() == 2, "recovery retries loader");
        require(cache.get("absent", key -> null) == null, "null result");
        require(cache.getIfPresent("absent") == null, "null result is not a stored value");
        require("later".equals(cache.get("absent", key -> "later")), "null does not prevent later load");
        System.out.println("PASS failure propagates, recovery loads again, null does not populate");
    }

    public static void main(String[] args) throws Exception {
        concurrent(false, false, 1);
        concurrent(true, false, 2);
        concurrent(false, true, 2);
        failureAndNull();
    }
}
