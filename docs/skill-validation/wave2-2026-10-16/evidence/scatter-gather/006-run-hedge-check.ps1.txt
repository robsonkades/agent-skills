$ErrorActionPreference = 'Stop'
$scatterEvidenceRoot = $PSScriptRoot
$scatterCache = 'C:\Users\robso\.m2\repository'
$scatterRelativeJars = @(
    'io\grpc\grpc-api\1.62.2\grpc-api-1.62.2.jar',
    'io\grpc\grpc-core\1.62.2\grpc-core-1.62.2.jar',
    'io\grpc\grpc-context\1.62.2\grpc-context-1.62.2.jar',
    'io\grpc\grpc-inprocess\1.62.2\grpc-inprocess-1.62.2.jar',
    'com\google\guava\guava\32.1.3-jre\guava-32.1.3-jre.jar',
    'com\google\guava\failureaccess\1.0.1\failureaccess-1.0.1.jar',
    'com\google\code\gson\gson\2.10.1\gson-2.10.1.jar',
    'io\perfmark\perfmark-api\0.26.0\perfmark-api-0.26.0.jar'
)
$scatterJars = foreach ($scatterRelativeJar in $scatterRelativeJars) {
    $scatterJar = Join-Path $scatterCache $scatterRelativeJar
    if (-not (Test-Path -LiteralPath $scatterJar)) { throw "Missing cached jar: $scatterJar" }
    $scatterJar
}
$scatterClasses = Join-Path $scatterEvidenceRoot 'classes'
New-Item -ItemType Directory -Path $scatterClasses -Force | Out-Null
$scatterClasspath = $scatterJars -join ';'
Get-FileHash -LiteralPath $scatterJars -Algorithm SHA256 | Select-Object Path,Hash | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $scatterEvidenceRoot 'classpath-hashes.json')
& javac --release 21 -cp $scatterClasspath -d $scatterClasses (Join-Path $scatterEvidenceRoot 'HedgeCommitCheck.java') 2>&1 | Tee-Object -FilePath (Join-Path $scatterEvidenceRoot 'compile.log')
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
& java -ea -cp ($scatterClasses + ';' + $scatterClasspath) HedgeCommitCheck 2>&1 | Tee-Object -FilePath (Join-Path $scatterEvidenceRoot 'runtime.log')
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
