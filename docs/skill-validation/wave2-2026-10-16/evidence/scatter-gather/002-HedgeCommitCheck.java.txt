import io.grpc.CallOptions;
import io.grpc.ClientCall;
import io.grpc.ManagedChannel;
import io.grpc.Metadata;
import io.grpc.MethodDescriptor;
import io.grpc.Server;
import io.grpc.ServerCall;
import io.grpc.ServerServiceDefinition;
import io.grpc.Status;
import io.grpc.inprocess.InProcessChannelBuilder;
import io.grpc.inprocess.InProcessServerBuilder;
import java.io.ByteArrayInputStream;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Map;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicReference;

// Isolated protocol-order controls using real cached grpc-java 1.62.2 libraries.
// In-process transport deliberately tests commitment, not replica placement or performance.
public final class HedgeCommitCheck {
    private static final MethodDescriptor.Marshaller<String> TEXT = new MethodDescriptor.Marshaller<>() {
        public InputStream stream(String value) {
            return new ByteArrayInputStream(value.getBytes(StandardCharsets.UTF_8));
        }
        public String parse(InputStream input) {
            try {
                return new String(input.readAllBytes(), StandardCharsets.UTF_8);
            } catch (java.io.IOException e) {
                throw new IllegalStateException(e);
            }
        }
    };
    private static final MethodDescriptor<String, String> METHOD = MethodDescriptor.<String, String>newBuilder()
            .setType(MethodDescriptor.MethodType.UNARY)
            .setFullMethodName("audit.Quote/Get")
            .setRequestMarshaller(TEXT).setResponseMarshaller(TEXT).build();

    private record Result(Status status, String watermark) {}
    private static final class Attempt {
        final ServerCall<String, String> call;
        final CountDownLatch cancelled = new CountDownLatch(1);
        Attempt(ServerCall<String, String> call) { this.call = call; }
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private static void run(String scenario) throws Exception {
        List<Attempt> attempts = Collections.synchronizedList(new ArrayList<>());
        CountDownLatch bothStarted = new CountDownLatch(2);
        var definition = ServerServiceDefinition.builder("audit.Quote")
                .addMethod(METHOD, (call, headers) -> {
                    Attempt attempt = new Attempt(call);
                    call.request(1);
                    return new ServerCall.Listener<String>() {
                        public void onHalfClose() {
                            attempts.add(attempt);
                            bothStarted.countDown();
                        }
                        public void onCancel() { attempt.cancelled.countDown(); }
                    };
                }).build();
        String name = InProcessServerBuilder.generateName();
        Server server = InProcessServerBuilder.forName(name).directExecutor().addService(definition).build().start();
        Map<String, Object> policy = Map.of(
                "maxAttempts", 2.0,
                "hedgingDelay", "0s",
                "nonFatalStatusCodes", List.of("UNAVAILABLE"));
        Map<String, Object> serviceConfig = Map.of("methodConfig", List.of(Map.of(
                "name", List.of(Map.of("service", "audit.Quote", "method", "Get")),
                "hedgingPolicy", policy)));
        ManagedChannel channel = null;
        try {
            channel = InProcessChannelBuilder.forName(name).directExecutor().enableRetry()
                    .defaultServiceConfig(serviceConfig).build();
            CompletableFuture<Result> result = new CompletableFuture<>();
            AtomicReference<String> body = new AtomicReference<>();
            ClientCall<String, String> clientCall = channel.newCall(METHOD,
                    CallOptions.DEFAULT.withDeadlineAfter(10, TimeUnit.SECONDS));
            clientCall.start(new ClientCall.Listener<>() {
                public void onMessage(String message) { body.set(message); }
                public void onClose(Status status, Metadata trailers) {
                    result.complete(new Result(status, body.get()));
                }
            }, new Metadata());
            clientCall.request(1);
            clientCall.sendMessage("minimum-watermark=8");
            clientCall.halfClose();
            require(bothStarted.await(5, TimeUnit.SECONDS), "native hedge did not start both attempts");
            require(attempts.size() == 2, "expected exactly two server attempts");
            Attempt first = attempts.get(0);
            Attempt second = attempts.get(1);
            if (scenario.equals("stale_payload_after_headers")) {
                first.call.sendHeaders(new Metadata());
                require(second.cancelled.await(5, TimeUnit.SECONDS), "loser was not cancelled on response headers");
                require(!result.isDone(), "payload/terminal result was not deliberately withheld");
                first.call.sendMessage("7");
                first.call.close(Status.OK, new Metadata());
                Result actual = result.get(5, TimeUnit.SECONDS);
                require(actual.status().isOk() && actual.watermark().equals("7"), "unexpected native winner");
                require(Integer.parseInt(actual.watermark()) < 8, "application check should reject this result");
                System.out.println("PASS stale_payload_after_headers: competitor cancelled before body; native OK watermark=7 fails application minimum=8");
            } else if (scenario.equals("nonfatal_failure_after_headers")) {
                first.call.sendHeaders(new Metadata());
                require(second.cancelled.await(5, TimeUnit.SECONDS), "loser was not cancelled on response headers");
                first.call.close(Status.UNAVAILABLE.withDescription("below minimum watermark"), new Metadata());
                Result actual = result.get(5, TimeUnit.SECONDS);
                require(actual.status().getCode() == Status.Code.UNAVAILABLE, "late nonfatal error should terminate committed call");
                System.out.println("PASS nonfatal_failure_after_headers: configured UNAVAILABLE cannot restore cancelled competitor after commitment");
            } else if (scenario.equals("nonfatal_failure_before_headers")) {
                first.call.close(Status.UNAVAILABLE.withDescription("below minimum watermark"), new Metadata());
                require(second.cancelled.getCount() == 1, "eligible second attempt was cancelled");
                require(!result.isDone(), "nonfatal trailers-only failure should leave second attempt eligible");
                second.call.sendHeaders(new Metadata());
                second.call.sendMessage("8");
                second.call.close(Status.OK, new Metadata());
                Result actual = result.get(5, TimeUnit.SECONDS);
                require(actual.status().isOk() && actual.watermark().equals("8"), "eligible second result did not win");
                System.out.println("PASS nonfatal_failure_before_headers: second attempt remains eligible and returns watermark=8");
            } else {
                throw new IllegalArgumentException(scenario);
            }
        } finally {
            if (channel != null) {
                channel.shutdownNow();
                require(channel.awaitTermination(5, TimeUnit.SECONDS), "channel did not terminate");
            }
            server.shutdownNow();
            require(server.awaitTermination(5, TimeUnit.SECONDS), "server did not terminate");
        }
    }

    public static void main(String[] args) throws Exception {
        System.out.println("Runtime=" + System.getProperty("java.version") + "; grpc-java=1.62.2; javac --release 21; in-process transport");
        run("stale_payload_after_headers");
        run("nonfatal_failure_after_headers");
        run("nonfatal_failure_before_headers");
        System.out.println("PASS 3 protocol-order controls; all servers/channels terminated");
    }
}
