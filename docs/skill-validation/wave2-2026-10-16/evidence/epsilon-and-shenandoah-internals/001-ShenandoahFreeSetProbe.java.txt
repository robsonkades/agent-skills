public class ShenandoahFreeSetProbe {
    private static volatile byte[] sink;

    public static void main(String[] args) {
        for (int i = 0; i < 16; i++) {
            sink = new byte[1024 * 1024];
            sink[0] = (byte) i;
        }
        System.gc();
        System.out.println("bounded probe completed: " + sink[0]);
    }
}
