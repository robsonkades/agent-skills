import io.github.resilience4j.circuitbreaker.CircuitBreaker;
import io.github.resilience4j.circuitbreaker.CircuitBreakerConfig;
import java.time.Duration;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

// Review-only probe, using cached Resilience4j 2.2.0; not a claim of 2.3.0 execution.
public final class AsyncBoundaryProbe {
    private static int assertions;

    public static void main(String[] args) throws Exception {
        outerTimeout();
        observedTimeout();
        outerCancellation();
        System.out.println("PASS: three completion-boundary cases, " + assertions + " assertions");
    }

    private static CircuitBreaker breaker(String name) {
        return CircuitBreaker.of(name, CircuitBreakerConfig.custom()
            .slidingWindowSize(1).minimumNumberOfCalls(1).failureRateThreshold(50)
            .slowCallDurationThreshold(Duration.ofMinutes(1))
            .recordException(error -> error instanceof TimeoutException).build());
    }

    private static void check(boolean condition, String message) {
        assertions++;
        if (!condition) throw new AssertionError(message);
    }

    private static void expectTimeout(CompletableFuture<String> result) throws Exception {
        try {
            result.get(2, TimeUnit.SECONDS);
            throw new AssertionError("Expected the configured future timeout");
        } catch (ExecutionException expected) {
            check(expected.getCause() instanceof TimeoutException, "Expected TimeoutException cause");
        }
        // A get() waiting timeout is deliberately not caught: it fails this check.
    }

    private static void outerTimeout() throws Exception {
        CircuitBreaker breaker = breaker("outer-timeout");
        CompletableFuture<String> source = new CompletableFuture<>();
        CompletableFuture<String> result = CircuitBreaker.decorateCompletionStage(breaker, () -> source)
            .get().toCompletableFuture();
        try {
            check(result != source, "Decorator returns a separate promise");
            check(result.orTimeout(30, TimeUnit.MILLISECONDS) == result, "orTimeout returns its receiver");
            expectTimeout(result);
            check(!source.isDone(), "Outer timeout must not complete the source");
            check(breaker.getMetrics().getNumberOfBufferedCalls() == 0, "No source outcome has been recorded");
            source.complete("late");
            check(breaker.getMetrics().getNumberOfSuccessfulCalls() == 1, "Late success is still recorded");
            check(breaker.getMetrics().getNumberOfFailedCalls() == 0, "Caller timeout is not a source failure");
            check(breaker.getState() == CircuitBreaker.State.CLOSED, "Late success keeps the breaker closed");
            expectTimeout(result);
        } finally {
            source.complete("cleanup");
        }
    }

    private static void observedTimeout() throws Exception {
        CircuitBreaker breaker = breaker("observed-timeout");
        CompletableFuture<String> source = new CompletableFuture<>();
        CompletableFuture<String> result = CircuitBreaker.decorateCompletionStage(breaker,
            () -> source.orTimeout(30, TimeUnit.MILLISECONDS)).get().toCompletableFuture();
        try {
            expectTimeout(result);
            check(source.isCompletedExceptionally(), "The observed stage timed out");
            check(breaker.getMetrics().getNumberOfFailedCalls() == 1, "Source timeout reaches the predicate");
            check(breaker.getState() == CircuitBreaker.State.OPEN, "Recorded failure opens this one-sample breaker");
            check(!source.complete("late"), "Late completion cannot replace source timeout");
        } finally {
            source.complete("cleanup");
        }
    }

    private static void outerCancellation() {
        CircuitBreaker breaker = breaker("outer-cancel");
        CompletableFuture<String> source = new CompletableFuture<>();
        CompletableFuture<String> result = CircuitBreaker.decorateCompletionStage(breaker, () -> source)
            .get().toCompletableFuture();
        try {
            check(result.cancel(true), "Outer promise cancels");
            check(!source.isDone(), "Outer cancellation does not complete the source");
            check(breaker.getMetrics().getNumberOfBufferedCalls() == 0, "Cancellation has no source sample");
            source.complete("late");
            check(breaker.getMetrics().getNumberOfSuccessfulCalls() == 1, "Late success still records after cancellation");
            check(result.isCancelled(), "Source completion does not replace caller cancellation");
        } finally {
            source.complete("cleanup");
        }
    }
}
