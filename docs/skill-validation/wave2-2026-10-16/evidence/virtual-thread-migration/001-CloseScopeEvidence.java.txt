import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

public final class CloseScopeEvidence {
    private static final class GateTask implements Callable<String> {
        final CountDownLatch entered = new CountDownLatch(1);
        final CountDownLatch mayFinish = new CountDownLatch(1);
        final CountDownLatch interrupted = new CountDownLatch(1);
        final CountDownLatch finished = new CountDownLatch(1);
        volatile boolean virtual;
        public String call() {
            virtual = Thread.currentThread().isVirtual();
            entered.countDown();
            boolean wasInterrupted = false;
            try {
                for (;;) {
                    try {
                        mayFinish.await();
                        return "done";
                    } catch (InterruptedException expected) {
                        // Deliberately model a dependency that does not stop at interruption.
                        wasInterrupted = true;
                        interrupted.countDown();
                    }
                }
            } finally {
                finished.countDown();
                if (wasInterrupted) Thread.currentThread().interrupt();
            }
        }
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private static void await(CountDownLatch latch, String label) throws InterruptedException {
        require(latch.await(5, TimeUnit.SECONDS), "Timed out waiting for " + label);
    }

    private static void join(Thread owner, AtomicReference<Throwable> failure) throws Exception {
        owner.join(5_000);
        require(!owner.isAlive(), "Owner did not terminate after controlled release");
        if (failure.get() != null) throw new AssertionError("Owner failed", failure.get());
    }

    private static void snippetReturnBoundary() throws Exception {
        GateTask gate = new GateTask();
        AtomicReference<Future<String>> result = new AtomicReference<>();
        AtomicReference<Throwable> failure = new AtomicReference<>();
        CountDownLatch returned = new CountDownLatch(1);
        Thread owner = Thread.ofPlatform().name("snippet-owner").start(() -> {
            try {
                result.set(MigrationSnippet.start(gate));
            } catch (Throwable error) {
                failure.set(error);
            } finally {
                returned.countDown();
            }
        });
        try {
            await(gate.entered, "snippet task entry");
            require(gate.virtual, "Snippet did not run a virtual task");
            require(gate.finished.getCount() == 1, "Task unexpectedly finished");
            require(returned.getCount() == 1 && result.get() == null,
                    "Method returned its Future before the held task terminated");
            System.out.println("lexical-return: task active, Future not delivered to caller");
        } finally {
            gate.mayFinish.countDown();
            join(owner, failure);
        }
        require("done".equals(result.get().get()), "Unexpected completed result");
        require(gate.finished.getCount() == 0, "Missing task cleanup");
        System.out.println("lexical-return: after release, method returns completed Future");
    }

    private static void timeoutThenCloseBoundary() throws Exception {
        GateTask gate = new GateTask();
        AtomicReference<Future<String>> handle = new AtomicReference<>();
        AtomicReference<Throwable> failure = new AtomicReference<>();
        AtomicBoolean interruptRestored = new AtomicBoolean();
        CountDownLatch timeoutCaught = new CountDownLatch(1);
        CountDownLatch returned = new CountDownLatch(1);
        Thread owner = Thread.ofPlatform().name("timed-owner").start(() -> {
            try {
                try (var executor = Executors.newVirtualThreadPerTaskExecutor()) {
                    Future<String> future = executor.submit(gate);
                    handle.set(future);
                    await(gate.entered, "timed task entry");
                    try {
                        future.get(20, TimeUnit.MILLISECONDS);
                        throw new AssertionError("Gated task completed before release");
                    } catch (TimeoutException expected) {
                        require(future.cancel(true), "Cancellation was not accepted");
                        timeoutCaught.countDown();
                    }
                }
                interruptRestored.set(Thread.currentThread().isInterrupted());
            } catch (Throwable error) {
                failure.set(error);
            } finally {
                returned.countDown();
            }
        });
        try {
            await(timeoutCaught, "timed get timeout");
            await(gate.interrupted, "task cancellation request");
            require(handle.get().isCancelled() && handle.get().isDone(), "Future not cancelled/done");
            require(gate.finished.getCount() == 1, "Cancellation unexpectedly ended task");
            require(returned.getCount() == 1, "Method escaped close before task termination");
            owner.interrupt(); // close must still await actual execution termination.
            require(returned.getCount() == 1, "Interrupted close returned while task active");
            System.out.println("timeout-close: timeout caught; Future done/cancelled; task active; caller not returned");
        } finally {
            gate.mayFinish.countDown();
            join(owner, failure);
        }
        require(interruptRestored.get(), "Close did not restore owner interrupt status");
        System.out.println("timeout-close: after task release, close returns and owner interrupt is restored");
    }

    private static void componentOwnerBoundary() throws Exception {
        GateTask gate = new GateTask();
        try (var componentExecutor = Executors.newVirtualThreadPerTaskExecutor()) {
            Future<String> future = componentExecutor.submit(gate);
            try {
                await(gate.entered, "component task entry");
                require(!future.isDone() && gate.finished.getCount() == 1,
                        "Task did not remain active after Future was delivered");
                System.out.println("component-owner: Future delivered while task remains active");
            } finally {
                gate.mayFinish.countDown();
            }
            require("done".equals(future.get(5, TimeUnit.SECONDS)), "Unexpected component result");
        }
        require(gate.finished.getCount() == 0, "Component did not drain task");
    }

    public static void main(String[] args) throws Exception {
        Thread.ofPlatform().daemon().name("validation-watchdog").start(() -> {
            try {
                Thread.sleep(30_000);
                System.err.println("FAIL: validation exceeded watchdog bound");
                Runtime.getRuntime().halt(2);
            } catch (InterruptedException ignored) { }
        });
        snippetReturnBoundary();
        timeoutThenCloseBoundary();
        componentOwnerBoundary();
        System.out.println("PASS: method return, Future state and actual task termination are distinct boundaries");
    }
}
