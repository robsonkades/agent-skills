import fs from 'node:fs';
import crypto from 'node:crypto';

const dir = 'C:/Users/robso/AppData/Local/Temp/skills-wave2-validation-Bu1zNV/poison-messages-and-dlq';
fs.mkdirSync(`${dir}/sources`, { recursive: true });
const queries = [
  ['kafka-design.html', 'https://kafka.apache.org/41/design/design/'],
  ['kafka-consumer.html', 'https://kafka.apache.org/41/javadoc/org/apache/kafka/clients/consumer/KafkaConsumer.html'],
  ['kafka-headers.html', 'https://kafka.apache.org/41/javadoc/org/apache/kafka/common/header/Headers.html'],
  ['kafka-header.html', 'https://kafka.apache.org/41/javadoc/org/apache/kafka/common/header/Header.html'],
  ['spring-tombstones.html', 'https://docs.spring.io/spring-kafka/reference/3.3/kafka/tombstones.html'],
  ['spring-error-handling.html', 'https://docs.spring.io/spring-kafka/reference/3.3/kafka/annotation-error-handling.html'],
  ['sqs-dlq.html', 'https://docs.aws.amazon.com/AWSSimpleQueueService/latest/SQSDeveloperGuide/sqs-dead-letter-queues.html'],
  ['sqs-redrive.html', 'https://docs.aws.amazon.com/AWSSimpleQueueService/latest/SQSDeveloperGuide/sqs-configure-dead-letter-queue-redrive.html'],
  ['s3-lifecycle.html', 'https://docs.aws.amazon.com/AmazonS3/latest/userguide/object-lifecycle-mgmt.html'],
  ['s3-presigned.html', 'https://docs.aws.amazon.com/AmazonS3/latest/userguide/using-presigned-url.html'],
  ['pubsub-dead-letter.html', 'https://cloud.google.com/pubsub/docs/dead-letter-topics'],
];
const results = await Promise.allSettled(queries.map(async ([file, url]) => {
  const response = await fetch(url, { signal: AbortSignal.timeout(25000) });
  const content = await response.text();
  fs.writeFileSync(`${dir}/sources/${file}`, content, { flag: 'wx' });
  return { file, url, status: response.status, finalUrl: response.url, bytes: Buffer.byteLength(content), sha256: crypto.createHash('sha256').update(content).digest('hex') };
}));
const sources = results.map((result, i) => result.status === 'fulfilled' ? result.value : { file: queries[i][0], url: queries[i][1], error: String(result.reason) });
fs.writeFileSync(`${dir}/source-index.json`, JSON.stringify(sources, null, 2) + '\n');
console.log(JSON.stringify(sources, null, 2));
if (sources.some((source) => source.status !== 200)) process.exitCode = 1;
