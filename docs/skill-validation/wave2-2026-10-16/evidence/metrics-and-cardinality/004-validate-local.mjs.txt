import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { pathToFileURL } from 'node:url';

const repo = resolve(process.argv[2] ?? 'C:/git/agent-skills');
const skill = resolve(repo, 'skills/metrics-and-cardinality');
const { parseManifest } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/skill-document.ts')));
const { manifest, issues } = parseManifest(readFileSync(resolve(skill, 'skill.yaml'), 'utf8'), { strict: true });
const document = parseSkillDocument(readFileSync(resolve(skill, 'SKILL.md'), 'utf8'));
assert.deepEqual(issues, []);
assert.equal(documentName(document), manifest.name);
assert.equal(documentDescription(document), manifest.description);
assert.equal(manifest.name, 'metrics-and-cardinality');
assert.equal(manifest.version, '1.3.5');
for (const file of manifest.files) assert.ok(existsSync(resolve(skill, file)), `Missing declared path ${file}`);
const baseline = {
  "SKILL.md": "2c06a11dc5b658dfd2db160b75154a4eec9b9c7f1cfece45f7c837257d2ddc07",
  "skill.yaml": "e68731b9ecbdc19529040f4dd9ce5cb02dae566a3257af98bffcf1cf94dea79f",
  "references/cardinality-budget.md": "b0fe2f7922b4d932d558090d1cfd3d9a3b92d276da6f85ea863575b2de608389",
  "references/instrument-selection.md": "f40b3a7b878d832c3b41dfb76f161523bb275bb8878d77601b14a272ca7d165a",
  "references/micrometer-and-prometheus.md": "f91cf39f4ba0deac3bc7a986a30fb82b1b331926dc4de485fdef8a8a157a36c4"
};
const permittedChanges = ["SKILL.md","skill.yaml","references/micrometer-and-prometheus.md"];
const actualChanges = [];
let relativeLinks = 0;
for (const [file, hash] of Object.entries(baseline)) {
  const path = resolve(skill, file);
  const content = readFileSync(path);
  if (createHash('sha256').update(content).digest('hex') !== hash) actualChanges.push(file);
  if (!file.endsWith('.md')) continue;
  for (const match of content.toString('utf8').matchAll(/\]\(([^\s)]+)\)/g)) {
    const target = match[1].replace(/^<|>$/g, '');
    if (/^(https?:|#)/.test(target)) continue;
    const [resource] = target.split('#');
    assert.ok(existsSync(resolve(dirname(path), resource)), `Missing resource ${target}`);
    relativeLinks++;
  }
}
assert.deepEqual(actualChanges.sort(), permittedChanges.sort());
console.log(JSON.stringify({ status: 'PASS', scope: manifest.name, manifestIssues: issues.length,
  metadataAgreement: true, version: manifest.version, declaredPaths: manifest.files.length,
  relativeLinks, changedFromAssignment: actualChanges, unchangedReferences: 2 }, null, 2));
