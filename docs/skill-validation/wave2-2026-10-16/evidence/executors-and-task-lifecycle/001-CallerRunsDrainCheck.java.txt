import java.util.List;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.RejectedExecutionException;
import java.util.concurrent.RejectedExecutionHandler;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;

public final class CallerRunsDrainCheck {
    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private static void await(CountDownLatch latch, String message) throws InterruptedException {
        require(latch.await(5, TimeUnit.SECONDS), message);
    }

    private static void verify(RejectedExecutionHandler handler, boolean inlineExpected)
            throws InterruptedException {
        var pool = new ThreadPoolExecutor(1, 1, 0, TimeUnit.MILLISECONDS,
                new ArrayBlockingQueue<Runnable>(1), handler);
        var workerStarted = new CountDownLatch(1);
        var releaseWorker = new CountDownLatch(1);
        var bodyStarted = new CountDownLatch(1);
        var releaseBody = new CountDownLatch(1);
        var bodyFinished = new CountDownLatch(1);
        var submitterDone = new CountDownLatch(1);
        var rejected = new AtomicBoolean();
        var callerInterrupted = new AtomicBoolean();
        var queuedRan = new AtomicBoolean();
        var failure = new AtomicReference<Throwable>();
        var submitter = new Thread(() -> {
            try {
                pool.execute(() -> {
                    bodyStarted.countDown();
                    try {
                        releaseBody.await();
                    } catch (InterruptedException interrupted) {
                        callerInterrupted.set(true);
                        Thread.currentThread().interrupt();
                    } finally {
                        bodyFinished.countDown();
                    }
                });
            } catch (RejectedExecutionException expected) {
                rejected.set(true);
            } catch (Throwable unexpected) {
                failure.set(unexpected);
            } finally {
                submitterDone.countDown();
            }
        }, "external-submitter");

        try {
            pool.execute(() -> {
                workerStarted.countDown();
                try {
                    releaseWorker.await();
                } catch (InterruptedException interrupted) {
                    Thread.currentThread().interrupt();
                }
            });
            await(workerStarted, "worker did not start");
            pool.execute(() -> queuedRan.set(true));
            submitter.start();
            if (inlineExpected) {
                await(bodyStarted, "caller-run body did not start");
            } else {
                await(submitterDone, "AbortPolicy did not return refusal");
                require(rejected.get(), "AbortPolicy did not reject");
                require(bodyStarted.getCount() == 1, "refused body executed");
            }

            List<Runnable> neverStarted = pool.shutdownNow();
            boolean terminated = pool.awaitTermination(5, TimeUnit.SECONDS);
            require(terminated, "pool did not terminate");
            require(neverStarted.size() == 1, "queued task not returned");
            require(!queuedRan.get(), "queued task unexpectedly ran");
            boolean callerActive = bodyStarted.getCount() == 0 && bodyFinished.getCount() != 0;
            require(callerActive == inlineExpected, "unexpected inline physical completion state");
            require(!callerInterrupted.get(), "shutdownNow interrupted external submitter");
            if (inlineExpected) require(submitter.isAlive(), "inline submitter already exited");

            releaseBody.countDown();
            await(submitterDone, "submitter did not complete after release");
            submitter.join(5000);
            require(!submitter.isAlive(), "submitter remains alive");
            require(failure.get() == null, "unexpected submitter failure: " + failure.get());
            if (inlineExpected) {
                require(bodyFinished.getCount() == 0, "body not physically complete after release");
                require(!rejected.get(), "caller-run submission refused");
            }
            System.out.printf(
                    "{\"policy\":\"%s\",\"poolTerminated\":%s,\"callerActiveAtTermination\":%s,"
                            + "\"callerInterruptedByShutdownNow\":%s,\"queuedReturned\":%d,"
                            + "\"refused\":%s,\"submitterJoined\":true}%n",
                    handler.getClass().getSimpleName(), terminated, callerActive,
                    callerInterrupted.get(), neverStarted.size(), rejected.get());
        } finally {
            releaseWorker.countDown();
            releaseBody.countDown();
            pool.shutdownNow();
            require(pool.awaitTermination(5, TimeUnit.SECONDS), "cleanup pool remains active");
            if (submitter.getState() != Thread.State.NEW) {
                submitter.join(5000);
                require(!submitter.isAlive(), "cleanup submitter remains active");
            }
        }
    }

    public static void main(String[] args) throws InterruptedException {
        verify(new ThreadPoolExecutor.CallerRunsPolicy(), true);
        verify(new ThreadPoolExecutor.AbortPolicy(), false);
    }
}
