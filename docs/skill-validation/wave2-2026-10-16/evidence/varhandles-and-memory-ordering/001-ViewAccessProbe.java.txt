import java.lang.invoke.MethodHandles;
import java.lang.invoke.VarHandle;
import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import java.nio.ReadOnlyBufferException;

public class ViewAccessProbe {
    private static final VarHandle ARRAY =
            MethodHandles.byteArrayViewVarHandle(int[].class, ByteOrder.LITTLE_ENDIAN);
    private static final VarHandle BUFFER =
            MethodHandles.byteBufferViewVarHandle(int[].class, ByteOrder.LITTLE_ENDIAN);

    public static void main(String[] args) {
        System.out.println("Java=" + Runtime.version()
                + " VM=" + System.getProperty("java.vm.name")
                + " arch=" + System.getProperty("os.arch"));
        System.out.println("GET_ACQUIRE type=" + BUFFER.accessModeType(VarHandle.AccessMode.GET_ACQUIRE));
        System.out.println("buffer GET_ACQUIRE supported="
                + BUFFER.isAccessModeSupported(VarHandle.AccessMode.GET_ACQUIRE));
        if (args.length == 1) {
            // Deliberately incomplete adapter: handle-level support is its only preflight.
            ByteBuffer storage = switch (args[0]) {
                case "claim-direct" -> ByteBuffer.allocateDirect(32).alignedSlice(Integer.BYTES);
                case "claim-heap" -> ByteBuffer.allocate(32);
                default -> throw new IllegalArgumentException("Unknown case: " + args[0]);
            };
            if (!BUFFER.isAccessModeSupported(VarHandle.AccessMode.GET_ACQUIRE)) {
                throw new UnsupportedOperationException("GET_ACQUIRE unsupported");
            }
            System.out.println("candidate direct=" + storage.isDirect() + " byteOffset=0");
            int observed = (int) BUFFER.getAcquire(storage, 0);
            require(observed == 0, "fresh storage must read zero");
            System.out.println("PASS: requested coordinate accepted");
            return;
        }

        byte[] bytes = new byte[32];
        ARRAY.set(bytes, 0, 41);
        require((int) ARRAY.get(bytes, 0) == 41, "byte-array plain access");
        require(!ARRAY.isAccessModeSupported(VarHandle.AccessMode.GET_ACQUIRE),
                "byte-array acquire is unsupported on this JDK 25 runtime");
        expect(UnsupportedOperationException.class, "byte-array acquire", () -> {
            int ignored = (int) ARRAY.getAcquire(bytes, 0);
        });

        ByteBuffer heap = ByteBuffer.allocate(32);
        BUFFER.set(heap, 0, 41);
        require((int) BUFFER.get(heap, 0) == 41, "heap-buffer plain access");
        require(BUFFER.isAccessModeSupported(VarHandle.AccessMode.GET_ACQUIRE),
                "buffer handle supports acquire for eligible coordinates");
        expect(IllegalStateException.class, "heap-buffer acquire at offset 0", () -> {
            int ignored = (int) BUFFER.getAcquire(heap, 0);
        });

        ByteBuffer direct = ByteBuffer.allocateDirect(32).alignedSlice(Integer.BYTES);
        require(direct.alignmentOffset(0, Integer.BYTES) == 0, "aligned direct coordinate");
        BUFFER.setRelease(direct, 0, 42);
        require((int) BUFFER.getAcquire(direct, 0) == 42, "aligned direct release/acquire");
        System.out.println("PASS: aligned direct setRelease/getAcquire = 42");
        expect(IllegalStateException.class, "direct acquire at byte offset 1", () -> {
            int ignored = (int) BUFFER.getAcquire(direct, 1);
        });

        ByteBuffer readOnly = direct.asReadOnlyBuffer();
        require((int) BUFFER.getAcquire(readOnly, 0) == 42, "read-only direct acquire");
        expect(ReadOnlyBufferException.class, "read-only direct release", () -> {
            BUFFER.setRelease(readOnly, 0, 43);
        });
        require((int) BUFFER.getAcquire(direct, 0) == 42, "rejected write preserved value");
        System.out.println("PASS: every observed support/coordinate result matches the JDK 25 factory contract");
    }

    private static void require(boolean condition, String message) {
        if (!condition) {
            throw new AssertionError(message);
        }
    }

    private static void expect(Class<? extends Throwable> expected, String label, Runnable action) {
        try {
            action.run();
        } catch (Throwable observed) {
            if (!expected.isInstance(observed)) {
                throw new AssertionError(label + ": expected " + expected.getName()
                        + ", got " + observed, observed);
            }
            System.out.println("PASS: " + label + " -> " + observed.getClass().getSimpleName());
            return;
        }
        throw new AssertionError(label + ": expected " + expected.getName() + ", no exception");
    }
}
