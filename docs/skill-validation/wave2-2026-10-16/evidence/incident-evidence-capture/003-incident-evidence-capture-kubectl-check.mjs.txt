import assert from 'node:assert/strict';
import { createServer } from 'node:http';
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { mkdirSync, mkdtempSync, writeFileSync } from 'node:fs';
import path from 'node:path';

const run = promisify(execFile);
const kubectl = 'C:/Program Files/Rancher Desktop/resources/resources/win32/bin/kubectl.exe';
const root = mkdtempSync(path.join(path.dirname(new URL(import.meta.url).pathname.replace(/^\/(\w:)/, '$1')), 'incident-evidence-capture-client-'));
mkdirSync(path.join(root, 'cache'));
const requests = [];
const pod = { apiVersion: 'v1', kind: 'Pod',
  metadata: { name: 'probe-0', namespace: 'incident', uid: 'synthetic-pod-uid', labels: { app: 'probe' } },
  spec: { containers: [{ name: 'app', image: 'synthetic' }] },
  status: { phase: 'Running', containerStatuses: [{ name: 'app', ready: true, restartCount: 1, containerID: 'synthetic://current', state: { running: { startedAt: '2026-10-16T10:00:00Z' } } }] }
};
const lines = Array.from({ length: 25 }, (_, i) => '2026-10-16T10:00:' + String(i).padStart(2, '0') + 'Z marker-' + i);
const server = createServer((req, res) => {
  const url = new URL(req.url, 'http://localhost');
  requests.push({ method: req.method, pathname: url.pathname, query: Object.fromEntries(url.searchParams) });
  res.setHeader('Content-Type', 'application/json');
  let body;
  if (url.pathname === '/api') body = { kind: 'APIVersions', apiVersion: 'v1', versions: ['v1'], serverAddressByClientCIDRs: [] };
  else if (url.pathname === '/apis') body = { kind: 'APIGroupList', apiVersion: 'v1', groups: [] };
  else if (url.pathname === '/api/v1') body = { kind: 'APIResourceList', apiVersion: 'v1', groupVersion: 'v1', resources: [{ name: 'pods', singularName: 'pod', namespaced: true, kind: 'Pod', verbs: ['get', 'list'] }, { name: 'pods/log', singularName: '', namespaced: true, kind: 'Pod', verbs: ['get'] }] };

  else if (url.pathname === '/api/v1/namespaces/incident/pods/probe-0') body = pod;
  else if (url.pathname === '/api/v1/namespaces/incident/pods/probe-0/log') {
    res.setHeader('Content-Type', 'text/plain');
    const tail = url.searchParams.get('tailLines');
    const selected = tail == null ? lines : lines.slice(-Number(tail));
    res.end(selected.join('\n') + '\n');
    return;
  } else {
    res.statusCode = 404;
    body = { kind: 'Status', apiVersion: 'v1', status: 'Failure', reason: 'NotFound', message: url.pathname, code: 404 };
  }
  res.end(JSON.stringify(body));

const config = path.join(root, 'config.json');

const env = { ...process.env, KUBECONFIG: config, HOME: root, USERPROFILE: root, KUBECTL_KUBERC: 'false', HTTP_PROXY: '', HTTPS_PROXY: '', ALL_PROXY: '', NO_PROXY: '127.0.0.1,localhost' };
const base = ['--kubeconfig=' + config, '--cache-dir=' + path.join(root, 'cache'), '--request-timeout=3s', '-n', 'incident', 'logs'];
const checks = [];
try {
  const version = JSON.parse((await run(kubectl, ['version', '--client', '--output=json'], { env, windowsHide: true, timeout: 10000 })).stdout).clientVersion;

    ['previous-container', ['probe-0', '--previous'], undefined, 'true']

    requests.length = 0;

    const logRequests = requests.filter(r => r.pathname.endsWith('/log'));
    assert.equal(logRequests.length, 1, name);
    const request = logRequests[0];
    assert.equal(request.query.tailLines, expectedTail, name);
    assert.equal(request.query.previous, expectedPrevious, name);
    assert.equal(request.query.container, 'app', name);
    assert.equal(request.query.sinceTime, '2026-10-16T09:59:00Z', name);

  }
  const report = { status: 'PASS', scope: 'Real kubectl client request construction against synthetic localhost API only; no kubelet, rotation, retention, cluster or application was tested.', clientVersion: version, artifacts: root, checks };

  console.log(JSON.stringify(report));

}
