import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Set;
import java.util.TreeSet;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.RejectedExecutionException;
import java.util.concurrent.RejectedExecutionHandler;
import java.util.concurrent.Semaphore;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicBoolean;

// Technical control of local ownership only. No broker behavior is simulated or proved.
public final class QueueDeliveryControl {
    private final Semaphore permits = new Semaphore(3);
    private final Set<String> received = ConcurrentHashMap.newKeySet();
    private final Set<String> completed = ConcurrentHashMap.newKeySet();
    private final Set<String> recovered = ConcurrentHashMap.newKeySet();
    private final Set<String> duplicates = ConcurrentHashMap.newKeySet();
    private final List<String> executionThreads = Collections.synchronizedList(new ArrayList<>());
    private int normalSubmissionReturns;

    private final class Delivery implements Runnable {
        private final String id;
        private final CountDownLatch started;
        private final CountDownLatch unblock;
        private final AtomicBoolean settled = new AtomicBoolean();

        private Delivery(String id, CountDownLatch started, CountDownLatch unblock) {
            this.id = id;
            this.started = started;
            this.unblock = unblock;
        }

        @Override public void run() {
            executionThreads.add(id + "@" + Thread.currentThread().getName());
            try {
                if (started != null) started.countDown();
                if (unblock != null && !unblock.await(5, TimeUnit.SECONDS)) {
                    throw new AssertionError("fixture did not release first handler");
                }
                settle(completed);
            } catch (InterruptedException failure) {
                Thread.currentThread().interrupt();
                settle(recovered);
            }
        }

        private void recover() { settle(recovered); }

        private void settle(Set<String> destination) {
            if (settled.compareAndSet(false, true)) {
                destination.add(id);
                permits.release();
            } else {
                duplicates.add(id);
            }
        }
    }

    private void submit(ThreadPoolExecutor pool, String id,
                        CountDownLatch started, CountDownLatch unblock) {
        if (!permits.tryAcquire()) throw new AssertionError("fixture exceeded intake capacity");
        received.add(id);
        Delivery delivery = new Delivery(id, started, unblock);
        try {
            pool.execute(delivery);
            normalSubmissionReturns++;
        } catch (RejectedExecutionException rejected) {
            delivery.recover();
        }
    }

    private static RejectedExecutionHandler policy(String name) {
        switch (name) {
            case "abort": return new ThreadPoolExecutor.AbortPolicy();
            case "discard": return new ThreadPoolExecutor.DiscardPolicy();
            case "discard-oldest": return new ThreadPoolExecutor.DiscardOldestPolicy();
            case "caller-runs": return new ThreadPoolExecutor.CallerRunsPolicy();
            case "recover":
                // Intentional non-throwing handler: explicit delivery recovery owns settlement.
                return (task, pool) -> ((QueueDeliveryControl.Delivery) task).recover();
            default: throw new IllegalArgumentException(name);
        }
    }

    private void run(String policyName, String phase) throws Exception {
        ThreadPoolExecutor pool = new ThreadPoolExecutor(
                1, 1, 0, TimeUnit.MILLISECONDS, new ArrayBlockingQueue<>(1),
                policy(policyName));
        CountDownLatch started = new CountDownLatch(1);
        CountDownLatch unblock = new CountDownLatch(1);
        try {
            if (phase.equals("saturation")) {
                submit(pool, "first", started, unblock);
                if (!started.await(5, TimeUnit.SECONDS)) throw new AssertionError("handler not started");
                submit(pool, "second", null, null);
                submit(pool, "third", null, null);
            } else if (phase.equals("shutdown")) {
                pool.shutdown();
                submit(pool, "first", null, null);
            } else {
                throw new IllegalArgumentException(phase);
            }
        } finally {
            unblock.countDown();
            pool.shutdown();
            if (!pool.awaitTermination(5, TimeUnit.SECONDS)) {
                pool.shutdownNow();
                throw new AssertionError("fixture executor did not terminate");
            }
        }
        Set<String> unresolved = new TreeSet<>(received);
        unresolved.removeAll(completed);
        unresolved.removeAll(recovered);
        Set<String> overlap = new TreeSet<>(completed);
        overlap.retainAll(recovered);
        System.out.println("policy=" + policyName + " phase=" + phase
                + " normalReturns=" + normalSubmissionReturns
                + " received=" + new TreeSet<>(received)
                + " completed=" + new TreeSet<>(completed)
                + " recovered=" + new TreeSet<>(recovered)
                + " unresolved=" + unresolved
                + " permits=" + permits.availablePermits()
                + " executorTerminated=" + pool.isTerminated()
                + " threads=" + executionThreads);
        if (!unresolved.isEmpty() || !overlap.isEmpty() || !duplicates.isEmpty()
                || permits.availablePermits() != 3) {
            throw new AssertionError("every received item must complete or recover exactly once; "
                    + "unresolved=" + unresolved + " duplicateSettlement=" + duplicates);
        }
    }

    public static void main(String[] args) throws Exception {
        new QueueDeliveryControl().run(args[0], args[1]);
    }
}
