import java.nio.file.Path;
import java.time.Duration;
import java.util.Map;
import java.util.TreeMap;
import jdk.jfr.Recording;
import jdk.jfr.consumer.RecordedEvent;
import jdk.jfr.consumer.RecordingFile;

public class InflationProbe {
    static final class TargetLock { int value; }
    static final class OtherLock {}

    public static void main(String[] args) throws Exception {
        if (args.length != 2 || !(args[0].equals("consecutive") || args[0].equals("interleaved"))) {
            throw new IllegalArgumentException("mode and output path required");
        }
        String mode = args[0];
        Path output = Path.of(args[1]);
        long checksum = 0;
        try (Recording recording = new Recording()) {
            recording.enable("jdk.JavaMonitorInflate").withThreshold(Duration.ZERO).withStackTrace();
            recording.start();
            for (int i = 0; i < 1000; i++) {
                TargetLock a = new TargetLock();
                OtherLock b = new OtherLock();
                synchronized (a) {
                    if (mode.equals("interleaved")) {
                        synchronized (b) {
                            synchronized (a) { a.value++; }
                        }
                    } else {
                        synchronized (a) { a.value++; }
                    }
                }
                checksum += a.value;
            }
            recording.stop();
            recording.dump(output);
        }
        long count = 0;
        long totalNanos = 0;
        Map<String, Long> causes = new TreeMap<>();
        Map<String, Long> threads = new TreeMap<>();
        for (RecordedEvent event : RecordingFile.readAllEvents(output)) {
            if (!event.getEventType().getName().equals("jdk.JavaMonitorInflate")
                    || !event.getClass("monitorClass").getName().equals(TargetLock.class.getName())) {
                continue;
            }
            count++;
            totalNanos += event.getDuration().toNanos();
            causes.merge(event.getString("cause"), 1L, Long::sum);
            threads.merge(event.getThread().getJavaName(), 1L, Long::sum);
        }
        System.out.println("runtime=" + Runtime.version());
        System.out.println("mode=" + mode + ", iterations=1000, checksum=" + checksum);
        System.out.println("targetInflations=" + count + ", totalInflationNanos=" + totalNanos
                + ", causes=" + causes + ", threads=" + threads);
        if (checksum != 1000) throw new AssertionError("Incomplete workload");
        if (mode.equals("interleaved") && causes.getOrDefault("Monitor Enter", 0L) == 0) {
            throw new AssertionError("Expected Monitor Enter inflation in the selected implementation");
        }
        if (mode.equals("consecutive") && count != 0) {
            throw new AssertionError("Expected no target inflation for the selected fast-recursive control");
        }
    }
}
