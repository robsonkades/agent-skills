import java.lang.foreign.Arena;
import java.lang.foreign.MemorySegment;
import java.lang.foreign.ValueLayout;
import java.nio.ByteBuffer;
import java.nio.ReadOnlyBufferException;

public final class LeaseScopeEvidenceCheck {
    // Isolated counterexample: checked handles coexist with deliberately escaped views.
    // This one-slot sequential fixture is not a production pool implementation.
    static final class OneSlotPool implements AutoCloseable {
        private final Arena arena = Arena.ofShared();
        private final MemorySegment block = arena.allocate(1);
        private boolean leased;
        private long generation;

        synchronized Lease acquire(byte value) {
            if (leased) throw new IllegalStateException("slot is leased");
            block.set(ValueLayout.JAVA_BYTE, 0, value);
            leased = true;
            return new Lease(this, ++generation);
        }

        private void requireCurrent(Lease lease) {
            if (!leased || lease.returned || lease.generation != generation) {
                throw new IllegalStateException("lease is no longer current");
            }
        }

        synchronized byte read(Lease lease) {
            requireCurrent(lease);
            return block.get(ValueLayout.JAVA_BYTE, 0);
        }

        synchronized MemorySegment view(Lease lease) {
            requireCurrent(lease);
            return block.asSlice(0, 1).asReadOnly();
        }

        synchronized byte[] copy(Lease lease) {
            requireCurrent(lease);
            return block.toArray(ValueLayout.JAVA_BYTE);
        }

        synchronized void release(Lease lease) {
            requireCurrent(lease);
            block.fill((byte) 0);
            lease.returned = true;
            leased = false;
        }

        @Override
        public synchronized void close() {
            if (leased) throw new IllegalStateException("outstanding lease");
            arena.close();
        }
    }

    static final class Lease implements AutoCloseable {
        private final OneSlotPool pool;
        private final long generation;
        private boolean returned;
        Lease(OneSlotPool pool, long generation) {
            this.pool = pool;
            this.generation = generation;
        }
        byte readChecked() { return pool.read(this); }
        MemorySegment readOnlyView() { return pool.view(this); }
        byte[] copy() { return pool.copy(this); }
        @Override public void close() { pool.release(this); }
    }

    public static void main(String[] args) throws Exception {
        Thread deadline = new Thread(() -> {
            try {
                Thread.sleep(30_000);
                System.err.println("Validation exceeded its 30-second deadline");
                Runtime.getRuntime().halt(124);
            } catch (InterruptedException finished) {
                Thread.currentThread().interrupt();
            }
        });
        deadline.setDaemon(true);
        deadline.start();
        try {
            verify();
        } finally {
            deadline.interrupt();
        }
    }

    static void verify() {
        MemorySegment escaped;
        ByteBuffer buffer;
        byte[] copy;
        try (OneSlotPool pool = new OneSlotPool()) {
            Lease first = pool.acquire((byte) 11);
            escaped = first.readOnlyView();
            buffer = escaped.asByteBuffer();
            copy = first.copy();
            require(escaped.get(ValueLayout.JAVA_BYTE, 0) == 11, "initial segment");
            require(buffer.get(0) == 11, "initial buffer");
            first.close();
            expectIllegalState(first::readChecked, "checked read after release");
            require(escaped.scope().isAlive(), "arena remains alive");
            require(escaped.get(ValueLayout.JAVA_BYTE, 0) == 0, "zeroed slot still visible");
            try (Lease second = pool.acquire((byte) 99)) {
                require(second.readChecked() == 99, "current lease");
                require(escaped.get(ValueLayout.JAVA_BYTE, 0) == 99, "stale segment aliases reused block");
                require(buffer.get(0) == 99, "stale buffer aliases reused block");
                require(copy[0] == 11, "copy keeps prior value");
                try {
                    buffer.put(0, (byte) 42);
                    throw new AssertionError("read-only view allowed write");
                } catch (ReadOnlyBufferException expected) {
                    // The restriction blocks writes, but does not isolate later contents.
                }
                System.out.println("afterRelease: checkedRead=REJECTED, scopeAlive=true");
                System.out.println("afterReuse: staleReadOnlySegment=99, staleReadOnlyBuffer=99, independentCopy=11");
            }
        }
        MemorySegment finalEscaped = escaped;
        ByteBuffer finalBuffer = buffer;
        expectIllegalState(() -> finalEscaped.get(ValueLayout.JAVA_BYTE, 0), "segment after arena close");
        expectIllegalState(() -> finalBuffer.get(0), "buffer after arena close");
        require(!escaped.scope().isAlive(), "arena closed");
        System.out.println("afterArenaClose: segmentRead=REJECTED, bufferRead=REJECTED");
        System.out.println("PASS: logical lease return and arena close have distinct access boundaries");
    }

    @FunctionalInterface
    interface Access { byte read(); }

    static void expectIllegalState(Access access, String label) {
        try {
            access.read();
            throw new AssertionError(label + " unexpectedly succeeded");
        } catch (IllegalStateException expected) {
            // Expected controlled rejection.
        }
    }

    static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
