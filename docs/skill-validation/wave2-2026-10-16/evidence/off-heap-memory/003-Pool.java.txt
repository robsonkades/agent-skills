import java.lang.foreign.Arena;
import java.lang.foreign.MemorySegment;
import java.lang.foreign.ValueLayout;

public final class Pool implements AutoCloseable {
    private final Arena arena = Arena.ofShared();
    private final MemorySegment block = arena.allocate(1);
    private boolean leased;
    private long generation;

    public synchronized Lease acquire(byte value) {
        if (leased) throw new IllegalStateException("slot is leased");
        block.set(ValueLayout.JAVA_BYTE, 0, value);
        leased = true;
        return new Lease(this, ++generation);
    }

    private void requireCurrent(Lease lease) {
        if (!leased || lease.returned || lease.generation != generation) {
            throw new IllegalStateException("lease is no longer current");
        }
    }

    private synchronized byte read(Lease lease) {
        requireCurrent(lease);
        return block.get(ValueLayout.JAVA_BYTE, 0);
    }

    private synchronized MemorySegment view(Lease lease) {
        requireCurrent(lease);
        return block.asSlice(0, 1).asReadOnly();
    }

    private synchronized byte[] copy(Lease lease) {
        requireCurrent(lease);
        return block.toArray(ValueLayout.JAVA_BYTE);
    }

    private synchronized void release(Lease lease) {
        requireCurrent(lease);
        block.fill((byte) 0);
        lease.returned = true;
        leased = false;
    }

    @Override
    public synchronized void close() {
        if (leased) throw new IllegalStateException("outstanding lease");
        arena.close();
    }

    public static final class Lease implements AutoCloseable {
        private final Pool pool;
        private final long generation;
        private boolean returned;
        private Lease(Pool pool, long generation) {
            this.pool = pool;
            this.generation = generation;
        }
        public byte readChecked() { return pool.read(this); }
        public MemorySegment readOnlyView() { return pool.view(this); }
        public byte[] copy() { return pool.copy(this); }
        @Override public void close() { pool.release(this); }
    }
}
