import java.lang.foreign.ValueLayout;

public final class Consumer {
    static byte consume(Pool pool, byte value) {
        try (Pool.Lease lease = pool.acquire(value)) {
            return lease.readOnlyView().get(ValueLayout.JAVA_BYTE, 0);
        }
    }

    public static void main(String[] args) {
        try (Pool pool = new Pool()) {
            System.out.println("first=" + consume(pool, (byte) 11));
            System.out.println("second=" + consume(pool, (byte) 99));
        }
    }
}
