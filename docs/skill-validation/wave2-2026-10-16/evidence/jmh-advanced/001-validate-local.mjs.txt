import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { pathToFileURL } from 'node:url';

const repo = resolve(process.argv[2] ?? 'C:/git/agent-skills');
const skill = resolve(repo, 'skills/jmh-advanced');
const { parseManifest } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(
  pathToFileURL(resolve(repo, 'packages/core/src/domain/skill-document.ts'))
);
const { manifest, issues } = parseManifest(readFileSync(resolve(skill, 'skill.yaml'), 'utf8'), { strict: true });
const document = parseSkillDocument(readFileSync(resolve(skill, 'SKILL.md'), 'utf8'));
assert.deepEqual(issues, []);
assert.equal(documentName(document), manifest.name);
assert.equal(documentDescription(document), manifest.description);
assert.equal(manifest.name, 'jmh-advanced');
assert.equal(manifest.version, '1.2.4');
for (const entry of manifest.files) assert.ok(existsSync(resolve(skill, entry)), `Missing declared file ${entry}`);

const baseline = {
  "SKILL.md": "057fd37d8e4bd505a2c5997b47d0e76fae479fdcd1855c598be3c5b76a5d5914",
  "skill.yaml": "b621eb65fdb8eacab84e154e01b115ea127a0fb4bce2f551fa208078af5d9ad1",
  "references/configuration-recipes.md": "635d312e729023cf3c482e014459c02c4b914083a852f3a9ce93a6779b83f305",
  "references/profilers-and-hsdis.md": "ae6da2594d67e6217f8eb2929697e6d2741b51a4523f44d52ee3d94edd7e8459"
};
let relativeLinks = 0;
let anchors = 0;
for (const [file, hash] of Object.entries(baseline)) {
  const path = resolve(skill, file);
  const bytes = readFileSync(path);
  assert.equal(createHash('sha256').update(bytes).digest('hex'), hash, `Package changed: ${file}`);
  if (!file.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\]\(([^\s)]+)\)/g)) {
    const target = match[1].replace(/^<|>$/g, '');
    if (/^(https?:|#)/.test(target)) continue;
    const [resource, fragment] = target.split('#');
    const resourcePath = resolve(dirname(path), resource);
    assert.ok(existsSync(resourcePath), `Missing resource ${target}`);
    relativeLinks++;
    if (fragment) {
      const headings = [...readFileSync(resourcePath, 'utf8').matchAll(/^#{1,6}\s+(.+)$/gm)]
        .map(value => value[1].trim().toLowerCase().replace(/[^\w\s-]/g, '').replace(/\s/g, '-'));
      assert.ok(headings.includes(fragment), `Missing heading ${target}`);
      anchors++;
    }
  }
}
console.log(JSON.stringify({ status: 'PASS', scope: manifest.name, manifestIssues: issues.length,
  metadataAgreement: true, version: manifest.version, declaredPaths: manifest.files.length,
  relativeLinks, anchors, allFourAssignmentHashesUnchanged: true }, null, 2));
