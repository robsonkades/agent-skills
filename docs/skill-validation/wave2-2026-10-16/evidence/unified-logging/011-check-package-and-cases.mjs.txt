import assert from 'node:assert/strict';
import { readFile, access } from 'node:fs/promises';
import { dirname, resolve, isAbsolute } from 'node:path';
import { fileURLToPath } from 'node:url';
import { loadPackageFromDirectory } from 'file:///C:/git/agent-skills/packages/core/src/application/package-loader.ts';
import { validatePackage } from 'file:///C:/git/agent-skills/packages/core/src/application/validate-package.ts';
import { NodeFileSystem } from 'file:///C:/git/agent-skills/packages/node/src/file-system.ts';

const artifacts = dirname(fileURLToPath(import.meta.url));
const root = 'C:/git/agent-skills/skills/unified-logging';
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), root, { strict: true });
const report = validatePackage(loaded.pkg, { directoryName: 'unified-logging' });
assert.deepEqual([...loaded.issues, ...report.issues], []);
assert.equal(loaded.pkg.manifest.version, '1.2.4');
let localLinks = 0;
for (const relative of ['SKILL.md', 'references/async-and-cost.md', 'references/legacy-flags.md', 'references/outputs-and-rotation.md', 'references/production-and-troubleshooting.md', 'references/runtime-reconfiguration.md', 'references/selection-syntax.md']) {
  const path = resolve(root, relative);
  for (const match of (await readFile(path, 'utf8')).matchAll(/\]\(([^)]+)\)/g)) {
    const target = match[1].split('#')[0];
    if (!target || /^[a-z]+:/i.test(target) || isAbsolute(target)) continue;
    await access(resolve(dirname(path), target));
    localLinks++;
  }
}
const runtimeReference = await readFile(resolve(root, 'references/runtime-reconfiguration.md'), 'utf8');
assert.ok(runtimeReference.includes('## Command composition and action scope'));
const composed = await readFile(resolve(artifacts, 'cases/composed-list/observe-output.ps1'), 'utf8');
const standalone = await readFile(resolve(artifacts, 'cases/standalone-list/observe-output.ps1'), 'utf8');
assert.equal(composed.replace(', ("output=" + $OutputName)', ''), standalone);
for (const name of ['context.txt', 'before-list.txt', 'TASK.md']) {
  assert.equal(await readFile(resolve(artifacts, 'cases/composed-list', name), 'utf8'), await readFile(resolve(artifacts, 'cases/standalone-list', name), 'utf8'));
}
const technical = JSON.parse(await readFile(resolve(artifacts, 'validated-final/results.json'), 'utf8'));
assert.equal(technical.cases.length, 5);
assert.ok(technical.cases.every(test => test.assertions === 'PASS'));
assert.equal(technical.allOwnedJvmsStopped, true);
console.log(JSON.stringify({ node: process.version, packagedFiles: loaded.pkg.files.length, issues: [], version: loaded.pkg.manifest.version, localLinks, caseDifference: 'one output=... argument only', recordedTechnicalConditionsPassed: technical.cases.length, behavioralEvaluations: 0 }, null, 2));
