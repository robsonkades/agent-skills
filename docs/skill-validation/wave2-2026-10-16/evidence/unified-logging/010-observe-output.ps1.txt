param(
    [Parameter(Mandatory)][string]$Jcmd,
    [Parameter(Mandatory)][int]$TargetPid,
    [Parameter(Mandatory)][string]$OutputName
)

$diagnosticArguments = @([string]$TargetPid, "VM.log", "list")
$listing = & $Jcmd @diagnosticArguments
if ($LASTEXITCODE -ne 0) {
    throw "VM.log inspection failed"
}
$listing | Where-Object {
    $_ -match '^\s*#\d+:' -and $_.Contains($OutputName + ' ')
}
