import assert from 'node:assert/strict';
import { spawn, spawnSync } from 'node:child_process';
import { readFile, writeFile, mkdir, readdir } from 'node:fs/promises';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = dirname(fileURLToPath(import.meta.url));
const bin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const childEnv = { ...process.env };
for (const name of ['JDK_JAVA_OPTIONS', 'JAVA_TOOL_OPTIONS', '_JAVA_OPTIONS']) delete childEnv[name];
const records = [];
const run = async (exe, args, cwd, name) => {
  const r = spawnSync(resolve(bin, exe + '.exe'), args, { cwd, env: childEnv, windowsHide: true, encoding: 'utf8', timeout: 10000 });
  const record = { executable: resolve(bin, exe + '.exe'), args, cwd, status: r.status, signal: r.signal, stdout: r.stdout, stderr: r.stderr, error: r.error?.message };
  await writeFile(resolve(cwd, name + '.json'), JSON.stringify(record, null, 2) + '\n');
  assert.equal(r.status, 0, name + ': ' + (r.error?.message ?? r.stderr));
  return record;
};
await run('java', ['-version'], root, 'java-version');
await run('javac', ['--release', '25', '-d', root, resolve(root, 'LoggingProbe.java')], root, 'javac');
await run('java', ['-Xlog:help'], root, 'xlog-help');
const cases = [
  { name: 'list-only', action: ['list'] },
  { name: 'list-with-output', action: ['output=file=a.log', 'list'] },
  { name: 'rotate-with-output', action: ['output=file=a.log', 'what=gc=info', 'decorators=time,level,tags', 'rotate'] },
  { name: 'rotate-false', action: ['rotate=false'] },
  { name: 'disable-false', action: ['disable=false'] },
];
const configuration = (text) => text.split(/\r?\n/).map(line => line.trim()).filter(line => /^#\d+:/.test(line));
const output = (lines, name) => lines.find(line => line.includes('file=' + name + ' '));
for (const test of cases) {
  const cwd = resolve(root, test.name);
  await mkdir(cwd);
  const args = ['-Xms16m', '-Xmx32m', '-XX:+UseSerialGC',
    '-Xlog:gc=info:file=a.log:time,level,tags:filecount=2,filesize=0',
    '-Xlog:class+load=info:file=b.log:time,level,tags:filecount=2,filesize=0',
    '-cp', root, 'LoggingProbe'];
  const vm = spawn(resolve(bin, 'java.exe'), args, { cwd, env: childEnv, windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] });
  let stdout = '', stderr = '';
  vm.stdout.on('data', data => { stdout += data.toString(); });
  vm.stderr.on('data', data => { stderr += data.toString(); });
  const exit = new Promise((done, reject) => { vm.once('exit', (code, signal) => done({ code, signal })); vm.once('error', reject); });
  try {
    await new Promise((done, reject) => {
      const timer = setTimeout(() => reject(new Error('probe readiness timeout')), 5000);
      const accept = data => {
        if (!data.toString().includes('READY ')) return;
        clearTimeout(timer);
        vm.stdout.off('data', accept);
        done();
      };
      vm.stdout.on('data', accept);
    });
    await run('jcmd', [String(vm.pid), 'help', 'VM.log'], cwd, 'help');
    const before = await run('jcmd', [String(vm.pid), 'VM.log', 'list'], cwd, 'before-list');
    const initial = configuration(before.stdout);
    assert.ok(output(initial, 'a.log')?.includes('gc=info'));
    assert.ok(output(initial, 'b.log')?.includes('class+load=info'));
    assert.ok((await readFile(resolve(cwd, 'a.log'), 'utf8')).includes('[gc]'));
    assert.ok((await readFile(resolve(cwd, 'b.log'), 'utf8')).includes('[class,load]'));
    const initialFiles = await readdir(cwd);
    assert.ok(!initialFiles.includes('a.log.0') && !initialFiles.includes('b.log.0'));
    const action = await run('jcmd', [String(vm.pid), 'VM.log', ...test.action], cwd, 'action');
    const after = await run('jcmd', [String(vm.pid), 'VM.log', 'list'], cwd, 'after-list');
    const final = configuration(after.stdout);
    const finalFiles = await readdir(cwd);
    if (test.name === 'list-only') {
      assert.deepEqual(final, initial);
      assert.ok(!finalFiles.includes('a.log.0') && !finalFiles.includes('b.log.0'));
    } else if (test.name === 'list-with-output') {
      assert.ok(output(final, 'a.log')?.includes('all=info'));
      assert.ok(output(final, 'a.log')?.includes('uptime,level,tags'));
      assert.equal(output(final, 'b.log'), output(initial, 'b.log'));
      assert.notEqual(output(final, 'a.log'), output(initial, 'a.log'));
    } else if (test.name.startsWith('rotate')) {
      assert.deepEqual(final, initial);
      assert.ok(finalFiles.includes('a.log.0') && finalFiles.includes('b.log.0'));
      assert.ok((await readFile(resolve(cwd, 'a.log.0'), 'utf8')).includes('[gc]'));
      assert.ok((await readFile(resolve(cwd, 'b.log.0'), 'utf8')).includes('[class,load]'));
    } else {
      assert.ok(!output(final, 'a.log') && !output(final, 'b.log'));
      assert.equal(final.length, 2);
      assert.ok(final.every(line => line.includes('all=off')));
    }
    records.push({ name: test.name, targetPid: vm.pid, startupArgs: args, actionArgs: action.args, beforeConfiguration: initial, afterConfiguration: final, beforeFiles: initialFiles, afterFiles: finalFiles, assertions: 'PASS' });
  } finally {
    vm.stdin.end('stop\n');
    const shutdown = await Promise.race([exit, new Promise(done => { const timer = setTimeout(() => done(null), 5000); timer.unref(); })]);
    if (!shutdown) vm.kill();
    const stopped = shutdown ?? await exit;
    await writeFile(resolve(cwd, 'target-stdout.log'), stdout);
    await writeFile(resolve(cwd, 'target-stderr.log'), stderr);
    await writeFile(resolve(cwd, 'target-exit.json'), JSON.stringify({ pid: vm.pid, ...stopped }, null, 2) + '\n');
    assert.equal(stopped.code, 0, 'Owned JVM did not exit cleanly');
  }
}
await writeFile(resolve(root, 'results.json'), JSON.stringify({ clearedChildVariables: ['JDK_JAVA_OPTIONS', 'JAVA_TOOL_OPTIONS', '_JAVA_OPTIONS'], compilerTarget: 25, cases: records, allOwnedJvmsStopped: true, behavioralEvaluation: false }, null, 2) + '\n');
console.log(JSON.stringify({ cases: records.map(({ name, assertions }) => ({ name, assertions })), allOwnedJvmsStopped: true }, null, 2));
