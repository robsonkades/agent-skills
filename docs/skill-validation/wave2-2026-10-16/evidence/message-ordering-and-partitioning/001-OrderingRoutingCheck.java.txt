import java.nio.charset.StandardCharsets;
import java.util.Arrays;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.Future;
import java.util.concurrent.TimeUnit;

public final class OrderingRoutingCheck {
    private static final int LANE_COUNT = 8;

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private static List<Integer> run(byte[] firstKey, byte[] secondKey, boolean contentHash)
            throws Exception {
        ExecutorService[] workers = new ExecutorService[LANE_COUNT];
        for (int i = 0; i < workers.length; i++) workers[i] = Executors.newSingleThreadExecutor();
        CountDownLatch firstStarted = new CountDownLatch(1);
        CountDownLatch releaseFirst = new CountDownLatch(1);
        List<Integer> effects = new CopyOnWriteArrayList<>();
        try {
            int firstLane = Math.floorMod(contentHash ? Arrays.hashCode(firstKey) : firstKey.hashCode(), workers.length);
            int secondLane = Math.floorMod(contentHash ? Arrays.hashCode(secondKey) : secondKey.hashCode(), workers.length);
            require(contentHash ? firstLane == secondLane : firstLane != secondLane, "unexpected routing witness");
            Future<?> first = workers[firstLane].submit(() -> {
                firstStarted.countDown();
                try {
                    require(releaseFirst.await(5, TimeUnit.SECONDS), "first effect release timed out");
                    effects.add(1);
                } catch (InterruptedException ex) {
                    Thread.currentThread().interrupt();
                    throw new RuntimeException(ex);
                }
            });
            require(firstStarted.await(5, TimeUnit.SECONDS), "first task did not start");
            Future<?> second = workers[secondLane].submit(() -> effects.add(2));
            if (!contentHash) {
                second.get(5, TimeUnit.SECONDS);
                require(effects.equals(List.of(2)), "negative control did not overtake the held effect");
            }
            releaseFirst.countDown();
            first.get(5, TimeUnit.SECONDS);
            second.get(5, TimeUnit.SECONDS);
            require(effects.equals(contentHash ? List.of(1, 2) : List.of(2, 1)), "wrong effect order " + effects);
            return List.copyOf(effects);
        } finally {
            releaseFirst.countDown();
            for (ExecutorService worker : workers) worker.shutdownNow();
            for (ExecutorService worker : workers) require(worker.awaitTermination(5, TimeUnit.SECONDS), "worker did not stop");
        }
    }

    public static void main(String[] args) throws Exception {
        byte[] first = "customer-123".getBytes(StandardCharsets.UTF_8);
        byte[] second = null;
        int attempts = 0;
        for (; attempts < 10000; attempts++) {
            byte[] candidate = first.clone();
            if (Math.floorMod(candidate.hashCode(), LANE_COUNT) != Math.floorMod(first.hashCode(), LANE_COUNT)) {
                second = candidate;
                break;
            }
        }
        require(second != null, "inconclusive: no distinct-lane identity-hash witness found");
        require(first != second && Arrays.equals(first, second), "keys must be distinct equal-value arrays");
        require(Arrays.hashCode(first) == Arrays.hashCode(second), "equal key contents must hash equally");
        List<Integer> negative = run(first, second, false);
        List<Integer> corrected = run(first, second, true);
        String valueA = new String("customer-123");
        String valueB = new String("customer-123");
        require(valueA != valueB && valueA.hashCode() == valueB.hashCode(), "String value-hash alternative failed");
        System.out.println("{\"identityHashEffectOrder\":" + negative + ",\"contentHashEffectOrder\":" + corrected
                + ",\"equalDistinctArrayKeys\":true,\"immutableStringAlternative\":true,\"witnessAttempts\":" + (attempts + 1)
                + ",\"javaVersion\":\"" + System.getProperty("java.version") + "\",\"allExecutorsTerminated\":true}");
    }
}
