import java.nio.file.Path;
import java.time.Duration;
import java.util.List;
import jdk.jfr.Event;
import jdk.jfr.EventType;
import jdk.jfr.Label;
import jdk.jfr.Name;
import jdk.jfr.Recording;
import jdk.jfr.Throttle;
import jdk.jfr.consumer.RecordingFile;

public final class JfrThrottleCoverage {
    private static final String EVENT_NAME = "quality13.ControlledWork";
    private static final int OPERATIONS = 100;

    @Name(EVENT_NAME)
    @Label("Controlled work")
    @Throttle
    static final class ControlledWork extends Event {
        int operationId;
    }

    record Result(String name, String throttle, Duration threshold,
                  int attempted, int enabled, long recorded) {}

    private static Result record(Path directory, String name, String throttle,
                                 Duration threshold) throws Exception {
        Path recordingPath = directory.resolve(name + ".jfr");
        try (Recording recording = new Recording()) {
            recording.setName(name);
            recording.setToDisk(true);
            recording.enable(ControlledWork.class)
                    .with("throttle", throttle)
                    .withThreshold(threshold)
                    .withoutStackTrace();
            recording.start();
            int enabled = 0;
            for (int i = 0; i < OPERATIONS; i++) {
                ControlledWork event = new ControlledWork();
                if (event.isEnabled()) {
                    enabled++;
                }
                event.operationId = i;
                event.begin();
                event.end();
                event.commit();
            }
            recording.stop();
            recording.dump(recordingPath);
            long observed = RecordingFile.readAllEvents(recordingPath).stream()
                    .filter(event -> event.getEventType().getName().equals(EVENT_NAME)).count();
            return new Result(name, throttle, threshold, OPERATIONS, enabled, observed);
        }
    }

    private static void require(boolean condition, String message) {
        if (!condition) {
            throw new AssertionError(message);
        }
    }

    public static void main(String[] args) throws Exception {
        Path directory = Path.of(args[0]);
        require(EventType.getEventType(ControlledWork.class).getSettingDescriptors().stream()
                        .anyMatch(setting -> setting.getName().equals("throttle")),
                "The target event must actually expose throttle");
        Result suppressed = record(directory, "throttle-zero-threshold-zero", "0/s", Duration.ZERO);
        Result unthrottled = record(directory, "throttle-off-threshold-zero", "off", Duration.ZERO);
        Result durationFiltered = record(directory, "throttle-off-threshold-high", "off", Duration.ofDays(1));
        for (Result result : List.of(suppressed, unthrottled, durationFiltered)) {
            require(result.enabled() == OPERATIONS,
                    "Enablement is intentionally the same in every control: " + result);
            System.out.println(result);
        }
        require(suppressed.recorded() == 0, "Throttle 0/s must suppress eligible events");
        require(unthrottled.recorded() == OPERATIONS, "With both filters open every emitted event is retained");
        require(durationFiltered.recorded() == 0, "Removing throttle must not remove a duration threshold");
        System.out.println("PASS: metadata plus 3 recording controls; enabled=true does not establish coverage");
    }
}
