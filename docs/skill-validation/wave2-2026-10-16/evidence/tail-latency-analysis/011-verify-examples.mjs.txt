import assert from 'node:assert/strict';

const checks = [];
const check = (name, work) => { work(); checks.push(name); };
const q = (values, p = 0.99) => [...values].sort((a, b) => a - b)[Math.ceil(p * values.length) - 1];
const sequential = (aIndices, bIndices) => {
  const a = Array.from({ length: 1000 }, (_, i) => aIndices.has(i + 1) ? 100 : 0);
  const b = Array.from({ length: 1000 }, (_, i) => bIndices.has(i + 1) ? 100 : 0);
  const total = a.map((n, i) => n + b[i]);
  return { a: q(a), b: q(b), total: q(total) };
};
const range = (start, end) => new Set(Array.from({ length: end - start + 1 }, (_, i) => start + i));
const disjointRare = sequential(range(1, 6), range(7, 12));
const coincidentRare = sequential(range(1, 6), range(1, 6));
const disjointFrequent = sequential(range(1, 20), range(21, 40));
check('rare disjoint stage tails cannot be added', () => assert.deepEqual(disjointRare, { a: 0, b: 0, total: 100 }));
check('same marginals with co-occurrence change total p99', () => assert.deepEqual(coincidentRare, { a: 0, b: 0, total: 0 }));
check('total p99 can also be below sum of component p99s', () => assert.deepEqual(disjointFrequent, { a: 100, b: 100, total: 100 }));
check('nonnegative matched-stage marginal lower bound is retained', () => {
  for (const result of [disjointRare, coincidentRare, disjointFrequent]) {
    assert.ok(result.total >= result.a && result.total >= result.b);
  }
});
const independent = [1, 10, 100, 1000].map(n => 1 - Math.pow(0.99, n));
check('independent fan-out table rounded probabilities', () => {
  assert.deepEqual(independent.slice(0, 3).map(p => Math.round(p * 1000) / 10), [1, 9.6, 63.4]);
  assert.ok(independent[3] > 0.9999 && independent[3] < 1);
});
const anyRate = predicate => Array.from({ length: 100 }, (_, world) =>
  Array.from({ length: 100 }, (_, leaf) => predicate(world, leaf)).some(Boolean)).filter(Boolean).length / 100;
check('perfectly correlated leaves reach lower union bound', () => assert.equal(anyRate(world => world === 0), 0.01));
check('mutually exclusive leaf stragglers reach upper union bound', () => assert.equal(anyRate((world, leaf) => world === leaf), 1));
const fifo = (arrivals, unavailableUntil) => {
  let available = unavailableUntil;
  return arrivals.map(arrival => {
    const start = Math.max(available, arrival);
    available = start + 10;
    return { arrival, start, finish: available, wait: start - arrival, latency: available - arrival };
  });
};
const backlog = fifo([10, 30, 50, 70, 90, 110], 100);
check('FIFO constructed backlog completions', () => assert.deepEqual(backlog.map(r => r.finish), [110, 120, 130, 140, 150, 160]));
check('later request latency is wait plus service, not earlier pause duration', () => assert.deepEqual(backlog.at(-1), { arrival: 110, start: 150, finish: 160, wait: 40, latency: 50 }));
check('same earlier pause without prior arrivals leaves no backlog', () => assert.equal(fifo([110], 100)[0].latency, 10));
console.log(JSON.stringify({ status: 'PASS', checks, numericalExamples: { disjointRare, coincidentRare, disjointFrequent, independent, backlog }, limitations: 'Deterministic calculations of teaching examples, not production measurements, a queue-model validation or an agent evaluation.' }, null, 2));
