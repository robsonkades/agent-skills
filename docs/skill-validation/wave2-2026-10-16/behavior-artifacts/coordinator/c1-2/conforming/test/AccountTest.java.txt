import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;
public class AccountTest {
@RepeatedTest(20)                       // more sampled executions, not exhaustive coverage
@Timeout(30)
void balanceConservesSuccessfulWithdrawalsAfterWorkersFinish() throws Exception {
    Account account = new Account(10_000);
    int threads = 32, opsPerThread = 1_000;

    CyclicBarrier start = new CyclicBarrier(threads);
    AtomicInteger succeeded = new AtomicInteger();
    AtomicInteger refused = new AtomicInteger();

    var exec = Executors.newVirtualThreadPerTaskExecutor();
    List<Future<?>> tasks = new ArrayList<>();
    assertAll("workers and teardown",
        () -> {
            try {
                for (int t = 0; t < threads; t++) {
                    tasks.add(exec.submit(() -> {
                        start.await(5, TimeUnit.SECONDS); // overlap with a bounded barrier
                        for (int i = 0; i < opsPerThread; i++) {
                            if (account.withdraw(10)) succeeded.incrementAndGet();
                            else refused.incrementAndGet();
                        }
                        return null;
                    }));
                }
                long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(20);
                assertAll("worker outcomes", tasks.stream().map(task -> () ->
                        task.get(Math.max(0, deadline - System.nanoTime()), TimeUnit.NANOSECONDS)));
            } finally {
                tasks.forEach(task -> task.cancel(true));
                exec.shutdownNow();
            }
        },
        () -> assertTrue(exec.awaitTermination(2, TimeUnit.SECONDS), "executor survived teardown"));

    // Invariants, not schedules:
    assertTrue(account.balance() >= 0);
    assertEquals(10_000 - succeeded.get() * 10, account.balance());       // conservation
    assertEquals(threads * opsPerThread, succeeded.get() + refused.get()); // accounting
    // Refusal is legal only after all 1,000 affordable withdrawals have succeeded.
    assertEquals(1_000, succeeded.get(), "all affordable withdrawals must succeed");
    assertEquals(threads * opsPerThread - 1_000, refused.get());
    assertEquals(0, account.balance(), "the attempts must exhaust the available funds");
}

@Test
@Timeout(15)
void affordableWithdrawalSucceedsDespiteGateContention() throws Exception {
    Account account = new Account(10);
    CountDownLatch gateHeld = new CountDownLatch(1);
    CountDownLatch releaseGate = new CountDownLatch(1);
    CountDownLatch withdrawalStarted = new CountDownLatch(1);
    AtomicReference<Thread> withdrawalThread = new AtomicReference<>();
    var exec = Executors.newVirtualThreadPerTaskExecutor();
    List<Future<?>> tasks = new ArrayList<>();

    assertAll("contended withdrawal and teardown",
        () -> {
            try {
                assertAll("contention checkpoint and worker outcomes",
                    () -> {
                        try {
                            tasks.add(exec.submit(() -> {
                                account.holdGate(gateHeld, releaseGate);
                                return null;
                            }));
                            assertTrue(gateHeld.await(2, TimeUnit.SECONDS), "gate owner did not acquire the gate");
                            Future<?> withdrawal = exec.submit(() -> {
                                withdrawalThread.set(Thread.currentThread());
                                withdrawalStarted.countDown();
                                assertTrue(account.withdraw(10), "contention is not insufficient funds");
                                return null;
                            });
                            tasks.add(withdrawal);
                            assertTrue(withdrawalStarted.await(2, TimeUnit.SECONDS), "withdrawal did not start");

                            // Startup alone is not evidence of contention. After that checkpoint the
                            // worker's only blocking operation is withdraw. Observe it waiting on the
                            // held gate, or retain an early result/failure (such as a tryLock refusal).
                            long attemptDeadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(2);
                            boolean attempted = false;
                            while (System.nanoTime() < attemptDeadline) {
                                if (withdrawal.isDone() || withdrawalThread.get().getState() == Thread.State.WAITING) {
                                    attempted = true;
                                    break;
                                }
                                Thread.sleep(1); // bounded polling backoff, not evidence of progress
                            }
                            assertTrue(attempted, "withdrawal neither waited on the held gate nor completed");
                        } finally {
                            releaseGate.countDown();
                        }
                    },
                    () -> {
                        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(2);
                        assertAll("worker outcomes", tasks.stream().map(task -> () ->
                                task.get(Math.max(0, deadline - System.nanoTime()), TimeUnit.NANOSECONDS)));
                    });
            } finally {
                releaseGate.countDown(); // independent release, even if an assertion fails
                tasks.forEach(task -> task.cancel(true));
                exec.shutdownNow();
            }
        },
        () -> assertTrue(exec.awaitTermination(2, TimeUnit.SECONDS), "executor survived teardown"));

    assertEquals(0, account.balance(), "the successful withdrawal must subtract the amount");
    assertFalse(account.withdraw(1), "an exhausted account must refuse a withdrawal");
    assertEquals(0, account.balance(), "a refused withdrawal must leave the balance unchanged");
}
}
