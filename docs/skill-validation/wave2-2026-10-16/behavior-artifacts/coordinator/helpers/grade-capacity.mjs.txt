import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
const report='C:/git/agent-skills/docs/skill-validation/wave2-2026-10-16';
const m=JSON.parse(fs.readFileSync(report+'/behavior-manifest.json'));
for(const run of m.runs.filter(x=>['C5','C6'].includes(x.case))){
 const c=JSON.parse(fs.readFileSync(path.join(run.directory,'cohort.json')));
 const s=JSON.parse(fs.readFileSync(path.join(run.directory,'slo.json')));
 if(c.knownGood+c.knownBad+c.missingTerminalRecords!==c.eligibleOperations)throw new Error('Population mismatch');
 const lower=c.knownGood/c.eligibleOperations,upper=(c.knownGood+c.missingTerminalRecords)/c.eligibleOperations;
 const requiredGood=Math.ceil(s.requiredGoodFraction*c.eligibleOperations);
 const possibilities=Array.from({length:c.missingTerminalRecords+1},(_,x)=>({unknownGood:x,passes:c.knownGood+x>=requiredGood}));
 const result={at:new Date().toISOString(),run:run.id,classification:'Independent finite-cohort arithmetic; decision semantics inspected by coordinator separately',command:['node','grade-capacity.mjs'],lower,upper,requiredGood,possibleOutcomeCounts:possibilities.length,passingPossibilities:possibilities.filter(x=>x.passes).length,failingPossibilities:possibilities.filter(x=>!x.passes).length,expectedDecision:lower>=s.requiredGoodFraction?'PASS':upper<s.requiredGoodFraction?'FAIL':'UNRESOLVED',decisionSha256:crypto.createHash('sha256').update(fs.readFileSync(path.join(run.directory,'decision.md'))).digest('hex'),limits:'Accounting bounds only; no new load test, outcome reconstruction or future capacity proof.'};
 const dest=path.join(m.root,'grader',run.id);
 fs.mkdirSync(dest,{recursive:true});
 fs.writeFileSync(path.join(dest,'coordinator-checks.json'),JSON.stringify(result,null,2)+'\n');
 console.log(JSON.stringify(result));
}
