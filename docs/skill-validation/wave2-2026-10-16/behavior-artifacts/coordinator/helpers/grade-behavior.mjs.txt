import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {spawnSync} from 'node:child_process';
const report='C:/git/agent-skills/docs/skill-validation/wave2-2026-10-16';
const m=JSON.parse(fs.readFileSync(report+'/behavior-manifest.json'));
const sha=x=>crypto.createHash('sha256').update(x).digest('hex');
const runId=process.argv[2];
const treatment=m.runs.find(x=>x.id===runId);
if(!treatment) throw new Error('Unknown run '+runId);
const spec=m.cases.find(x=>x.id===treatment.case);
const original=treatment.directory;
const grading=path.join(m.root,'grader',runId);
fs.mkdirSync(grading,{recursive:true});
const comparisons=Object.entries(spec.fixtures).map(([p,v])=>({path:p,originalSha256:sha(v),finalSha256:fs.existsSync(path.join(original,p))?sha(fs.readFileSync(path.join(original,p))):null}));
const record={run:runId,condition:treatment.condition,at:new Date().toISOString(),classification:'Coordinator artifact execution; rubric decisions recorded separately',fixtureComparisons:comparisons,checks:[]};
function copy(variant){
 const dest=path.join(grading,variant);
 if(fs.existsSync(dest))throw new Error('Preserve first execution; directory exists '+dest);
 fs.mkdirSync(dest,{recursive:true});
 for(const f of fs.readdirSync(original,{withFileTypes:true})) {
  if(['classes','resources','.resources','.git','node_modules'].includes(f.name))continue;
  fs.cpSync(path.join(original,f.name),path.join(dest,f.name),{recursive:true});
 }
 return dest;
}
function execute(dest,label,command=process.execPath,args=['check.mjs']){
 const startedAt=new Date().toISOString();
 const r=spawnSync(command,args,{cwd:dest,encoding:'utf8',timeout:120000,windowsHide:true});
 const stdout=r.stdout||'',stderr=r.stderr||'';
 fs.writeFileSync(path.join(dest,'coordinator.stdout.log'),stdout);
 fs.writeFileSync(path.join(dest,'coordinator.stderr.log'),stderr);
 const check={label,command:[command,...args],cwd:dest,startedAt,finishedAt:new Date().toISOString(),exit:r.status,signal:r.signal,error:r.error?.message,stdout,stderr};
 record.checks.push(check);return check;
}
if(['C1','C2'].includes(spec.id)){
 execute(copy('conforming'),'conforming implementation');
 const mutant=copy('always-refuse');
 const source=path.join(mutant,'src','Account.java');
 let code=fs.readFileSync(source,'utf8');
 const needle='if (amount <= 0) throw new IllegalArgumentException("amount");';
 if(!code.includes(needle)) throw new Error('Account changed; inspect before mutation');
 code=code.replace(needle,needle+'\n    if (amount > 0) return false; // evaluator control: positive withdrawals never act');
 fs.writeFileSync(source,code);
 execute(mutant,'always-refuse mutation; must compile and fail relevant assertion');
} else if(['C3','C4'].includes(spec.id)) {
 execute(copy('delivered'),'delivered compatibility checks');
} else throw new Error('Capacity cases require decision review, not generated execution');
fs.writeFileSync(path.join(grading,'coordinator-checks.json'),JSON.stringify(record,null,2)+'\n');
console.log(JSON.stringify({run:runId,changed:comparisons.filter(x=>x.originalSha256!==x.finalSha256).map(x=>x.path),checks:record.checks.map(x=>({label:x.label,exit:x.exit,error:x.error,stdout:x.stdout,stderr:x.stderr}))},null,2));
