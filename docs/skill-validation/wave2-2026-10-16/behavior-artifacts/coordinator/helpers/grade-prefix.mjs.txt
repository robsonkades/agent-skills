import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave2-behavior-6orq6B';
const id=process.argv[2];
if(!/^c[34]-[123]$/.test(id))throw new Error('Routing cases only');
const dest=path.join(root,'grader',id,'prefix-control');
if(fs.existsSync(dest))throw new Error('Preserve first execution');
fs.cpSync(path.join(root,'grader',id,'delivered'),dest,{recursive:true,filter:p=>!p.includes(path.sep+'classes')&&!p.includes(path.sep+'.resources')&&!p.endsWith('.log')});
const file=path.join(dest,'candidate.properties');
const before=fs.readFileSync(file,'utf8');
const prefix=id.startsWith('c4-')?'cache:':'';
if(!/^keyPrefix=/m.test(before))throw new Error('Inspect configuration syntax');
fs.writeFileSync(file,before.replace(/^keyPrefix=.*$/m,'keyPrefix='+prefix));
const startedAt=new Date().toISOString();
const r=spawnSync(process.execPath,['check.mjs'],{cwd:dest,encoding:'utf8',timeout:120000,windowsHide:true});
const record={run:id,label:'Coordinator changes only candidate prefix to '+JSON.stringify(prefix),scope:'Disposable copy; does not count as actor executing a missing control',startedAt,finishedAt:new Date().toISOString(),command:[process.execPath,'check.mjs'],cwd:dest,exit:r.status,error:r.error?.message,stdout:r.stdout||'',stderr:r.stderr||''};
fs.writeFileSync(path.join(dest,'coordinator.stdout.log'),record.stdout);
fs.writeFileSync(path.join(dest,'coordinator.stderr.log'),record.stderr);
fs.writeFileSync(path.join(dest,'coordinator-checks.json'),JSON.stringify(record,null,2)+'\n');
console.log(JSON.stringify(record,null,2));
