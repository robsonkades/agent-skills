import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashSet;
import java.util.List;
import java.util.Properties;

/** Runs the frozen accepted vector through the actual request-to-owner path. */
public final class RoutingAgreementCheck {
  private static Properties config(String file) throws Exception {
    Properties config = new Properties();
    try (var in = Files.newInputStream(Path.of(file))) {
      config.load(in);
    }
    return config;
  }

  private static void agreement(String label, ClientRouter router, String key,
      String expectedPrimary, List<String> expectedReplicas) {
    String primary = router.owner(key);
    List<String> replicas = router.owners(key);
    if (!expectedPrimary.equals(primary) || !expectedReplicas.equals(replicas)
        || replicas.size() != 3 || new HashSet<>(replicas).size() != 3
        || !primary.equals(replicas.get(0))) {
      throw new AssertionError(label + ": key=" + key + ", expected primary="
          + expectedPrimary + ", replicas=" + expectedReplicas + "; actual primary="
          + primary + ", replicas=" + replicas);
    }
    System.out.println(label + " agrees: key=" + key + ", primary=" + primary
        + ", replicas=" + replicas);
  }

  public static void main(String[] args) throws Exception {
    if (args.length != 7) throw new IllegalArgumentException("Run node check.mjs");
    String[] nodes = args[2].split(",");
    String[] weights = args[3].split(",");
    if (nodes.length != weights.length) throw new AssertionError("fixture topology");
    HashRing ring = new HashRing(Integer.parseInt(args[1]));
    for (int i = 0; i < nodes.length; i++) ring.add(nodes[i], Integer.parseInt(weights[i]));
    String key = args[4], primary = args[5];
    List<String> replicas = List.of(args[6].split(","));
    Properties accepted = config("accepted.properties"), candidate = config("candidate.properties");

    // Routing comparisons precede the explicit raw-key contract check so that a
    // changed prefix cannot hide behind identical virtual-node positions.
    agreement("Accepted epoch " + args[0], new ClientRouter(ring, accepted), key, primary, replicas);
    agreement("Candidate epoch " + args[0], new ClientRouter(ring, candidate), key, primary, replicas);
    if (!accepted.getProperty("keyPrefix", "").isEmpty()
        || !candidate.getProperty("keyPrefix", "").isEmpty()) {
      throw new AssertionError("The accepted raw-key contract requires an empty keyPrefix");
    }

    // Negative control changes only one client's key construction; the same ring
    // object guarantees that membership and all node positions stay unchanged.
    Properties control = new Properties();
    control.putAll(candidate);
    control.setProperty("keyPrefix", "cache:");
    boolean rejected = false;
    try {
      agreement("Prefix negative control", new ClientRouter(ring, control), key, primary, replicas);
    } catch (AssertionError expected) {
      rejected = true;
      System.out.println("Negative control rejected as required: " + expected.getMessage());
    }
    if (!rejected) throw new AssertionError("Golden vector did not detect the changed key prefix");

    control.setProperty("keyPrefix", candidate.getProperty("keyPrefix", ""));
    agreement("Restored prefix control", new ClientRouter(ring, control), key, primary, replicas);
    System.out.println("ClientRouter compatibility check passed (1 frozen vector, prefix control).");
  }
}
