import java.lang.reflect.Field;
import java.nio.file.*;
import java.util.*;
public final class CurrentChecks {
  static HashRing ring(String... nodes) { HashRing r=new HashRing(1024); for(String n:nodes) r.add(n,1); return r; }
  static void checkPlacement(HashRing ring, String configFile, String key,
                             String expectedPrimary, List<String> expectedReplicas) throws Exception {
    Properties config=new Properties();
    try(var in=Files.newInputStream(Path.of(configFile))){config.load(in);}
    ClientRouter router=new ClientRouter(ring,config);
    String primary=router.owner(key);
    List<String> replicas=router.owners(key);
    System.out.println(configFile+" key="+key+" primary="+primary+" replicas="+replicas);
    if(!primary.equals(expectedPrimary)||!replicas.equals(expectedReplicas)) {
      throw new AssertionError(configFile+" is incompatible with accepted-vectors.json: expected primary="
          +expectedPrimary+" replicas="+expectedReplicas+"; actual primary="+primary+" replicas="+replicas);
    }
  }
  public static void main(String[] args) throws Exception {
    if(args.length!=5)throw new IllegalArgumentException("Expected frozen key, primary and three ordered replicas from check.mjs");
    HashRing accepted=ring("n1","n2","n3","n4"), candidate=ring("n4","n3","n2","n1"), joined=ring("n1","n2","n3","n4","n5");
    Field f=HashRing.class.getDeclaredField("ring");f.setAccessible(true);
    if(!f.get(accepted).equals(f.get(candidate))) throw new AssertionError("points differ");
    Properties config=new Properties();try(var in=Files.newInputStream(Path.of("candidate.properties"))){config.load(in);}
    ClientRouter before=new ClientRouter(candidate,config),after=new ClientRouter(joined,config);int moved=0;
    for(int i=0;i<10000;i++){String k="user:"+i;if(!before.owner(k).equals(after.owner(k))){moved++;if(!after.owner(k).equals("n5"))throw new AssertionError("monotonicity");}}
    if(moved<1800||moved>2200)throw new AssertionError("movement="+moved);
    System.out.println("Existing point and join checks passed; moved="+moved);
    // Independently frozen evidence, read by check.mjs; never derived from either router.
    List<String> expectedReplicas=List.of(args[2],args[3],args[4]);
    checkPlacement(accepted,"accepted.properties",args[0],args[1],expectedReplicas);
    checkPlacement(candidate,"candidate.properties",args[0],args[1],expectedReplicas);
  }
}
