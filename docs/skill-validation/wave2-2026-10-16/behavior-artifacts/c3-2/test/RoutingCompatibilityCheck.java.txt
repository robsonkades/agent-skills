import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashSet;
import java.util.List;
import java.util.Properties;

/** Compares the actual client path with independent, frozen accepted placement. */
public final class RoutingCompatibilityCheck {
    public static void main(String[] args) throws Exception {
        if (args.length != 8) {
            throw new IllegalArgumentException(
                    "config epoch V nodes weights requestKey expectedPrimary expectedReplicas");
        }
        String[] nodes = args[3].split(",");
        String[] weights = args[4].split(",");
        if (nodes.length != weights.length) {
            throw new IllegalArgumentException("node and weight counts differ");
        }
        HashRing ring = new HashRing(Integer.parseInt(args[2]));
        for (int i = 0; i < nodes.length; i++) {
            ring.add(nodes[i], Integer.parseInt(weights[i]));
        }
        Properties config = new Properties();
        try (var input = Files.newInputStream(Path.of(args[0]))) {
            config.load(input);
        }
        ClientRouter router = new ClientRouter(ring, config);
        String primary = router.owner(args[5]);
        List<String> replicas = router.owners(args[5]);
        List<String> expectedReplicas = List.of(args[7].split(","));
        System.out.println("epoch=" + args[1] + " config=" + args[0]
                + " requestKey=" + args[5] + " keyPrefix=" + config.getProperty("keyPrefix", ""));
        System.out.println("expected primary=" + args[6] + " replicas=" + expectedReplicas);
        System.out.println("actual   primary=" + primary + " replicas=" + replicas);
        if (!primary.equals(args[6]) || !replicas.equals(expectedReplicas)
                || replicas.size() != 3 || new HashSet<>(replicas).size() != 3
                || !primary.equals(replicas.get(0))) {
            throw new AssertionError("ClientRouter placement differs from accepted epoch " + args[1]);
        }
        System.out.println("Routing compatibility passed for " + args[0]);
    }
}
