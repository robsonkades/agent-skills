import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';

const root = path.resolve(import.meta.dirname, '..');
const sourcePath = path.join(root, 'src/Account.java');
const original = fs.readFileSync(sourcePath, 'utf8');
const digest = value => createHash('sha256').update(value).digest('hex');
const mutations = [
  ['always-refuses',
    'try { if (amount > balance) return false; balance -= amount; return true; }',
    'try { return false; }'],
  ['refuses-contention', '    gate.lock();',
    '    if (!gate.tryLock()) return false;'],
  ['wrong-debit', 'balance -= amount;', 'balance -= amount - 1;'],
];
let failed = false;
for (const [name, before, after] of mutations) {
  if (!original.includes(before)) throw new Error(`Mutation target absent: ${name}`);
  const destination = path.join(import.meta.dirname, name);
  for (const directory of ['src', 'test', 'tools']) {
    fs.mkdirSync(path.join(destination, directory), { recursive: true });
  }
  for (const file of ['check.mjs', 'test/AccountTest.java', 'tools/TestRunner.java']) {
    fs.copyFileSync(path.join(root, file), path.join(destination, file));
  }
  fs.writeFileSync(path.join(destination, 'src/Account.java'), original.replace(before, after));
  const result = spawnSync(process.execPath, [path.join(destination, 'check.mjs')], {
    encoding: 'utf8', timeout: 60000, windowsHide: true,
  });
  const output = `${result.stdout ?? ''}${result.stderr ?? ''}`;
  fs.writeFileSync(path.join(import.meta.dirname, `${name}.log`), output);
  const testsStarted = output.match(/\[\s*(\d+) tests started\s*\]/)?.[1];
  const testsFailed = output.match(/\[\s*(\d+) tests failed\s*\]/)?.[1];
  const detected = !result.error && result.status === 1 &&
    Number(testsStarted) === 21 && Number(testsFailed) > 0;
  console.log(`${name}: exit=${result.status}, started=${testsStarted}, failed=${testsFailed}, detected=${detected}`);
  if (result.error) console.log(result.error.message);
  if (!detected) failed = true;
}
const unchanged = digest(original) === digest(fs.readFileSync(sourcePath, 'utf8'));
console.log(`Production source unchanged: ${unchanged}; SHA-256 ${digest(original)}`);
if (!unchanged || failed) process.exitCode = 1;
