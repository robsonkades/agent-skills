import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, existsSync, readdirSync, writeFileSync } from 'node:fs';
import { resolve, dirname, relative } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
const here = dirname(fileURLToPath(import.meta.url));
const repo = 'C:/git/agent-skills';
const skill = resolve(repo, 'skills/startup-cds-crac-leyden');
const { parseManifest } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(resolve(repo, 'packages/core/src/domain/skill-document.ts')));
const { manifest, issues } = parseManifest(readFileSync(resolve(skill, 'skill.yaml'), 'utf8'), { strict: true });
const document = parseSkillDocument(readFileSync(resolve(skill, 'SKILL.md'), 'utf8'));
assert.deepEqual(issues, []);
assert.equal(documentName(document), manifest.name);
assert.equal(documentDescription(document), manifest.description);
assert.equal(manifest.name, 'startup-cds-crac-leyden');
assert.equal(manifest.version, '2.1.5');
for (const entry of manifest.files) assert.ok(existsSync(resolve(skill, entry)), entry);
const baseline = JSON.parse(readFileSync(resolve(here, 'baseline.json'), 'utf8').replace(/^\uFEFF/, ''));
const files = readdirSync(skill, { recursive: true, withFileTypes: true }).filter(e => e.isFile())
  .map(e => relative(skill, resolve(e.parentPath, e.name)).replaceAll('\\', '/')).sort();
assert.deepEqual(files, baseline.files.map(e => e.path).sort());
let relativeLinks = 0;
const changed = [];
for (const file of baseline.files) {
  const full = resolve(skill, file.path);
  const bytes = readFileSync(full);
  if (createHash('sha256').update(bytes).digest('hex') !== file.sha256) changed.push(file.path);
  if (!file.path.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\]\(([^\s)]+)\)/g)) {
    const target = match[1].replace(/^<|>$/g, '');
    if (/^(https?:|#)/.test(target)) continue;
    assert.ok(existsSync(resolve(dirname(full), target.split('#')[0])), target);
    relativeLinks++;
  }
}
assert.deepEqual(changed.sort(), ['SKILL.md', 'references/technique-selection.md', 'skill.yaml']);
const result = { status: 'PASS', skill: manifest.name, strictManifestIssues: 0,
  metadataAgreement: true, version: manifest.version, packageFiles: files.length,
  declaredPaths: manifest.files.length, relativeLinks, changedFromAssignment: changed };
writeFileSync(resolve(here, 'structure-results.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify(result, null, 2));
