import assert from 'node:assert/strict';
import { mkdirSync, writeFileSync, readFileSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
const here = dirname(fileURLToPath(import.meta.url));
const java = 'C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe';
const env = { ...process.env };
for (const key of ['JAVA_TOOL_OPTIONS', 'JDK_JAVA_OPTIONS', '_JAVA_OPTIONS', 'JDK_AOT_VM_OPTIONS']) delete env[key];
const results = [];
function run(label, args, expected) {
  const r = spawnSync(java, args, { cwd: here, env, encoding: 'utf8', timeout: 15000 });
  const check = { label, executable: java, args, cwd: here, expectedExit: expected, status: r.status,
    stdout: r.stdout, stderr: r.stderr, error: r.error?.message ?? null,
    clearedChildEnvironment: ['JAVA_TOOL_OPTIONS', 'JDK_JAVA_OPTIONS', '_JAVA_OPTIONS', 'JDK_AOT_VM_OPTIONS'] };
  results.push(check);
  writeFileSync(resolve(here, 'runtime-results.json'), JSON.stringify(results, null, 2) + '\n');
  assert.equal(r.status, expected, JSON.stringify(check));
  return r.stdout + r.stderr;
}
const version = run('actual runtime identity', ['-Xinternalversion'], 0);
assert.match(version, /25\.0\.3\+9/);
const flags = run('successful producer then capability filtering', ['-XX:+PrintFlagsFinal', '-version'], 0);
assert.match(flags, /AOTCacheOutput\s*=/);
assert.match(flags, /AOTCache\s*=/);
assert.match(flags, /AOTMode\s*=/);
assert.doesNotMatch(flags, /CRaCCheckpointTo/);
const unsupported = run('negative CRaC capability on standard build', ['-XX:CRaCCheckpointTo=not-created', '-version'], 1);
assert.match(unsupported, /Unrecognized VM option.*CRaCCheckpointTo/);
const cds = run('default CDS mapping only', ['-Xlog:cds', '-version'], 0);
assert.match(cds, /classes\.jsa/);
assert.match(cds, /Mapped static/);
