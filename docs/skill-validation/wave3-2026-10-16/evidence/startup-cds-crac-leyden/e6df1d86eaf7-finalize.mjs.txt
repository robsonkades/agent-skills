function run(executable, args, cwd = repo) {
  const result = spawnSync(executable, args, { cwd, encoding: 'utf8', timeout: 15000 });
  const check = { executable, args, cwd, status: result.status, stdout: result.stdout,
    stderr: result.stderr, error: result.error?.message ?? null };
  checks.push(check);
  writeFileSync(resolve(here, 'final-check-results.json'), JSON.stringify(checks, null, 2) + '\n');
  assert.equal(result.status, 0, JSON.stringify(check));
  return result.stdout;
}
