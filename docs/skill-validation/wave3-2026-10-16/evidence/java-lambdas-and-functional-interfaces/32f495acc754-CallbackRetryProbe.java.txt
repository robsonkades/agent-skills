import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ConcurrentLinkedQueue;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicReference;

/** Controlled correctness probe, not a performance or all-interleavings test. */
public final class CallbackRetryProbe {
    static void await(CountDownLatch latch) {
        try {
            if (!latch.await(5, TimeUnit.SECONDS)) {
                throw new AssertionError("Timed out waiting for controlled phase");
            }
        } catch (InterruptedException failure) {
            Thread.currentThread().interrupt();
            throw new AssertionError("Probe interrupted", failure);
        }
    }

    public static void main(String[] args) throws Exception {
        if (args.length != 1) throw new IllegalArgumentException("Expected one mode");
        String mode = args[0];
        if (mode.equals("audit-failure")) {
            AtomicInteger value = new AtomicInteger();
            int committed = value.updateAndGet(previous -> previous + 1);
            boolean caught = false;
            try {
                failingAudit(committed);
            } catch (IllegalStateException expected) {
                caught = true;
            }
            if (!caught || value.get() != 1) throw new AssertionError("Wrong failure state");
            System.out.println("PASS audit-failure committed=1 auditThrew=true counter=1");
            return;
        }
        if (!(mode.equals("stable") || mode.equals("retry") || mode.equals("after-commit"))) {
            throw new IllegalArgumentException("Unknown mode");
        }

        boolean interfere = !mode.equals("stable");
        boolean auditAfterCommit = mode.equals("after-commit");
        AtomicInteger value = new AtomicInteger();
        AtomicInteger calls = new AtomicInteger();
        AtomicInteger result = new AtomicInteger(-1);
        AtomicReference<Throwable> failure = new AtomicReference<>();
        ConcurrentLinkedQueue<Integer> audits = new ConcurrentLinkedQueue<>();
        CountDownLatch entered = new CountDownLatch(1);
        CountDownLatch release = new CountDownLatch(1);

        Thread updater = new Thread(() -> {
            try {
                int committed = value.updateAndGet(previous -> {
                    // Instrumentation/gates deliberately expose invocation attempts.
                    int attempt = calls.incrementAndGet();
                    if (!auditAfterCommit) audits.add(previous + 1);
                    if (attempt == 1) {
                        entered.countDown();
                        await(release);
                    }
                    return previous + 1;
                });
                if (auditAfterCommit) audits.add(committed);
                result.set(committed);
            } catch (Throwable problem) {
                failure.set(problem);
            }
        }, "callback-probe-updater");
        updater.setDaemon(true);
        updater.start();
        try {
            await(entered);
            if (interfere && value.incrementAndGet() != 1) {
                throw new AssertionError("Unexpected interfering update");
            }
        } finally {
            release.countDown();
            updater.join(6_000);
        }
        if (updater.isAlive()) throw new AssertionError("Updater did not terminate");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());

        int expectedCalls = interfere ? 2 : 1;
        int expectedValue = interfere ? 2 : 1;
        int expectedAudits = auditAfterCommit ? 1 : expectedCalls;
        List<Integer> observed = new ArrayList<>(audits);
        if (calls.get() != expectedCalls || value.get() != expectedValue
                || result.get() != expectedValue || observed.size() != expectedAudits) {
            throw new AssertionError("Unexpected result: calls=" + calls + ", value=" + value
                    + ", result=" + result + ", audits=" + observed);
        }
        if (auditAfterCommit && !observed.get(0).equals(result.get())) {
            throw new AssertionError("Audit did not receive committed value");
        }
        System.out.println("PASS " + mode + " callbackCalls=" + calls.get()
                + " result=" + result.get() + " auditValues=" + observed);
    }

    static void failingAudit(int committed) {
        throw new IllegalStateException("Audit failed after committed value " + committed);
    }
}
