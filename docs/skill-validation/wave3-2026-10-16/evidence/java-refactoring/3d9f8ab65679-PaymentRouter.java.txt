import java.time.LocalDate;
public class PaymentRouter {
  static final LocalDate TODAY = LocalDate.of(2026, 10, 16);
  sealed interface Payment permits Card, Transfer {}
  record Card(String number, LocalDate expiry) implements Payment {}
  record Transfer(String account) implements Payment {}
  static String route(Payment payment) {
    if (payment instanceof Card card && card.expiry().isBefore(TODAY)) return "expired";
    if (payment instanceof Card card) return "authorised";
    if (payment instanceof Transfer transfer) return "settled";
    return "no payment";
  }
  // Existing caller with its own failure policy.
  static String response(Payment payment) {
    try { return route(payment); }
    catch (IllegalStateException unavailable) { return "retry later"; }
  }
}
