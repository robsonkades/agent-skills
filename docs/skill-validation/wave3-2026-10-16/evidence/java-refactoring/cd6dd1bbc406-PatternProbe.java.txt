import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.function.Supplier;
public class PatternProbe {
  static final List<String> calls = new ArrayList<>();
  static int assertions;
  static final LocalDate today = LocalDate.of(2026, 10, 16);
  sealed interface Payment permits Card, Transfer {}
  record Card(String number, LocalDate expiry) implements Payment {
    public LocalDate expiry() {
      calls.add("expiry");
      if (expiry == null) throw new IllegalStateException("expiry unavailable");
      return expiry;
    }
  }
  record Transfer(String account) implements Payment {}
  record Result(String value) {
    static Result rejected(String reason) { return new Result("rejected:" + reason); }
  }
  record Seen(String value, String exception, String cause, List<String> effects) {}
  static Result authorise(Card card) { calls.add("authorise"); return new Result("authorised"); }
  static Result settle(Transfer transfer) { calls.add("settle"); return new Result("settled"); }
  static Result before(Payment payment) {
    if (payment instanceof Card card && card.expiry().isBefore(today)) return Result.rejected("expired");
    if (payment instanceof Card card) return authorise(card);
    if (payment instanceof Transfer transfer) return settle(transfer);
    return Result.rejected("no payment");
  }
  static Result extracted(Payment payment) {
    return switch (payment) {
    case null -> Result.rejected("no payment");
    case Card card when card.expiry().isBefore(today) -> Result.rejected("expired");
    case Card card -> authorise(card);
    case Transfer transfer -> settle(transfer);
};
  }
  static Seen observe(Supplier<Result> action) {
    calls.clear();
    try { return new Seen(action.get().value(), "", "", List.copyOf(calls)); }
    catch (RuntimeException failure) {
      return new Seen("", failure.getClass().getName(), failure.getCause() == null ? "" :
        failure.getCause().getClass().getName(), List.copyOf(calls));
    }
  }
  static void check(boolean condition, String message) {
    assertions++;
    if (!condition) throw new AssertionError(message);
  }
  record Payload(String text) {}
  record Envelope(Payload payload) {}
  static String explicit(Envelope envelope) { return envelope.payload() == null ? "absent" : envelope.payload().text(); }
  static String nested(Envelope envelope) { return switch (envelope) { case Envelope(Payload(var text)) -> text; }; }
  static String bound(Envelope envelope) { return switch (envelope) { case Envelope(var payload) -> payload == null ? "absent" : payload.text(); }; }
  public static void main(String[] args) {
    Payment[] inputs = {null, new Card("n", today.minusDays(1)), new Card("n", today),
      new Card("n", today.plusDays(1)), new Card("n", null), new Transfer("a")};
    int mismatches = 0;
    for (int i = 0; i < inputs.length; i++) {
      Payment payment = inputs[i];
      Seen original = observe(() -> before(payment));
      Seen actual = observe(() -> extracted(payment));
      boolean knownDifference = false && i == 4;
      if (knownDifference) {
        check(original.exception().equals("java.lang.IllegalStateException"), "before failure " + original);
        check(actual.exception().equals("java.lang.MatchException"), "record pattern failure " + actual);
        check(actual.cause().equals(original.exception()), "original failure must be the cause");
        check(actual.effects().equals(original.effects()), "control must isolate exception wrapping");
        mismatches++;
      } else {
        check(actual.equals(original), "row " + i + " before=" + original + " actual=" + actual);
      }
      System.out.println("row=" + i + " before=" + original + " actual=" + actual);
    }
    Envelope absent = new Envelope(null);
    check(explicit(absent).equals("absent"), "explicit null accepted");
    check(bound(absent).equals("absent"), "component binding retains null");
    try { nested(absent); throw new AssertionError("nested null should not match"); }
    catch (MatchException expected) { check(expected.getCause() == null, "exhaustive miss differs from accessor failure"); }
    check(nested(new Envelope(new Payload("ok"))).equals("ok"), "nested normal value");
    System.out.println("PASS form=revised accessor=fallible assertions=" + assertions + " observedContractDifferences=" + mismatches);
  }
}
