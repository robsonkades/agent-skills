package org.openrewrite.maven;
import java.nio.file.*; import java.nio.charset.*; import java.util.*; import java.lang.reflect.*;
import org.openrewrite.*;
public final class MethodControls {
  static int checks;
  static final String A="class A {}
", B="class B {}
", LOCAL="keep this unrelated local work
";
  static void check(boolean ok,String label){if(!ok)throw new AssertionError(label);checks++;System.out.println("PASS "+label);}
  static String read(Path p)throws Exception{return new String(Files.readAllBytes(p),StandardCharsets.UTF_8);}
  static void write(Path p,String s)throws Exception{Files.write(p,s.getBytes(StandardCharsets.UTF_8));}
  static Path setup(Path root,String name)throws Exception{Path p=root.resolve(name);Files.createDirectories(p);write(p.resolve("local-notes.txt"),LOCAL);return p;}
  static final class Text implements SourceFile {
    final Path path; final String text; Text(String p,String t){path=Paths.get(p);text=t;}
    public Path getSourcePath(){return path;} public Charset getCharset(){return StandardCharsets.UTF_8;}
    public FileAttributes getFileAttributes(){return null;} public String printAll(PrintOutputCapture<?> capture){return text;}
  }
  static AbstractRewriteBaseRunMojo.ResultsContainer changes(Path root){
    AbstractRewriteBaseRunMojo.ResultsContainer r=new AbstractRewriteBaseRunMojo.ResultsContainer(root);
    r.generated.add(new Result(null,new Text("A.java",A)));r.generated.add(new Result(null,new Text("B.java",B)));return r;
  }
  static void snapshot(Path root,AbstractRewriteBaseRunMojo mojo,Throwable failure)throws Exception{
    List<String> lines=new ArrayList<>(mojo.logger.lines);lines.add("OBSERVED state="+mojo.state);
    lines.add("OBSERVED failure="+(failure==null?"none":failure.getClass().getName()+": "+failure.getMessage()));
    lines.add("OBSERVED A.exists="+Files.isRegularFile(root.resolve("A.java")));
    lines.add("OBSERVED B.regularFile="+Files.isRegularFile(root.resolve("B.java")));
    lines.add("OBSERVED B.directory="+Files.isDirectory(root.resolve("B.java")));
    lines.add("OBSERVED localNotes="+read(root.resolve("local-notes.txt")).trim());
    Files.write(root.resolve("capture.log"),lines,StandardCharsets.UTF_8);
  }
  public static void main(String[] args)throws Exception{
    Path root=Paths.get(args[0]);
    System.out.println("CONTROL ONLY: unchanged pinned mojo methods with supplied Result objects and stubbed collaborators; no Maven or recipe engine executed.");
    Path success=setup(root,"success"); AbstractRewriteRunMojo good=new AbstractRewriteRunMojo();good.supplied=changes(success);good.execute();
    check(read(success.resolve("A.java")).equals(A)&&read(success.resolve("B.java")).equals(B),"successful application writes both generated files");
    check(good.state==AbstractRewriteBaseRunMojo.State.PROCESSED,"successful application reaches processed");
    check(read(success.resolve("local-notes.txt")).equals(LOCAL),"success preserves unrelated local file");snapshot(success,good,null);

    Path early=setup(root,"early-error"); AbstractRewriteRunMojo before=new AbstractRewriteRunMojo();before.supplied=changes(early);
    before.supplied.firstException=new IllegalStateException("injected result error before file application");Throwable earlyFailure=null;
    try{before.execute();}catch(RuntimeException e){earlyFailure=e;}
    check(earlyFailure!=null,"result error surfaces before file application");
    check(!Files.exists(early.resolve("A.java"))&&!Files.exists(early.resolve("B.java")),"early error creates neither result file");
    check(read(early.resolve("local-notes.txt")).equals(LOCAL),"early error preserves local file");snapshot(early,before,earlyFailure);

    Path late=setup(root,"write-error"); Files.createDirectory(late.resolve("B.java"));write(late.resolve("B.java/keep.txt"),LOCAL);
    AbstractRewriteRunMojo after=new AbstractRewriteRunMojo();after.supplied=changes(late);Throwable lateFailure=null;
    try{after.execute();}catch(RuntimeException e){lateFailure=e;}
    check(lateFailure!=null,"later target conflict surfaces a file-application failure");
    check(Files.isRegularFile(late.resolve("A.java"))&&read(late.resolve("A.java")).equals(A),"earlier generated file remains after failure");
    check(Files.isDirectory(late.resolve("B.java"))&&!Files.isRegularFile(late.resolve("B.java")),"later result was not written as a source file");
    check(after.logger.lines.stream().anyMatch(s->s.contains("Generated new file B.java")),"log announces later file before its write fails");
    check(after.state!=AbstractRewriteBaseRunMojo.State.PROCESSED,"failed application does not reach processed");
    check(read(late.resolve("local-notes.txt")).equals(LOCAL)&&read(late.resolve("B.java/keep.txt")).equals(LOCAL),"failure preserves unrelated local files");snapshot(late,after,lateFailure);

    Field fail=AbstractRewriteDryRunMojo.class.getDeclaredField("failOnDryRunResults");fail.setAccessible(true);
    Field output=AbstractRewriteDryRunMojo.class.getDeclaredField("reportOutputDirectory");output.setAccessible(true);
    Path preview=setup(root,"preview-changes");AbstractRewriteDryRunMojo dry=new AbstractRewriteDryRunMojo();dry.supplied=changes(preview);
    output.set(dry,preview.resolve("reports").toString());fail.setBoolean(dry,true);Throwable gateFailure=null;
    try{dry.execute();}catch(Exception e){gateFailure=e;}
    check(gateFailure!=null&&gateFailure.getMessage().contains("would make changes"),"positive preview gate fails for detected results");
    check(Files.isRegularFile(preview.resolve("reports/rewrite.patch")),"positive preview retains report despite expected gate failure");
    check(!Files.exists(preview.resolve("A.java"))&&!Files.exists(preview.resolve("B.java")),"positive preview does not apply source results");snapshot(preview,dry,gateFailure);

    Path clean=setup(root,"preview-clean");AbstractRewriteDryRunMojo noChange=new AbstractRewriteDryRunMojo();noChange.supplied=new AbstractRewriteBaseRunMojo.ResultsContainer(clean);
    output.set(noChange,clean.resolve("reports").toString());fail.setBoolean(noChange,true);noChange.execute();
    check(!Files.exists(clean.resolve("reports/rewrite.patch")),"clean preview creates no patch");
    check(noChange.state==AbstractRewriteBaseRunMojo.State.PROCESSED,"clean preview completes with gate enabled");snapshot(clean,noChange,null);

    Path stale=setup(root,"preview-stale");Files.createDirectory(stale.resolve("reports"));write(stale.resolve("reports/rewrite.patch"),"stale report
");
    AbstractRewriteDryRunMojo reused=new AbstractRewriteDryRunMojo();reused.supplied=new AbstractRewriteBaseRunMojo.ResultsContainer(stale);
    output.set(reused,stale.resolve("reports").toString());fail.setBoolean(reused,true);reused.execute();
    check(read(stale.resolve("reports/rewrite.patch")).equals("stale report
"),"clean invocation does not delete a pre-existing report");snapshot(stale,reused,null);
    System.out.println("SUMMARY checks="+checks);
  }
}
