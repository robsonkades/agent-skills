import java.util.concurrent.CompletableFuture;
import java.util.concurrent.CompletionException;

public final class CompletionContractProbe {
    record Receipt(int units) {}
    enum Timing { PENDING, PRECOMPLETED }
    enum Input { VALID, INVALID, UPSTREAM_FAILURE }

    static final class Loader {
        final CompletableFuture<Receipt> upstream = new CompletableFuture<>();
        int checks;

        CompletableFuture<Receipt> load() { return upstream; }

        void checkReceipt(Receipt receipt) {
            checks++;
            if (receipt.units() <= 0) {
                throw new IllegalStateException("invalid receipt");
            }
        }

        // The partial snippet shipped in contracts-in-java.md.
        CompletableFuture<Receipt> original() {
            CompletableFuture<Receipt> loaded = load();
            loaded.thenAccept(this::checkReceipt);
            return loaded;
        }

        CompletableFuture<Receipt> checked() {
            return load().thenApply(receipt -> {
                checkReceipt(receipt);
                return receipt;
            });
        }
    }

    static void require(boolean ok, String message) {
        if (!ok) throw new AssertionError(message);
    }

    static void complete(Loader loader, Input input, Receipt receipt,
                         IllegalStateException failure) {
        if (input == Input.UPSTREAM_FAILURE) {
            require(loader.upstream.completeExceptionally(failure), "first failure completion");
        } else {
            require(loader.upstream.complete(receipt), "first normal completion");
        }
    }

    static Throwable failureOf(CompletableFuture<?> future) {
        require(future.isDone(), "must not block checking an incomplete future");
        try {
            future.join();
            throw new AssertionError("expected exceptional completion");
        } catch (CompletionException error) {
            return error.getCause();
        }
    }

    static void scenario(boolean checked, Timing timing, Input input) {
        Loader loader = new Loader();
        Receipt receipt = new Receipt(input == Input.INVALID ? 0 : 7);
        IllegalStateException upstreamFailure = new IllegalStateException("load failed");
        if (timing == Timing.PRECOMPLETED) {
            complete(loader, input, receipt, upstreamFailure);
        }

        // Any synchronous throw here fails the probe. A callback failure on a
        // precompleted source must still be observed through the dependent future.
        CompletableFuture<Receipt> observed = checked ? loader.checked() : loader.original();
        require(observed != null, "immediate return contract");
        if (timing == Timing.PENDING) {
            require(!observed.isDone(), "method returned before load completion");
            complete(loader, input, receipt, upstreamFailure);
        }

        String outcome;
        require(observed.isDone(), "completion after the controlled producer completes");
        if (input == Input.UPSTREAM_FAILURE) {
            require(failureOf(observed) == upstreamFailure, "upstream cause preserved");
            require(loader.checks == 0, "success check not run on upstream failure");
            outcome = "UPSTREAM_FAILURE";
        } else if (input == Input.INVALID && checked) {
            Throwable failure = failureOf(observed);
            require(failure instanceof IllegalStateException, "checker failure type");
            require("invalid receipt".equals(failure.getMessage()), "checker failure cause");
            require(loader.upstream.join() == receipt, "dependent failure does not change source");
            require(loader.checks == 1, "checker invoked once");
            outcome = "CHECK_FAILURE";
        } else {
            require(observed.join() == receipt, "normal result identity");
            require(loader.checks == 1, "checker invoked once");
            outcome = input == Input.INVALID ? "NORMAL_WITH_REJECTED_RECEIPT" : "VALID";
        }
        System.out.printf("%s %s %s -> %s%n",
                checked ? "DEPENDENT" : "ORIGINAL", timing, input, outcome);
    }

    public static void main(String[] args) {
        int cases = 0;
        for (boolean checked : new boolean[] { false, true }) {
            for (Timing timing : Timing.values()) {
                for (Input input : Input.values()) {
                    scenario(checked, timing, input);
                    cases++;
                }
            }
        }
        System.out.printf("PASS: %d scenarios; assertions=%s; runtime=%s%n",
                cases, CompletionContractProbe.class.desiredAssertionStatus(),
                System.getProperty("java.runtime.version"));
    }
}
