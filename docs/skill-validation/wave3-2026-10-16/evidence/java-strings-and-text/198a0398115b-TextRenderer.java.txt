final class TextRenderer {
    // Existing consumer-visible rendering; keep this compatibility oracle unchanged.
    static String legacy(char[][] fragments) {
        String result = "";
        for (char[] fragment : fragments) result += fragment + "|";
        return result;
    }

    // Candidate change to avoid copying an ever-growing prefix.
    static String render(char[][] fragments) {
        StringBuilder result = new StringBuilder();
        for (Object fragment : fragments) result.append(fragment).append('|');
        return result.toString();
    }
}
