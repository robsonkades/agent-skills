import {spawnSync} from 'node:child_process';
import {mkdirSync, writeFileSync} from 'node:fs';
import {resolve, dirname} from 'node:path';
import {fileURLToPath} from 'node:url';
import assert from 'node:assert/strict';
const root = dirname(fileURLToPath(import.meta.url));
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const results = [];
for (const [id, expectedStatus] of [['A', 0], ['B', 1]]) {
    const cwd = resolve(root, 'case-inputs', id);
    mkdirSync(resolve(cwd, 'classes'), {recursive: true});
    for (const [stage, executable, args, status] of [
        ['compile', jdk + '/javac.exe', ['--release', '25', '-d', 'classes', 'TextRenderer.java', 'ConsumerContract.java'], 0],
        ['run', jdk + '/java.exe', ['-cp', 'classes', 'ConsumerContract'], expectedStatus],
    ]) {
        const r = spawnSync(executable, args, {cwd, encoding: 'utf8', timeout: 30000, maxBuffer: 1_000_000});
        writeFileSync(resolve(root, 'case-' + id + '-' + stage + '.stdout.txt'), r.stdout ?? '');
        writeFileSync(resolve(root, 'case-' + id + '-' + stage + '.stderr.txt'), r.stderr ?? '');
        const result = {id, stage, executable, args, cwd, status:r.status, error:r.error?.message ?? null, stdout:r.stdout ?? '', stderr:r.stderr ?? ''};
        results.push(result);
        writeFileSync(resolve(root, 'case-fixture-controls.json'), JSON.stringify(results, null, 2) + '\n');
        console.log(JSON.stringify(result));
        assert.equal(r.status, status, id + ' ' + stage);
        if (stage === 'run') {
            if (id === 'A') assert(r.stdout.includes('PASS 6 compatibility cases'));
            else {
                assert(r.stderr.includes('5 of 6 compatibility cases failed'));
                assert(r.stdout.includes('null fragment: new NullPointerException'));
                assert(r.stdout.includes('nonempty fragment: changed output'));
            }
        }
    }
}
