public final class PublishedSnippet { public static void main(String[] args) {
char[] chars = {'A', 'B'};
String original = "" + chars;                              // array object representation
String preserved = new StringBuilder().append((Object) chars).toString();
String contents = new StringBuilder().append(chars).toString(); // "AB": different contract

if (!original.equals(preserved)) throw new AssertionError("object conversion changed");
if (!contents.equals("AB") || original.equals(contents)) throw new AssertionError("content contract not distinguished");
System.out.println("PASS actual published snippet: object conversion preserved; content alternative distinguished");
} }
