import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { resolve } from 'node:path';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';

const root = 'C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL/java-date-and-time-check';
const repo = 'C:/git/agent-skills';
const bin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const document = readFileSync(resolve(repo, 'skills/java-date-and-time/references/clocks-and-tests.md'), 'utf8');
const examples = [...document.matchAll(/```java\r?\n([\s\S]*?)\r?\n```/g)];
if (examples.length !== 1) throw new Error('Expected exactly one complete Java example');
const original = examples[0][1] + '\n';
const method = /    static ZonedDateTime parseConsistent\(String text\) \{[\s\S]*?\n    \}/g;
if ([...original.matchAll(method)].length !== 1) throw new Error('Expected exactly one parseConsistent method');
const replacement = '    static ZonedDateTime parseConsistent(String text) {\n        return ZonedDateTime.parse(text);\n    }';
const mutant = original.replace(method, replacement);
const sha256 = text => createHash('sha256').update(text).digest('hex');
const results = [];
function run(id, executable, args, cwd, expectedExitCode) {
  const result = spawnSync(executable, args, { cwd, encoding: 'utf8' });
  const entry = { id, executable, args, cwd, expectedExitCode, exitCode: result.status,
    signal: result.signal, stdout: result.stdout ?? '', stderr: result.stderr ?? '',
    error: result.error ? String(result.error) : null };
  results.push(entry);
  writeFileSync(resolve(root, id + '.json'), JSON.stringify(entry, null, 2) + '\n');
  if (entry.exitCode !== expectedExitCode) throw new Error(id + ' unexpected exit: ' + entry.exitCode);
  return entry;
}
mkdirSync(root, { recursive: true });
for (const [name, source] of [['original', original], ['normalizing-mutant', mutant]]) {
  const cwd = resolve(root, name);
  mkdirSync(resolve(cwd, 'classes'), { recursive: true });
  writeFileSync(resolve(cwd, 'TemporalChecks.java'), source);
  run(name + '-compile', resolve(bin, 'javac.exe'), ['--release', '17', '-Xlint:all', '-Werror', '-d', 'classes', 'TemporalChecks.java'], cwd, 0);
}
