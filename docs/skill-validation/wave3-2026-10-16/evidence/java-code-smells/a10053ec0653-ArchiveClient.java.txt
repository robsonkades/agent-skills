import java.io.*;

public final class ArchiveClient {
    public static void main(String[] args) throws Exception {
        var entry = new ArchiveEntry("INV-27", "customer requested email");
        var bytes = new ByteArrayOutputStream();
        try (var out = new ObjectOutputStream(bytes)) {
            out.writeObject(entry);
        }
        ArchiveEntry restored;
        try (var in = new ObjectInputStream(new ByteArrayInputStream(bytes.toByteArray()))) {
            restored = (ArchiveEntry) in.readObject();
        }
        if (!entry.id().equals(restored.id()) || !entry.note().equals(restored.note())) {
            throw new AssertionError("archive must retain id and note: restored note=" + restored.note());
        }
        System.out.println("PASS round trip: id=" + restored.id() + ", note=" + restored.note());
    }
}
