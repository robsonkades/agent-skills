import fs from 'node:fs';
import {spawnSync} from 'node:child_process';
import assert from 'node:assert/strict';
const dir='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL/c2-sea-of-nodes';
const executable='C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe';
const results=[];
for(const [id,flags,expected] of [
 ['medium-64',['-XX:InlineSmallCode=64'],'already compiled into a medium method'],
 ['medium-64-bytecode-1000',['-XX:InlineSmallCode=64','-XX:MaxInlineSize=1000'],'already compiled into a medium method'],
 ['medium-64-frequency-0',['-XX:InlineSmallCode=64','-XX:InlineFrequencyRatio=0'],'inline (hot)'],
]) {
 const args=['-Xbatch','-XX:-TieredCompilation','-XX:CompileThreshold=1000','-XX:+UnlockDiagnosticVMOptions','-XX:+LogCompilation',`-XX:LogFile=${dir}/logs/${id}.xml`,'-XX:CompileCommand=exclude,InlineGate::main','-XX:CompileCommand=PrintInlining,InlineGate::caller',...flags,'-cp',`${dir}/classes`,'InlineGate'];
 const r=spawnSync(executable,args,{cwd:dir,encoding:'utf8',timeout:30000,maxBuffer:5_000_000});
 fs.writeFileSync(`${dir}/logs/${id}.stdout.txt`,r.stdout??'');
 fs.writeFileSync(`${dir}/logs/${id}.stderr.txt`,r.stderr??'');
 const entry={id,executable,args,cwd:dir,status:r.status,error:r.error?.message,expected,stdout:r.stdout,stderr:r.stderr};
 results.push(entry);
}
fs.writeFileSync(`${dir}/gate-controls.json`,JSON.stringify(results,null,2));
for(const r of results) {
 console.log(JSON.stringify({id:r.id,status:r.status,expected:r.expected,output:r.stdout+r.stderr}));
 assert.equal(r.status,0,r.id);
 assert.ok(r.stdout.includes(r.expected),r.id);
}
