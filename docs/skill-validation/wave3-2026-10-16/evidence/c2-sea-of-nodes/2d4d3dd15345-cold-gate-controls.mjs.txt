import fs from 'node:fs';
import {spawnSync} from 'node:child_process';
import assert from 'node:assert/strict';
const dir='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL/c2-sea-of-nodes';
const executable='C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe';
// Frozen expectations before these fresh runs. Earlier exploratory failures remain retained.
const cases=[
 ['cold-medium-128',['-XX:InlineSmallCode=128'],'already compiled into a medium method'],
 ['cold-medium-128-bytecode-1000',['-XX:InlineSmallCode=128','-XX:MaxInlineSize=1000'],'already compiled into a medium method'],
 ['cold-medium-512',['-XX:InlineSmallCode=512'],'inline (hot)'],
];
const results=[];
for(const [id,flags,expected] of cases) {
 const args=['-Xbatch','-XX:-TieredCompilation','-XX:CompileThreshold=1000','-XX:+UnlockDiagnosticVMOptions','-XX:InlineFrequencyRatio=2.0','-XX:+LogCompilation',`-XX:LogFile=${dir}/logs/${id}.xml`,'-XX:CompileCommand=exclude,InlineGate::main','-XX:CompileCommand=PrintInlining,InlineGate::caller',...flags,'-cp',`${dir}/classes`,'InlineGate'];
 const r=spawnSync(executable,args,{cwd:dir,encoding:'utf8',timeout:30000,maxBuffer:5_000_000});
 fs.writeFileSync(`${dir}/logs/${id}.stdout.txt`,r.stdout??'');
 fs.writeFileSync(`${dir}/logs/${id}.stderr.txt`,r.stderr??'');
 results.push({id,executable,args,cwd:dir,status:r.status,error:r.error?.message,expected,stdout:r.stdout,stderr:r.stderr});
}
fs.writeFileSync(`${dir}/cold-gate-controls.json`,JSON.stringify(results,null,2));
for(const r of results) {
 console.log(JSON.stringify({id:r.id,status:r.status,expected:r.expected,output:r.stdout+r.stderr}));
 assert.equal(r.status,0,r.id);
 assert.ok(r.stdout.includes(r.expected),r.id);
 assert.match(r.stdout,/12399400384\r?\n$/);
}
