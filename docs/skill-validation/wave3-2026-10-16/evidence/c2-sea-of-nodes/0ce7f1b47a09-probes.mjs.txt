import fs from 'node:fs';
import assert from 'node:assert/strict';

fs.mkdirSync(`${dir}/logs`,{recursive:true});
const runs=[];
function run(id, executable, args) {
 const r=spawnSync(executable,args,{cwd:dir,encoding:'utf8',timeout:30000,maxBuffer:10_000_000});
 const stdout=`${dir}/logs/${id}.stdout.txt`, stderr=`${dir}/logs/${id}.stderr.txt`;
 fs.writeFileSync(stdout,r.stdout ?? ''); fs.writeFileSync(stderr,r.stderr ?? '');
 const entry={id,executable,args,cwd:dir,status:r.status,signal:r.signal,error:r.error?.message,stdout,stderr};
 runs.push(entry);
 return {...entry,output:(r.stdout ?? '')+(r.stderr ?? '')};
}
