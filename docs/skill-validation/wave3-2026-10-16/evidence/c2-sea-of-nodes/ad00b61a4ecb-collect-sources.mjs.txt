import fs from 'node:fs';

import crypto from 'node:crypto';
import assert from 'node:assert/strict';

const root = 'C:/git/agent-skills';
const dir = 'C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL/c2-sea-of-nodes';
const baseline = JSON.parse(fs.readFileSync(`${root}/docs/skill-validation/wave3-2026-10-16/baseline.json`, 'utf8')).packages.find(p => p.name === 'c2-sea-of-nodes');
const sha256 = bytes => crypto.createHash('sha256').update(bytes).digest('hex');
const hashes = baseline.files.map(file => {
  const currentSha256 = sha256(fs.readFileSync(`${root}/${file.path}`));
  assert.equal(currentSha256, file.sha256);
  return {path:file.path, assignmentSha256:file.sha256, currentSha256};
});
fs.writeFileSync(`${dir}/assignment-baseline.json`, JSON.stringify({committedVersion:baseline.committedVersion, initialVersion:baseline.initialVersion, initialDiff:baseline.initialDiff, initialStatus:baseline.initialStatus,files:hashes},null,2));

fs.mkdirSync(`${dir}/sources`, {recursive:true});
const results = await Promise.all(sources.map(async ([relativePath,purpose]) => {
 const url = `https://raw.githubusercontent.com/openjdk/jdk25u/jdk-25.0.3%2B9/${relativePath}`;
 const file = path.basename(relativePath);
 try {
  const response = await fetch(url);
  if(!response.ok) return {file,url,purpose,status:response.status};
  const bytes = Buffer.from(await response.arrayBuffer());
  fs.writeFileSync(`${dir}/sources/${file}`,bytes);
  return {file,url,purpose,status:response.status,bytes:bytes.length,sha256:sha256(bytes)};
 } catch(error) {return {file,url,purpose,error:error.message};}
}));
