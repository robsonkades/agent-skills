import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { mkdirSync, writeFileSync, readFileSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = dirname(fileURLToPath(import.meta.url));
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const logs = join(root, 'logs');
mkdirSync(logs, { recursive: true });
const commands = [];
function run(id, executable, args, expectedExit = 0) {
    const result = spawnSync(join(jdk, executable + '.exe'), args, {
        cwd: root, encoding: 'utf8', timeout: 30000, windowsHide: true
    });
    const stdoutPath = join(logs, id + '.stdout.txt');
    const stderrPath = join(logs, id + '.stderr.txt');
    writeFileSync(stdoutPath, result.stdout ?? '');
    writeFileSync(stderrPath, result.stderr ?? '');
    const command = { id, executable: join(jdk, executable + '.exe'), args, cwd: root,
        expectedExit, actualExit: result.status, signal: result.signal, error: result.error?.message ?? null,
        stdoutPath, stderrPath };
    commands.push(command);
    writeFileSync(join(root, 'commands.json'), JSON.stringify(commands, null, 2) + '\n');
    assert.equal(result.error, undefined, id + ': process error');
    assert.equal(result.status, expectedExit, id + ': unexpected exit');
    return { stdout: result.stdout, stderr: result.stderr };
}
const src = join(root, 'actor/common/src/FirstChoice.java');
const test = join(root, 'actor/common/src/FirstChoiceTest.java');
const original = join(root, 'actor/common/artifact');
const release17 = join(root, 'outputs/release17-rejection');
const release21 = join(root, 'outputs/release21');
const compatible17 = join(root, 'outputs/compatible17');
const wrong17 = join(root, 'outputs/wrong17');
for (const path of [original, release17, release21, compatible17, wrong17]) mkdirSync(path, { recursive: true });
const javaVersion = run('01-java-version', 'java', ['-version']);
const javacVersion = run('02-javac-version', 'javac', ['-version']);
const originalCompile = run('03-source-target17', 'javac', ['-source', '17', '-target', '17', '-d', original, src, test]);
const originalBytecode = run('04-original-bytecode', 'javap', ['-c', '-verbose', '-classpath', original, 'FirstChoice']);
assert.match(originalBytecode.stdout, /major version: 61/);
assert.match(originalBytecode.stdout, /java\/util\/List\.getFirst/);
const originalRun = run('05-original-on25', 'java', ['-cp', original, 'FirstChoiceTest']);
assert.match(originalRun.stdout, /PASS first element and empty-list contract/);
const rejected17 = run('06-release17-rejected', 'javac', ['--release', '17', '-d', release17, src, test], 1);
assert.match(rejected17.stderr, /cannot find symbol/);
assert.match(rejected17.stderr, /getFirst/);
run('07-release21', 'javac', ['--release', '21', '-d', release21, src, test]);
const release21Bytecode = run('08-release21-bytecode', 'javap', ['-c', '-verbose', '-classpath', release21, 'FirstChoice']);
assert.match(release21Bytecode.stdout, /major version: 65/);
const release21Run = run('09-release21-on25', 'java', ['-cp', release21, 'FirstChoiceTest']);
assert.match(release21Run.stdout, /PASS first element and empty-list contract/);
run('10-compatible17', 'javac', ['--release', '17', '-d', compatible17, join(root, 'controls/compatible17/FirstChoice.java'), test]);
const compatible17Run = run('11-compatible17-on25', 'java', ['-cp', compatible17, 'FirstChoiceTest']);
assert.match(compatible17Run.stdout, /PASS first element and empty-list contract/);
run('12-wrong17', 'javac', ['--release', '17', '-d', wrong17, join(root, 'controls/wrong17/FirstChoice.java'), test]);
const wrong17Run = run('13-wrong17-on25', 'java', ['-cp', wrong17, 'FirstChoiceTest'], 1);
assert.match(wrong17Run.stderr, /IndexOutOfBoundsException/);
const summary = {
    javaVersion: javaVersion.stderr.trim(), javacVersion: javacVersion.stdout.trim(),
    executedCommands: commands.length, expectedNegativeExits: 2,
    original: { compilerLanguageAndClassTarget: 17, linkedPlatformAPI: 'JDK25 default; List.getFirst since21',
        classMajor: 61, runtimeActuallyExercised: 25, tests: 'PASS first and empty-list behavior',
        compilerWarning: originalCompile.stderr.trim() },
    release17: 'Original rejected at compile time: getFirst unavailable',
    release21: { compilation: 'PASS', classMajor: 65, runtimeActuallyExercised: 25, tests: 'PASS' },
    compatible17: { compilation: 'PASS --release17', runtimeActuallyExercised: 25, tests: 'PASS including NoSuchElementException on empty' },
    wrong17: { compilation: 'PASS --release17', runtimeActuallyExercised: 25,
        tests: 'EXPECTED FAILURE: IndexOutOfBoundsException instead of NoSuchElementException' },
    limitations: ['No Java17 or21 VM was executed.', 'This is a technical fixture; no agent evaluation or performance experiment.',
        'The original class major61 alone does not establish runtime17 compatibility.']
};
writeFileSync(join(root, 'technical-results.json'), JSON.stringify(summary, null, 2) + '\n');
console.log(JSON.stringify(summary, null, 2));
