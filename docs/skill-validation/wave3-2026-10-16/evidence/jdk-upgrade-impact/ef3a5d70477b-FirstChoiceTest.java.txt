import java.util.List;
import java.util.NoSuchElementException;

public final class FirstChoiceTest {
    public static void main(String[] args) {
        String result = FirstChoice.first(List.of("north", "south"));
        if (!"north".equals(result)) throw new AssertionError("first element: " + result);
        try {
            FirstChoice.first(List.of());
            throw new AssertionError("empty list must throw NoSuchElementException");
        } catch (NoSuchElementException expected) {
            // Public behavior promised in runtime-contract.json.
        }
        System.out.println("PASS first element and empty-list contract");
    }
}
