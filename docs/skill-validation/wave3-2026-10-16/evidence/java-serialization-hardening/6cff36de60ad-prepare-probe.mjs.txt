const header = `package fixture;
import java.io.*;
public final class RetryPolicy implements Serializable {
    @Serial private static final long serialVersionUID = 1L;
    private int maxRetries;
    public RetryPolicy(int value) { maxRetries = value; }
    public int maxRetries() { return maxRetries; }
`;
const baseline = `    @Serial private void readObject(ObjectInputStream s) throws IOException, ClassNotFoundException {
        s.defaultReadObject();
        if (maxRetries < 0 || maxRetries > 10) throw new InvalidObjectException("invalid maxRetries");
    }
`;
const compat = `    @Serial private void readObject(ObjectInputStream s) throws IOException, ClassNotFoundException {
        ObjectInputStream.GetField fields = s.readFields();
        int candidate = fields.defaulted("maxRetries") ? 3 : fields.get("maxRetries", 0);
        if (candidate < 0 || candidate > 10) throw new InvalidObjectException("invalid maxRetries");
        maxRetries = candidate;
    }
`;
const denied = `package fixture;
import java.io.*;
public final class DeniedPayload implements Serializable {

    public static int readHooks;

const writer = `import fixture.*;

    }

`;

const reader = `import fixture.*;
import java.io.*;
import java.nio.file.*;
public final class Reader {
    private static int assertions;
    public static void main(String[] args) throws Exception {
        String mode = args[0];
        Path inputs = Path.of(args[1]);
        equal(0, read(inputs.resolve("value-0.ser")).maxRetries(), "explicit zero retained");
        equal(5, read(inputs.resolve("value-5.ser")).maxRetries(), "ordinary value retained");
        equal(10, read(inputs.resolve("value-10.ser")).maxRetries(), "upper bound retained");
        rejects(InvalidObjectException.class, () -> read(inputs.resolve("value--1.ser")), "negative value rejected");
        rejects(InvalidObjectException.class, () -> read(inputs.resolve("value-11.ser")), "over-limit value rejected");
        if (mode.equals("required")) rejects(InvalidObjectException.class, () -> read(inputs.resolve("missing.ser")), "omitted required field rejected");
        else equal(mode.equals("legacy") ? 3 : 0, read(inputs.resolve("missing.ser")).maxRetries(), mode + " omitted-field control");
        rejects(InvalidClassException.class, () -> read(inputs.resolve("denied.ser")), "unlisted class rejected before hook");

        rejects(InvalidObjectException.class, () -> read(inputs.resolve("null.ser")), "null root rejected");
        System.out.println("PASS: mode=" + mode + ", assertions=" + assertions + ", runtime=" + Runtime.version());
    }
    public static RetryPolicy read(Path path) throws Exception {
        if (Files.size(path) > 4096) throw new InvalidObjectException("input too large");
        byte[] bytes = Files.readAllBytes(path);
        try (ObjectInputStream in = new ObjectInputStream(new ByteArrayInputStream(bytes))) {
            in.setObjectInputFilter(ObjectInputFilter.Config.createFilter("maxdepth=6;maxrefs=20;maxarray=16;maxbytes=4096;fixture.RetryPolicy;!*"));
            Object value = in.readObject();
            if (!(value instanceof RetryPolicy)) throw new InvalidObjectException("unexpected root");
            return (RetryPolicy) value;
        }

    private static void equal(int expected, int actual, String name) {
        if (expected != actual) throw new AssertionError(name + ": " + expected + " != " + actual);

        System.out.println("PASS: " + name);
    }
    private static void rejects(Class<? extends Exception> expected, Action action, String name) throws Exception {
        try { action.run(); } catch (Exception failure) {
            if (!expected.isInstance(failure)) throw failure;
            assertions++;
            System.out.println("PASS: " + name + " -> " + failure.getClass().getSimpleName());
            return;
        }
        throw new AssertionError("Expected rejection: " + name);
    }
    @FunctionalInterface private interface Action { void run() throws Exception; }
}
