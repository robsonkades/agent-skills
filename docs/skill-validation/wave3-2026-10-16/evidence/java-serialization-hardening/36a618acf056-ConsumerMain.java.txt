import java.io.*;
import java.nio.file.*;
import fixture.RetryPolicy;
public final class ConsumerMain {
    public static void main(String[] args) throws Exception {
        Path path = Path.of(args[0]);
        if (Files.size(path) > 4096) throw new InvalidObjectException("input too large");
        byte[] bytes = Files.readAllBytes(path);
        try (ObjectInputStream in = new ObjectInputStream(new ByteArrayInputStream(bytes))) {
            in.setObjectInputFilter(ObjectInputFilter.Config.createFilter("maxdepth=6;maxrefs=20;maxarray=16;maxbytes=4096;fixture.RetryPolicy;!*"));
            Object value = in.readObject();
            if (!(value instanceof RetryPolicy)) throw new InvalidObjectException("unexpected root");
            System.out.println(((RetryPolicy) value).maxRetries());
        }
    }
}
