import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
const root=path.dirname(fileURLToPath(import.meta.url));
const get=(variant,file)=>readFile(path.join(root,'cases',variant,file),'utf8');
for (const file of ['task.md','Operation.java','StoredFailureResource.java','Probe.java']) {
  assert.equal(await get('shared',file),await get('distinct',file));
}
assert.equal((await get('shared','FixtureFactory.java')).replace('new StoredFailureResource(workFailure)','new StoredFailureResource(new IOException("storage failure"))'),await get('distinct','FixtureFactory.java'));
const criteria=JSON.parse(await readFile(path.join(root,'case-criteria.json'),'utf8'));
assert.equal(criteria.length,2);
assert.equal(new Set(criteria.map(c=>c.id)).size,2);
console.log(JSON.stringify({checks:7,result:'PASS',scope:'Case fixture integrity only; not an agent evaluation.'},null,2));
