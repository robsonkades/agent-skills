import java.io.IOException;
import java.util.Objects;

public final class SuppressionContractMutant {
    private static int checks;
    @FunctionalInterface interface Work { void run() throws Throwable; }

    static final class Resource implements AutoCloseable {
        final Exception closeFailure;
        int closeCalls;
        Resource(Exception closeFailure) { this.closeFailure = closeFailure; }
        @Override public void close() throws Exception {
            closeCalls++;
            if (closeFailure != null) throw closeFailure;
        }
    }

    static final class SuppressionDisabled extends RuntimeException {
        SuppressionDisabled() { super("disabled", null, false, false); }
    }

    static Throwable capture(Work work) {
        try { work.run(); return null; }
        catch (Throwable failure) { return failure; }
    }

    static void twr(Resource resource, Exception workFailure) throws Exception {
        try (resource) {
            if (workFailure != null) throw workFailure;
        }
    }

    static void manual(Resource resource, Exception workFailure) throws Throwable {
        Throwable primary = null;
        try {
            if (workFailure != null) throw workFailure;
        } catch (Throwable failure) {
            primary = failure;
            throw failure;
        } finally {
            if (primary == null) {
                resource.close();
            } else {
                try { resource.close(); }
                catch (Throwable secondary) {
                    primary.addSuppressed(secondary);
                }
            }
        }
    }

    static void check(boolean condition, String name) {
        checks++;
        if (!condition) throw new AssertionError(name);
        System.out.println("PASS " + name);
    }

    public static void main(String[] args) {
        IOException body1 = new IOException("operation");
        IOException close1 = new IOException("cleanup");
        Resource distinct = new Resource(close1);
        Throwable actual1 = capture(() -> twr(distinct, body1));
        check(actual1 == body1, "distinct TWR retains body identity");
        check(actual1.getSuppressed().length == 1 && actual1.getSuppressed()[0] == close1,
              "distinct TWR retains cleanup identity");
        check(distinct.closeCalls == 1, "distinct TWR closes once");

        IOException shared = new IOException("operation");
        Resource same = new Resource(shared);
        Throwable actual2 = capture(() -> twr(same, shared));
        check(actual2 instanceof IllegalArgumentException, "same TWR masks primary with self-suppression error");
        check(actual2.getCause() == shared, "self-suppression error retains original as cause on this JDK");
        check(shared.getSuppressed().length == 0, "same object not added as suppressed");
        check(same.closeCalls == 1, "same TWR still closes once");

        IOException body3 = new IOException("operation");
        Resource manualShared = new Resource(body3);
        Throwable actual3 = capture(() -> manual(manualShared, body3));
        check(actual3 == body3 && body3.getSuppressed().length == 0, "guarded manual cleanup retains shared primary");
        check(manualShared.closeCalls == 1, "guarded shared cleanup attempted once");

        IOException body4 = new IOException("operation");
        IOException close4 = new IOException("cleanup");
        Resource manualDistinct = new Resource(close4);
        Throwable actual4 = capture(() -> manual(manualDistinct, body4));
        check(actual4 == body4 && actual4.getSuppressed().length == 1
              && actual4.getSuppressed()[0] == close4, "guard still records distinct cleanup failure");

        IOException closeOnly = new IOException("cleanup");
        Resource successThenCloseFailure = new Resource(closeOnly);
        check(capture(() -> manual(successThenCloseFailure, null)) == closeOnly,
              "manual cleanup failure primary when work succeeds");
        Resource allSuccess = new Resource(null);
        check(capture(() -> manual(allSuccess, null)) == null && allSuccess.closeCalls == 1,
              "manual successful cleanup retains normal completion");

        SuppressionDisabled disabled = new SuppressionDisabled();
        IOException distinctDisabledClose = new IOException("cleanup");
        check(capture(() -> twr(new Resource(distinctDisabledClose), disabled)) == disabled
              && disabled.getSuppressed().length == 0, "disabled suppression loses distinct cleanup evidence");
        Throwable disabledSame = capture(() -> twr(new Resource(disabled), disabled));
        check(disabledSame instanceof IllegalArgumentException,
              "disabled suppression does not allow self-suppression");

        IOException sameTextBody = new IOException("same");
        IOException sameTextClose = new IOException("same");
        Throwable sameTextActual = capture(() -> twr(new Resource(sameTextClose), sameTextBody));
        check(sameTextActual == sameTextBody && sameTextActual.getSuppressed()[0] == sameTextClose,
              "identity matters; same text on distinct instances is valid");
        System.out.println("CHECKS=" + checks);
        System.out.println("Runtime=" + Runtime.version());
        System.out.println("Scope=isolated contract checks compiled with --release 21; runtime is not JDK 21");
    }
}
