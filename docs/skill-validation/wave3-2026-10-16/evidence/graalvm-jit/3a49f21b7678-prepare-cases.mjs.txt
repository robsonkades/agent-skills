import { mkdirSync, writeFileSync, readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { createHash } from 'node:crypto';

const root = 'C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL/graalvm-jit';
const runtime = {
  fixtureKind: 'Synthetic development evidence; not production measurements or the technical JFR probe',
  application: 'QuoteService.compute is the representative CPU hot path under fixed request mix',
  runtime: 'GraalVM CE 25.0.2+10-jvmci-b01, Windows x64',
  comparison: 'Three independent fresh processes per treatment on the same binary and resource limits; only UseJVMCICompiler differs',
  correctness: 'Representative response contracts passed on both treatments; preserved accepted evidence',
  load: '2000 requests/second, 10-second windows, identical request mix, errors=0, no omitted requests',
  observation: 'Metrics timestamps are seconds since process start; all starts use equivalent cache/data state; 60-150 seconds is the proposed measurement region',
  diagnostics: 'Separate equivalent diagnostic runs; no compilation capture threshold or relevant dropped events',
  scope: 'Review the measurement window; production adoption, licensing and fleet readiness remain a later decision',
};
const compilation = {
  graalConfiguration: 'Graal Community compiler with Truffle extensions loaded from a Native Image shared library (gc=Serial GC)',
  processes: ['graal-1', 'graal-2', 'graal-3', 'c2-1', 'c2-2', 'c2-3'].map((process) => ({
    process,
    relevantSuccessfulCompilation: { timestampSeconds: 25, method: 'QuoteService.compute', compileLevel: 4, succeded: true, compiler: process.startsWith('graal') ? 'jvmci' : 'c2', isOsr: false },
    subsequentInvalidationOrFailure: false,
    compilerStatistics: [30, 60, 90, 120, 150].map((timestampSeconds) => ({ timestampSeconds, compileCount: 5000, totalTimeSpentMillis: 300 })),
  })),
};
const draft = `# Proposed measurement decision

The compiler statistics stop changing at 30 seconds, so Graal and the application are fully warm then. The 300 ms totalTimeSpent is compiler CPU cost. Use the 60-150 second window as steady-state evidence for the C2/Graal comparison. All runtime-selection and correctness checks listed in runtime.json remain accepted.
`;
const inputs = [];
const files = ['runtime.json', 'compilation-evidence.json', 'metrics.csv', 'decision.md'];
for (const id of ['case-a', 'case-b']) {
  const path = resolve(root, 'cases', id);
  mkdirSync(path, { recursive: true });
  writeFileSync(resolve(path, 'runtime.json'), JSON.stringify(runtime, null, 2) + '\n');
  writeFileSync(resolve(path, 'compilation-evidence.json'), JSON.stringify(compilation, null, 2) + '\n');
  writeFileSync(resolve(path, 'decision.md'), draft);
  const rows = ['process,windowStartSeconds,windowEndSeconds,requests,errors,p99Millis'];
  for (const treatment of ['graal', 'c2']) {
    for (let process = 1; process <= 3; process++) {
      for (let start = 0; start < 150; start += 10) {
        const offset = process - 2 + ((start / 10) % 2);
        let p99;
        if (start < 60) p99 = (treatment === 'graal' ? 190 : 160) - start + offset;
        else if (treatment === 'c2') p99 = 100 + offset;
        else p99 = (id === 'case-a' ? 80 : 130 - (start - 60) / 2) + offset;
        rows.push(`${treatment}-${process},${start},${start + 10},20000,0,${p99}`);
      }
    }
  }
  writeFileSync(resolve(path, 'metrics.csv'), rows.join('\n') + '\n');
  inputs.push({
    id,
    fixtureDirectory: path,
    files,
    task: 'Review decision.md against the supplied artifacts and update it in your isolated case copy. Decide whether the proposed 60-150 second window supports a steady-state C2/Graal comparison, correct unsupported interpretations, preserve valid runtime/correctness evidence, and state the smallest missing evidence or next step. Do not install tools or change infrastructure. Treat the fixture as a supplied observation, not a request to reproduce a performance benchmark.',
  });
}
writeFileSync(resolve(root, 'actor-inputs.json'), JSON.stringify({ cases: inputs }, null, 2) + '\n');
const criteria = {
  classification: 'Known development cases, not holdouts; no actor evaluation has been run',
  isolationRequirement: 'Copy only one actor fixture plus the selected original/revised skill into the actor environment. Keep this evaluator file, other-case fixtures and review artifacts unavailable to the actor. Folder separation here is procedural, not access control.',
  decisiveConstraint: 'Only metrics.csv changes: Graal application p99 after second 60 is approximately stationary in case-a and systematically declining in case-b. The compiler counters and all other artifacts are identical.',
  common: [
    'Inspect the runtime, compilation and metric artifacts; update decision.md rather than merely list questions.',
    'Remove the inference that flat CompilerStatistics proves the application became warm at second 30.',
    'Do not label totalTimeSpent as compiler CPU cost or process uptime; identify cumulative process accounting with no per-method/per-compiler field.',
    'Keep accepted runtime selection, successful relevant tier-4 compilation and correctness evidence; do not manufacture a missing-Graal diagnosis.',
    'Do not promote this measurement review to a production migration or readiness approval, or run an unrequested installation/benchmark.',
  ],
  cases: [
    {
      id: 'case-a',
      expected: [
        'Permit the proposed 60-150 second window as an observed approximately stable region under the supplied repeated workload, with uncertainty and scope explicit.',
        'Use the application trace, not quiet compiler counters, for that decision; Graal p99 is about 79-82 ms and C2 about 99-102 ms per observed window.',
        'Do not demand more warm-up solely because no universal convergence threshold was supplied; exact steady-state onset and future workload phases remain unproven.',
      ],
      observableFailure: ['Reject all existing measurement evidence merely because compiler statistics cannot prove stability.', 'Claim an exact universal warm-up duration or statistical significance from the supplied summaries.'],
    },
    {
      id: 'case-b',
      expected: [
        'Reject the current steady-state window claim: all three Graal traces continue declining through the final window despite unchanged compiler statistics.',
        'Preserve the stable C2 baseline and runtime evidence; identify remaining application/cache/dependency behavior as a hypothesis to investigate, not a proved compiler fault.',
        'Request/plan only the observation needed to establish a later stable region or report the workload as non-steady; do not average away the drift or blindly extend a universal warm-up count.',
      ],
      observableFailure: ['Accept the window because compilation totals are flat.', 'Replace the drift with an average or the best final sample and call it a steady-state speedup.'],
    },
  ],
};
writeFileSync(resolve(root, 'evaluator-criteria.json'), JSON.stringify(criteria, null, 2) + '\n');
const differences = files.filter((file) => readFileSync(resolve(root, 'cases/case-a', file)).compare(readFileSync(resolve(root, 'cases/case-b', file))) !== 0);
if (JSON.stringify(differences) !== JSON.stringify(['metrics.csv'])) throw new Error(`Unexpected case differences: ${differences}`);
const hashes = inputs.map((entry) => ({ id: entry.id, files: files.map((file) => ({ file, sha256: createHash('sha256').update(readFileSync(resolve(entry.fixtureDirectory, file))).digest('hex') })) }));
writeFileSync(resolve(root, 'case-integrity.json'), JSON.stringify({ differences, hashes, inputCriteriaSeparated: true }, null, 2) + '\n');
