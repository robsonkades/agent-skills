import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.*;
public final class CoreStreamControls {
    static void check(boolean condition, String label) { if (!condition) throw new AssertionError(label); }
    public static void main(String[] args) {
        AtomicInteger peek = new AtomicInteger(), terminal = new AtomicInteger();
        long count = List.of(1, 2, 3).stream().peek(x -> peek.incrementAndGet()).count();
        check(count == 3 && peek.get() == 0, "observed sized count elision on this build");
        List.of(1, 2, 3).stream().forEach(x -> terminal.incrementAndGet());
        check(terminal.get() == 3, "required terminal actions");
        try { Stream.of("a", "a").collect(Collectors.toMap(x -> x, x -> 1)); throw new AssertionError("duplicate accepted"); }
        catch (IllegalStateException expected) {}
        var removedThenInserted = Stream.of(1, 2, 3).collect(Collectors.toMap(x -> "k", x -> x, (a, b) -> null));
        check(removedThenInserted.equals(Map.of("k", 3)), "null merge removes, third reinserts");
        AtomicInteger outerClosed = new AtomicInteger(), opened = new AtomicInteger(), innerClosed = new AtomicInteger();
        try (Stream<Integer> outer = Stream.of(1, 2).onClose(outerClosed::incrementAndGet)) {
            var value = outer.flatMap(i -> { opened.incrementAndGet(); return Stream.of(i).onClose(innerClosed::incrementAndGet); }).findFirst();
            check(value.orElseThrow() == 1 && opened.get() == 1 && innerClosed.get() == 1, "mapped closure and lazy opening");
            check(outerClosed.get() == 0, "terminal does not close outer");
        }
        check(outerClosed.get() == 1, "explicit outer owner closes");
        AtomicInteger failureInner = new AtomicInteger(), failureOuter = new AtomicInteger();
        try (Stream<Integer> outer = Stream.of(1).onClose(failureOuter::incrementAndGet)) {
            outer.flatMap(i -> Stream.of(i).onClose(failureInner::incrementAndGet)).forEach(i -> { throw new IllegalArgumentException("fixture"); });
        } catch (IllegalArgumentException expected) { check(expected.getMessage().equals("fixture"), "expected failure"); }
        check(failureInner.get() == 1 && failureOuter.get() == 1, "close under terminal failure");
        System.out.println("PASS sized-count elision observed; forEach=3; duplicate and null-merge controls; inner/outer closure and failure cleanup");
    }
}
