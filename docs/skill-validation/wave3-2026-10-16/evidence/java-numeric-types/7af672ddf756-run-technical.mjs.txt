const results=[];
function run(name,exe,args) {
 const r=spawnSync(path.join(jdk,exe+'.exe'),args,{cwd:root,encoding:'utf8',timeout:30000});
 results.push({name,executable:path.join(jdk,exe+'.exe'),args,cwd:root,status:r.status,stdout:r.stdout,stderr:r.stderr,error:r.error?.message??null});
 fs.writeFileSync(path.join(root,'results.json'),JSON.stringify(results,null,2)+'\n');
 assert.equal(r.status,0,name+': '+r.stderr);
 console.log(name+': PASS');
 if(name==='probe')console.log(r.stdout);
}
