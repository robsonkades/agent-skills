import java.lang.foreign.Arena;
import java.lang.foreign.MemorySegment;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.Future;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.concurrent.TimeUnit;

public final class WorkerBoundary {
    public static int execute(NativeFixture nativeFixture) throws Exception {
        try (Arena requestArena = RequestArenaFactory.open()) {
            MemorySegment payload = requestArena.allocate(1);
            ThreadPoolExecutor worker = new ThreadPoolExecutor(
                    1, 1, 0L, TimeUnit.MILLISECONDS,
                    new ArrayBlockingQueue<>(1),
                    new ThreadPoolExecutor.AbortPolicy());
            try {
                Future<Integer> pending = worker.submit(() -> nativeFixture.call(payload));
                return pending.get();
            } finally {
                // All fixture callbacks finish immediately. Preserve arena lifetime
                // through actual completion, including an interrupted caller.
                worker.shutdown();
                boolean interrupted = false;
                while (!worker.isTerminated()) {
                    try {
                        worker.awaitTermination(1, TimeUnit.SECONDS);
                    } catch (InterruptedException failure) {
                        interrupted = true;
                    }
                }
                if (interrupted) Thread.currentThread().interrupt();
            }
        }
    }
}
