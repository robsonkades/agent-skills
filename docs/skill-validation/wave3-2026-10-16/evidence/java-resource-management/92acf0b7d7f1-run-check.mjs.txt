import { writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';

const root = dirname(fileURLToPath(import.meta.url));
const javaBin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const commands = [];
for (const [name, executable, args] of [
  ['compile', join(javaBin, 'javac.exe'), ['--release', '21', '-d', '.', 'SuppressionChecks.java']],
  ['run', join(javaBin, 'java.exe'), ['-cp', '.', 'SuppressionChecks']]
]) {
  const result = spawnSync(executable, args, { cwd: root, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = { name, executable, args, workingDirectory: root, exitCode: result.status,
    signal: result.signal, error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  commands.push(record);
  writeFileSync(join(root, name + '.log'), JSON.stringify(record, null, 2) + '\n');
  if (result.status !== 0 || result.error || result.signal) throw new Error(name + ' failed; see log');
}
const report = { status: 'PASS', target: 'Java 21 (--release 21); no preview',
  runtime: 'Temurin 25.0.3+9-LTS, not a Java 21 runtime test',
  scenarios: 9, commands, limitations: ['test doubles, no driver/pool', 'no concurrency or durability proof', 'not an agent evaluation'] };
writeFileSync(join(root, 'result.json'), JSON.stringify(report, null, 2) + '\n');
console.log(JSON.stringify(report, null, 2));
