import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Stream;

public final class SuppressionChecks {
    static final class Failure extends RuntimeException {
        Failure(String message, boolean suppression) {
            super(message, null, suppression, true);
        }
    }
    static final class Resource implements AutoCloseable {
        final String name;
        final List<String> attempts;
        final Failure failure;
        Resource(String name, List<String> attempts, Failure failure) {
            this.name = name;
            this.attempts = attempts;
            this.failure = failure;
        }
        @Override public void close() {
            attempts.add(name);
            if (failure != null) throw failure;
        }
    }
    static void require(boolean ok, String message) {
        if (!ok) throw new AssertionError(message);
    }
    static Throwable twr(Failure primary, Failure closing, Failure earlier, List<String> attempts) {
        try (var first = new Resource("first", attempts, earlier);
             var second = new Resource("second", attempts, closing)) {
            throw primary;
        } catch (Throwable thrown) {
            return thrown;
        }
    }
    static void matrix(boolean suppression, boolean sameInstance) {
        var primary = new Failure("operation", suppression);
        var closing = sameInstance ? primary : new Failure("operation", true);
        var earlier = new Failure("earlier-close", true);
        var attempts = new ArrayList<String>();
        Throwable thrown = twr(primary, closing, earlier, attempts);
        require(attempts.equals(List.of("second", "first")), "not all close attempts in reverse order");
        if (sameInstance) {
            require(thrown instanceof IllegalArgumentException, "expected self-suppression exception");
            require(thrown.getCause() == primary, "runtime should retain original as cause");
            require(Arrays.equals(thrown.getSuppressed(), new Throwable[] {earlier}), "earlier close failure not attached");
        } else {
            require(thrown == primary, "primary identity changed");
            Throwable[] expected = suppression ? new Throwable[] {closing, earlier} : new Throwable[0];
            require(Arrays.equals(thrown.getSuppressed(), expected), "incorrect suppressed failures");
        }
        System.out.println("PASS matrix suppression=" + suppression + " sameInstance=" + sameInstance
            + " escaped=" + thrown.getClass().getSimpleName() + " attempts=" + attempts
            + " suppressed=" + thrown.getSuppressed().length);
    }
    static void ordinaryCloseOnly() {
        var failure = new Failure("close-only", true);
        var attempts = new ArrayList<String>();
        Throwable actual = null;
        try (var resource = new Resource("resource", attempts, failure)) {
            // body succeeds
        } catch (Throwable thrown) { actual = thrown; }
        require(actual == failure, "close-only failure lost");
        require(attempts.equals(List.of("resource")), "close-only attempt missing");
        System.out.println("PASS successful body propagates close failure");
    }
    static void partialAcquisition() {
        var failure = new Failure("acquisition", true);
        var attempts = new ArrayList<String>();
        Throwable actual = null;
        try (var first = new Resource("first", attempts, null);
             var second = failAcquisition(failure)) {
            throw new AssertionError("body should not run");
        } catch (Throwable thrown) { actual = thrown; }
        require(actual == failure, "acquisition failure changed");
        require(attempts.equals(List.of("first")), "acquired resource not released");
        System.out.println("PASS partial acquisition closes only acquired resource");
    }
    static Resource failAcquisition(Failure failure) { throw failure; }
    static void guardedManual(boolean sameInstance) {
        var primary = new Failure("operation", true);
        var closing = sameInstance ? primary : new Failure("close", true);
        var attempts = new ArrayList<String>();
        var resource = new Resource("manual", attempts, closing);
        try {
            throw primary;
        } catch (RuntimeException | Error failure) {
            try {
                resource.close();
            } catch (Throwable closeFailure) {
                if (closeFailure != failure) failure.addSuppressed(closeFailure);
            }
            require(failure == primary, "manual original changed");
            require(Arrays.equals(failure.getSuppressed(), sameInstance ? new Throwable[0] : new Throwable[] {closing}),
                "manual suppression incorrect");
        }
        require(attempts.equals(List.of("manual")), "manual release skipped");
        System.out.println("PASS existing factory identity guard sameInstance=" + sameInstance);
    }
    static void streamHandlers() {
        var failure = new Failure("same-handler-failure", true);
        var attempts = new ArrayList<String>();
        var stream = Stream.of(1)
            .onClose(() -> { attempts.add("handler-1"); throw failure; })
            .onClose(() -> { attempts.add("handler-2"); throw failure; });
        Throwable actual = null;
        try { stream.close(); } catch (Throwable thrown) { actual = thrown; }
        require(actual == failure && actual.getSuppressed().length == 0, "stream handler control should guard identity");
        require(attempts.equals(List.of("handler-1", "handler-2")), "stream close handler missing");
        System.out.println("PASS stream handler aggregation has its own identity guard");
    }
    public static void main(String[] args) {
        matrix(true, false);
        matrix(true, true);
        matrix(false, false);
        matrix(false, true);
        ordinaryCloseOnly();
        partialAcquisition();
        guardedManual(false);
        guardedManual(true);
        streamHandlers();
        System.out.println("PASS: 9 technical scenarios; Java 21 target, actual runtime " + Runtime.version());
    }
}
