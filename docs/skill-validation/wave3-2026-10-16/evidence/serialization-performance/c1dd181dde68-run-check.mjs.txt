import {mkdirSync, writeFileSync} from 'node:fs';
import {spawnSync} from 'node:child_process';
import {dirname, join} from 'node:path';
import {fileURLToPath} from 'node:url';
const root=dirname(fileURLToPath(import.meta.url));
const classes=join(root,'classes');
mkdirSync(classes,{recursive:true});
const jdk='C:/Users/robso/.jdks/temurin-25.0.3/bin';
const commands=[
 {name:'toolchain',exe:join(jdk,'java.exe'),args:['-version'],expectedExit:0},
 {name:'compile',exe:join(jdk,'javac.exe'),args:['--release','17','-d',classes,join(root,'CompressionReview.java'),join(root,'FixtureChecks.java')],expectedExit:0},
 {name:'controls',exe:join(jdk,'java.exe'),args:['-cp',classes,'FixtureChecks'],expectedExit:0},
 {name:'stream-case',exe:join(jdk,'java.exe'),args:['-cp',classes,'CompressionReview',join(root,'stream.properties')],expectedExit:0},
 {name:'frame-case',exe:join(jdk,'java.exe'),args:['-cp',classes,'CompressionReview',join(root,'frame.properties')],expectedExit:1}
];
const results=[];
for (const command of commands) {
 const result=spawnSync(command.exe,command.args,{cwd:root,encoding:'utf8',timeout:20000,windowsHide:true});
 const output={...command,cwd:root,exitCode:result.status,signal:result.signal,stdout:result.stdout,stderr:result.stderr,error:result.error?.message};
 writeFileSync(join(root,command.name+'.log'),JSON.stringify(output,null,2)+'\n');
 results.push(output);
 if(result.status!==command.expectedExit) throw new Error('Unexpected result: '+command.name);
 if(command.name==='controls' && result.stdout.split('\n').filter(line=>line.startsWith('PASS ')).length!==4) throw new Error('Missing controls');
 if(command.name==='frame-case' && !result.stderr.includes('EOFException')) throw new Error('Unexpected failure mechanism');
}
writeFileSync(join(root,'result.json'),JSON.stringify({kind:'technical fixture validation, not agent or performance evaluation',compilerRelease:17,results},null,2)+'\n');
console.log(JSON.stringify({status:'PASS',controls:4,streamCaseExit:0,frameCaseExit:1,frameFailure:'Expected EOFException',runtime:'Temurin 25.0.3+9-LTS',compilerRelease:17,result:join(root,'result.json')},null,2));
