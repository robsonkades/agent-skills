import java.io.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.*;
import java.util.zip.*;

public class CompressionReview {
    record Transmission(List<byte[]> chunks, byte[] tail) {}

    static final class CandidateEncoder implements AutoCloseable {
        private final ByteArrayOutputStream sink = new ByteArrayOutputStream();
        private final GZIPOutputStream gzip = new GZIPOutputStream(sink, true);

        CandidateEncoder() throws IOException {}

        byte[] encode(String record) throws IOException {
            gzip.write((record + "\n").getBytes(StandardCharsets.UTF_8));
            gzip.flush();
            return drain();
        }

        byte[] finish() throws IOException {
            gzip.finish();
            return drain();
        }

        private byte[] drain() {
            byte[] bytes = sink.toByteArray();
            sink.reset();
            return bytes;
        }

        public void close() throws IOException {
            gzip.close();
        }
    }

    interface Receiver {
        List<String> receive(Transmission transmission) throws IOException;
    }

    static Receiver receiver(Properties properties) {
        return switch (properties.getProperty("receiver.mode")) {
            case "stream" -> new StreamReceiver();
            case "frame" -> new FrameReceiver();
            default -> throw new IllegalArgumentException("Unknown receiver.mode");
        };
    }

    static final class StreamReceiver implements Receiver {
        public List<String> receive(Transmission transmission) throws IOException {
            ByteArrayOutputStream stream = new ByteArrayOutputStream();
            for (byte[] chunk : transmission.chunks()) stream.write(chunk);
            stream.write(transmission.tail());
            return records(decompress(stream.toByteArray()));
        }
    }

    static final class FrameReceiver implements Receiver {
        public List<String> receive(Transmission transmission) throws IOException {
            List<String> result = new ArrayList<>();
            for (byte[] frame : transmission.chunks()) {
                List<String> decoded = records(decompress(frame));
                if (decoded.size() != 1) throw new IOException("One record per frame required");
                result.addAll(decoded);
            }
            if (transmission.tail().length != 0) throw new IOException("Unexpected out-of-frame tail");
            return result;
        }
    }

    static String decompress(byte[] bytes) throws IOException {
        try (GZIPInputStream gzip = new GZIPInputStream(new ByteArrayInputStream(bytes))) {
            return new String(gzip.readAllBytes(), StandardCharsets.UTF_8);
        }
    }

    static List<String> records(String text) throws IOException {
        if (!text.endsWith("\n")) throw new IOException("Incomplete record");
        return text.lines().toList();
    }

    static Transmission encode(List<String> records) throws IOException {
        try (CandidateEncoder encoder = new CandidateEncoder()) {
            List<byte[]> chunks = new ArrayList<>();
            for (String record : records) chunks.add(encoder.encode(record));
            return new Transmission(List.copyOf(chunks), encoder.finish());
        }
    }

    public static void main(String[] args) throws IOException {
        Properties properties = new Properties();
        try (Reader reader = Files.newBufferedReader(Path.of(args[0]))) {
            properties.load(reader);
        }
        List<String> expected = List.of("order-17:USD:1200", "order-18:EUR:2500");
        List<String> actual = receiver(properties).receive(encode(expected));
        if (!actual.equals(expected)) throw new AssertionError("Wrong record count, order or values");
        System.out.println(actual);
    }
}
