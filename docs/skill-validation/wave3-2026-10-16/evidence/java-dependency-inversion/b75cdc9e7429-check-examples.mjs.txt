function put(path, content) {
  const full = join(root, path);
  mkdirSync(dirname(full), { recursive: true });
  writeFileSync(full, content);
  return full;
}
function run(name, tool, args, expected = 0, pattern) {
  const result = spawnSync(join(jdk, tool + '.exe'), args, { cwd: root, env: childEnv, encoding: 'utf8', timeout: 30000 });
  const record = { name, command: join(jdk, tool + '.exe'), args, cwd: root, exitCode: result.status, stdout: result.stdout, stderr: result.stderr, error: result.error?.message };
  runs.push(record);
  writeFileSync(join(root, name + '.log'), JSON.stringify(record, null, 2) + '\n');
  assert.equal(result.status, expected, name + ': ' + result.stderr);
  if (pattern) assert.match(result.stdout + result.stderr, pattern, name);
  return result.stdout;
}
function compile(name, output, files, classpath, expected = 0, pattern) {
  const args = ['-J-Xms16m', '-J-Xmx128m', '--release', '17', '-d', join(root, output)];
  if (classpath) args.push('-classpath', classpath);
  return run(name, 'javac', [...args, ...files], expected, pattern);
}
function execute(name, paths, main, expected = 0, pattern) {
  return run(name, 'java', ['-Xms16m', '-Xmx64m', '-cp', paths.map(p => join(root, p)).join(delimiter), main], expected, pattern);
}
function sdk(version, breaking = false) {
  const message = put(`src/sdk-${version}/shop/smtp/SmtpMessage.java`, 'package shop.smtp; public record SmtpMessage(String recipient, String subject, String body) {}\n');
  const body = breaking ? 'public void send(String message) { System.out.println("breaking"); }' : `public void send(SmtpMessage message) { System.out.println("${version}|" + message.recipient() + "|" + message.subject() + "|" + message.body()); }`;
  const client = put(`src/sdk-${version}/shop/smtp/SmtpClient.java`, `package shop.smtp; public final class SmtpClient { public SmtpClient(String host, int port) {} ${body} }\n`);
  compile('compile-sdk-' + version, 'sdk-' + version, [message, client]);
}
