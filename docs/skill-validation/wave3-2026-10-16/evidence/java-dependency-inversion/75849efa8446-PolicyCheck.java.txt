package shop.orders;
import java.math.BigDecimal;
public final class PolicyCheck {
 public static void main(String[] args) {
  Order order = new Order("A-1", new BigDecimal("12.50"), "customer@example.test");
  RecordingSender recording = new RecordingSender();
  new OrderConfirmer(recording).confirm(order);
  if (!recording.sent.equals(java.util.List.of(new Confirmation("customer@example.test", "A-1", "Order A-1 confirmed, total 12.50")))) throw new AssertionError("policy outcome");
  RuntimeException expected = new RuntimeException("synthetic transport failure");
  int[] calls = {0};
  try { new OrderConfirmer(c -> { calls[0]++; throw expected; }).confirm(order); throw new AssertionError("unexpected success"); }
  catch (RuntimeException failure) { if (failure != expected) throw new AssertionError("failure changed", failure); }
  if (calls[0] != 1) throw new AssertionError("policy retried");
  System.out.println("PASS: recording outcome and synchronous failure without retry");
 }
}
