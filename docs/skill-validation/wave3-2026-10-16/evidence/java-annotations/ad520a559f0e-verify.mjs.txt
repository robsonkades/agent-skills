import assert from 'node:assert/strict';
import {spawnSync} from 'node:child_process';
import {readFileSync,writeFileSync,mkdirSync,mkdtempSync,readdirSync} from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import {fileURLToPath} from 'node:url';
const root=path.dirname(fileURLToPath(import.meta.url));
const config=JSON.parse(readFileSync(path.join(root,'project.json'),'utf8'));
const jdk=process.env.REVIEW_JAVA_BIN || 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const executable=name=>path.join(jdk,name+(process.platform==='win32'?'.exe':''));
const work=mkdtempSync(path.join(os.tmpdir(),'annotation-reader-case-'));
const output=name=>{const p=path.join(work,name);mkdirSync(p);return p};
const api=output('api'), app=output('app');
const sources=group=>readdirSync(path.join(root,'src',group,'example')).filter(f=>f.endsWith('.java')).map(f=>path.join(root,'src',group,'example',f));
const runs=[];
function run(label,exe,args,expected=0,text) {
 const r=spawnSync(executable(exe),args,{cwd:root,encoding:'utf8',timeout:30000});
 const record={label,executable:executable(exe),args,cwd:root,status:r.status,stdout:r.stdout,stderr:r.stderr,error:r.error?.message??null};
 runs.push(record);
 writeFileSync(path.join(work,'results.json'),JSON.stringify(runs,null,2)+'\n');
 assert.equal(r.status,expected,label+': '+r.stderr);
 if(text) assert.ok((r.stdout+r.stderr).includes(text),label+': missing '+text);
 console.log(label+': PASS');
}
console.log('Evidence: '+work);
assert.equal(config.release,17);
assert.ok(['processor','runtime'].includes(config.reader));
run('api','javac',['--release',String(config.release),'-proc:none','-d',api,...sources('api')]);
if(config.reader==='processor') {
 const processors=output('processors');
 run('processor','javac',['--release',String(config.release),'-proc:none','-cp',api,'-d',processors,path.join(root,'src/processors/probe/CodecProcessor.java')]);
 const compile=(name,extra=[])=>['--release',String(config.release),'-cp',api,'-processorpath',processors+path.delimiter+api,'-processor','probe.CodecProcessor','-d',output(name),...sources('app'),...extra];
 run('accepted','javac',compile('accepted'));
 run('rejected-value','javac',compile('rejected-value',sources('invalid-value')),1,'CODEC_REQUIRED');
 run('rejected-array','javac',compile('rejected-array',sources('invalid-array')),1,'CODEC_REQUIRED');
} else {
 run('application','javac',['--release',String(config.release),'-proc:none','-cp',api,'-d',app,...sources('app'),...sources('runtime'),...sources('invalid-value'),...sources('invalid-array')]);
 const launch=names=>['-cp',api+path.delimiter+app,'example.RuntimeReader',...names];
 run('accepted','java',launch(['example.ExplicitUse','example.DefaultUse']));
 run('rejected-value','java',launch(['example.InvalidUse']),1,'CODEC_REQUIRED');
 run('rejected-array','java',launch(['example.InvalidArrayUse']),1,'CODEC_REQUIRED');
}
console.log('Selected reader checks passed: '+config.reader);
