package probe;
import example.*;
import java.util.Set;
import javax.annotation.processing.*;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.*;

@SupportedAnnotationTypes("example.UseCodec")
@SupportedSourceVersion(SourceVersion.RELEASE_17)
public final class BadArrayProcessor extends AbstractProcessor {
    @Override public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment round) {
        for (Element element : round.getElementsAnnotatedWith(UseCodec.class)) {
            Class<?>[] ignored = element.getAnnotation(UseCodec.class).alternatives();
        }
        return false;
    }
}
