import assert from 'node:assert/strict';
import {spawnSync} from 'node:child_process';
import {readFileSync,writeFileSync,mkdirSync,existsSync,readdirSync} from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
const root=path.dirname(fileURLToPath(import.meta.url));
const jdk='C:/Users/robso/.jdks/temurin-25.0.3/bin';
const javac=path.join(jdk,'javac.exe'), java=path.join(jdk,'java.exe');
const results=[];
const output=n=>{const p=path.join(root,'out',n);mkdirSync(p,{recursive:true});return p};
const api=output('api'), processors=output('processors');
const sources=d=>readdirSync(path.join(root,'src',d,'example')).filter(f=>f.endsWith('.java')).map(f=>path.join(root,'src',d,'example',f));
const processorSources=readdirSync(path.join(root,'src/processors/probe')).map(f=>path.join(root,'src/processors/probe',f));
mkdirSync(path.join(root,'logs'),{recursive:true});
function run(name,exe,args,expectedStatus=0,expectedText=[]) {
  const r=spawnSync(exe,args,{cwd:root,encoding:'utf8',timeout:30000});
  const stdout=r.stdout??'',stderr=r.stderr??'';
  writeFileSync(path.join(root,'logs',name+'.stdout.txt'),stdout);
  writeFileSync(path.join(root,'logs',name+'.stderr.txt'),stderr);
  const result={name,executable:exe,args,cwd:root,status:r.status,signal:r.signal,error:r.error?.message??null,expectedStatus,expectedText,stdout,stderr};
  results.push(result);writeFileSync(path.join(root,'results.json'),JSON.stringify(results,null,2)+'\n');
  assert.equal(r.status,expectedStatus,name+': '+stderr);
  for(const text of expectedText) assert.ok((stdout+stderr).includes(text),name+': missing '+text);
  console.log(name+': PASS (exit '+r.status+')');
}
run('java-version',java,['-version']);
run('javac-version',javac,['-version']);
run('compile-api',javac,['--release','17','-proc:none','-d',api,...sources('api')]);
run('compile-processors',javac,['--release','17','-proc:none','-cp',api,'-d',processors,...processorSources]);
assert.equal(existsSync(path.join(api,'example/LocalCodec.class')),false);
assert.equal(existsSync(path.join(processors,'example/LocalCodec.class')),false);
writeFileSync(path.join(root,'loader-boundary.json'),JSON.stringify({appClassOnApiPath:false,appClassOnProcessorPath:false,processorPath:[processors,api],applicationInput:'src/app/example/LocalCodec.java'},null,2)+'\n');

const explicitApp=sources('app').filter(p=>!p.endsWith('DefaultUse.java'));

run('mirror-negative-value',javac,processArgs('MirrorProcessor','mirror-negative-value',sources('invalid-value')),1,['CODEC_REQUIRED: java.lang.String in value']);
run('mirror-negative-array',javac,processArgs('MirrorProcessor','mirror-negative-array',sources('invalid-array')),1,['CODEC_REQUIRED: java.lang.String in alternatives']);

 'CAUGHT_OK: example.LocalCodec in value','CAUGHT_OK: example.DefaultCodec in value','CAUGHT_OK: example.LocalCodec in alternatives'
