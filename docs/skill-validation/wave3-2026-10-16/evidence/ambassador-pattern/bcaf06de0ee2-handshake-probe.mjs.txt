import assert from 'node:assert/strict';
import net from 'node:net';
import { once } from 'node:events';
import { writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

// A synthetic byte protocol, not Envoy, MySQL, TLS or production routing.
// Each case owns loopback listeners, sockets, and a bounded caller guard.
const artifactDir = dirname(fileURLToPath(import.meta.url));
const results = [];
const cases = [
  { name: 'server-first-deferred', firstSpeaker: 'server', defer: true, expected: 'timeout', expectedUpstreamConnections: 0 },
  { name: 'server-first-immediate', firstSpeaker: 'server', defer: false, expected: 'complete', expectedUpstreamConnections: 1 },
  { name: 'client-first-deferred', firstSpeaker: 'client', defer: true, expected: 'complete', expectedUpstreamConnections: 1 }
];

async function runCase(config) {
  const sockets = new Set();
  const events = [];
  let upstreamConnections = 0;
  let proxyConnections = 0;
  let firstClientBytes = 0;
  const track = socket => {
    sockets.add(socket);
    socket.on('error', error => events.push({ event: 'socket-error', code: error.code }));
    socket.once('close', () => sockets.delete(socket));
    return socket;
  };
  const upstream = net.createServer(socket => {
    track(socket);
    upstreamConnections++;
    events.push({ event: 'upstream-accepted' });
    if (config.firstSpeaker === 'server') {
      events.push({ event: 'upstream-sent-greeting' });
      socket.write('HELLO\n');
    }
    let received = '';
    socket.on('data', chunk => {
      received += chunk.toString();
      if (received.includes('PING\n')) {
        events.push({ event: 'upstream-received-ping' });
        socket.end('PONG\n');
      }
    });
  });
  upstream.listen(0, '127.0.0.1');
  await once(upstream, 'listening');
  const upstreamPort = upstream.address().port;
  const proxy = net.createServer(downstream => {
    track(downstream);
    proxyConnections++;
    events.push({ event: 'proxy-accepted-client' });
    const openUpstream = firstChunk => {
      events.push({ event: 'proxy-started-upstream-connect' });
      const peer = track(net.createConnection({ host: '127.0.0.1', port: upstreamPort }));
      peer.pipe(downstream);
      peer.once('connect', () => {
        if (firstChunk) peer.write(firstChunk);
        downstream.pipe(peer);
      });
      downstream.once('close', () => peer.destroy());
      peer.once('error', () => downstream.destroy());
    };
    if (config.defer) {
      downstream.once('data', chunk => {
        firstClientBytes = chunk.length;
        events.push({ event: 'proxy-received-first-client-bytes', bytes: chunk.length });
        downstream.pause();
        openUpstream(chunk);
      });
    } else openUpstream();
  });
  proxy.listen(0, '127.0.0.1');
  await once(proxy, 'listening');
  const proxyPort = proxy.address().port;
  let callerTimer;
  let outcome;
  try {
    outcome = await new Promise(resolveOutcome => {
      let done = false;
      const finish = value => {
        if (done) return;
        done = true;
        clearTimeout(callerTimer);
        resolveOutcome(value);
      };
      const client = track(net.createConnection({ host: '127.0.0.1', port: proxyPort }));
      let received = '';
      let pingSent = false;
      const ping = () => {
        pingSent = true;
        events.push({ event: 'client-sent-ping' });
        client.write('PING\n');
      };
      client.once('connect', () => {
        events.push({ event: 'client-tcp-connected' });
        if (config.firstSpeaker === 'client') ping();
      });
      client.on('data', chunk => {
        received += chunk.toString();
        if (!pingSent && received.includes('HELLO\n')) ping();
        if (received.includes('PONG\n')) finish('complete');
      });
      client.once('error', error => finish(`error:${error.code}`));
      // Timeout is an execution guard, not a measured production latency claim.
      callerTimer = setTimeout(() => finish('timeout'), config.expected === 'timeout' ? 200 : 3000);
    });
  } finally {
    clearTimeout(callerTimer);
    const closes = [...sockets].map(socket => new Promise(resolveClose => {
      socket.once('close', resolveClose);
      socket.destroy();
    }));
    await Promise.all(closes);
    await Promise.all([upstream, proxy].map(server => new Promise((resolveClose, rejectClose) => {
      server.close(error => error ? rejectClose(error) : resolveClose());
    })));
  }
  return { ...config, outcome, proxyConnections, upstreamConnections, firstClientBytes,
    socketsRemaining: sockets.size, listenersClosed: !upstream.listening && !proxy.listening, events };
}

let assertions = 0;
function equal(actual, expected, label) {
  assertions++;
  assert.equal(actual, expected, label);
}
for (const config of cases) {
  const result = await runCase(config);
  results.push(result);
  writeFileSync(resolve(artifactDir, 'handshake-results.json'), JSON.stringify({ runtime: process.version, results }, null, 2));
  equal(result.outcome, config.expected, `${config.name}: outcome`);
  equal(result.upstreamConnections, config.expectedUpstreamConnections, `${config.name}: upstream count`);
  equal(result.proxyConnections, 1, `${config.name}: local listener accepted`);
  equal(result.events.some(event => event.event === 'client-tcp-connected'), true, `${config.name}: TCP connected`);
  equal(result.socketsRemaining, 0, `${config.name}: socket cleanup`);
  equal(result.listenersClosed, true, `${config.name}: listeners closed`);
  if (config.name === 'server-first-deferred') {
    equal(result.firstClientBytes, 0, 'No client data available to trigger upstream connection');
    equal(result.events.some(event => event.event === 'proxy-started-upstream-connect'), false, 'No upstream attempt before guard expiry');
  }
}
const summary = { status: 'PASS', runtime: process.version, assertions, cases: results.length,
  observations: results.map(({ name, outcome, upstreamConnections, socketsRemaining, listenersClosed }) =>
    ({ name, outcome, upstreamConnections, socketsRemaining, listenersClosed })) };
writeFileSync(resolve(artifactDir, 'handshake-summary.json'), JSON.stringify(summary, null, 2));
console.log(JSON.stringify(summary, null, 2));
