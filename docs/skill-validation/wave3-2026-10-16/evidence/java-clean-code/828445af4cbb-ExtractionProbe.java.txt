import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

public final class ExtractionProbe {
  static String original(char[] label) {
    return new StringBuilder().append(label).toString();
  }

  static String widened(char[] label) {
    return renderObject(label);
  }

  static String renderObject(Object value) {
    return new StringBuilder().append(value).toString();
  }

  static String renderChars(char[] value) {
    return new StringBuilder().append(value).toString();
  }

  static String originalObject(Object value) {
    return new StringBuilder().append(value).toString();
  }

  record Receipt(String id) {}

  static final class Audit {
    final List<String> events = new ArrayList<>();
    void record(Receipt receipt) { events.add("receipt:" + receipt.id()); }
    void record(Object value) { events.add("generic:" + value); }
  }

  static void originalAudit(Audit audit, Receipt receipt) {
    audit.record(receipt);
  }

  static void originalGenericAudit(Audit audit, Object receipt) {
    audit.record(receipt);
  }

  static void extractedAudit(Audit audit, Object receipt) {
    audit.record(receipt);
  }

  static void extractedTypedAudit(Audit audit, Receipt receipt) {
    audit.record(receipt);
  }

  static void equal(Object expected, Object actual) {
    if (!Objects.equals(expected, actual)) {
      throw new AssertionError(expected + " != " + actual);
    }
  }

  public static void main(String[] args) {
    System.out.println("java=" + System.getProperty("java.runtime.version"));
    for (char[] label : new char[][] {{'O', 'K'}, {}}) {
      equal(new String(label), original(label));
      equal(original(label), renderChars(label));
      equal(label.toString(), widened(label));
      equal(false, original(label).equals(widened(label)));
      equal(originalObject(label), renderObject(label));
      System.out.println("chars=" + label.length + " original=" + original(label)
          + " widened=" + widened(label) + " typed=" + renderChars(label)
          + " originalObjectPreserved=" + originalObject(label).equals(renderObject(label)));
    }
    equal("OK", originalObject("OK"));
    equal(originalObject("OK"), renderObject("OK"));
    Receipt receipt = new Receipt("r-7");
    Audit original = new Audit();
    Audit widened = new Audit();
    Audit typed = new Audit();
    Audit originalGeneric = new Audit();
    originalAudit(original, receipt);
    extractedAudit(widened, receipt);
    extractedTypedAudit(typed, receipt);
    originalGenericAudit(originalGeneric, receipt);
    equal(List.of("receipt:r-7"), original.events);
    equal(List.of("generic:" + receipt), widened.events);
    equal(original.events, typed.events);
    equal(originalGeneric.events, widened.events);
    System.out.println("typedOriginal=" + original.events + " widened=" + widened.events
        + " typedHelper=" + typed.events + " objectOriginal=" + originalGeneric.events);
    System.out.println("PASS: widened helper changes typed overload; typed helper preserves it; original Object contract stays generic.");
  }
}
