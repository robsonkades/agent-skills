import java.util.*;
import java.time.*;
import java.nio.*;
public final class ImmutabilityProbe {
    record RawSignature(byte[] bytes) {}
    record CopyOnlySignature(byte[] bytes) {
        CopyOnlySignature { bytes = bytes.clone(); }
        public byte[] bytes() { return bytes.clone(); }
    }
    record Shallow(List<byte[]> values) { Shallow { values = List.copyOf(values); } }
    static void check(boolean condition, String label) { if (!condition) throw new AssertionError(label); }
    static void unsupported(Runnable action) {
        try { action.run(); throw new AssertionError("mutation accepted"); }
        catch (UnsupportedOperationException expected) {}
    }
    public static void main(String[] args) {
        byte[] original = {1, 2, 3};
        Signature a = new Signature(original);
        Signature b = new Signature(new byte[]{1, 2, 3});
        int stableHash = a.hashCode();
        Set<Signature> keys = new HashSet<>(); keys.add(a);
        original[0] = 9;
        byte[] exposed = a.bytes(); exposed[1] = 9;
        check(Arrays.equals(a.bytes(), new byte[]{1, 2, 3}), "defensive directions");
        check(a.equals(b) && b.equals(a) && a.hashCode() == b.hashCode(), "content equality");
        check(a.equals(new Signature(a.bytes())), "record reconstruction");
        check(a.hashCode() == stableHash && keys.contains(b), "stable map key");
        check(a.toString().equals(b.toString()) && a.toString().contains("<redacted>"), "diagnostic contract");
        byte[] shared = {1}; RawSignature raw = new RawSignature(shared); shared[0] = 9;
        check(raw.bytes()[0] == 9 && !raw.equals(new RawSignature(new byte[]{9})), "raw negative control");
        CopyOnlySignature copied = new CopyOnlySignature(new byte[]{1});
        check(!copied.equals(new CopyOnlySignature(copied.bytes())), "copy-only reconstruction negative");
        byte[] element = {1}; Shallow shallow = new Shallow(new ArrayList<>(List.of(element))); element[0] = 9;
        check(shallow.values().get(0)[0] == 9, "shallow element negative control");
        List<String> mutable = new ArrayList<>(List.of("12A"));
        BeforeReservation before = new BeforeReservation("r", mutable);
        Reservation reservation = new Reservation("r", mutable);
        mutable.add("14C");
        check(before.seatIds().size() == 2 && reservation.seatIds().equals(List.of("12A")), "reservation inbound copy");
        unsupported(() -> reservation.seatIds().add("15D"));
        check(!reservation.equals(new Reservation("r", List.of("12A"))), "identity contract");
        List<OrderLine> lines = new ArrayList<>(List.of(new OrderLine("A", 2)));
        Order order = new Order("o", lines, Instant.EPOCH); lines.clear();
        check(order.lines().size() == 1, "order snapshot");
        unsupported(() -> order.lines().add(new OrderLine("B", 3)));
        List<OrderLine> replacement = new ArrayList<>(List.of(new OrderLine("B", 3)));
        Order changed = order.withLines(replacement); replacement.clear();
        check(changed.lines().size() == 1 && order.lines().get(0).sku().equals("A"), "wither copies and preserves original");
        byte[] bufferBytes = {1, 2, 3, 4};
        ByteBuffer buffer = ByteBuffer.wrap(bufferBytes).order(ByteOrder.LITTLE_ENDIAN);
        ByteBuffer readOnly = buffer.asReadOnlyBuffer();
        check(readOnly.order() == ByteOrder.BIG_ENDIAN, "read-only default order");
        check(buffer.duplicate().order() == ByteOrder.BIG_ENDIAN && buffer.slice().order() == ByteOrder.BIG_ENDIAN, "view default order");
        bufferBytes[0] = 9; check(readOnly.get(0) == 9, "shared backing negative control");
        int hashBefore = readOnly.hashCode(); readOnly.get();
        check(readOnly.hashCode() != hashBefore, "cursor changes hash negative control");
        System.out.println("PASS signature inbound/outbound, record reconstruction/hash/text, negative alias and equality controls, reservation identity, order/wither, buffer backing/cursor/order controls");
    }
}
