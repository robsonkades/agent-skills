import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.util.Arrays;

public final class OwnershipProbe {
    private static int checks;
    private static final byte[] PAYLOAD = { 17, 42, 63 };

    private static class TrackingInput extends InputStream {
        private int cursor;
        private final boolean fail;
        boolean closed;
        int closeCalls;
        final IOException failure = new IOException("synthetic read failure");

        TrackingInput(boolean fail) { this.fail = fail; }

        @Override public int read() throws IOException {
            if (closed) throw new IOException("closed");
            if (fail) throw failure;
            return cursor < PAYLOAD.length ? PAYLOAD[cursor++] & 0xff : -1;
        }

        @Override public void close() {
            closeCalls++;
            closed = true;
        }
    }

    private static final class ClosingReadAll extends TrackingInput {
        ClosingReadAll() { super(false); }
        @Override public byte[] readAllBytes() throws IOException {
            try (this) {
                return super.readAllBytes();
            }
        }
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private static void emit(String label, boolean contractSatisfied, int closes, String observation) {
        checks++;
        System.out.printf("{\"case\":\"%s\",\"contractSatisfied\":%s,"
                + "\"closeCalls\":%d,\"observation\":\"%s\"}%n",
                label, contractSatisfied, closes, observation);
    }

    public static void main(String[] args) throws Exception {
        boolean transferred = args.length == 1 && args[0].equals("transferred");
        System.out.printf("{\"runtime\":\"%s\",\"vendor\":\"%s\",\"contract\":\"%s\"}%n",
                System.getProperty("java.runtime.version"), System.getProperty("java.vendor"),
                transferred ? "transferred" : "borrowed");
        PayloadReader reader = new StreamPayloadReader();

        TrackingInput successful = new TrackingInput(false);
        byte[] actual = reader.read(successful);
        require(Arrays.equals(PAYLOAD, actual), "fixture returned wrong bytes");
        require(successful.closed && successful.closeCalls == 1, "fixture did not close once");
        boolean successContract = transferred == successful.closed;
        require(successContract == transferred, "wrong ownership interpretation");
        emit("fixture-success", successContract, successful.closeCalls,
                transferred ? "legitimate-transferred-control" : "borrowed-violation-reproduced");

        TrackingInput failed = new TrackingInput(true);
        try {
            reader.read(failed);
            throw new AssertionError("missing read failure");
        } catch (IOException expected) {
            require(expected == failed.failure, "fixture changed failure identity");
        }
        require(failed.closed && failed.closeCalls == 1, "failure path did not close once");
        boolean failureContract = transferred == failed.closed;
        require(failureContract == transferred, "wrong failure ownership interpretation");
        emit("fixture-read-failure", failureContract, failed.closeCalls,
                transferred ? "legitimate-transferred-control" : "borrowed-violation-reproduced");

        TrackingInput remainsOpen = new TrackingInput(false);
        require(Arrays.equals(PAYLOAD, remainsOpen.readAllBytes()), "control bytes differ");
        require(!remainsOpen.closed && remainsOpen.closeCalls == 0 && remainsOpen.read() == -1,
                "borrowed control closed input");
        emit("borrowed-control-success", true, remainsOpen.closeCalls, "legitimate-borrowed-control");
        remainsOpen.close();

        TrackingInput failureRemainsOpen = new TrackingInput(true);
        try {
            failureRemainsOpen.readAllBytes();
            throw new AssertionError("missing control failure");
        } catch (IOException expected) {
            require(expected == failureRemainsOpen.failure, "control changed failure identity");
        }
        require(!failureRemainsOpen.closed && failureRemainsOpen.closeCalls == 0,
                "borrowed control closed on failure");
        emit("borrowed-control-read-failure", true, failureRemainsOpen.closeCalls,
                "legitimate-borrowed-control");
        failureRemainsOpen.close();

        ByteArrayInputStream maskingFixture = new ByteArrayInputStream(PAYLOAD);
        require(Arrays.equals(PAYLOAD, reader.read(maskingFixture)), "byte-array bytes differ");
        maskingFixture.reset();
        require(maskingFixture.read() == PAYLOAD[0], "byte-array close unexpectedly prevented reuse");
        emit("byte-array-fixture", transferred, -1, "reuse-check-cannot-detect-close");

        ClosingReadAll subtype = new ClosingReadAll();
        InputStream supertype = subtype;
        require(Arrays.equals(PAYLOAD, supertype.readAllBytes()), "subtype bytes differ");
        require(subtype.closed && subtype.closeCalls == 1, "subtype did not close");
        emit("jdk-readAllBytes-override", false, subtype.closeCalls,
                "inherited-no-close-contract-violation-reproduced");

        require(checks == 6, "unexpected check count");
        System.out.printf("{\"checks\":%d,\"result\":\"PASS\"}%n", checks);
    }
}
