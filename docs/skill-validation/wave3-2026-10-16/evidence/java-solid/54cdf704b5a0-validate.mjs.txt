const results = { node: process.version, technical: [], processes: [], structural: null };
function run(label, command, args, cwd = repository) {
  const result = spawnSync(command,args,{cwd,env,windowsHide:true,encoding:'utf8',timeout:30000});
  const stdoutPath = resolve(root,label+'.stdout.log');
  const stderrPath = resolve(root,label+'.stderr.log');
  writeFileSync(stdoutPath,result.stdout ?? '');
  writeFileSync(stderrPath,result.stderr ?? '');
  results.processes.push({label,command,args,cwd,status:result.status,signal:result.signal,
    error:result.error?.message ?? null,stdoutPath,stderrPath});
  writeFileSync(resolve(root,'validation-results.json'),JSON.stringify(results,null,2)+'\n');
  assert.equal(result.status,0,label+': '+result.error+'\n'+result.stderr);
  return result.stdout ?? '';
}
