import java.math.BigDecimal;
public class AccountCheck {
    private static BigDecimal money(String value) { return new BigDecimal(value); }
    private static void equal(BigDecimal actual, String expected) {
        if (actual.compareTo(money(expected)) != 0) throw new AssertionError(actual + " != " + expected);
    }
    private static void throwsType(Class<? extends Throwable> type, Runnable action) {
        try { action.run(); } catch (Throwable error) {
            if (type.isInstance(error)) return;
            throw new AssertionError("Wrong exception", error);
        }
        throw new AssertionError("Missing " + type.getName());
    }
    public static void main(String[] args) {
        Account account = new Account(money("10.00"), money("20.00"));
        BigDecimal earlier = account.balance();
        var atFloor = account.withdraw(money("30.00"));
        if (!(atFloor instanceof Account.Withdrawn done)) throw new AssertionError("Expected approval at floor");
        equal(done.newBalance(), "-20.00");
        equal(account.balance(), "-20.00");
        equal(earlier, "10.00");
        var refused = account.withdraw(money("0.01"));
        if (!(refused instanceof Account.Refused failure)) throw new AssertionError("Expected refusal below floor");
        equal(failure.shortfall(), "0.01");
        equal(account.balance(), "-20.00");
        for (String amount : new String[] { "0", "-1" }) {
            throwsType(IllegalArgumentException.class, () -> account.withdraw(money(amount)));
            equal(account.balance(), "-20.00");
        }
        throwsType(NullPointerException.class, () -> account.withdraw(null));
        equal(account.balance(), "-20.00");
        throwsType(IllegalArgumentException.class, () -> new Account(money("-20.01"), money("20")));
        throwsType(IllegalArgumentException.class, () -> new Account(money("0"), money("-1")));
        throwsType(NullPointerException.class, () -> new Account(null, money("0")));
        throwsType(NullPointerException.class, () -> new Account(money("0"), null));
        Account noOverdraft = new Account(money("0"), money("0"));
        if (!(noOverdraft.withdraw(money("0.01")) instanceof Account.Refused)) throw new AssertionError("Zero limit bypass");
        equal(noOverdraft.balance(), "0");
        System.out.println("PASS: floor approval, refusal/shortfall, no mutation on invalid/refused input, constructor guards, immutable balance observation");
    }
}
`);

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
public class QueryAliasCheck {
    static final class Item { int quantity; Item(int quantity) { this.quantity = quantity; } }
    public static void main(String[] args) {
        Item first = new Item(1);
        ArrayList<Item> backing = new ArrayList<>();
        backing.add(first);
        List<Item> snapshot = List.copyOf(backing);
        List<Item> view = Collections.unmodifiableList(backing);
        backing.add(new Item(2));
        if (snapshot.size() != 1 || view.size() != 2) throw new AssertionError("Membership contract");
        for (List<Item> result : List.of(snapshot, view)) {
            try { result.clear(); throw new AssertionError("Mutation allowed"); }
            catch (UnsupportedOperationException expected) {}
        }
        first.quantity = 7;
        if (snapshot.get(0).quantity != 7 || view.get(0).quantity != 7) throw new AssertionError("Unexpected deep copy");
        try { List.copyOf(Arrays.asList(first, null)); throw new AssertionError("Null accepted"); }
        catch (NullPointerException expected) {}
        System.out.println("PASS: membership snapshot versus live view; both expose mutable element; mutation rejected and copyOf rejects null");
    }
}
`);
