import assert from 'node:assert/strict';
import { mkdirSync, writeFileSync, readFileSync, readdirSync } from 'node:fs';
import { resolve, join, dirname, delimiter } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';

const root = dirname(fileURLToPath(import.meta.url));
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const logs = join(root, 'logs');
mkdirSync(logs, { recursive: true });
const commands = [];
function run(id, tool, args, expectedExit = 0) {
    const executable = join(jdk, tool + '.exe');
    const result = spawnSync(executable, args, { cwd: root, encoding: 'utf8', timeout: 30000, windowsHide: true });
    const stdout = result.stdout ?? '';
    const stderr = result.stderr ?? '';
    writeFileSync(join(logs, id + '.stdout.log'), stdout);
    writeFileSync(join(logs, id + '.stderr.log'), stderr);
    commands.push({ id, executable, args, cwd: root, expectedExit, actualExit: result.status, error: result.error?.message ?? null, stdout, stderr });
    assert.equal(result.error, undefined, id + ': process error');
    assert.equal(result.status, expectedExit, id + ': unexpected exit');
    return result;
}
function compile(id, output, sources, classpath = []) {
    mkdirSync(output, { recursive: true });
    const args = ['--release', '25', '-d', output];
    if (classpath.length) args.push('-cp', classpath.join(delimiter));
    args.push(...sources);
    run(id, 'javac', args);
}
function hashes(rootPath) {
    return readdirSync(rootPath, { recursive: true, withFileTypes: true })
        .filter(entry => entry.isFile())
        .map(entry => {
            const path = join(entry.parentPath, entry.name);
            return { path, sha256: createHash('sha256').update(readFileSync(path)).digest('hex') };
        });
}

    run('01-java-version', 'java', ['-version']);
    run('02-javac-version', 'javac', ['-version']);
    const open = join(root, 'actor-fixtures/extensible');
    const correct = join(root, 'controls/correct-lib');
    compile('03-open-v1-library', join(open, 'v1-lib'), [join(open, 'v1-src/api/Labeler.java')]);

    compile('06-open-old-consumers', join(open, 'legacy-bin'), [
        join(open, 'consumer-src/client/PartnerLabeler.java'),

    ], [join(open, 'v1-lib')]);
    const openBefore = hashes(join(open, 'legacy-bin'));
    compile('07-open-new-consumer', join(root, 'controls/open-new-bin'), [join(open, 'new-src/client/NewClient.java')], [correct, join(open, 'legacy-bin')]);
    const oldRun = (id, library, expected = 0) => run(id, 'java', ['-cp', [library, join(open, 'legacy-bin')].join(delimiter), 'client.OldClient'], expected);

    oldRun('09-open-old-candidate-negative', join(open, 'candidate-lib'), 1);
    oldRun('10-open-old-correct', correct);
    const newRun = (id, library, expected = 0) => run(id, 'java', ['-cp', [library, join(open, 'legacy-bin'), join(root, 'controls/open-new-bin')].join(delimiter), 'client.NewClient'], expected);
    newRun('11-open-new-candidate-negative', join(open, 'candidate-lib'), 1);
    newRun('12-open-new-correct', correct);
    run('13-open-candidate-signatures', 'javap', ['-classpath', join(open, 'candidate-lib'), '-p', '-s', 'api.Labeler']);
    run('14-open-correct-signatures', 'javap', ['-classpath', correct, '-p', '-s', 'api.Labeler']);
    assert.deepEqual(hashes(join(open, 'legacy-bin')), openBefore, 'Legacy open consumer binaries changed');

    const closed = join(root, 'actor-fixtures/final');
    compile('15-final-v1-library', join(closed, 'v1-lib'), [join(closed, 'v1-src/api/Labeler.java')]);
    compile('16-final-candidate-library', join(closed, 'candidate-lib'), [join(closed, 'candidate-src/api/Labeler.java')]);
