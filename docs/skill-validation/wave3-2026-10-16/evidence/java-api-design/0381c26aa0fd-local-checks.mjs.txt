import assert from 'node:assert/strict';
import { readFileSync, existsSync, readdirSync, statSync } from 'node:fs';
import { join, resolve, dirname } from 'node:path';
import { pathToFileURL, fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';

const repo = resolve('C:/git/agent-skills');
const skill = join(repo, 'skills/java-api-design');
const artifactRoot = dirname(fileURLToPath(import.meta.url));
const { parseManifest } = await import(pathToFileURL(join(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(join(repo, 'packages/core/src/domain/skill-document.ts')));
const manifestText = readFileSync(join(skill, 'skill.yaml'), 'utf8');
const { manifest, issues } = parseManifest(manifestText, { strict: true });
assert.deepEqual(issues, []);
const doc = parseSkillDocument(readFileSync(join(skill, 'SKILL.md'), 'utf8'));
assert.equal(documentName(doc), 'java-api-design');
assert.equal(documentName(doc), manifest.name);
assert.equal(documentDescription(doc), manifest.description);
assert.match(manifestText, /^version: 1\.2\.5$/m);
for (const entry of manifest.files) assert.ok(existsSync(join(skill, entry)), entry);
const files = [join(skill, 'SKILL.md'), join(skill, 'skill.yaml'), ...readdirSync(join(skill, 'references')).map(file => join(skill, 'references', file))];
assert.equal(files.length, 5);
const localLinks = [];
for (const file of files.filter(file => file.endsWith('.md'))) {
    for (const match of readFileSync(file, 'utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
        const href = match[1].replace(/^<|>$/g, '');
        if (/^[a-z][a-z\d+.-]*:/i.test(href)) continue;
        const [relative, anchor] = href.split('#');
        const target = relative ? resolve(dirname(file), relative) : file;
        assert.ok(existsSync(target), href);
        if (anchor) {
            const headings = [...readFileSync(target, 'utf8').matchAll(/^#+\s+(.+)$/gm)]
                .map(match => match[1].trim().toLowerCase().replace(/[^\p{L}\p{N}_\s-]/gu, '').replace(/\s/g, '-'));
            assert.ok(headings.includes(anchor), href);
        }
        localLinks.push({ source: file, href, target });
    }
}
for (const ref of readdirSync(join(skill, 'references'))) assert.ok(doc.body.includes('references/' + ref), ref);
const baseline = JSON.parse(readFileSync(join(artifactRoot, 'baseline-hashes.json'), 'utf8'));
const finalFiles = files.map(path => ({ path, bytes: statSync(path).size, sha256: createHash('sha256').update(readFileSync(path)).digest('hex').toUpperCase() }));
for (const file of finalFiles) {
    assert.equal(file.sha256, baseline.find(old => resolve(old.Path) === resolve(file.path)).SHA256, file.path);
}
console.log(JSON.stringify({
    scope: 'Read-only structural check of the assigned skill only.',
    strictManifestIssues: 0, descriptionMatches: true, version: '1.2.5', shippedFiles: files.length,
    localLinks, allReferencesRouted: true, changedFilesFromAssignment: [], finalFiles
}, null, 2));
