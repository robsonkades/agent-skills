import fs from 'node:fs';
import path from 'node:path';
export function render(root) {
  const policy=JSON.parse(fs.readFileSync(path.join(root,'policy.json'),'utf8'));
  if(!Number.isInteger(policy.maxItems)||policy.maxItems<1)throw Error('Invalid policy');
  for(const name of ['UiLimit','ApiLimit']) {
    const source=`package example.limits;\n\n@javax.annotation.processing.Generated("fixture.PolicyRenderer")\npublic final class ${name} {\n    private ${name}() {}\n    public static boolean accepts(int count) {\n        return count >= 1 && count <= ${policy.maxItems};\n    }\n}\n`;
    fs.writeFileSync(path.join(root,'src/example/limits',name+'.java'),source);
  }
}
