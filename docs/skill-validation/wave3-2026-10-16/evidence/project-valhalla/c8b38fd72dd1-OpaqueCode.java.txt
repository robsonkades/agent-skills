public value class OpaqueCode {
    private final int code;

    private OpaqueCode(int code) {
        if (code < 0 || code > 9999) {
            throw new IllegalArgumentException("code outside fixture range");
        }
        this.code = code;
    }

    public static OpaqueCode of(int code) {
        return new OpaqueCode(code);
    }

    // The public API deliberately offers no code-sensitive equality or hash.
    @Override
    public boolean equals(Object other) {
        return other instanceof OpaqueCode;
    }

    @Override
    public int hashCode() {
        return 0;
    }

    @Override
    public String toString() {
        return "OpaqueCode[redacted]";
    }
}
