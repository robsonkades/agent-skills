import java.util.function.IntFunction;

public final class ChosenCandidateProbe {
    private static int recover(Object target, IntFunction<Object> candidates) {
        // This is the hostile caller boundary: no field access, reflection or secret parameter.
        for (int candidate = 0; candidate <= 9999; candidate++) {
            if (target == candidates.apply(candidate)) {
                return candidate;
            }
        }
        return -1;
    }

    public static void main(String[] args) {
        boolean expectValue = args.length == 1 && args[0].equals("value");
        int[] syntheticCodes = {0, 137, 4321, 9999};
        int freshMatches = 0;
        int canonicalMatches = 0;
        OpaqueCode[] canonical = new OpaqueCode[10000];
        for (int i = 0; i < canonical.length; i++) {
            canonical[i] = OpaqueCode.of(i);
        }

        for (int secret : syntheticCodes) {
            OpaqueCode target = OpaqueCode.of(secret);
            if (!target.equals(OpaqueCode.of((secret + 1) % 10000))
                    || target.hashCode() != 0
                    || !target.toString().equals("OpaqueCode[redacted]")) {
                throw new AssertionError("Fixture accidentally exposes a public discriminator");
            }
            int recovered = recover(target, OpaqueCode::of);
            if (recovered != (expectValue ? secret : -1)) {
                throw new AssertionError("Unexpected fresh-candidate result: " + recovered);
            }
            if (recovered >= 0) {
                freshMatches++;
            }

            // Positive sensitivity control: identity alone is insufficient if the factory interns.
            int canonicalRecovered = recover(canonical[secret], candidate -> canonical[candidate]);
            if (canonicalRecovered != secret) {
                throw new AssertionError("Hostile probe failed its canonicalization control");
            }
            canonicalMatches++;
        }

        System.out.println("runtime=" + Runtime.version());
        System.out.println("fresh candidate recoveries=" + freshMatches + "/4");
        System.out.println("canonicalized identity/value control recoveries=" + canonicalMatches + "/4");
        System.out.println("PASS: public-opacity checks and 8 chosen-candidate observations");
    }
}
