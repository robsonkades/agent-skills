import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const root = path.dirname(new URL(import.meta.url).pathname).replace(/^\/(.:\/)/, '$1');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3';
const env = { ...process.env };
for (const key of ['JDK_JAVA_OPTIONS', 'JAVA_TOOL_OPTIONS', '_JAVA_OPTIONS']) delete env[key];
const results = [];
function run(label, bin, args, cwd, timeout = 25000) {
  const r = spawnSync(`${jdk}/bin/${bin}.exe`, args, { cwd, env, encoding: 'utf8', timeout, maxBuffer: 32 * 1024 * 1024, windowsHide: true });
  fs.writeFileSync(`${root}/${label}.stdout.txt`, r.stdout ?? '');
  fs.writeFileSync(`${root}/${label}.stderr.txt`, r.stderr ?? '');
  const result = { label, executable: `${jdk}/bin/${bin}.exe`, args, cwd, status: r.status, signal: r.signal, error: r.error?.message ?? null, stdout: `${root}/${label}.stdout.txt`, stderr: `${root}/${label}.stderr.txt` };
  results.push(result);
  fs.writeFileSync(`${root}/commands.json`, JSON.stringify(results, null, 2) + '\n');
  console.log(JSON.stringify(result));
  if (r.error || r.status !== 0) throw new Error(`${label} failed: ${r.stderr ?? r.error}`);
  return r.stdout;
}

const source = `import java.lang.management.ManagementFactory;
import com.sun.management.HotSpotDiagnosticMXBean;
import java.time.Instant;

public class CompilerStatsLab {
  static volatile int sink;
  static int work(int value) {
${Array.from({ length: 200 }, () => '    value = (value * 1664525 + 1013904223) ^ (value >>> 13);').join('\n')}
    return value;
  }
  public static void main(String[] args) throws Exception {
    var bean = ManagementFactory.getPlatformMXBean(HotSpotDiagnosticMXBean.class);
    System.out.println("runtime=" + System.getProperty("java.runtime.version"));
    System.out.println("pid=" + ProcessHandle.current().pid() + " start=" + ManagementFactory.getRuntimeMXBean().getStartTime());
    for (String flag : new String[] {"TieredCompilation", "CompileThreshold", "MaxNodeLimit", "NodeLimitFudgeFactor", "BackgroundCompilation"})
      System.out.println(flag + "=" + bean.getVMOption(flag).getValue());
    System.out.println("before=" + Instant.now());
    Thread.sleep(1400);
    int sum = 0;
    for (int i = 0; i < 10000; i++) sum ^= work(i);
    sink = sum;
    System.out.println("after=" + Instant.now() + " checksum=" + sink);
    Thread.sleep(1400);
  }
}
`;
fs.writeFileSync(`${root}/CompilerStatsLab.java`, source, { flag: 'wx' });
fs.mkdirSync(`${root}/classes`, { recursive: true });
run('javac-version', 'javac', ['-version'], root);
run('java-version', 'java', ['-version'], root);
run('compile', 'javac', ['--release', '25', '-d', `${root}/classes`, `${root}/CompilerStatsLab.java`], root);
for (const [label, nodes] of [['restricted', 1000], ['normal', 80000]]) {
  const cwd = `${root}/${label}`;
  fs.mkdirSync(cwd, { recursive: true });
  const args = ['-Xbatch', '-XX:-TieredCompilation', '-XX:CompileThreshold=100', '-XX:+UnlockDiagnosticVMOptions', `-XX:MaxNodeLimit=${nodes}`, '-XX:NodeLimitFudgeFactor=200', '-XX:CompileCommand=compileonly,CompilerStatsLab::work', '-XX:+LogCompilation', `-XX:LogFile=${cwd}/hotspot.xml`, '-XX:+PrintCompilation', `-XX:StartFlightRecording=filename=${cwd}/capture.jfr,settings=default,dumponexit=true,jdk.CompilerStatistics#period=100ms`, '-cp', `${root}/classes`, 'CompilerStatsLab'];
  run(label, 'java', args, cwd);
  run(`${label}-summary`, 'jfr', ['summary', `${cwd}/capture.jfr`], cwd);
  const output = run(`${label}-statistics`, 'jfr', ['print', '--json', '--events', 'jdk.CompilerStatistics', `${cwd}/capture.jfr`], cwd);
  const samples = JSON.parse(output).recording.events.map(e => e.values);
  fs.writeFileSync(`${cwd}/statistics.json`, JSON.stringify(samples, null, 2) + '\n');
  console.log(JSON.stringify({ label, samples: samples.length, first: samples[0], last: samples.at(-1) }));
}
