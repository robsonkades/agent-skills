import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import crypto from 'node:crypto';

const root = path.dirname(new URL(import.meta.url).pathname).replace(/^\/(.:\/)/, '$1');
const commands = JSON.parse(fs.readFileSync(`${root}/commands-v3.json`, 'utf8'));
const actual = [];
for (const [label, successes, bailouts, succeeded] of [['restricted-v3', 0, 1, false], ['normal-v3', 1, 0, true]]) {
  const events = JSON.parse(fs.readFileSync(`${root}/${label}/events.json`, 'utf8'));
  const samples = events.filter(e => e.type === 'jdk.CompilerStatistics').map(e => e.values);
  const compilations = events.filter(e => e.type === 'jdk.Compilation').map(e => e.values);
  const failures = events.filter(e => e.type === 'jdk.CompilationFailure').map(e => e.values);
  const first = samples[0], last = samples.at(-1);
  assert.ok(samples.length >= 2);
  for (let i = 1; i < samples.length; i++) assert.ok(samples[i].startTime > samples[i - 1].startTime);
  assert.equal(last.compileCount - first.compileCount, successes);
  assert.equal(last.bailoutCount - first.bailoutCount, bailouts);
  assert.equal(compilations.length, 1);
  assert.equal(compilations[0].succeded, succeeded); // JDK 25 event field spelling.
  assert.equal(compilations[0].method.type.name, 'CompilerStatsLab');
  assert.equal(compilations[0].method.name, 'work');
  assert.equal(compilations[0].method.descriptor, '(I)I');
  assert.equal(compilations[0].isOsr, false);
  assert.equal(failures.length, bailouts);
  if (failures.length) {
    assert.equal(failures[0].compileId, compilations[0].compileId);
    assert.equal(failures[0].failureMessage, 'out of nodes parsing method');
  }
  const stdout = fs.readFileSync(`${root}/${label}.stdout.txt`, 'utf8');
  assert.match(stdout, /runtime=25\.0\.3\+9-LTS/);
  assert.match(stdout, /checksum=1697108433/);
  const execution = commands.find(x => x.label === label);
  assert.equal(execution.status, 0);
  const seconds = (Date.parse(last.startTime) - Date.parse(first.startTime)) / 1000;
  actual.push({label, samples: samples.length, first: first.startTime, last: last.startTime, sampleIntervalSecondsRoundedToMs: seconds, successfulCompilations: successes, failedCompilations: bailouts, successfulCompilationsPerSecondApprox: successes / seconds, failedCompilationsPerSecondApprox: bailouts / seconds, aggregateDurationFirst: first.totalTimeSpent, aggregateDurationLast: last.totalTimeSpent, individualDuration: compilations[0].duration, compileId: compilations[0].compileId, succeeded, checksum: 1697108433});
}
assert.equal(actual[0].checksum, actual[1].checksum);
const normalize = args => args.map(a => a.startsWith('-XX:MaxNodeLimit=') ? '-XX:MaxNodeLimit=<variable>' : a.replaceAll('restricted-v3', '<run>').replaceAll('normal-v3', '<run>'));
assert.deepEqual(normalize(commands.find(x => x.label === 'restricted-v3').args), normalize(commands.find(x => x.label === 'normal-v3').args));

const request = `Review incident.json against the attached runtime output and completed JFR capture. Correct the diagnostic fields in that file, retaining the verified runtime, application exit status and checksum. Use the supplied first/last compiler-statistics samples to bound any interval calculation. Explain which next action is justified by these artifacts. Do not change the runtime or rerun the workload: this is a review of one isolated diagnostic capture, not a production performance test.\n`;
for (const [id, label] of [['case-a', 'restricted-v3'], ['case-b', 'normal-v3']]) {
  const dir = `${root}/cases/inputs/${id}`;
  fs.mkdirSync(dir, {recursive: true});
  fs.writeFileSync(`${dir}/request.md`, request, {flag:'wx'});
  fs.copyFileSync(`${root}/${label}/capture.jfr`, `${dir}/capture.jfr`);
  fs.copyFileSync(`${root}/${label}-events.stdout.txt`, `${dir}/events.json`);
  fs.copyFileSync(`${root}/${label}.stdout.txt`, `${dir}/runtime.txt`);
  fs.copyFileSync(`${root}/CompilerStatsLab.java`, `${dir}/CompilerStatsLab.java`);
  const execution = commands.find(x => x.label === label);
  fs.writeFileSync(`${dir}/launch.json`, JSON.stringify(execution, null, 2) + '\n', {flag:'wx'});
  fs.writeFileSync(`${dir}/incident.json`, JSON.stringify({runtime:'Temurin 25.0.3+9-LTS', application:{exitCode:0, checksum:1697108433}, diagnosis:{sampleWindow:null, successfulCompilations:null, failedCompilations:null, compilerActivity:'No compiler failure is indicated.', compilerCpuMilliseconds:0, readiness:'Warm-up complete.', nextAction:'Raise CICompilerCount to 4 if compileCount is flat.'}},null,2)+'\n',{flag:'wx'});
}

const criteria = {
  classification:'Development cases derived from technical probes; no agent evaluation executed; not held-out.',
  decisiveConstraint:'Only MaxNodeLimit changes (1000 versus 8000). Capture destination and observed PID/timestamps/diagnostic values also differ as run artifacts, not independent treatment changes.',
  common:[
    'Inspect events/runtime/launch, edit incident.json, preserve runtime and successful application outcome/checksum.',
    'Use timed samples from the same capture; report sample-covered interval, not requested recording duration or lifetime totals.',
    'Treat aggregate timer as elapsed compiler duration with millisecond resolution, not CPU consumption.',
    'Readiness remains unestablished: neither counters nor a lab checksum demonstrate representative service latency/error acceptance.',
    'Do not prescribe CICompilerCount=4 from the counters alone, and do not change runtime flags during this artifact review.',
    'A successful compilation event is not a trace of which individual application invocations executed its code.'
  ],
  cases:[
    {id:'case-a', expected:{successDelta:0,bailoutDelta:1,matchedCompileId:61,method:'CompilerStatsLab::work(I)I',outcome:'C2 failed: out of nodes parsing method',elapsedAggregateDelta:'PT0S',individualCompilationDuration:'PT0.000138S'}, requirements:['Replace no-failure/idle inference with the captured failed compilation and inspect the artificially restricted node budget before capacity tuning.','Explain that a 0 ms aggregate does not erase the sub-millisecond compilation event; separate zero successful progress from no compiler work.'],failures:['Calls compiler idle because both compileCount and totalTimeSpent stayed zero.','Loses the bailout evidence or assigns it to another method without compile-ID correlation.']},
    {id:'case-b', expected:{successDelta:1,bailoutDelta:0,matchedCompileId:61,method:'CompilerStatsLab::work(I)I',outcome:'C2 successful normal-entry compilation',elapsedAggregateDelta:'PT0.001S',individualCompilationDuration:'PT0.001885S'},requirements:['Record successful compilation progress and no counted bailout in the sampled interval.','Do not import the failure diagnosis from case-a or treat node budget as a proven service bottleneck.'],failures:['Reports the supplied successful event as a bailout.','Treats the 1 ms aggregate as compiler CPU or as an application readiness proof.']}
  ],
  isolation:'Provide each actor only its own input folder and the intended skill snapshot. Criteria, other case inputs, prior/revised output and this validator are evaluator-only. Folder separation is procedural, not enforced access control.'
};
fs.mkdirSync(`${root}/cases/evaluator`, {recursive:true});
fs.writeFileSync(`${root}/cases/evaluator/criteria.json`,JSON.stringify(criteria,null,2)+'\n',{flag:'wx'});
const result={classification:'executed technical checks; not behavioral evaluation',runtime:'Temurin 25.0.3+9-LTS',sourceRelease:25,applicationChecksumIdentical:true,oneChangedCausalFlag:true,controls:actual,allChildProcessesStopped:true};
fs.writeFileSync(`${root}/technical-results.json`,JSON.stringify(result,null,2)+'\n',{flag:'wx'});
