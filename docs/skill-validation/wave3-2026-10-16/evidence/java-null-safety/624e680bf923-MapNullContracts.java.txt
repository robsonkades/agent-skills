import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;

public final class MapNullContracts {
    private static int checks;

    private static void equal(Object actual, Object expected, String label) {
        checks++;
        if (!Objects.equals(actual, expected)) {
            throw new AssertionError(label + ": actual=" + actual + ", expected=" + expected);
        }
    }

    private static String cachedAbsence(Map<String, String> values, String key,
            Function<String, String> loader) {
        if (values.containsKey(key)) return values.get(key);
        String loaded = loader.apply(key);
        values.put(key, loaded);
        return loaded;
    }

    public static void main(String[] args) {
        Map<String, String> values = new HashMap<>();
        values.put("negative", null);
        values.put("present", "stored");
        equal(values.getOrDefault("missing", "default"), "default", "absent default");
        equal(values.getOrDefault("negative", "default"), null, "mapped null preserved");
        equal(values.getOrDefault("present", "default"), "stored", "present preserved");
        int[] calls = {0};
        Function<String, String> loader = key -> { calls[0]++; return "loaded"; };
        equal(values.computeIfAbsent("negative", loader), "loaded", "null entry recomputed");
        equal(calls[0], 1, "null entry invokes loader");
        equal(values.computeIfAbsent("present", loader), "stored", "non-null hit");
        equal(calls[0], 1, "non-null hit skips loader");

        values.put("negative", null);
        equal(values.putIfAbsent("negative", "replacement"), null, "putIfAbsent old value");
        equal(values.get("negative"), "replacement", "putIfAbsent replaces null");

        calls[0] = 0;
        Function<String, String> missingLoader = key -> { calls[0]++; return null; };
        equal(values.computeIfAbsent("unknown", missingLoader), null, "first missing result");
        equal(values.computeIfAbsent("unknown", missingLoader), null, "second missing result");
        equal(values.containsKey("unknown"), false, "null result creates no mapping");
        equal(calls[0], 2, "null result is recomputed");
        values.put("negative", null);
        calls[0] = 0;
        values.computeIfAbsent("negative", missingLoader);
        values.computeIfAbsent("negative", missingLoader);
        equal(values.containsKey("negative"), true, "existing null mapping remains");
        equal(calls[0], 2, "existing null still recomputed");

        values.clear();
        values.put("negative", null);
        calls[0] = 0;
        equal(cachedAbsence(values, "negative", loader), null, "cached negative preserved");
        equal(calls[0], 0, "cached negative skips loader");
        equal(cachedAbsence(values, "unknown", missingLoader), null, "new negative loaded");
        equal(cachedAbsence(values, "unknown", missingLoader), null, "new negative cached");
        equal(values.containsKey("unknown"), true, "explicit negative mapping recorded");
        equal(calls[0], 1, "completed absence loaded once");

        values.clear();
        calls[0] = 0;
        Function<String, String> eventuallyFound = key -> ++calls[0] == 1 ? null : "found";
        equal(values.computeIfAbsent("retry", eventuallyFound), null, "unresolved first attempt");
        equal(values.computeIfAbsent("retry", eventuallyFound), "found", "legitimate retry succeeds");
        equal(calls[0], 2, "retry contract invokes twice");
        values.clear();
        calls[0] = 0;
        cachedAbsence(values, "retry", eventuallyFound);
        equal(cachedAbsence(values, "retry", eventuallyFound), null, "wrong permanent-negative policy hides later value");
        equal(calls[0], 1, "counterexample to universal negative caching");
        System.out.println("PASS " + checks + " checks; HashMap confined to main thread; no concurrency guarantee tested");
    }
}
