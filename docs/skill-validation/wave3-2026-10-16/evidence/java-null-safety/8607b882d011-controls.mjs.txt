import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
import assert from 'node:assert/strict';
const root = path.dirname(fileURLToPath(import.meta.url));
const bin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
fs.mkdirSync(path.join(root, 'classes'), { recursive: true });
fs.mkdirSync(path.join(root, 'logs'), { recursive: true });
const commands = [];
function run(id, executable, args) {
  const r = spawnSync(executable, args, { cwd: root, encoding: 'utf8', timeout: 30000 });
  const record = { id, executable, args, cwd: root, exitCode: r.status, signal: r.signal, stdout: r.stdout, stderr: r.stderr };
  commands.push(record);
  fs.writeFileSync(path.join(root, 'logs', id + '.json'), JSON.stringify(record, null, 2));
  assert.equal(r.status, 0, id + ': ' + r.stderr);
  return r.stdout;
}
run('java-version', path.join(bin, 'java.exe'), ['-version']);
run('javac-version', path.join(bin, 'javac.exe'), ['-version']);
run('compile', path.join(bin, 'javac.exe'), ['--release', '25', '-d', path.join(root, 'classes'), path.join(root, 'MapNullContracts.java')]);
const result = run('run', path.join(bin, 'java.exe'), ['-cp', path.join(root, 'classes'), 'MapNullContracts']);
assert.match(result, /PASS 26 checks/);
const classMajor = fs.readFileSync(path.join(root, 'classes', 'MapNullContracts.class')).readUInt16BE(6);
assert.equal(classMajor, 69);
const executables = ['java.exe', 'javac.exe'].map(name => ({ path: path.join(bin, name), sha256: crypto.createHash('sha256').update(fs.readFileSync(path.join(bin, name))).digest('hex') }));
fs.writeFileSync(path.join(root, 'result.json'), JSON.stringify({ commands, classMajor, executables, result }, null, 2));
console.log(JSON.stringify({ result: result.trim(), classMajor, checks: commands.map(({id,exitCode}) => ({id,exitCode})) }));
