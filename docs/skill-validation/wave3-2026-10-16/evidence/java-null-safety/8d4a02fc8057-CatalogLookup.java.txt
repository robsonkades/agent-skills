import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;

public final class CatalogLookup {
    private final Map<String, String> values;
    private final Function<String, String> loader;

    public CatalogLookup(Map<String, String> seed, Function<String, String> loader) {
        this.values = new HashMap<>(Objects.requireNonNull(seed, "seed"));
        this.loader = Objects.requireNonNull(loader, "loader");
    }

    public String lookup(String sku) {
        Objects.requireNonNull(sku, "sku");
        return values.computeIfAbsent(sku, loader);
    }

    public void invalidate(String sku) {
        values.remove(Objects.requireNonNull(sku, "sku"));
    }
}
