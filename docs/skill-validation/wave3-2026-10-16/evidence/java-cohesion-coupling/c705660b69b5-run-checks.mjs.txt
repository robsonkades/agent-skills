import { spawnSync } from 'node:child_process';

import { parseManifest } from 'file:///C:/git/agent-skills/packages/core/src/domain/manifest.ts';
import { parseSkillDocument, documentName, documentDescription } from 'file:///C:/git/agent-skills/packages/core/src/domain/skill-document.ts';

const root = path.dirname(fileURLToPath(import.meta.url));
const repo = 'C:/git/agent-skills';

const commands = [];
const checks = [];
fs.mkdirSync(path.join(root, 'logs'), { recursive: true });
function run(label, executable, args, options = {}) {
  const result = spawnSync(executable, args, { cwd: repo, encoding: 'utf8', timeout: 30000 });
  const record = { label, executable, args, cwd: repo, exitCode: result.status, signal: result.signal, stdout: result.stdout ?? '', stderr: result.stderr ?? '', error: result.error?.message ?? null };
  commands.push(record);
  fs.writeFileSync(path.join(root, 'logs', `${label}.json`), JSON.stringify(record, null, 2) + '\n');
  assert.equal(result.error, undefined, label);
  assert.equal(result.status, options.expectedExit ?? 0, label);
  return record;
}
function write(relative, content) {
  const filename = path.join(root, relative);
  fs.mkdirSync(path.dirname(filename), { recursive: true });
  fs.writeFileSync(filename, content.trim() + '\n');
  return filename;
}
function sourcePaths(relative) {
  const directory = path.join(root, relative);
  return fs.readdirSync(directory, { recursive: true }).filter(p => p.endsWith('.java')).map(p => path.join(directory, p));
}
