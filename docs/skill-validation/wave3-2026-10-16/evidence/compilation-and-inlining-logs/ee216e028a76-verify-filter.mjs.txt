import { readFileSync, writeFileSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import assert from 'node:assert/strict';

const grep = 'C:/Program Files/Git/usr/bin/grep.exe';
const oldPattern = String.raw`\]\[jit,compilation\] +[0-9]+ [ %s!bn]{5} 4 `;
const reference = readFileSync('C:/git/agent-skills/skills/compilation-and-inlining-logs/references/printcompilation-format.md', 'utf8');
const correctedPattern = /^grep -E '([^']+)' jit.log$/m.exec(reference)?.[1];
assert(correctedPattern, 'Read the actual shipped unified-log extraction command');
const lines = path => readFileSync(path, 'utf8').split(/\r?\n/).filter(Boolean);
function key(parts) { return JSON.stringify([parts[1], parts[2], parts[3], parts[4]]); }
function printOracle(path) {
  // Different recording path: PrintCompilation supplies the full structural fields.
  return lines(path).flatMap(line => {
    const match = /^ *\d+ +(\d+) ([ %s!bn]{5}) ([0-9-]) +(.*)$/.exec(line);
    return match && match[3] === '4' ? [key(match)] : [];
  }).sort();
}
function select(pattern, path, outputFile) {
  const result = spawnSync(grep, ['-E', pattern, path], {encoding:'utf8'});
  writeFileSync(outputFile, result.stdout);
  assert([0,1].includes(result.status), result.stderr);
  return {exit:result.status, lines:result.stdout.split(/\r?\n/).filter(Boolean)};
}
function ulKeys(selected) {
  return selected.map(line => {
    const body = line.replace(/^(?:\[[^\]]*\])+ +/, '');
    const match = /^(\d+) ([ %s!bn]{5}) ([0-9-]) +(.*)$/.exec(body);
    assert(match, 'Unrecognized selected record: ' + line);
    return key(match);
  }).sort();
}
const results = [];
for (const mode of ['jit-only','mixed-loader']) {
  const oracle = printOracle(mode+'.print.log');
  assert(oracle.length > 0, 'Fixture must capture tier-4 records');
  assert(lines(mode+'.print.log').some(line => line.startsWith('completed=')));
  const old = select(oldPattern, mode+'.ul.log', mode+'.old-selected.log');
  const fixed = select(correctedPattern, mode+'.ul.log', mode+'.fixed-selected.log');
  assert.deepEqual(ulKeys(fixed.lines), oracle, 'Corrected filter must preserve every tier-4 header/status');
  if(mode==='jit-only') assert.deepEqual(ulKeys(old.lines),oracle);
  else assert(old.lines.length < oracle.length, 'Old filter must expose padding loss');
  results.push({mode,oracleRecords:oracle.length,oldRecords:old.lines.length,oldExit:old.exit,fixedRecords:fixed.lines.length,fixedExit:fixed.exit});
}
const negative = select(correctedPattern, 'negative-tags.log', 'negative-selected.log');
assert.equal(negative.exit,1);
assert.equal(negative.lines.length,0);
results.push({negativeTags:true,selected:0,exit:negative.exit});
console.log(JSON.stringify({grep,oldPattern,correctedPattern,results},null,2));
