public final class CompilationProbe {
    static volatile long sink;
    static int hot(int value) {
        int x = value * 31 + 7;
        return (x ^ (x >>> 3)) + value;
    }
    public static void main(String[] args) {
        System.gc();
        long sum = 0;
        for (int i = 0; i < 2_000_000; i++) sum += hot(i);
        sink = sum;
        System.out.println("completed=" + sink);
    }
}
