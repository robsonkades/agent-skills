// Checks retained controls produced by quality03; this does not rerun a JVM.
import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { resolve, dirname } from 'node:path';
const root = dirname(fileURLToPath(import.meta.url));
const results = JSON.parse(readFileSync(resolve(root, 'cold-gate-controls.json'), 'utf8'));
const expected = new Map([
  ['cold-medium-128', { tag: 'inline_fail', reason: 'already compiled into a medium method' }],
  ['cold-medium-128-bytecode-1000', { tag: 'inline_fail', reason: 'already compiled into a medium method' }],
  ['cold-medium-512', { tag: 'inline_success', reason: 'inline (hot)' }],
]);
const checked = [];
for (const [id, verdict] of expected) {
  const recorded = results.find(r => r.id === id);
  assert(recorded, id);
  assert.equal(recorded.status, 0, id);
  const stdout = readFileSync(resolve(root, 'logs', id + '.stdout.txt'), 'utf8');
  const stderr = readFileSync(resolve(root, 'logs', id + '.stderr.txt'), 'utf8');
  assert.equal(stdout, recorded.stdout);
  assert.equal(stderr, '');
  assert(stdout.includes('InlineGate::callee (9 bytes)'));
  assert(stdout.includes(verdict.reason));
  assert.equal(stdout.trim().split(/\r?\n/).at(-1), '12399400384');
  const xml = readFileSync(resolve(root, 'logs', id + '.xml'), 'utf8');
  assert(xml.includes('25.0.3+9-LTS'));
  const tasks = [...xml.matchAll(/<task\b[^>]*method='InlineGate caller \(I\)I'[^>]*>[\s\S]*?<\/task>/g)];
  assert.equal(tasks.length, 1, id);
  const task = tasks[0][0];
  const compileId = /compile_id='([^']+)'/.exec(task)[1];
  assert(task.includes('<' + verdict.tag + " reason='" + verdict.reason + "'/>"));
  assert(task.includes("<task_done success='1'"));
  assert(/<method\b[^>]*name='callee'[^>]*bytes='9'[^>]*compiler='c2'/.test(task));
  const nmethods = [...xml.matchAll(/<nmethod\b[^>]*\/>/g)].map(match => match[0]);
  assert(nmethods.some(nm =>
    nm.includes("compile_id='" + compileId + "'") &&
    nm.includes("compiler='c2'") && nm.includes("method='InlineGate caller (I)I'")));
  checked.push({
    id, recordedJvmExit: recorded.status, verdict: verdict.reason, compileId,
    recordedCompiler: 'c2', calleeBytecodes: 9, callerTaskSucceeded: true,
    callerNmethodPresent: true, checksum: '12399400384',
  });
}
const report = {
  kind: 'Reinspection of retained technical controls, not a new JVM or behavioral evaluation',
  originalProducer: '/root/quality03',
  checked,
  limits: [
    'Compiler decision and successful caller task/nmethod are demonstrated for these controlled executions.',
    'No application performance, prevalence, production impact or agent improvement measured.',
    'Earlier exploratory gate-controls.json/probes.json are not used as medium-branch evidence.',
  ],
};
writeFileSync(resolve(root, 'retained-control-verification.json'), JSON.stringify(report, null, 2) + '\n');
console.log(JSON.stringify(report, null, 2));
