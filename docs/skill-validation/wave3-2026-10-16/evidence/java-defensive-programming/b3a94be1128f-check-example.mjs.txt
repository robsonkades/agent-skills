import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, mkdirSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';

const repo = resolve(process.argv[2] ?? 'C:/git/agent-skills');
const mode = process.argv[3] ?? 'baseline';
assert.ok(['baseline', 'fixed'].includes(mode));
const temp = dirname(fileURLToPath(import.meta.url));
const output = resolve(temp, mode);
mkdirSync(resolve(output, 'classes'), { recursive: true });
const source = readFileSync(resolve(repo, 'skills/java-defensive-programming/references/hardening-example.md'), 'utf8');
const blocks = [...source.matchAll(/```java\r?\n([\s\S]*?)```/g)].map(match => match[1]);
const exception = blocks.find(block => block.includes('public final class RefundInputException'));
const account = blocks.find(block => block.includes('public record AccountId'));
const records = blocks.find(block => block.includes('public record Money'));
assert.ok(exception && account && records);
const boundary = records.indexOf('public record RefundRequest');
assert.ok(boundary > 0);
const imports = 'import java.math.*;\nimport java.util.*;\nimport java.util.regex.*;\n';
for (const [name, body] of Object.entries({ RefundInputException: exception, AccountId: account,
  Money: records.slice(0, boundary), RefundRequest: records.slice(boundary) })) {
  const target = resolve(output, `${name}.java`);
  // Baseline files were extracted before the package edit and are retained for replay.
  // Never overwrite that evidence with the current (revised) Markdown.
  if (mode === 'baseline') assert.ok(existsSync(target), `Missing retained baseline: ${target}`);
  else writeFileSync(target, imports + body);
}
const commands = [
  ['javac', ['--release', '16', '-d', resolve(output, 'classes'),
    ...['RefundInputException','AccountId','Money','RefundRequest'].map(name => resolve(output, `${name}.java`)), resolve(temp, 'RefundProbe.java')]],
  ['java', ['-da', '-cp', resolve(output, 'classes'), 'RefundProbe', mode]],
  ['java', ['-ea', '-cp', resolve(output, 'classes'), 'RefundProbe', mode]]
];
const results = [];
for (const [command, args] of commands) {
  const result = spawnSync(command, args, { cwd: output, encoding: 'utf8', timeout: 15000 });
  results.push({ command, args, cwd: output, status: result.status, signal: result.signal,
    stdout: result.stdout, stderr: result.stderr, error: result.error?.message });
  writeFileSync(resolve(output, 'results.json'), JSON.stringify(results, null, 2));
  console.log(JSON.stringify(results.at(-1), null, 2));
  assert.equal(result.status, 0, `${command} failed`);
}
