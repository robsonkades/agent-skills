import assert from 'node:assert/strict';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { pathToFileURL } from 'node:url';
import { createHash } from 'node:crypto';

const repo = 'C:/git/agent-skills';
const root = resolve(repo, 'skills/code-cache-segments');
const { loadPackageFromDirectory } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/package-loader.ts')).href);
const { validatePackage } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/validate-package.ts')).href);
const { NodeFileSystem } = await import(pathToFileURL(resolve(repo, 'packages/node/src/file-system.ts')).href);
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), root, { strict: true });
const report = validatePackage(loaded.pkg, { directoryName: 'code-cache-segments', strict: true });
const issues = [...loaded.issues, ...report.issues];
assert.equal(issues.length, 0, JSON.stringify(issues));
const baseline = {
  'SKILL.md': '31D23EB1987227DB9F19FF95C1EF8E672344A224CD1D14A5FFA461185729BB63',
  'skill.yaml': '0A6E1360030BDF68168D6DF9F504F4825A697BC41A7AE27A19C265D425A38CEF',
  'references/segments-and-sizing.md': '6B00FA330DCA2B3D27862F0F4213F798AA6669BA41352B88A805B68AE8635FCD',
  'references/unloading-and-gc.md': 'A9616D90028DE8E1CB85CE5A229AEAECA73901D74396C4DA6E4C3E9493CF1F20',
  'references/diagnosing-exhaustion.md': '3ECC525F6243619BCD534690F45D7B4AFBA0D7D0C0012E6CF7C8D2EBA4517714'
};
assert.equal(loaded.pkg.files.length, Object.keys(baseline).length);
const links = [];
const hashes = {};
const changed = [];
for (const [file, expected] of Object.entries(baseline)) {
  const absolute = resolve(root, file);
  const bytes = readFileSync(absolute);
  hashes[file] = createHash('sha256').update(bytes).digest('hex').toUpperCase();
  if (hashes[file] !== expected) changed.push(file);
  if (!file.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
    const target = match[1];
    if (/^<?[a-z]+:/i.test(target)) continue;
    const [path, anchor] = target.split('#');
    const linked = resolve(dirname(absolute), path);
    assert.ok(existsSync(linked), `${file}: ${target}`);
    if (anchor) {
      const headings = [...readFileSync(linked, 'utf8').matchAll(/^#+\s+(.+)$/gm)].map(heading => heading[1].trim().toLowerCase().replace(/[`*_]/g, '').replace(/[^\p{L}\p{N}\s-]/gu, '').replace(/\s/g, '-'));
      assert.ok(headings.includes(anchor), `${file}: unresolved anchor ${target}`);
    }
    links.push({ file, target });
  }
}
assert.deepEqual(changed.sort(), ['SKILL.md', 'references/diagnosing-exhaustion.md', 'skill.yaml'].sort());
const referrals = ['jit-compilation', 'jvm-memory-regions', 'deoptimization', 'container-awareness', 'gc-log-analysis', 'metaspace-internals'];
for (const name of referrals) assert.ok(existsSync(resolve(repo, 'skills', name, 'SKILL.md')), name);
assert.equal(loaded.pkg.manifest.version, '2.1.5');
console.log(JSON.stringify({ runtime: process.version, name: loaded.pkg.manifest.name, version: loaded.pkg.manifest.version, packageFiles: loaded.pkg.files.length, issues, localMarkdownLinks: links, referrals, baselineHashes: baseline, finalHashes: hashes, changed }, null, 2));
