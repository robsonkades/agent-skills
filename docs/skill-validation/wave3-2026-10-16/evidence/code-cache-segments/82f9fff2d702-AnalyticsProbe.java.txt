import java.lang.ref.WeakReference;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.function.IntUnaryOperator;

public final class AnalyticsProbe {
    private static final List<IntUnaryOperator> retained = new ArrayList<>();
    private static final List<WeakReference<ClassLoader>> loaders = new ArrayList<>();
    private static volatile int sink;

    private static final class IsolatedLoader extends ClassLoader {
        Class<?> define(byte[] bytes) {
            return defineClass("HotWork", bytes, 0, bytes.length);
        }
    }

    private static void prepare(byte[] bytes) throws Exception {
        for (int index = 0; index < 80; index++) {
            IsolatedLoader loader = new IsolatedLoader();
            IntUnaryOperator operation = (IntUnaryOperator) loader.define(bytes).getDeclaredConstructor().newInstance();
            for (int invocation = 0; invocation < 4000; invocation++) {
                sink = operation.applyAsInt(invocation);
            }
            loaders.add(new WeakReference<>(loader));
            if ((index & 1) != 0) retained.add(operation);
        }
    }

    public static void main(String[] arguments) throws Exception {
        prepare(Files.readAllBytes(Path.of("HotWork.class")));
        System.gc();
        System.gc();
        long cleared = loaders.stream().filter(reference -> reference.get() == null).count();
        System.out.println("READY pid=" + ProcessHandle.current().pid() + " unloadedLoaders=" + cleared + " retained=" + retained.size());
        System.out.flush();
        System.in.read();
    }
}
