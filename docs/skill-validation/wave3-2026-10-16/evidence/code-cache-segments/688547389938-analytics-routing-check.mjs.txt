import assert from 'node:assert/strict';
import { spawn, spawnSync } from 'node:child_process';
import { writeFileSync, readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const directory = fileURLToPath(new URL('.', import.meta.url));
const original = JSON.parse(readFileSync(resolve(directory, 'probe-process.json'), 'utf8'));
const bin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const records = [];
const target = spawn(original.executable, original.arguments, { cwd: directory, windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] });
let stdout = '';
let stderr = '';
let resolveReady;
let rejectReady;
let readyTimer;
const ready = new Promise((resolve, reject) => { resolveReady = resolve; rejectReady = reject; });
target.stdout.on('data', chunk => { stdout += chunk.toString(); if (stdout.includes('READY pid=')) { clearTimeout(readyTimer); resolveReady(); } });
target.stderr.on('data', chunk => { stderr += chunk.toString(); });
target.on('error', error => { clearTimeout(readyTimer); rejectReady(error); });
const finished = new Promise(resolveExit => target.once('exit', (code, signal) => {
  clearTimeout(readyTimer);
  if (!stdout.includes('READY pid=')) rejectReady(new Error('Probe exited before READY: ' + code));
  resolveExit({ code, signal });
}));
readyTimer = setTimeout(() => rejectReady(new Error('Probe READY timeout')), 20000);
function run(id, args) {
  const commandArgs = [String(target.pid), ...args];
  const result = spawnSync(resolve(bin, 'jcmd.exe'), commandArgs, { cwd: directory, encoding: 'utf8', windowsHide: true, timeout: 20000, maxBuffer: 4 * 1024 * 1024 });
  writeFileSync(resolve(directory, id + '.stdout.log'), result.stdout ?? '');
  writeFileSync(resolve(directory, id + '.stderr.log'), result.stderr ?? '');
  records.push({ id, executable: resolve(bin, 'jcmd.exe'), arguments: commandArgs, cwd: directory, status: result.status, error: result.error?.message ?? null, stdout: id + '.stdout.log', stderr: id + '.stderr.log' });
  writeFileSync(resolve(directory, 'routing-executions.json'), JSON.stringify(records, null, 2) + '\n');
  assert.equal(result.error, undefined, id);
  assert.equal(result.status, 0, id);
  return result.stdout;
}
try {
  await ready;
  assert.match(stdout, /unloadedLoaders=40 retained=40/);
  const help = run('routing-help', ['help', 'Compiler.CodeHeap_Analytics']);
  assert.match(help, /MethodNames/);
  run('routing-aggregate', ['Compiler.CodeHeap_Analytics', 'aggregate']);
  const names = run('routing-methodnames', ['Compiler.CodeHeap_Analytics', 'MethodNames']);
  assert.match(names, /Function 'MethodNames' is only available as part of function 'all'/);
  assert.doesNotMatch(names, /M E T H O D   N A M E S/);
  const all = run('routing-all', ['Compiler.CodeHeap_Analytics', 'all']);
  assert.match(all, /M E T H O D   N A M E S/);
  assert.match(all, /HotWork/);
  const after = run('routing-after-all', ['Compiler.CodeHeap_Analytics', 'FreeSpace']);
  assert.match(after, /No aggregated/);
  assert.doesNotMatch(after, /Top Ten Free Blocks/);
  const summary = { helpListsMethodNames: true, standaloneMethodNamesExitZeroButNoNames: true, allIncludesMethodNames: true, allDiscardsSnapshot: true };
  writeFileSync(resolve(directory, 'routing-comparison.json'), JSON.stringify(summary, null, 2) + '\n');
  console.log(JSON.stringify(summary, null, 2));
} finally {
  clearTimeout(readyTimer);
  target.stdin.end('\n');
  const deadline = setTimeout(() => target.kill(), 5000);
  const exit = await finished;
  clearTimeout(deadline);
  writeFileSync(resolve(directory, 'routing-probe.stdout.log'), stdout);
  writeFileSync(resolve(directory, 'routing-probe.stderr.log'), stderr);
  writeFileSync(resolve(directory, 'routing-probe-process.json'), JSON.stringify({ executable: original.executable, arguments: original.arguments, cwd: directory, pid: target.pid, ...exit }, null, 2) + '\n');
  assert.equal(exit.code, 0, 'Owned probe terminated normally');
}
