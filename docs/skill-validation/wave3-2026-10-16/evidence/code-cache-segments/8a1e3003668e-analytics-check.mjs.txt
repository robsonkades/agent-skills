import assert from 'node:assert/strict';
import { spawn, spawnSync } from 'node:child_process';
import { writeFileSync, readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const directory = fileURLToPath(new URL('.', import.meta.url));
const bin = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const executions = [];
function run(id, executable, args) {
  const result = spawnSync(resolve(bin, executable), args, { cwd: directory, encoding: 'utf8', timeout: 20000, maxBuffer: 4 * 1024 * 1024, windowsHide: true });
  writeFileSync(resolve(directory, id + '.stdout.log'), result.stdout ?? '');
  writeFileSync(resolve(directory, id + '.stderr.log'), result.stderr ?? '');
  const record = { id, executable: resolve(bin, executable), arguments: args, cwd: directory, status: result.status, error: result.error?.message ?? null, stdout: id + '.stdout.log', stderr: id + '.stderr.log' };
  executions.push(record);
  writeFileSync(resolve(directory, 'analytics-executions.json'), JSON.stringify(executions, null, 2) + '\n');
  assert.equal(record.error, null, id);
  assert.equal(result.status, 0, id);
  return result.stdout;
}

run('compile', 'javac.exe', ['--release', '25', 'HotWork.java', 'AnalyticsProbe.java']);
const args = ['-Xms16m', '-Xmx64m', '-XX:+UseSerialGC', '-Xbatch', '-XX:TieredStopAtLevel=1', '-XX:CICompilerCount=2', '-XX:CompileThresholdScaling=0.01', '-XX:CompileCommand=compileonly,HotWork::applyAsInt', '-XX:+SegmentedCodeCache', '-XX:ReservedCodeCacheSize=64m', '-XX:NonNMethodCodeHeapSize=8m', '-Xlog:gc=info,class+unload=info', 'AnalyticsProbe'];
const target = spawn(resolve(bin, 'java.exe'), args, { cwd: directory, windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] });
let stdout = '';
let stderr = '';
let resolveReady;
let rejectReady;
let readyTimer;
const ready = new Promise((resolve, reject) => { resolveReady = resolve; rejectReady = reject; });
target.stdout.on('data', chunk => {
  stdout += chunk.toString();
  if (stdout.includes('READY pid=')) { clearTimeout(readyTimer); resolveReady(); }
});
target.stderr.on('data', chunk => { stderr += chunk.toString(); });
target.on('error', error => { clearTimeout(readyTimer); rejectReady(error); });
const finished = new Promise(resolveExit => target.once('exit', (code, signal) => {
  clearTimeout(readyTimer);
  if (!stdout.includes('READY pid=')) rejectReady(new Error('Probe exited before READY: ' + code));
  resolveExit({ code, signal });
}));
readyTimer = setTimeout(() => rejectReady(new Error('Probe READY timeout')), 20000);

try {
  await ready;
  assert.match(stdout, /unloadedLoaders=40 retained=40/);
  run('codecache-before', 'jcmd.exe', [String(target.pid), 'Compiler.codecache']);
  run('aggregate-first', 'jcmd.exe', [String(target.pid), 'Compiler.CodeHeap_Analytics', 'aggregate']);
  run('freespace-first', 'jcmd.exe', [String(target.pid), 'Compiler.CodeHeap_Analytics', 'FreeSpace']);
  run('freespace-repeated', 'jcmd.exe', [String(target.pid), 'Compiler.CodeHeap_Analytics', 'FreeSpace']);
  run('aggregate-refresh', 'jcmd.exe', [String(target.pid), 'Compiler.CodeHeap_Analytics', 'aggregate']);
  run('freespace-refreshed', 'jcmd.exe', [String(target.pid), 'Compiler.CodeHeap_Analytics', 'FreeSpace']);
  run('codecache-after', 'jcmd.exe', [String(target.pid), 'Compiler.codecache']);
  run('discard', 'jcmd.exe', [String(target.pid), 'Compiler.CodeHeap_Analytics', 'discard']);
} finally {
  clearTimeout(readyTimer);
  target.stdin.end('\n');
  const deadline = setTimeout(() => target.kill(), 5000);
  const exit = await finished;
  clearTimeout(deadline);
  writeFileSync(resolve(directory, 'probe.stdout.log'), stdout);
  writeFileSync(resolve(directory, 'probe.stderr.log'), stderr);
  writeFileSync(resolve(directory, 'probe-process.json'), JSON.stringify({ executable: resolve(bin, 'java.exe'), arguments: args, cwd: directory, pid: target.pid, ...exit }, null, 2) + '\n');
  assert.equal(exit.code, 0, 'Owned probe terminated normally');
}

function freeBlocks(file) {
  const text = readFileSync(resolve(directory, file), 'utf8');
  return [...text.matchAll(/(0x[0-9a-f]+): Len\[\s*(\d+)\] = (0x[0-9a-f]+)/gi)].map(match => ({ address: match[1], index: Number(match[2]), bytes: Number.parseInt(match[3], 16) }));
}
const first = freeBlocks('freespace-first.stdout.log');
const repeated = freeBlocks('freespace-repeated.stdout.log');
const refreshed = freeBlocks('freespace-refreshed.stdout.log');
assert.ok(first.length > 2, 'Control must expose multiple reclaimed free-list blocks');
assert.equal(first.length, repeated.length);
assert.equal(first.length, refreshed.length);
const changed = first.map((block, index) => ({ ...block, repeatedBytes: repeated[index].bytes, refreshedBytes: refreshed[index].bytes })).filter(block => block.bytes !== block.repeatedBytes);
assert.ok(changed.length > 0, 'Repeated printing must expose the source mutation');
assert.deepEqual(refreshed, first, 'Fresh aggregation restores the actual stable free-block snapshot');
assert.ok(changed.every(block => block.repeatedBytes > block.bytes));
writeFileSync(resolve(directory, 'analytics-comparison.json'), JSON.stringify({ observedFreeBlocks: first.length, changedBlocks: changed.length, first, repeated, refreshed, changed, freshAggregationRestoredSnapshot: true }, null, 2) + '\n');
