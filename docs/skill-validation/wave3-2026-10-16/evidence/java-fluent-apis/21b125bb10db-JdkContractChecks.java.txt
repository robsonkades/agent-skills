package sample;

import java.net.URI;
import java.net.http.HttpRequest;
import java.time.DateTimeException;
import java.time.Duration;
import java.time.LocalDate;
import java.util.List;

public final class JdkContractChecks {
    public static void main(String[] args) {
        LocalDate original = LocalDate.of(2026, 10, 16);
        original.withDayOfMonth(17);
        require(original.getDayOfMonth() == 16, "ignored immutable result");
        LocalDate updated = original.withDayOfMonth(17);
        require(updated.getDayOfMonth() == 17 && original.getDayOfMonth() == 16, "retained immutable result");
        try {
            original.withDayOfMonth(32);
            throw new AssertionError("Invalid date accepted");
        } catch (DateTimeException expected) {
            require(original.getDayOfMonth() == 16, "failed immutable update changed original");
        }

        HttpRequest.Builder builder = HttpRequest.newBuilder(URI.create("https://example.invalid"));
        HttpRequest.Builder alias = builder.header("X-Example", "one");
        require(alias == builder, "setter identity");
        HttpRequest first = builder.build();
        builder.header("X-Example", "two");
        require(first.headers().allValues("X-Example").equals(List.of("one")), "product snapshot");
        require(builder.build().headers().allValues("X-Example").equals(List.of("one", "two")), "header accumulation");
        HttpRequest.Builder copy = builder.copy().setHeader("X-Example", "replacement");
        require(copy.build().headers().allValues("X-Example").equals(List.of("replacement")), "header replacement");
        require(builder.build().headers().allValues("X-Example").equals(List.of("one", "two")), "copy independence");
        try {
            builder.header("X-Earlier", "retained").timeout(Duration.ZERO);
            throw new AssertionError("Invalid timeout accepted");
        } catch (IllegalArgumentException expected) {
            require(builder.build().headers().firstValue("X-Earlier").orElseThrow().equals("retained"), "earlier chain mutation");
        }
        System.out.println("PASS: immutable results, mutable identity, snapshots, copy isolation, repeated options and failed-chain state");
        System.out.println("No HTTP request was sent.");
    }

    private static void require(boolean condition, String name) {
        if (!condition) throw new AssertionError(name);
    }
}
