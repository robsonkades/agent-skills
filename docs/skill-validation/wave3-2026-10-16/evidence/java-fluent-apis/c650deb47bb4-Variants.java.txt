package sample;

import java.util.List;

public final class Variants {
    private Variants() {}

    public static List<Product> build(Options base, boolean audited) {
        if (audited) {
            base.withAudit(true);
        }
        Options left = base.withLabel("left");
        Options right = base.withLabel("right");
        return List.of(left.build(), right.build());
    }
}
