import assert from 'node:assert/strict';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { pathToFileURL, fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';

const repo = 'C:/git/agent-skills';
const skill = 'java-reflection-and-method-handles';
const root = resolve(repo, 'skills', skill);
const artifacts = fileURLToPath(new URL('.', import.meta.url));
const { loadPackageFromDirectory } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/package-loader.ts')).href);
const { validatePackage } = await import(pathToFileURL(resolve(repo, 'packages/core/src/application/validate-package.ts')).href);
const { NodeFileSystem } = await import(pathToFileURL(resolve(repo, 'packages/node/src/file-system.ts')).href);
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), root, { strict: true });
const report = validatePackage(loaded.pkg, { directoryName: skill, strict: true });
const issues = [...loaded.issues, ...report.issues];
assert.equal(issues.length, 0, JSON.stringify(issues));
const baseline = JSON.parse(readFileSync(resolve(artifacts, 'baseline-hashes.json'), 'utf8').replace(/^\uFEFF/, ''));
assert.equal(loaded.pkg.files.length, baseline.length);
const links = [];
const hashes = {};
const changed = [];
for (const { path: file, sha256: expected } of baseline) {
  const absolute = resolve(root, file);
  const bytes = readFileSync(absolute);
  hashes[file] = createHash('sha256').update(bytes).digest('hex').toUpperCase();
  if (hashes[file] !== expected) changed.push(file);
  if (!file.endsWith('.md')) continue;
  for (const match of bytes.toString('utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
    const target = match[1];
    if (/^<?[a-z]+:/i.test(target)) continue;
    const [path, anchor] = target.split('#');
    const linked = resolve(dirname(absolute), path);
    assert.ok(existsSync(linked), `${file}: ${target}`);
    assert.equal(anchor, undefined, 'No local anchors expected in this package');
    links.push({ file, target });
  }
}
assert.deepEqual(changed.sort(), ['SKILL.md', 'references/method-handles-and-encapsulation.md', 'skill.yaml'].sort());
const referrals = ['compilation-and-inlining-logs', 'graalvm-native-image', 'java-annotations', 'java-dependency-inversion',
  'java-exception-design', 'java-serialization-hardening', 'java-test-design', 'jmh-microbenchmarks',
  'jni-and-ffm', 'jvm-class-loading', 'off-heap-memory', 'varhandles-and-memory-ordering'];
for (const name of referrals) assert.ok(existsSync(resolve(repo, 'skills', name, 'SKILL.md')), name);
assert.equal(loaded.pkg.manifest.version, '1.4.6');
console.log(JSON.stringify({ runtime: process.version, name: loaded.pkg.manifest.name,
  version: loaded.pkg.manifest.version, packageFiles: loaded.pkg.files.length,
  issues, localMarkdownLinks: links, referrals, baselineHashes: baseline, finalHashes: hashes, changed }, null, 2));
