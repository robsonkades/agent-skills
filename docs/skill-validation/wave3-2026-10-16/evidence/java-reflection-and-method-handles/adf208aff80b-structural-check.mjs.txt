import { spawnSync } from 'node:child_process';
import { writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import assert from 'node:assert/strict';

const root = fileURLToPath(new URL('.', import.meta.url));
const cwd = 'C:/git/agent-skills';
const packagePath = 'skills/java-reflection-and-method-handles';
const files = ['SKILL.md', 'skill.yaml', 'references/method-handles-and-encapsulation.md',
  'references/reflection-cost-model.md', 'references/when-reflection-is-justified.md'];
const commands = [
  ['package-check', process.execPath, [resolve(root, 'package-check.mjs')]],
  ['prettier-check', process.execPath, ['node_modules/prettier/bin/prettier.cjs', '--check', ...files.map(file => packagePath + '/' + file)]],
  ['diff-check', 'git', ['diff', '--check', '--', packagePath]],
  ['final-diff', 'git', ['diff', '--', packagePath]],
  ['final-status', 'git', ['status', '--short', '--', packagePath]],
  ['staged-diff', 'git', ['diff', '--cached', '--', packagePath]]
];
const results = [];
for (const [name, executable, args] of commands) {
  const result = spawnSync(executable, args, { cwd, encoding: 'utf8', windowsHide: true, timeout: 15000 });
  writeFileSync(resolve(root, 'logs', name + '.stdout.txt'), result.stdout || '');
  writeFileSync(resolve(root, 'logs', name + '.stderr.txt'), result.stderr || '');
  results.push({ name, command: [executable, ...args], cwd, status: result.status,
    signal: result.signal, error: result.error?.message ?? null,
    stdout: 'logs/' + name + '.stdout.txt', stderr: 'logs/' + name + '.stderr.txt' });
  writeFileSync(resolve(root, 'structural-results.json'), JSON.stringify(results, null, 2) + '\n');
  assert.equal(result.error, undefined, name);
  assert.equal(result.status, 0, name + '\n' + result.stdout + '\n' + result.stderr);
  if (name === 'staged-diff') assert.equal(result.stdout, '', 'No staged changes in owned package');
  console.log(name + ': PASS');
}
