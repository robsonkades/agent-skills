import java.util.Arrays;
import jdk.incubator.vector.*;

public final class RecipeChecks {
    private static int checks;
    private static final float[] SPECIAL = {
        Float.NaN, Float.POSITIVE_INFINITY, Float.NEGATIVE_INFINITY, 0f, -0f,
        Float.MIN_VALUE, -Float.MIN_VALUE, Float.MAX_VALUE, -Float.MAX_VALUE, 1f, -1f
    };

    public static void main(String[] args) {
        int lanes = VectorAPILab.FSPECIES.length();
        for (int n = 0; n <= 2 * lanes + 1; n++) {
            float[] a = new float[n], b = new float[n], expected = new float[n];
            float[] doubled = new float[n];
            for (int i = 0; i < n; i++) {
                a[i] = SPECIAL[i % SPECIAL.length];
                b[i] = SPECIAL[(i + 3) % SPECIAL.length];
                expected[i] = a[i] + b[i];
                doubled[i] = a[i] + a[i];
            }
            for (boolean masked : new boolean[] {false, true}) {
                float[] out = new float[n];
                add(masked, a, b, out);
                same(expected, out, "distinct arrays");
                float[] left = a.clone();
                add(masked, left, b, left);
                same(expected, left, "output aliases a");
                float[] right = b.clone();
                add(masked, a, right, right);
                same(expected, right, "output aliases b");
                float[] all = a.clone();
                add(masked, all, all, all);
                same(doubled, all, "all three aliases");
            }
            for (float threshold : new float[] {Float.NaN, 0f, Float.POSITIVE_INFINITY,
                    Float.NEGATIVE_INFINITY}) {
                int expectedCount = 0;
                for (float value : a) if (value > threshold) expectedCount++;
                check(VectorAPILab.countAboveVector(a, threshold) == expectedCount,
                        "count with special values and tail");
            }
            float[] ones = new float[n], twos = new float[n];
            Arrays.fill(ones, 1f); Arrays.fill(twos, 2f);
            check(VectorAPILab.dotProductVector(ones, twos) == 2f * n,
                    "dot product for exactly representable noncancelling inputs");
        }
        for (boolean masked : new boolean[] {false, true}) {
            float[] sentinel = {42f, 43f};
            try {
                add(masked, new float[1], new float[2], sentinel);
                throw new AssertionError("length mismatch accepted");
            } catch (IllegalArgumentException expected) { checks++; }
            same(new float[] {42f, 43f}, sentinel, "mismatch before writes");
        }
        VectorSpecies<Integer> is = IntVector.SPECIES_PREFERRED;
        VectorMask<Integer> active = is.indexInRange(0, 1);
        IntVector x = IntVector.fromArray(is, new int[] {8}, 0, active);
        IntVector y = IntVector.fromArray(is, new int[] {2}, 0, active);
        int[] out = new int[is.length()];
        Arrays.fill(out, 99);
        try {
            x.div(y).intoArray(out, 0, active);
            throw new AssertionError("unmasked inactive divisor did not throw");
        } catch (ArithmeticException expected) { checks++; }
        check(Arrays.stream(out).allMatch(value -> value == 99), "throw before masked store");
        x.div(y, active).intoArray(out, 0, active);
        check(out[0] == 4 && Arrays.stream(out).skip(1).allMatch(value -> value == 99),
                "masked divide and sentinel preservation");
        try {
            x.div(IntVector.zero(is), active);
            throw new AssertionError("active zero divisor accepted");
        } catch (ArithmeticException expected) { checks++; }
        check(IntVector.broadcast(is, Integer.MAX_VALUE).add(1).lane(0) == Integer.MIN_VALUE,
                "integer vector addition wraps");
        int[] indexes = new int[is.length()];
        int[] bins = new int[is.length()];
        for (int i = 0; i < indexes.length; i++) indexes[i] = i;
        IntVector.fromArray(is, bins, 0, indexes, 0).add(1).intoArray(bins, 0, indexes, 0);
        check(Arrays.stream(bins).allMatch(value -> value == 1), "unique destinations");
        Arrays.fill(indexes, 0); Arrays.fill(bins, 0);
        IntVector.fromArray(is, bins, 0, indexes, 0).add(1).intoArray(bins, 0, indexes, 0);
        check(bins[0] == 1 && bins[0] != is.length(), "duplicate destination loses increments");

        float a = 0x1.000002p0f, b = 0x1.fffffcp-1f, c = -1f;
        FloatVector va = FloatVector.broadcast(VectorAPILab.FSPECIES, a);
        FloatVector vb = FloatVector.broadcast(VectorAPILab.FSPECIES, b);
        FloatVector vc = FloatVector.broadcast(VectorAPILab.FSPECIES, c);
        check(Float.floatToRawIntBits(va.mul(vb).add(vc).lane(0)) == 0, "separate operations");
        check(va.fma(vb, vc).lane(0) == -0x1p-46f, "fused result differs");
        int bound = VectorAPILab.FSPECIES.loopBound(Integer.MAX_VALUE);
        check(bound <= Integer.MAX_VALUE && (long) Integer.MAX_VALUE - bound < lanes,
                "upper loop bound without large allocation");
        System.out.printf("PASS checks=%d runtime=%s species=%s lengths=0..%d%n",
                checks, Runtime.version(), VectorAPILab.FSPECIES, 2 * lanes + 1);
        System.out.println("Functional checks only; no decoded instructions, benchmark or speedup claim.");
    }

    private static void add(boolean masked, float[] a, float[] b, float[] out) {
        if (masked) VectorAPILab.addArraysVectorMasked(a, b, out);
        else VectorAPILab.addArraysVector(a, b, out);
    }

    private static void same(float[] expected, float[] actual, String label) {
        if (expected.length != actual.length) throw new AssertionError(label + " length");
        for (int i = 0; i < expected.length; i++) {
            if (Float.floatToIntBits(expected[i]) != Float.floatToIntBits(actual[i]))
                throw new AssertionError(label + " at " + i);
        }
        checks++;
    }

    private static void check(boolean condition, String label) {
        if (!condition) throw new AssertionError(label);
        checks++;
    }
}
