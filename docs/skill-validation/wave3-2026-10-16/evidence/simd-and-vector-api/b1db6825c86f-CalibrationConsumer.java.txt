import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Properties;

final class CalibrationConsumer {
    private final String comparison;
    private final double absoluteTolerance;

    CalibrationConsumer(Path config) throws IOException {
        Properties p = new Properties();
        try (var input = Files.newInputStream(config)) { p.load(input); }
        comparison = p.getProperty("comparison");
        if (!comparison.equals("bitwise") && !comparison.equals("absolute"))
            throw new IllegalArgumentException("unknown comparison");
        absoluteTolerance = Double.parseDouble(p.getProperty("absoluteTolerance"));
        if (!Double.isFinite(absoluteTolerance) || absoluteTolerance < 0)
            throw new IllegalArgumentException("invalid tolerance");
    }

    boolean accepts(float expected, float actual) {
        if (!Float.isFinite(expected) || !Float.isFinite(actual)) return false;
        return switch (comparison) {
            case "bitwise" -> Float.floatToRawIntBits(expected) == Float.floatToRawIntBits(actual);
            case "absolute" -> Math.abs((double) expected - actual) <= absoluteTolerance;
            default -> throw new AssertionError(comparison);
        };
    }
}
