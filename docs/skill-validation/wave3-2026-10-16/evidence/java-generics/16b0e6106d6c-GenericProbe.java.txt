import java.lang.reflect.ParameterizedType;
import java.lang.reflect.Type;
import java.lang.reflect.TypeVariable;
import java.util.ArrayList;
import java.util.Collection;
import java.util.List;

public final class GenericProbe {
    private static int checks;
    static void check(boolean condition, String label) {
        if (!condition) throw new AssertionError(label);
        checks++;
    }
    static void expect(Class<? extends Throwable> expected, Runnable call, String label) {
        try { call.run(); }
        catch (Throwable actual) {
            if (!expected.isInstance(actual)) throw new AssertionError(label, actual);
            checks++;
            return;
        }
        throw new AssertionError(label + " did not throw " + expected.getName());
    }
    static <T> void copyFirst(List<? extends T> source, List<? super T> destination) {
        destination.add(source.get(0));
    }
    static <T extends Comparable<? super T>> T max(Collection<? extends T> values) {
        var it = values.iterator();
        T best = it.next();
        while (it.hasNext()) {
            T candidate = it.next();
            if (candidate.compareTo(best) > 0) best = candidate;
        }
        return best;
    }
    static class Base implements Comparable<Base> {
        final int rank;
        Base(int rank) { this.rank = rank; }
        public int compareTo(Base other) { return Integer.compare(rank, other.rank); }
    }
    static final class Sub extends Base { Sub(int rank) { super(rank); } }
    static <T> Type unresolvedToken() { return new TypeRef<List<T>>() {}.type(); }
    static void requiresObjectElements(ArrayList<Object> value) {}

    public static void main(String[] args) {
        System.out.println("runtime=" + Runtime.version() + "; compiler target=21");
        Attributes attributes = new Attributes();
        check(attributes.get(String.class).isEmpty(), "absent key");
        attributes.put(String.class, "billing");
        attributes.put(Integer.class, 7);
        check(attributes.get(String.class).orElseThrow().equals("billing"), "typed retrieval");
        check(attributes.get(Integer.class).orElseThrow() == 7, "independent class key");
        attributes.put(String.class, "shipping");
        check(attributes.get(String.class).orElseThrow().equals("shipping"), "replacement");
        check(attributes.get(CharSequence.class).isEmpty(), "exact key, no assignable search");
        attributes.put(String.class, null);
        check(attributes.get(String.class).isEmpty(), "stored null indistinguishable from absence");
        expect(ClassCastException.class, () -> attributes.put(int.class, 1), "primitive token");
        expect(NullPointerException.class, () -> attributes.put(null, "value"), "null key");

        ArrayList<String> mutable = new ArrayList<>(List.of("a", "b"));
        WildcardOps.swap(mutable, 0, 1);
        check(mutable.equals(List.of("b", "a")), "capture mutates same list safely");
        expect(UnsupportedOperationException.class,
                () -> WildcardOps.swap(List.of("a", "b"), 0, 1), "type acceptance is not writability");
        List<?> wildcard = mutable;
        wildcard.clear();
        check(mutable.isEmpty(), "wildcard does not mean read-only");
        List<Double> values = List.of(2.5);
        List<Number> destination = new ArrayList<>();
        copyFirst(values, destination);
        check(destination.get(0) == values.get(0), "PECS preserves original object");
        Sub high = new Sub(9);
        Sub result = max(List.of(new Sub(1), high));
        check(result == high, "inherited Comparable<Base> accepted for Sub");

        List<?>[] arrays = new List<?>[2];
        arrays[0] = List.of("text");
        arrays[1] = List.of(42);
        check(arrays[1].get(0).equals(42), "reifiable wildcard array");
        Object[] strings = new String[1];
        expect(ArrayStoreException.class, () -> strings[0] = 42, "reified array store");
        Object castList = List.class.cast(values);
        check(castList == values, "class token does not validate list argument");
        Type stringToken = new TypeRef<List<String>>() {}.type();
        check(((ParameterizedType) stringToken).getActualTypeArguments()[0] == String.class,
                "anonymous subclass declaration captures String");
        Type genericToken = unresolvedToken();
        check(((ParameterizedType) genericToken).getActualTypeArguments()[0] instanceof TypeVariable<?>,
                "generic method token retains T");
        List<String> known = new ArrayList<>();
        check(known instanceof ArrayList<String>, "checkable parameterized type test");
        var inferred = new ArrayList<>();
        requiresObjectElements(inferred);
        inferred.add("text");
        inferred.add(42);
        check(inferred.size() == 2, "var diamond has Object elements");
        System.out.println("PASS: " + checks + " targeted contract checks");
    }
}
