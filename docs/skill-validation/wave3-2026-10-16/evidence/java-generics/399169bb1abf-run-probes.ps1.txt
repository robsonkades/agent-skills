$checkResults = [System.Collections.Generic.List[object]]::new()
function Invoke-Probe {
    param([string]$Name, [string]$Tool, [string[]]$ToolArguments, [int]$ExpectedExit = 0)
    $output = & $Tool @ToolArguments 2>&1
    $status = $LASTEXITCODE
    $output | Set-Content -LiteralPath (Join-Path $base ($Name + '.log'))
    $record = [pscustomobject]@{name=$Name;tool=$Tool;arguments=$ToolArguments;cwd=$base;exitCode=$status;expectedExit=$ExpectedExit}
    $checkResults.Add($record)
    if ($status -ne $ExpectedExit) { throw ("Unexpected status for " + $Name + ": " + $status + [Environment]::NewLine + ($output -join [Environment]::NewLine)) }
    $record | ConvertTo-Json -Compress
}
