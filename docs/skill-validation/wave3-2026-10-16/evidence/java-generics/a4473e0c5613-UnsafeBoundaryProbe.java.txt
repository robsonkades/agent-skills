import java.util.List;

public final class UnsafeBoundaryProbe {
    // Intentionally unsafe controls: warnings are retained, not suppressed.
    static <T> T dangerous(List<T>... lists) {
        Object[] array = lists;
        array[0] = List.of(42);
        return lists[0].get(0);
    }
    static <T> T[] leakedArray() { return (T[]) new Object[1]; }
    public static void main(String[] args) {
        Attributes attributes = new Attributes();
        attributes.put(String.class, "preserved");
        Class raw = String.class;
        try {
            attributes.put(raw, 42);
            throw new AssertionError("raw mismatch was accepted");
        } catch (ClassCastException expected) {
            if (!attributes.get(String.class).orElseThrow().equals("preserved")) {
                throw new AssertionError("failed insertion changed prior value");
            }
            System.out.println("PASS: raw key mismatch rejected before mutation");
        }
        try {
            String value = dangerous(List.of("safe"));
            throw new AssertionError("pollution escaped caller cast: " + value);
        } catch (ClassCastException expected) {
            System.out.println("PASS: unsafe varargs caught at caller String cast");
        }
        Object[] broad = UnsafeBoundaryProbe.<String>leakedArray();
        if (broad.getClass() != Object[].class) throw new AssertionError("wrong control class");
        try {
            String[] narrow = UnsafeBoundaryProbe.<String>leakedArray();
            throw new AssertionError("array runtime class promise was accepted: " + narrow);
        } catch (ClassCastException expected) {
            System.out.println("PASS: Object[] consumer works, String[] consumer rejects exposed array");
        }
    }
}
