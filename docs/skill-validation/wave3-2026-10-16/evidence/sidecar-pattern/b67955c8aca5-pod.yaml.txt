apiVersion: v1
kind: ConfigMap

  name: app-settings

  application.yaml: |
    version: blue

apiVersion: v1
kind: Pod
metadata:
  name: config-reload-example
spec:
  initContainers:
    - name: config-reloader
      image: registry.example/config-reloader:1.0
      restartPolicy: Always
      volumeMounts:
        - name: app-config
          mountPath: /watch
          readOnly: true
  containers:
    - name: api
      image: registry.example/api:1.0
      args: ["--config=/etc/app-config/application.yaml"]
      volumeMounts:
        - name: app-config
          mountPath: /etc/app-config
          readOnly: true
  volumes:
    - name: app-config
      configMap:
        name: app-settings
