function yamlDocuments(text,label) {
  const documents=parseAllDocuments(text,{strict:true,uniqueKeys:true});
  for (const doc of documents) {
    assert.deepEqual(doc.errors,[],label+': YAML parse errors');
    assert.deepEqual(doc.warnings,[],label+': YAML parse warnings');
  }
  return documents.map(document=>document.toJS());
}

function run(label,command,args) {
  const result=spawnSync(command,args,{cwd:repository,windowsHide:true,timeout:30000,encoding:'utf8'});
  const stdoutPath=resolve(root,label+'.stdout.log');
  const stderrPath=resolve(root,label+'.stderr.log');
  writeFileSync(stdoutPath,result.stdout??'');
  writeFileSync(stderrPath,result.stderr??'');
  results.processes.push({label,command,args,cwd:repository,status:result.status,signal:result.signal,
    error:result.error?.message??null,stdoutPath,stderrPath});
  writeFileSync(resolve(root,'validation-results.json'),JSON.stringify(results,null,2)+'\n');
  assert.equal(result.status,0,label+': '+result.stdout+'\n'+result.stderr);
  return result.stdout??'';
}
