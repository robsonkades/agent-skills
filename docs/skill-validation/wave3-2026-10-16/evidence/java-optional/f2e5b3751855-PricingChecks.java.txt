import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.function.Supplier;

public class PricingChecks {
    private static int checks;
    record Price(BigDecimal amount) {
        Price { Objects.requireNonNull(amount, "amount"); }
    }
    static final class MissingPriceException extends RuntimeException {
        private static final long serialVersionUID = 1L;
        MissingPriceException(String sku) { super(sku); }
    }
    @interface Nullable {}
    interface Promotion { Optional<Price> priceFor(String sku); }
    interface Contract { Optional<Price> priceFor(String customerId, String sku); }
    interface Catalogue { Optional<Price> listPrice(String sku); }
    static <T> Supplier<T> eager(T value) { return () -> value; }
    static final class Pricing {
        private final Promotion promotions;
        private final Contract contracts;
        private final Catalogue catalogue;
        Pricing(Supplier<Optional<Price>> p, Supplier<Optional<Price>> c,
                Supplier<Optional<Price>> l, List<String> calls) {
            promotions = sku -> { calls.add("promotion"); return p.get(); };
            contracts = (customer, sku) -> { calls.add("contract"); return c.get(); };
            catalogue = sku -> { calls.add("catalogue"); return l.get(); };
        }
public BigDecimal effectivePrice(String customerId, String sku) {
    return promotions.priceFor(sku)
            .or(() -> contracts.priceFor(customerId, sku))  // runs only if no promotion
            .or(() -> catalogue.listPrice(sku))             // runs only if no contract
            .map(Price::amount)
            .orElseThrow(() -> new MissingPriceException(sku));
}
    }
    static final class CatalogueAdapter {
private final Map<String, Price> index;

CatalogueAdapter(Map<String, Price> loadedIndex) {
    this.index = Map.copyOf(loadedIndex);       // reject invalid entries at load time
}

@Nullable
private Price lookup(String sku) {              // private, hot, locally checked
    return index.get(sku);
}

public Optional<Price> listPrice(String sku) {  // Optional at the boundary only
    return Optional.ofNullable(lookup(sku));
}
    }
    static Optional<Price> price(String amount) {
        return Optional.of(new Price(new BigDecimal(amount)));
    }
    static void equal(Object expected, Object actual, String label) {
        if (!Objects.equals(expected, actual)) {
            throw new AssertionError(label + ": expected=" + expected + ", actual=" + actual);
        }
        checks++;
    }
    static Throwable rejects(Class<? extends Throwable> type, Runnable action) {
        try { action.run(); }
        catch (Throwable failure) {
            if (!type.isInstance(failure)) throw new AssertionError("Wrong failure", failure);
            checks++;
            return failure;
        }
        throw new AssertionError("Expected " + type.getSimpleName());
    }
    static void tier(Optional<Price> p, Optional<Price> c, Optional<Price> l,
                     String expected, List<String> expectedCalls) {
        List<String> calls = new ArrayList<>();
        Pricing pricing = new Pricing(() -> p, () -> c, () -> l, calls);
        equal(new BigDecimal(expected), pricing.effectivePrice("customer", "sku"), "price");
        equal(expectedCalls, calls, "consulted sources");
    }
    public static void main(String[] args) {
        tier(price("9"), price("8"), price("7"), "9", List.of("promotion"));
        tier(Optional.empty(), price("8"), price("7"), "8", List.of("promotion", "contract"));
        tier(Optional.empty(), Optional.empty(), price("7"), "7",
             List.of("promotion", "contract", "catalogue"));

        List<String> missingCalls = new ArrayList<>();
        Pricing missing = new Pricing(Optional::empty, Optional::empty, Optional::empty, missingCalls);
        rejects(MissingPriceException.class, () -> missing.effectivePrice("customer", "sku"));
        equal(List.of("promotion", "contract", "catalogue"), missingCalls, "missing source order");

        List<String> failedCalls = new ArrayList<>();
        RuntimeException outage = new IllegalStateException("promotion unavailable");
        Pricing failed = new Pricing(() -> { throw outage; }, () -> price("8"), () -> price("7"), failedCalls);
        equal(outage, rejects(IllegalStateException.class,
              () -> failed.effectivePrice("customer", "sku")), "same source failure");
        equal(List.of("promotion"), failedCalls, "no fallback after source failure");

        List<String> invalidCalls = new ArrayList<>();
        Pricing invalid = new Pricing(Optional::empty, () -> null, () -> price("7"), invalidCalls);
        rejects(NullPointerException.class, () -> invalid.effectivePrice("customer", "sku"));
        equal(List.of("promotion", "contract"), invalidCalls, "no fallback after null Optional");
        rejects(NullPointerException.class, () -> new Price(null));

        Map<String, Price> invalidIndex = new HashMap<>();
        invalidIndex.put("sku", null);
        rejects(NullPointerException.class, () -> new CatalogueAdapter(invalidIndex));
        Map<String, Price> nullKeyIndex = new HashMap<>();
        nullKeyIndex.put(null, price("9").orElseThrow());
        rejects(NullPointerException.class, () -> new CatalogueAdapter(nullKeyIndex));

        Map<String, Price> loaded = new HashMap<>();
        loaded.put("sku", price("9").orElseThrow());
        CatalogueAdapter snapshot = new CatalogueAdapter(loaded);
        loaded.put("sku", price("1").orElseThrow());
        loaded.put("new", price("2").orElseThrow());
        equal(new BigDecimal("9"), snapshot.listPrice("sku").orElseThrow().amount(), "snapshot value");
        equal(Optional.empty(), snapshot.listPrice("new"), "snapshot excludes later key");
        equal(Optional.empty(), snapshot.listPrice("unknown"), "valid absent key");
        System.out.println("Passed " + checks + " pricing checks");
        System.out.println("runtime=" + Runtime.version());
    }
}
