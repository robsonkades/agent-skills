import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { spawnSync } from 'node:child_process';
import assert from 'node:assert/strict';

const out = 'C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL/java-object-contracts';
const java = 'C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe';
const javac = 'C:/Users/robso/.jdks/temurin-25.0.3/bin/javac.exe';
const javap = 'C:/Users/robso/.jdks/temurin-25.0.3/bin/javap.exe';
const runs = [];
await mkdir(`${out}/logs`, { recursive: true });
await mkdir(`${out}/classes`, { recursive: true });

function run(name, executable, args, expectedExit, expectedPattern) {
  const result = spawnSync(executable, args, { cwd: out, encoding: 'utf8', timeout: 30000 });
  const stdout = result.stdout ?? '';
  const stderr = result.stderr ?? '';
  const entry = { name, executable, args, cwd: out, expectedExit,
    expectedPattern: expectedPattern?.source, actualExit: result.status, signal: result.signal,
    error: result.error?.message, stdout, stderr,
    stdoutPath: `${out}/logs/${name}.stdout.txt`, stderrPath: `${out}/logs/${name}.stderr.txt` };
  entry.passed = result.status === expectedExit && (!expectedPattern || expectedPattern.test(stdout + stderr));
  runs.push(entry);
  return entry;
}

try {
  assert(run('java-version', java, ['-version'], 0, /25\.0\.3/).passed);
  assert(run('compile-contracts', javac, ['--release', '16', '-d', `${out}/classes`, `${out}/ContractsProbe.java`], 0).passed);
  assert(run('run-contracts', java, ['-cp', `${out}/classes`, 'ContractsProbe'], 0, /SUMMARY checks=29/).passed);

  const common = `package api;\npublic final class Key {\n  private final String id;\n  public Key(String id) { this.id = java.util.Objects.requireNonNull(id); }\n  @Override public int hashCode() { return id.hashCode(); }\n`;
  const typed = `  public boolean equals(Key other) { return other != null && id.equals(other.id); }\n`;
  const object = `  @Override public boolean equals(Object other) { return other instanceof Key key && id.equals(key.id); }\n`;
  const bridge = `  public boolean equals(Key other) { return equals((Object) other); }\n`;
  const variants = {
    original: common + typed + '}\n',
    replaced: common + object + '}\n',
    bridged: common + object + bridge + '}\n',
    badOverride: common + '  @Override\n' + typed + '}\n',
  };
  for (const [name, source] of Object.entries(variants)) {
    await mkdir(`${out}/binary/${name}/src/api`, { recursive: true });
    await mkdir(`${out}/binary/${name}/classes`, { recursive: true });
    await writeFile(`${out}/binary/${name}/src/api/Key.java`, source);
    const expected = name === 'badOverride' ? 1 : 0;
    assert(run(`compile-${name}`, javac,
      ['--release', '16', '-d', `${out}/binary/${name}/classes`, `${out}/binary/${name}/src/api/Key.java`],
      expected, expected === 1 ? /does not override/ : undefined).passed);
  }
  const client = `import api.Key;\npublic final class LegacyClient {\n  public static void main(String[] args) {\n    boolean equal = new Key("A-1").equals(new Key("A-1"));\n    if (!equal) throw new AssertionError("typed equality");\n    System.out.println("typed=true");\n  }\n}\n`;
  const consumer = `import api.Key;\nimport java.util.*;\npublic final class ValueConsumer {\n  public static void main(String[] args) {\n    var one = new Key("A-1");\n    var two = new Key("A-1");\n    Map<Key, String> map = new HashMap<>();\n    map.put(one, "stored");\n    if (!Objects.equals(one, two) || !"stored".equals(map.get(two)))\n      throw new AssertionError("equal logical keys must resolve");\n    System.out.println("lookup=stored");\n  }\n}\n`;
  await mkdir(`${out}/binary/clients`, { recursive: true });
  await writeFile(`${out}/binary/LegacyClient.java`, client);
  await writeFile(`${out}/binary/ValueConsumer.java`, consumer);
  assert(run('compile-client-against-original', javac, ['--release', '16', '-cp', `${out}/binary/original/classes`,
    '-d', `${out}/binary/clients`, `${out}/binary/LegacyClient.java`, `${out}/binary/ValueConsumer.java`], 0).passed);
  assert(run('legacy-bytecode', javap, ['-c', '-classpath', `${out}/binary/clients`, 'LegacyClient'], 0,
    /api\/Key\.equals:\(Lapi\/Key;\)Z/).passed);

  for (const name of ['original', 'replaced', 'bridged']) {
    const cp = `${out}/binary/${name}/classes;${out}/binary/clients`;
    assert(run(`legacy-on-${name}`, java, ['-cp', cp, 'LegacyClient'], name === 'replaced' ? 1 : 0,
      name === 'replaced' ? /NoSuchMethodError/ : /typed=true/).passed);
    assert(run(`consumer-on-${name}`, java, ['-cp', cp, 'ValueConsumer'], name === 'original' ? 1 : 0,
      name === 'original' ? /equal logical keys must resolve/ : /lookup=stored/).passed);
  }
  await mkdir(`${out}/binary/recompiled-clients`, { recursive: true });
  assert(run('recompile-client-against-replacement', javac,
    ['--release', '16', '-cp', `${out}/binary/replaced/classes`, '-d', `${out}/binary/recompiled-clients`,
      `${out}/binary/LegacyClient.java`], 0).passed);
  assert(run('recompiled-client-on-replacement', java,
    ['-cp', `${out}/binary/replaced/classes;${out}/binary/recompiled-clients`, 'LegacyClient'], 0, /typed=true/).passed);
} finally {
  for (const entry of runs) {
    await writeFile(entry.stdoutPath, entry.stdout);
    await writeFile(entry.stderrPath, entry.stderr);
  }
  await writeFile(`${out}/probes.json`, JSON.stringify({
    scope: 'Technical Java controls, not agent evaluation or universal proof',
    release: 16, runtime: 'Temurin 25.0.3+9', runs,
  }, null, 2) + '\n');
}
assert(runs.every(run => run.passed));
const classBytes = await readFile(`${out}/classes/ContractsProbe.class`);
assert.equal(classBytes.readUInt16BE(6), 60);
console.log(JSON.stringify({ checks: 29, commands: runs.length, expectedNegativeControls: runs.filter(run => run.expectedExit !== 0).length,
  classMajorVersion: classBytes.readUInt16BE(6), allPassed: true }, null, 2));
