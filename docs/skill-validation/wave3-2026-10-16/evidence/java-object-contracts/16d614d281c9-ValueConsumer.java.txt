import api.Key;
import java.util.*;
public final class ValueConsumer {
  public static void main(String[] args) {
    var one = new Key("A-1");
    var two = new Key("A-1");
    Map<Key, String> map = new HashMap<>();
    map.put(one, "stored");
    if (!Objects.equals(one, two) || !"stored".equals(map.get(two)))
      throw new AssertionError("equal logical keys must resolve");
    System.out.println("lookup=stored");
  }
}
