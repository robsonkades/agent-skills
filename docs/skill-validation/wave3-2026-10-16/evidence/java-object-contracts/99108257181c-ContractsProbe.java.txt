import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.HashMap;
import java.util.HashSet;
import java.util.IdentityHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.TreeSet;

public final class ContractsProbe {
    private static int checks;

    private static void check(boolean condition, String label) {
        if (!condition) throw new AssertionError(label);
        checks++;
        System.out.println("PASS " + label);
    }

    static final class OverloadedKey {
        private final int value;
        OverloadedKey(int value) { this.value = value; }
        public boolean equals(OverloadedKey other) {
            return other != null && value == other.value;
        }
        @Override public int hashCode() { return Integer.hashCode(value); }
    }

    static final class ValueKey {
        private final int value;
        ValueKey(int value) { this.value = value; }
        @Override public boolean equals(Object other) {
            return other instanceof ValueKey key && value == key.value;
        }
        @Override public int hashCode() { return Integer.hashCode(value); }
    }

    record Measurement(double value) { }
    record RawPayload(byte[] bytes) { }
    record CopyOnly(byte[] bytes) {
        CopyOnly { bytes = bytes.clone(); }
        @Override public byte[] bytes() { return bytes.clone(); }
    }
    record ContentPayload(byte[] bytes) {
        ContentPayload { bytes = bytes.clone(); }
        @Override public byte[] bytes() { return bytes.clone(); }
        @Override public boolean equals(Object other) {
            return other instanceof ContentPayload payload && Arrays.equals(bytes, payload.bytes);
        }
        @Override public int hashCode() { return Arrays.hashCode(bytes); }
        @Override public String toString() { return "ContentPayload[redacted]"; }
    }

    record RankedItem(Integer score, long id) { }
    record UnsafeCredentials(String username, String password) { }
    record Credentials(String username, String password) {
        @Override public String toString() { return "Credentials[redacted]"; }
    }

    static class ShallowClone implements Cloneable {
        static int constructorCalls;
        final List<String> values = new ArrayList<>();
        ShallowClone() { constructorCalls++; }
        @Override public ShallowClone clone() {
            try { return (ShallowClone) super.clone(); }
            catch (CloneNotSupportedException e) { throw new AssertionError(e); }
        }
    }
    static final class CloneSubtype extends ShallowClone { }
    static final class ConstructorClone implements Cloneable {
        static int constructorCalls;
        ConstructorClone() { constructorCalls++; }
        @Override public ConstructorClone clone() { return new ConstructorClone(); }
    }

    record Leg(String name) { }
    static final class Route {
        private final List<Leg> legs;
        Route(Route other) { this(other.legs); }
        Route(List<Leg> legs) { this.legs = List.copyOf(legs); }
        Route withLeg(Leg leg) {
            var next = new ArrayList<>(legs);
            next.add(leg);
            return new Route(next);
        }
    }

    public static void main(String[] args) {
        var a = new OverloadedKey(7);
        var b = new OverloadedKey(7);
        check(a.equals(b), "typed overload sees value equality");
        check(a.hashCode() == b.hashCode(), "same hash does not establish Object equality");
        check(!((Object) a).equals(b), "Object call retains reference equality");
        check(!Objects.equals(a, b), "Objects.equals invokes Object signature");
        Map<OverloadedKey, String> wrong = new HashMap<>();
        wrong.put(a, "stored");
        check(wrong.get(b) == null, "HashMap does not select equals(OverloadedKey)");

        var x = new ValueKey(7);
        var y = new ValueKey(7);
        check(((Object) x).equals(y) && Objects.equals(x, y), "override reaches consumer Object calls");
        check(x.equals(x) && !x.equals(null) && !x.equals("7"), "override identity null and unrelated type");
        Map<ValueKey, String> values = new HashMap<>();
        values.put(x, "stored");
        check("stored".equals(values.get(y)), "equal distinct keys resolve in HashMap");
        Map<ValueKey, String> identities = new IdentityHashMap<>();
        identities.put(x, "first");
        check(identities.get(y) == null, "IdentityHashMap intentionally ignores value equality");
        identities.put(y, "second");
        check(identities.size() == 2 && "first".equals(identities.get(x)), "identity map preserves distinct equal representatives");

        check(new Measurement(Double.NaN).equals(new Measurement(Double.NaN)), "record NaN equivalence");
        check(!new Measurement(0.0).equals(new Measurement(-0.0)), "record signed zero distinction");
        check(!new RawPayload(new byte[] {1}).equals(new RawPayload(new byte[] {1})), "record arrays retain identity semantics");
        var copyOnly = new CopyOnly(new byte[] {1});
        check(!copyOnly.equals(new CopyOnly(copyOnly.bytes())), "copying alone violates record reconstruction");
        byte[] source = {1, 2};
        var content = new ContentPayload(source);
        var equivalent = new ContentPayload(new byte[] {1, 2});
        check(content.equals(equivalent) && content.hashCode() == equivalent.hashCode(), "array content equality and matching hash");
        check(content.equals(new ContentPayload(content.bytes())), "content record reconstruction");
        source[0] = 9;
        content.bytes()[0] = 8;
        check(content.equals(equivalent), "constructor and accessor preserve owned array contents");
        check(content.toString().equals(equivalent.toString()), "custom equal records have matching diagnostics");
        check(new HashSet<>(List.of(content)).contains(equivalent), "content key lookup in hash collection");

        var amounts = List.of(new BigDecimal("1.0"), new BigDecimal("1.00"));
        check(new HashSet<>(amounts).size() == 2 && new TreeSet<>(amounts).size() == 1,
                "sorted collection uses ordering rather than equals");
        Comparator<RankedItem> byRank = Comparator.comparing(RankedItem::score,
                Comparator.nullsLast(Comparator.reverseOrder())).thenComparingLong(RankedItem::id);
        var ranking = List.of(new RankedItem(1, 5), new RankedItem(null, 9),
                new RankedItem(2, 3), new RankedItem(2, 1));
        var sorted = new ArrayList<>(ranking);
        sorted.sort(byRank);
        check(sorted.stream().map(RankedItem::id).toList().equals(List.of(1L, 3L, 5L, 9L)),
                "descending score null last ascending tie");
        boolean laws = true;
        for (var first : ranking) for (var second : ranking) for (var third : ranking) {
            int ab = Integer.signum(byRank.compare(first, second));
            int ba = Integer.signum(byRank.compare(second, first));
            int bc = Integer.signum(byRank.compare(second, third));
            int ac = Integer.signum(byRank.compare(first, third));
            laws &= ab == -ba && !(ab > 0 && bc > 0 && ac <= 0) && !(ab == 0 && ac != bc);
        }
        check(laws, "comparator laws on the four-value fixture including ties and null score");

        String username = "user\nFORGED";
        String password = "secret\r\nTOKEN";
        String unsafe = new UnsafeCredentials(username, password).toString();
        check(unsafe.contains(username) && unsafe.contains(password), "generated diagnostic negative control discloses synthetic secrets");
        String safe = new Credentials(username, password).toString();
        check(!safe.contains(username) && !safe.contains(password) && !safe.contains("\n"), "redacted diagnostics exclude hostile components");

        var original = new CloneSubtype();
        original.values.add("shared");
        int before = ShallowClone.constructorCalls;
        var clone = original.clone();
        check(clone != original && clone.getClass() == original.getClass(), "Object.clone creates fresh same-runtime-class instance");
        check(ShallowClone.constructorCalls == before && clone.values == original.values,
                "Object.clone does not rerun constructor and shares reference fields");
        clone.values.add("alias");
        check(original.values.size() == 2, "shallow clone does not grant independent mutable contents");
        var constructed = new ConstructorClone();
        int count = ConstructorClone.constructorCalls;
        check(constructed.clone() != constructed && ConstructorClone.constructorCalls == count + 1,
                "an override named clone can invoke a constructor");
        var legs = new ArrayList<>(List.of(new Leg("first")));
        var route = new Route(legs);
        legs.add(new Leg("external"));
        var copy = new Route(route);
        var extended = route.withLeg(new Leg("second"));
        check(route.legs.size() == 1 && copy.legs.equals(route.legs) && extended.legs.size() == 2,
                "copy constructor and wither preserve immutable-element snapshot");
        System.out.println("SUMMARY checks=" + checks);
    }
}
