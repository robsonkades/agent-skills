async function visit(relative) {
  for (const entry of await readdir(`${repo}/${relative}`, { withFileTypes: true })) {
    const next = `${relative}/${entry.name}`;
    if (entry.isDirectory()) await visit(next);
    else {
      const bytes = await readFile(`${repo}/${next}`);
      files.push({ path: next, sha256: sha256(bytes), bytes: bytes.length, content: bytes.toString('utf8') });
    }
  }
}
