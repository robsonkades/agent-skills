import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.security.GeneralSecurityException;
import java.util.Arrays;
import javax.crypto.AEADBadTagException;
import javax.crypto.Cipher;
import javax.crypto.CipherOutputStream;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;

public final class AeadStreamProbe {
    private static int checks;
    private record Outcome(boolean accepted, int publications, byte[] output, String failure) {}

    private static Cipher cipher(int mode, SecretKey key, byte[] nonce, byte[] aad)
            throws GeneralSecurityException {
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        cipher.init(mode, key, new GCMParameterSpec(128, nonce));
        cipher.updateAAD(aad);
        return cipher;
    }

    // Model the application boundary: successful return publishes the result.
    private static Outcome consume(byte[] ciphertext, SecretKey key, byte[] nonce, byte[] aad,
                                   boolean stream) {
        try {
            Cipher cipher = cipher(Cipher.DECRYPT_MODE, key, nonce, aad);
            byte[] result;
            if (stream) {
                ByteArrayOutputStream sink = new ByteArrayOutputStream();
                try (CipherOutputStream output = new CipherOutputStream(sink, cipher)) {
                    output.write(ciphertext);
                }
                result = sink.toByteArray();
            } else {
                result = cipher.doFinal(ciphertext);
            }
            return new Outcome(true, 1, result, "");
        } catch (GeneralSecurityException | IOException e) {
            return new Outcome(false, 0, new byte[0], e.getClass().getSimpleName());
        }
    }

    private static void accept(String label, Outcome result, byte[] expected) {
        if (!result.accepted() || result.publications() != 1
                || !result.failure().isEmpty() || !Arrays.equals(result.output(), expected)) {
            throw new AssertionError(label + ": acceptance mismatch");
        }
        emit(label, result, "legitimate-control");
    }

    private static void reject(String label, Outcome result) {
        if (result.accepted() || result.publications() != 0 || result.output().length != 0
                || !AEADBadTagException.class.getSimpleName().equals(result.failure())) {
            throw new AssertionError(label + ": failed to reject an unauthenticated message");
        }
        emit(label, result, "adversarial-control");
    }

    private static void emit(String label, Outcome result, String observation) {
        checks++;
        System.out.printf("{\"case\":\"%s\",\"accepted\":%s,\"publications\":%d,"
                + "\"outputBytes\":%d,\"failure\":\"%s\",\"observation\":\"%s\"}%n",
                label, result.accepted(), result.publications(), result.output().length,
                result.failure(), observation);
    }

    private static byte[] nonce(int value) {
        byte[] result = new byte[12];
        result[11] = (byte) value;
        return result;
    }

    public static void main(String[] args) throws Exception {
        KeyGenerator generator = KeyGenerator.getInstance("AES");
        generator.init(128);
        SecretKey key = generator.generateKey(); // Ephemeral, synthetic fixture only.
        byte[] aad = "tenant-fixture:record-fixture:v1".getBytes(StandardCharsets.UTF_8);
        byte[] plaintext = "synthetic authenticated fixture; no real credentials"
                .getBytes(StandardCharsets.UTF_8);
        byte[] dataNonce = nonce(1);
        Cipher encryption = cipher(Cipher.ENCRYPT_MODE, key, dataNonce, aad);
        byte[] valid = encryption.doFinal(plaintext);
        System.out.printf("{\"runtime\":\"%s\",\"vendor\":\"%s\",\"provider\":\"%s %s\"}%n",
                System.getProperty("java.runtime.version"), System.getProperty("java.vendor"),
                encryption.getProvider().getName(), encryption.getProvider().getVersionStr());

        accept("direct-valid", consume(valid, key, dataNonce, aad, false), plaintext);
        accept("output-stream-valid", consume(valid, key, dataNonce, aad, true), plaintext);

        byte[] badTag = valid.clone();
        badTag[badTag.length - 1] ^= 1;
        reject("direct-tampered-tag", consume(badTag, key, dataNonce, aad, false));

        Outcome swallowed = consume(badTag, key, dataNonce, aad, true);
        if (!swallowed.accepted() || swallowed.publications() != 1
                || !swallowed.failure().isEmpty() || swallowed.output().length != 0) {
            throw new AssertionError("Expected SunJCE CipherOutputStream counterexample absent");
        }
        emit("output-stream-tampered-tag", swallowed, "counterexample-reproduced");

        byte[] badBody = valid.clone();
        badBody[0] ^= 1;
        reject("direct-tampered-ciphertext", consume(badBody, key, dataNonce, aad, false));

        byte[] badAad = aad.clone();
        badAad[0] ^= 1;
        reject("direct-wrong-aad", consume(valid, key, dataNonce, badAad, false));

        byte[] wrongKeyBytes = key.getEncoded();
        wrongKeyBytes[0] ^= 1;
        SecretKey wrongKey = new SecretKeySpec(wrongKeyBytes, "AES");
        reject("direct-wrong-key", consume(valid, wrongKey, dataNonce, aad, false));

        byte[] emptyNonce = nonce(2);
        byte[] empty = cipher(Cipher.ENCRYPT_MODE, key, emptyNonce, aad).doFinal(new byte[0]);
        accept("direct-valid-empty", consume(empty, key, emptyNonce, aad, false), new byte[0]);
        accept("output-stream-valid-empty", consume(empty, key, emptyNonce, aad, true), new byte[0]);

        byte[] encryptionNonce = nonce(3);
        ByteArrayOutputStream ciphertext = new ByteArrayOutputStream();
        try (CipherOutputStream output = new CipherOutputStream(ciphertext,
                cipher(Cipher.ENCRYPT_MODE, key, encryptionNonce, aad))) {
            output.write(plaintext);
        }
        accept("output-stream-encryption-round-trip",
                consume(ciphertext.toByteArray(), key, encryptionNonce, aad, false), plaintext);

        if (checks != 10) throw new AssertionError("Unexpected check count: " + checks);
        System.out.printf("{\"checks\":%d,\"result\":\"PASS\"}%n", checks);
    }
}
