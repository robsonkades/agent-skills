import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { spawnSync } from 'node:child_process';
import assert from 'node:assert/strict';
const root = import.meta.dirname;
const javaHome = 'C:/Users/robso/.jdks/temurin-25.0.3';
const env = { ...process.env };
for (const key of ['JDK_JAVA_OPTIONS', 'JAVA_TOOL_OPTIONS', '_JAVA_OPTIONS']) delete env[key];
mkdirSync(resolve(root, 'classes'), { recursive: true });
const runs = [];
function run(name, command, args) {
  const result = spawnSync(command, args, {
    cwd: root, env, windowsHide: true, timeout: 30000, encoding: 'utf8'
  });
  const stdoutPath = resolve(root, name + '.stdout.log');
  const stderrPath = resolve(root, name + '.stderr.log');
  writeFileSync(stdoutPath, result.stdout ?? '');
  writeFileSync(stderrPath, result.stderr ?? '');
  runs.push({ name, command, args, cwd: root, status: result.status, signal: result.signal,
    error: result.error?.message ?? null, stdoutPath, stderrPath });
  writeFileSync(resolve(root, 'technical-runs.json'), JSON.stringify({ node: process.version, runs }, null, 2) + '\n');
  assert.equal(result.status, 0, name + ': ' + result.error + '\n' + result.stderr);
  return result.stdout ?? '';
}
run('java-version', javaHome + '/bin/java.exe', ['-version']);
run('javac-release21', javaHome + '/bin/javac.exe', ['--release', '21', '-d', 'classes', 'AeadStreamProbe.java']);
const output = run('aead-probe', javaHome + '/bin/java.exe', ['-cp', 'classes', 'AeadStreamProbe']);
const lines = output.trim().split(/\r?\n/).map(line => JSON.parse(line));
const conclusion = lines.at(-1);
assert.equal(conclusion.checks, 10);
assert.equal(conclusion.result, 'PASS');
const classFile = readFileSync(resolve(root, 'classes/AeadStreamProbe.class'));
assert.equal(classFile.readUInt16BE(6), 65, 'Java 21 class-file target');
writeFileSync(resolve(root, 'technical-results.json'), JSON.stringify({
  classification: 'executed technical API/consumer checks, not agent behavioral evaluation',
  compileTarget: 'Java 21 (--release 21; class file major 65)',
  runtime: lines[0],
  checks: lines.slice(1, -1),
  result: conclusion,
  allSpawnedProcessesExited: true
}, null, 2) + '\n');
process.stdout.write(output);
