import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { resolve } from 'node:path';

const root = fileURLToPath(new URL('.', import.meta.url));
const fixture = JSON.parse(readFileSync(resolve(root, 'window-fixture.json'), 'utf8'));
const packetTotal = fixture.delivered - fixture.old_delivered;
const packetMarked = fixture.delivered_ce - fixture.old_delivered_ce;
const byteTotal = fixture.acked_bytes_total;
const byteMarked = fixture.acked_bytes_ecn;
assert.equal(fixture.deliveryGroups.reduce((sum, group) => sum + group.packets, 0), packetTotal);
assert.equal(fixture.deliveryGroups.filter(group => group.eceReported).reduce((sum, group) => sum + group.packets, 0), packetMarked);
assert.equal(fixture.deliveryGroups.reduce((sum, group) => sum + group.packets * group.payloadBytesPerPacket, 0), byteTotal);
assert.equal(fixture.deliveryGroups.filter(group => group.eceReported).reduce((sum, group) => sum + group.packets * group.payloadBytesPerPacket, 0), byteMarked);
assert.equal(fixture.dctcp_alpha, 0, 'Control intentionally uses zero prior state, so both source versions have zero decay contribution');
const scale = 1n << BigInt(10 - fixture.dctcp_shift_g);
const packetAlpha = Number(BigInt(packetMarked) * scale / BigInt(packetTotal));
const byteAlpha = Number(BigInt(byteMarked) * scale / BigInt(byteTotal));
assert.equal(packetAlpha, 12);
assert.equal(byteAlpha, 2);
const packetDiagnostics = { dctcp_ab_ecn: fixture.mss_cache * packetMarked, dctcp_ab_tot: fixture.mss_cache * packetTotal };
const byteDiagnostics = { dctcp_ab_ecn: byteMarked, dctcp_ab_tot: byteTotal };
assert.deepEqual(packetDiagnostics, { dctcp_ab_ecn: 30000, dctcp_ab_tot: 150000 });
assert.notDeepEqual(packetDiagnostics, byteDiagnostics);
const results = {
  classification: 'Source-derived arithmetic control only; not execution of Linux C code or live TCP',
  runtime: process.version,
  oldAlpha: 0,
  shiftG: fixture.dctcp_shift_g,
  packetFraction: packetMarked / packetTotal,
  payloadByteFraction: byteMarked / byteTotal,
  linux612: { diagnosticBeforeUpdate: packetDiagnostics, alphaAfterOneUpdate: packetAlpha },
  linux318: { diagnosticBeforeUpdate: byteDiagnostics, alphaAfterOneUpdate: byteAlpha },
  negativeControl: 'A byte-only interpretation differs from v6.12 packet accounting for this constructed unequal-size input.',
  legitimateAlternative: 'The byte-based calculation remains consistent with v3.18 for this restricted fixture.',
  limitations: ['No ACK/retransmission timing or kernel concurrency modeled', 'No generic overflow handling exercised', 'No network-performance or behavioral-agent inference']
};
writeFileSync(resolve(root, 'arithmetic-results.json'), JSON.stringify(results, null, 2) + '\n');
console.log(JSON.stringify(results, null, 2));
