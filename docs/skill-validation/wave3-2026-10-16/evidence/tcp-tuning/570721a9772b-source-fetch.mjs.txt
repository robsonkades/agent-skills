import { mkdirSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';

const root = fileURLToPath(new URL('.', import.meta.url));
mkdirSync(resolve(root, 'sources'), { recursive: true });
const sources = [
  ['linux-6.12-ip-sysctl.rst', 'https://raw.githubusercontent.com/torvalds/linux/v6.12/Documentation/networking/ip-sysctl.rst'],
  ['linux-6.12-tcp-dctcp.c', 'https://raw.githubusercontent.com/torvalds/linux/v6.12/net/ipv4/tcp_dctcp.c'],
  ['linux-3.18-tcp-dctcp.c', 'https://raw.githubusercontent.com/torvalds/linux/v3.18/net/ipv4/tcp_dctcp.c'],
  ['linux-6.12-dctcp.rst', 'https://raw.githubusercontent.com/torvalds/linux/v6.12/Documentation/networking/dctcp.rst'],
  ['linux-6.12-tcp-output.c', 'https://raw.githubusercontent.com/torvalds/linux/v6.12/net/ipv4/tcp_output.c'],
  ['linux-6.12-net-tcp.h', 'https://raw.githubusercontent.com/torvalds/linux/v6.12/include/net/tcp.h'],
  ['linux-6.12-linux-tcp.h', 'https://raw.githubusercontent.com/torvalds/linux/v6.12/include/linux/tcp.h'],
  ['jdk-17.0.16-net-util-md.c', 'https://raw.githubusercontent.com/openjdk/jdk17u/jdk-17.0.16-ga/src/java.base/unix/native/libnet/net_util_md.c'],
  ['jdk-17.0.16-net.c', 'https://raw.githubusercontent.com/openjdk/jdk17u/jdk-17.0.16-ga/src/java.base/unix/native/libnio/ch/Net.c'],
  ['java-17-socket.html', 'https://docs.oracle.com/en/java/javase/17/docs/api/java.base/java/net/Socket.html'],
  ['java-17-server-socket.html', 'https://docs.oracle.com/en/java/javase/17/docs/api/java.base/java/net/ServerSocket.html']
];
const results = await Promise.all(sources.map(async ([name, url]) => {
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(25000) });
    const bytes = Buffer.from(await response.arrayBuffer());
    writeFileSync(resolve(root, 'sources', name), bytes);
    return { url, status: response.status, artifact: 'sources/' + name, bytes: bytes.length,
      sha256: createHash('sha256').update(bytes).digest('hex') };
  } catch (error) {
    return { url, status: 'UNAVAILABLE', error: error.message };
  }
}));
writeFileSync(resolve(root, 'source-fetch-results.json'), JSON.stringify(results, null, 2) + '\n');
console.log(JSON.stringify(results, null, 2));
if (results.some(result => result.status !== 200)) process.exitCode = 1;
