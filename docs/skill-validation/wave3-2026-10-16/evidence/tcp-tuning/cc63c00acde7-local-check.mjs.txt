import { spawnSync } from 'node:child_process';
import { mkdirSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import assert from 'node:assert/strict';

const root = fileURLToPath(new URL('.', import.meta.url));
const cwd = 'C:/git/agent-skills';
const packagePath = 'skills/tcp-tuning';
mkdirSync(resolve(root, 'logs'), { recursive: true });
const files = ['SKILL.md', 'skill.yaml', 'references/sysctl-and-socket-options.md', 'references/diagnosis-recipes.md'];
const commands = [
  ['arithmetic-control', process.execPath, [resolve(root, 'arithmetic-control.mjs')], cwd],
  ['proposed-decoder', process.execPath, [resolve(root, 'proposed-decoder.mjs'), resolve(root, 'window-fixture.json')], cwd],
  ['package-check', process.execPath, [resolve(root, 'package-check.mjs')], cwd],
  ['prettier-check', process.execPath, ['node_modules/prettier/bin/prettier.cjs', '--check', ...files.map(file => packagePath + '/' + file)], cwd],
  ['diff-check', 'git', ['diff', '--check', '--', packagePath], cwd],
  ['final-diff', 'git', ['diff', '--', packagePath], cwd],
  ['final-status', 'git', ['status', '--short', '--', packagePath], cwd],
  ['staged-diff', 'git', ['diff', '--cached', '--', packagePath], cwd]
];
const results = [];
const startAt = process.argv.includes('--finish') ? 2 : 0;
for (const [name, executable, args, commandCwd] of commands.slice(startAt)) {
  const result = spawnSync(executable, args, { cwd: commandCwd, encoding: 'utf8', windowsHide: true, timeout: 15000 });
  writeFileSync(resolve(root, 'logs', name + '.stdout.txt'), result.stdout || '');
  writeFileSync(resolve(root, 'logs', name + '.stderr.txt'), result.stderr || '');
  results.push({ name, command: [executable, ...args], cwd: commandCwd, status: result.status,
    signal: result.signal, error: result.error?.message ?? null,
    stdout: 'logs/' + name + '.stdout.txt', stderr: 'logs/' + name + '.stderr.txt' });
  writeFileSync(resolve(root, 'local-results.json'), JSON.stringify(results, null, 2) + '\n');
  assert.equal(result.error, undefined, name);
  assert.equal(result.status, 0, name + '\n' + result.stdout + '\n' + result.stderr);
  if (name === 'staged-diff') assert.equal(result.stdout, '', 'No staged changes in owned package');
  console.log(name + ': PASS');
}
