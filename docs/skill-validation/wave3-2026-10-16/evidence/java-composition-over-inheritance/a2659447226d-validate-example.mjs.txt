import java.math.BigDecimal;
import java.math.RoundingMode;

public class FeeChecks {
    private static int mappedChecks;

    private static void same(String label, BigDecimal expected, BigDecimal actual) {
        if (!expected.equals(actual)) {
            throw new AssertionError(label + ": expected " + expected + " but got " + actual);
        }
    }

    public static void main(String[] args) {
        var none = new FeeSchedule(FeeAdjustment.none());
        var promo = new FeeSchedule(FeeAdjustment.promotionalRate(new BigDecimal("0.5")));
        var domestic = new Card("example", false);
        var international = new Card("example", true);
        for (var literal : new String[] {"0.00", "0.12", "1", "100.00", "1234.56", "2.5"}) {
            var amount = new BigDecimal(literal);
            same("domestic " + literal, new CardFee().total(amount), none.feeFor(domestic, amount));
            same("international " + literal, new InternationalCardFee().total(amount), none.feeFor(international, amount));
            same("promotion " + literal, new PromotionalCardFee().total(amount), promo.feeFor(domestic, amount));
            mappedChecks += 3;
        }
        var hundred = new BigDecimal("100.00");
        same("domestic known value", new BigDecimal("2.90"), none.feeFor(domestic, hundred));
        same("international known value", new BigDecimal("4.40"), none.feeFor(international, hundred));
        same("promotion known value", new BigDecimal("1.45"), promo.feeFor(domestic, hundred));
        var small = new BigDecimal("0.12");
        var correct = none.feeFor(international, small);
        var perComponent = small.multiply(new BigDecimal("0.029")).setScale(2, RoundingMode.HALF_EVEN)
            .add(small.multiply(new BigDecimal("0.015")).setScale(2, RoundingMode.HALF_EVEN));
        same("final rounding", new BigDecimal("0.01"), correct);
        same("component rounding counterexample", new BigDecimal("0.00"), perComponent);
        if (correct.equals(perComponent)) throw new AssertionError("rounding mutation survived");
        same("new full-fee promotional policy", new BigDecimal("2.20"), promo.feeFor(international, hundred));
        var baseOnlyPromotion = hundred.multiply(new BigDecimal("0.029")).multiply(new BigDecimal("0.5"))
            .add(hundred.multiply(new BigDecimal("0.015"))).setScale(2, RoundingMode.HALF_EVEN);
        same("alternative base-only policy", new BigDecimal("2.95"), baseOnlyPromotion);
        same("new boleto policy", new BigDecimal("3.49"), none.feeFor(new Boleto(), hundred));
        same("new pix policy", new BigDecimal("0.00"), none.feeFor(new Pix(), hundred));
        System.out.println("PASS: " + mappedChecks + " mapped comparisons; known values 2.90/4.40/1.45; rounding 0.01 versus 0.00; new policies separately 2.20/2.95/3.49/0.00");
    }
}

const commands = [];
function run(name, executable, args, cwd) {
    const result = spawnSync(executable, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
    const entry = { name, executable, args, cwd, exitCode: result.status, signal: result.signal,
        error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
    commands.push(entry);
    writeFileSync(join(root, name + '.log'), JSON.stringify(entry, null, 2) + '\n');
    if (result.error || result.signal) throw new Error(name + ' failed to execute');
    return entry;
}
