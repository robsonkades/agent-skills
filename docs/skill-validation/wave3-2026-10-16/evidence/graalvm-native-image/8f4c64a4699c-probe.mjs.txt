import assert from 'node:assert/strict';
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';

const root = path.dirname(fileURLToPath(import.meta.url));
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const ni = 'C:/Users/robso/.jdks/graalvm-ce-25.0.2/lib/svm/bin/native-image.exe';
for (const subdir of ['classes', 'tmp', 'native-home', 'logs']) fs.mkdirSync(path.join(root, subdir), { recursive: true });
const childEnv = {
  SystemRoot: process.env.SystemRoot,
  WINDIR: process.env.WINDIR,
  SystemDrive: process.env.SystemDrive,
  ComSpec: process.env.ComSpec,
  ProgramFiles: process.env.ProgramFiles,
  'ProgramFiles(x86)': process.env['ProgramFiles(x86)'],
  PATH: jdk + ';' + path.join(process.env.SystemRoot, 'System32'),
  TEMP: path.join(root, 'tmp'), TMP: path.join(root, 'tmp'),
  HOME: root, USERPROFILE: root,
  NATIVE_IMAGE_USER_HOME: path.join(root, 'native-home')
};
const commands = [];
function run(name, executable, args) {
  const result = spawnSync(executable, args, { cwd: root, env: childEnv, encoding: 'utf8', timeout: 30000 });
  const record = { name, executable, args, cwd: root, exitCode: result.status, signal: result.signal, error: result.error?.message, stdout: result.stdout, stderr: result.stderr };
  commands.push(record);
  fs.writeFileSync(path.join(root, 'logs', name + '.json'), JSON.stringify(record, null, 2));
  assert.equal(result.status, 0, name + ': ' + result.stderr);
  return result.stdout + '\n' + result.stderr;
}
const identity = { executable: ni, sha256: crypto.createHash('sha256').update(fs.readFileSync(ni)).digest('hex') };
identity.version = run('native-version', ni, ['--version']);
run('javac-version', path.join(jdk, 'javac.exe'), ['-version']);
run('compile', path.join(jdk, 'javac.exe'), ['--release', '17', '-d', path.join(root, 'classes'), path.join(root, 'src', 'Probe.java')]);
run('jar', path.join(jdk, 'jar.exe'), ['--create', '--file', path.join(root, 'probe.jar'), '-C', path.join(root, 'classes'), '.', '-C', path.join(root, 'resources'), '.']);
const before = run('option-before-classpath', ni, ['--verbose', '--dry-run', '-O2', '-cp', path.join(root, 'probe.jar'), 'Probe']);
const after = run('option-after-classpath', ni, ['--verbose', '--dry-run', '-cp', path.join(root, 'probe.jar'), '-O2', 'Probe']);
function optimizationSequence(output) {
  return [...output.matchAll(/-H:Optimize(?:@[^\r\n =]+)?=([0-9b])/g)].map(m => m[1]);
}
const sequences = { before: optimizationSequence(before), after: optimizationSequence(after) };
fs.writeFileSync(path.join(root, 'result.json'), JSON.stringify({ identity, childEnv, sequences, commands }, null, 2));
assert.ok(before.includes('META-INF/native-image/review/probe/native-image.properties'), 'embedded configuration provenance absent');
assert.deepEqual(sequences.before, ['2', '0']);
assert.deepEqual(sequences.after, ['0', '2']);
assert.equal(fs.readFileSync(path.join(root, 'classes', 'Probe.class')).readUInt16BE(6), 61);
assert.ok(!fs.existsSync(path.join(root, 'probe.exe')), 'dry-run unexpectedly generated native binary');
console.log(JSON.stringify({ identity, sequences, commands: commands.map(({ name, exitCode }) => ({ name, exitCode })), nativeBinaryGenerated: false, classMajor: 61 }, null, 2));
