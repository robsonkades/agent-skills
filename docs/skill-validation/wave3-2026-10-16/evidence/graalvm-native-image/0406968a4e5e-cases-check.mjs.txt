import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
import assert from 'node:assert/strict';
const root = path.dirname(fileURLToPath(import.meta.url));
const prior = JSON.parse(fs.readFileSync(path.join(root, 'result.json'), 'utf8'));
const results = [];
for (const id of ['a', 'b']) {
  const cwd = path.join(root, 'cases', id);
  const args = ['--verbose', '--dry-run', ...JSON.parse(fs.readFileSync(path.join(cwd, 'release-build.args.json'), 'utf8'))];
  const p = spawnSync(prior.identity.executable, args, { cwd, env: prior.childEnv, encoding: 'utf8', timeout: 30000 });
  const output = p.stdout + '\n' + p.stderr;
  const optimizationSequence = [...output.matchAll(/-H:Optimize(?:@[^\r\n =]+)?=([0-9b])/g)].map(m => m[1]);
  const record = { id, executable: prior.identity.executable, args, cwd, exitCode: p.status, signal: p.signal, stdout: p.stdout, stderr: p.stderr, optimizationSequence };
  fs.writeFileSync(path.join(root, 'logs', 'case-' + id + '.json'), JSON.stringify(record, null, 2));
  assert.equal(p.status, 0);
  assert.deepEqual(optimizationSequence, id === 'a' ? ['2', '0'] : ['0', '2']);
  assert.ok(!fs.existsSync(path.join(cwd, 'probe.exe')));
  results.push({ id, exitCode: p.status, optimizationSequence });
}
console.log(JSON.stringify({ results, nativeBinaryProduced: false }));
