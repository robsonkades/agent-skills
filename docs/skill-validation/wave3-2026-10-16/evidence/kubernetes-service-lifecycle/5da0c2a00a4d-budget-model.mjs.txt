const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
let assertions = 0;
const equal = (actual, expected, name) => {
  assert.deepEqual(actual, expected, name);
  assertions++;
};
const scale = (value, total, roundUp) => {
  if (Number.isInteger(value)) return value;
  assert.match(value, /^\d+%$/); // Bounded fixture model, not full API validation.
  const scaled = total * Number(value.slice(0, -1)) / 100;
  return roundUp ? Math.ceil(scaled) : Math.floor(scaled);
};
function rollout(n, unavailable, surge) {
  let u = scale(unavailable, n, false);
  const s = scale(surge, n, true);
  if (u === 0 && s === 0) u = 1; // v1.34 ResolveFenceposts fallback
  return { unavailable: Math.min(u, n), surge: s };
}
function pdb(n, healthy, spec) {
  // Positive desired scale, exactly one owner, all healthy input counts already
  // exclude deleting/already-disrupted pods. This is not a controller execution.
  assert.ok(n > 0);
  assert.notEqual(spec.minAvailable !== undefined, spec.maxUnavailable !== undefined);
  const desiredHealthy = spec.minAvailable !== undefined
    ? scale(spec.minAvailable, n, true)
    : Math.max(0, n - scale(spec.maxUnavailable, n, true));
  return { desiredHealthy, disruptionsAllowed: Math.max(0, healthy - desiredHealthy) };
}
function parseYaml(file) {
  const doc = YAML.parseDocument(readFileSync(file, "utf8"), { uniqueKeys: true });
  assert.equal(doc.errors.length, 0, file + ": " + doc.errors.map(e => e.message).join("; "));
  return doc.toJS();
}

    changedFiles.push(file);

  }

const cases = [];
let fixtureYamlDocuments = 0;
for (const id of ["one-replica", "four-replicas"]) {
  const input = path.join(root, "inputs", id);
  const deployment = parseYaml(path.join(input, "deployment.yaml"));
  const disruption = parseYaml(path.join(input, "pdb.yaml"));
  const service = parseYaml(path.join(input, "service.yaml"));
  fixtureYamlDocuments += 3;
  equal(disruption.spec.selector.matchLabels, deployment.spec.template.metadata.labels, "PDB selector");
  equal(service.spec.selector, deployment.spec.template.metadata.labels, "Service selector");
  equal(deployment.spec.selector.matchLabels, deployment.spec.template.metadata.labels, "Deployment selector");
  const n = deployment.spec.replicas;
  const resolvedPdb = pdb(n, n, disruption.spec);
  const resolvedRollout = rollout(n, deployment.spec.strategy.rollingUpdate.maxUnavailable, deployment.spec.strategy.rollingUpdate.maxSurge);
  equal(resolvedPdb, n === 1 ? { desiredHealthy: 0, disruptionsAllowed: 1 } : { desiredHealthy: 3, disruptionsAllowed: 1 }, id + " PDB");
  equal(resolvedRollout, n === 1 ? { unavailable: 0, surge: 1 } : { unavailable: 1, surge: 1 }, id + " rollout");
  cases.push({ id, desiredReplicas: n, modeledPdb: resolvedPdb, modeledRollout: resolvedRollout });
}
