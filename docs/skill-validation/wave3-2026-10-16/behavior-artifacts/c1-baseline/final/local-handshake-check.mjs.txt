import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFile } from 'node:fs/promises';
import net from 'node:net';
import { performance } from 'node:perf_hooks';
import { attachClient } from './client.mjs';
import { attachServer } from './server.mjs';

// This models only the documented connection-start gate. It is not the installed
// adapter, and does not implement or validate its inspection timer/buffer behavior.
// Use ephemeral loopback ports; do not bind the proposed production listener.
const originalPolicy = {
  listener: '127.0.0.1:15001',
  route: 'fixed-service-a',
  connect_start: 'first_client_data',
  inspection_timeout_ms: 1000,
  inspection_buffer_limit: 4096,
};
const protocol = JSON.parse(await readFile(new URL('./protocol.json', import.meta.url), 'utf8'));
const policy = JSON.parse(await readFile(new URL('./proxy-policy.json', import.meta.url), 'utf8'));
const caseBoundMs = 1200;
const hardBoundMs = 8000;
const hardStop = setTimeout(() => {
  console.error(JSON.stringify({ result: 'FAIL', reason: 'global watchdog expired', hardBoundMs }));
  process.exit(1);
}, hardBoundMs);

function log(value) {
  console.log(JSON.stringify(value));
}

async function scenario(name, connectionStart) {
  const started = performance.now();
  const sockets = new Set();
  const servers = [];
  const state = {
    name,
    connectionStart,
    clientConnected: false,
    downstreamAccepted: 0,
    proxyUpstreamAttempts: 0,
    upstreamAccepted: 0,
    completed: 0,
    clientReceived: '',
    upstreamReceived: '',
    errors: [],
  };
  let cleaning = false;
  let settle;
  const outcome = new Promise(resolve => { settle = resolve; });
  const caseTimer = setTimeout(() => settle('deadline'), caseBoundMs);

  function track(socket) {
    sockets.add(socket);
    socket.on('close', () => sockets.delete(socket));
    socket.on('error', error => {
      if (!cleaning) {
        state.errors.push(error.message);
        settle('socket_error');
      }
    });
    return socket;
  }

  async function listen(server) {
    servers.push(server);
    await new Promise((resolve, reject) => {
      server.once('error', reject);
      server.listen(0, '127.0.0.1', resolve);
    });
    return server.address().port;
  }

  try {
    const upstreamPort = await listen(net.createServer(socket => {
      track(socket);
      state.upstreamAccepted++;
      socket.on('data', bytes => { state.upstreamReceived += bytes.toString(); });
      attachServer(socket, protocol);
    }));

    let clientPort = upstreamPort;
    if (connectionStart !== 'direct') {
      clientPort = await listen(net.createServer(downstream => {
        track(downstream);
        state.downstreamAccepted++;

        function openUpstream(firstBytes) {
          state.proxyUpstreamAttempts++;
          const upstream = track(net.createConnection({ host: '127.0.0.1', port: upstreamPort }));
          if (firstBytes) upstream.write(firstBytes);
          downstream.pipe(upstream);
          upstream.pipe(downstream);
          downstream.once('close', () => upstream.destroy());
          upstream.once('close', () => downstream.destroy());
        }

        if (connectionStart === 'immediate') openUpstream();
        else if (connectionStart === 'first_client_data') downstream.once('data', openUpstream);
        else throw new Error(`Unsupported connection start: ${connectionStart}`);
      }));
    }

    const client = track(net.createConnection({ host: '127.0.0.1', port: clientPort }));
    client.once('connect', () => { state.clientConnected = true; });
    client.on('data', bytes => { state.clientReceived += bytes.toString(); });
    attachClient(client, protocol, () => {
      state.completed++;
      settle('completed');
    });

    state.outcome = await outcome;
    state.elapsedMs = Number((performance.now() - started).toFixed(1));
    return state;
  } finally {
    clearTimeout(caseTimer);
    cleaning = true;
    for (const socket of sockets) socket.destroy();
    await Promise.all(servers.map(server => new Promise(resolve => server.close(() => resolve()))));
    state.cleanup = 'all scenario sockets destroyed and listeners closed';
  }
}

try {
  log({ runtime: process.version, caseBoundMs, hardBoundMs, scope: 'local fixture + documented connection-start model' });
  assert.deepEqual(protocol, { first_message: 'server' });
  assert.deepEqual(policy, { ...originalPolicy, connect_start: 'immediate' });
  log({ check: 'policy diff', result: 'PASS', changedFields: ['connect_start'], from: 'first_client_data', to: 'immediate' });

  const protectedHashes = {
    'client.mjs': '03f3c11d53a2d6273fe791b7dc41a42ce03b99e33747bf785e480a27630bf043',
    'server.mjs': '9207275610e3e93e7f06b73f9dde844d56e519ff874cf6a1854f6d170192b7d0',
    'protocol.json': '093a5214ebd8a4ba26a9aa3588b6add210b22e21678d324d5e4f634430fdc2e9',
    'adapter-contract.md': '591f36efd3fb491c27ecf90fe637c61933beaff291858d60502e7c3a0c1aab29',
  };
  for (const [file, expected] of Object.entries(protectedHashes)) {
    const actual = createHash('sha256').update(await readFile(new URL(file, import.meta.url))).digest('hex');
    assert.equal(actual, expected, `${file} must remain intact`);
    log({ check: 'protected fixture', file, sha256: actual, result: 'PASS' });
  }

  const direct = await scenario('direct client/server control', 'direct');
  log(direct);
  assert.equal(direct.outcome, 'completed');
  assert.equal(direct.completed, 1);
  assert.equal(direct.upstreamAccepted, 1);
  assert.equal(direct.clientReceived, 'HELLO\nPONG\n');
  assert.equal(direct.upstreamReceived, 'PING\n');
  assert.deepEqual(direct.errors, []);

  const original = await scenario('original policy gate', originalPolicy.connect_start);
  log(original);
  assert.equal(original.clientConnected, true);
  assert.equal(original.downstreamAccepted, 1);
  assert.equal(original.outcome, 'deadline');
  assert.equal(original.completed, 0);
  assert.equal(original.proxyUpstreamAttempts, 0);
  assert.equal(original.upstreamAccepted, 0);
  assert.equal(original.clientReceived, '');
  assert.equal(original.upstreamReceived, '');
  assert.deepEqual(original.errors, []);

  const proposed = await scenario('proposed policy gate', policy.connect_start);
  log(proposed);
  assert.equal(proposed.clientConnected, true);
  assert.equal(proposed.downstreamAccepted, 1);
  assert.equal(proposed.outcome, 'completed');
  assert.equal(proposed.completed, 1);
  assert.equal(proposed.proxyUpstreamAttempts, 1);
  assert.equal(proposed.upstreamAccepted, 1);
  assert.equal(proposed.clientReceived, 'HELLO\nPONG\n');
  assert.equal(proposed.upstreamReceived, 'PING\n');
  assert.deepEqual(proposed.errors, []);
  log({ result: 'PASS', scenarios: 3, deploymentValidated: false, cleanup: 'complete' });
} finally {
  clearTimeout(hardStop);
}
