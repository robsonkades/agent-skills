import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import net from 'node:net';
import { attachClient } from './client.mjs';
import { attachServer } from './server.mjs';

// Local contract model only: do not import or operate an installed adapter.
const protocol = JSON.parse(await readFile(new URL('./protocol.json', import.meta.url)));
const policy = JSON.parse(await readFile(new URL('./proxy-policy.json', import.meta.url)));
assert.equal(protocol.first_message, 'client');
assert.equal(policy.connect_start, 'first_client_data');
assert.equal(policy.route, 'fixed-service-a');
assert.equal(policy.inspection_timeout_ms, 1000);
assert.equal(policy.inspection_buffer_limit, 4096);

const started = performance.now();
const events = [];
const sockets = new Set();
const servers = [];
const timers = new Set();
let serverBytes = '';
let clientBytes = '';
let completionCount = 0;
let upstreamStarts = 0;
let peakBufferedBytes = 0;
let resolveComplete;
let rejectComplete;
const completed = new Promise((resolve, reject) => {
  resolveComplete = resolve;
  rejectComplete = reject;
});
// Failures during listener setup must remain handled until the await below.
completed.catch(() => {});

function record(event, details = {}) {
  const entry = { ms: Math.round(performance.now() - started), event, ...details };
  events.push(entry);
  console.log(JSON.stringify(entry));
}

function boundedTimer(ms, label) {
  const timer = setTimeout(() => rejectComplete(new Error(label)), ms);
  timers.add(timer);
  return timer;
}

function track(socket) {
  sockets.add(socket);
  socket.on('error', rejectComplete);
  return socket;
}

async function listen(server) {
  servers.push(server);
  server.maxConnections = 1;
  server.on('error', rejectComplete);
  await Promise.race([
    new Promise(resolve => server.listen(0, '127.0.0.1', resolve)),
    completed,
  ]);
  const address = server.address();
  assert.ok(address && typeof address !== 'string');
  return address.port;
}

async function closeServer(server) {
  if (server.listening) {
    await new Promise((resolve, reject) => server.close(error => error ? reject(error) : resolve()));
  }
}

boundedTimer(3000, 'Overall handshake deadline exceeded (3000 ms)');
record('start', {
  node: process.version,
  protocol: protocol.first_message,
  connect_start: policy.connect_start,
  inspection_timeout_ms: policy.inspection_timeout_ms,
  inspection_buffer_limit: policy.inspection_buffer_limit,
  overall_deadline_ms: 3000,
});

try {
  // The trusted fixed route is represented by this ephemeral local server.
  const serverPort = await listen(net.createServer(socket => {
    track(socket);
    record('server_accept');
    socket.on('data', bytes => {
      serverBytes += bytes.toString();
      record('server_receive', { data: bytes.toString() });
    });
    attachServer(socket, protocol);
  }));

  const proxyPort = await listen(net.createServer(downstream => {
    track(downstream);
    record('downstream_accept');
    const inspectionTimer = boundedTimer(policy.inspection_timeout_ms, 'Client-data inspection timeout');
    let upstream;
    let connected = false;
    let pending = [];
    let pendingBytes = 0;

    downstream.on('data', bytes => {
      record('client_data', { data: bytes.toString() });
      if (connected) {
        upstream.write(bytes);
        return;
      }
      pendingBytes += bytes.length;
      peakBufferedBytes = Math.max(peakBufferedBytes, pendingBytes);
      if (pendingBytes > policy.inspection_buffer_limit) {
        rejectComplete(new Error('Inspection buffer limit exceeded'));
        return;
      }
      pending.push(bytes);
      if (upstream) return;

      clearTimeout(inspectionTimer);
      upstreamStarts++;
      record('upstream_connect_start');
      upstream = track(net.createConnection({ host: '127.0.0.1', port: serverPort }));
      upstream.once('connect', () => {
        record('upstream_connect');
        connected = true;
        for (const chunk of pending) upstream.write(chunk);
        pending = [];
        pendingBytes = 0;
      });
      upstream.on('data', bytes => downstream.write(bytes));
      upstream.on('end', () => downstream.end());
    });
  }));

  const client = track(new net.Socket());
  client.once('connect', () => record('client_connect'));
  client.on('data', bytes => {
    clientBytes += bytes.toString();
    record('client_receive', { data: bytes.toString() });
  });
  attachClient(client, protocol, () => {
    completionCount++;
    record('client_complete');
    resolveComplete();
  });
  client.connect({ host: '127.0.0.1', port: proxyPort });

  await completed;
  assert.equal(serverBytes, 'PING\n');
  assert.equal(clientBytes, 'PONG\n');
  assert.equal(completionCount, 1);
  assert.equal(upstreamStarts, 1);
  const eventIndex = name => events.findIndex(entry => entry.event === name);
  assert.ok(eventIndex('downstream_accept') < eventIndex('client_data'));
  assert.ok(eventIndex('client_data') < eventIndex('upstream_connect_start'));
  assert.ok(eventIndex('upstream_connect_start') < eventIndex('upstream_connect'));
  assert.ok(eventIndex('upstream_connect') < eventIndex('client_complete'));
  assert.ok(peakBufferedBytes <= policy.inspection_buffer_limit);
  assert.ok(performance.now() - started < 3000);
  record('PASS', { upstreamStarts, completionCount, peakBufferedBytes });
} finally {
  for (const timer of timers) clearTimeout(timer);
  for (const socket of sockets) socket.destroy();
  await Promise.all(servers.map(closeServer));
  assert.ok([...sockets].every(socket => socket.destroyed));
  assert.ok(servers.every(server => !server.listening));
  record('cleanup', { socketsDestroyed: sockets.size, listenersClosed: servers.length });
}
