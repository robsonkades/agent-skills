export function attachClient(socket, protocol, onComplete) {
  let received = '';
  let sent = false;
  function ping() { if (!sent) { sent = true; socket.write('PING\n'); } }
  socket.once('connect', () => { if (protocol.first_message === 'client') ping(); });
  socket.on('data', bytes => {
    received += bytes.toString();
    if (protocol.first_message === 'server' && received.includes('HELLO\n')) ping();
    if (received.includes('PONG\n')) onComplete();
  });
}
