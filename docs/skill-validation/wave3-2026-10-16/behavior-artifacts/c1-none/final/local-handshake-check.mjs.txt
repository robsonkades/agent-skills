import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import net from 'node:net';
import { performance } from 'node:perf_hooks';
import { attachClient } from './client.mjs';
import { attachServer } from './server.mjs';

// A bounded loopback model of the documented connection gate, not a vendor adapter.
const protocol = JSON.parse(await readFile(new URL('./protocol.json', import.meta.url), 'utf8'));
const policy = JSON.parse(await readFile(new URL('./proxy-policy.json', import.meta.url), 'utf8'));
assert.deepEqual(protocol, { first_message: 'server' });
assert.deepEqual(policy, {
  listener: '127.0.0.1:15001',
  route: 'fixed-service-a',
  connect_start: 'immediate',
  inspection_timeout_ms: 1000,
  inspection_buffer_limit: 4096,
});

const deadlineMs = 1200;
const hardDeadline = setTimeout(() => {
  console.error('FAIL: overall 7000 ms watchdog expired');
  process.exit(1);
}, 7000);

function listen(server) {
  return new Promise((resolve, reject) => {
    server.once('error', reject);
    server.listen(0, '127.0.0.1', () => {
      server.off('error', reject);
      resolve(server.address().port);
    });
  });
}

async function exercise(name, connectStart) {
  const sockets = new Set();
  const servers = [];
  const errors = [];
  const metrics = {
    downstreamTcpConnected: false,
    downstreamAccepted: false,
    upstreamAttempts: 0,
    upstreamAccepted: 0,
    clientToServer: '',
    serverToClient: '',
  };
  let settle;
  let timer;
  let outcome;
  const completion = new Promise(resolve => { settle = resolve; });
  const track = socket => {
    sockets.add(socket);
    socket.on('error', error => {
      errors.push(error.message);
      settle('socket-error');
    });
    socket.once('close', () => sockets.delete(socket));
    return socket;
  };
  const started = performance.now();
  try {
    const server = net.createServer(socket => {
      track(socket);
      metrics.upstreamAccepted += 1;
      if (connectStart === null) metrics.downstreamAccepted = true;
      socket.on('data', bytes => { metrics.clientToServer += bytes.toString(); });
      attachServer(socket, protocol);
    });
    servers.push(server);
    const upstreamPort = await listen(server);
    let clientPort = upstreamPort;

    if (connectStart !== null) {
      const proxy = net.createServer(downstream => {
        track(downstream);
        metrics.downstreamAccepted = true;
        let upstream;
        let ready = false;
        let pending = [];
        let pendingBytes = 0;
        const openUpstream = () => {
          if (upstream) return;
          metrics.upstreamAttempts += 1;
          upstream = track(net.createConnection({ host: '127.0.0.1', port: upstreamPort }));
          upstream.once('connect', () => {
            ready = true;
            for (const bytes of pending) upstream.write(bytes);
            pending = [];
            pendingBytes = 0;
          });
          upstream.on('data', bytes => downstream.write(bytes));
          upstream.once('end', () => downstream.end());
        };
        downstream.on('data', bytes => {
          if (ready) {
            upstream.write(bytes);
            return;
          }
          pendingBytes += bytes.length;
          if (pendingBytes > policy.inspection_buffer_limit) {
            errors.push('Model inspection buffer limit exceeded');
            settle('buffer-limit');
            return;
          }
          pending.push(bytes);
          if (connectStart === 'first_client_data') openUpstream();
        });
        downstream.once('end', () => upstream?.end());
        if (connectStart === 'immediate') openUpstream();
      });
      servers.push(proxy);
      clientPort = await listen(proxy);
    }

    const client = track(new net.Socket());
    attachClient(client, protocol, () => settle('handshake-complete'));
    client.once('connect', () => { metrics.downstreamTcpConnected = true; });
    client.on('data', bytes => { metrics.serverToClient += bytes.toString(); });
    timer = setTimeout(() => settle('local-observation-deadline'), deadlineMs);
    client.connect({ host: '127.0.0.1', port: clientPort });
    outcome = await completion;
  } finally {
    clearTimeout(timer);
    for (const socket of sockets) socket.destroy();
    await Promise.all(servers.map(server => new Promise(resolve => server.close(resolve))));
  }
  return {
    name,
    connectStart: connectStart ?? 'direct-control',
    outcome,
    elapsedMs: Math.round(performance.now() - started),
    ...metrics,
    errors,
    listenersClosed: servers.every(server => !server.listening),
  };
}

try {
  console.log(JSON.stringify({ node: process.version, deadlineMs, policy, protocol }));
  const results = [];
  for (const [name, mode] of [
    ['direct client/server control', null],
    ['original policy regression model', 'first_client_data'],
    ['proposed policy model', policy.connect_start],
  ]) {
    const result = await exercise(name, mode);
    results.push(result);
    console.log(JSON.stringify(result));
  }
  const [direct, original, proposed] = results;
  for (const result of results) {
    assert.equal(result.downstreamTcpConnected, true);
    assert.equal(result.downstreamAccepted, true);
    assert.deepEqual(result.errors, []);
    assert.equal(result.listenersClosed, true);
  }
  for (const result of [direct, proposed]) {
    assert.equal(result.outcome, 'handshake-complete');
    assert.equal(result.upstreamAccepted, 1);
    assert.equal(result.clientToServer, 'PING\n');
    assert.equal(result.serverToClient, 'HELLO\nPONG\n');
  }
  assert.equal(original.outcome, 'local-observation-deadline');
  assert.equal(original.upstreamAttempts, 0);
  assert.equal(original.upstreamAccepted, 0);
  assert.equal(original.clientToServer, '');
  assert.equal(original.serverToClient, '');
  assert.equal(proposed.upstreamAttempts, 1);
  console.log('PASS: direct control and immediate gate completed HELLO -> PING -> PONG; deferred gate accepted TCP but never connected upstream. All listeners closed.');
} finally {
  clearTimeout(hardDeadline);
}
