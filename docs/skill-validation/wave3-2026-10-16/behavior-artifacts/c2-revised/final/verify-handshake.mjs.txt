// Local model of adapter-contract.md, not the installed adapter or vendor config.
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import net from 'node:net';
import { performance } from 'node:perf_hooks';
import { attachClient } from './client.mjs';
import { attachServer } from './server.mjs';

const root = new URL('./', import.meta.url);
const policy = JSON.parse(await readFile(new URL('proxy-policy.json', root), 'utf8'));
const protocol = JSON.parse(await readFile(new URL('protocol.json', root), 'utf8'));
const logDirectory = new URL('logs/', root);
await mkdir(logDirectory, { recursive: true });

const suppliedHashes = {
  'client.mjs': '03f3c11d53a2d6273fe791b7dc41a42ce03b99e33747bf785e480a27630bf043',
  'server.mjs': '9207275610e3e93e7f06b73f9dde844d56e519ff874cf6a1854f6d170192b7d0',
  'adapter-contract.md': '591f36efd3fb491c27ecf90fe637c61933beaff291858d60502e7c3a0c1aab29',
  'protocol.json': '7c6e7302b359275ebc74d9b0bf95bc7e2d44719a535cd3e47531b0a7ba6ae994',
  'proxy-policy.json': '1bb64854cfb9a3ee1f16cd231fec9643363313964e4022517f1d725d53740c60',
};
const hashes = {};
for (const [file, expected] of Object.entries(suppliedHashes)) {
  hashes[file] = createHash('sha256').update(await readFile(new URL(file, root))).digest('hex');
  assert.equal(hashes[file], expected, `${file} must remain unchanged`);
}
assert.equal(protocol.first_message, 'client');
assert.equal(policy.connect_start, 'first_client_data');
assert.equal(policy.listener, '127.0.0.1:15001');
assert.equal(policy.route, 'fixed-service-a');
assert.equal(policy.inspection_timeout_ms, 1000);
assert.equal(policy.inspection_buffer_limit, 4096);

const results = [];
const globalTimer = setTimeout(() => {
  console.error('FAIL: 15-second whole-process safety deadline');
  process.exit(1);
}, 15000);

async function runCase(name, firstMessage, connectStart, expectComplete) {
  const start = performance.now();
  const events = [];
  const sockets = new Set();
  const timers = new Set();
  const failures = [];
  let upstreamAttempts = 0;
  let upstreamAccepted = 0;
  let clientBytes = '';
  let serverBytes = '';
  let completions = 0;
  let inspectionExpired = false;
  let finish;
  const outcome = new Promise(resolve => { finish = resolve; });
  const event = (kind, detail = {}) => events.push({ ms: +(performance.now() - start).toFixed(3), kind, ...detail });
  const track = socket => {
    sockets.add(socket);
    socket.on('error', error => {
      failures.push(error.message);
      event('socket-error', { message: error.message });
      finish('error');
    });
    return socket;
  };
  const setTimer = (callback, ms) => {
    const timer = setTimeout(callback, ms);
    timers.add(timer);
    return timer;
  };
  const upstream = net.createServer(socket => {
    track(socket);
    upstreamAccepted++;
    event('upstream-accepted');
    attachServer(socket, { first_message: firstMessage });
  });
  const proxy = net.createServer(downstream => {
    track(downstream);
    event('proxy-accepted');
    let remote;
    let connected = false;
    let buffered = [];
    let bufferedBytes = 0;
    let inspectionTimer;
    const connectUpstream = () => {
      if (remote) return;
      clearTimeout(inspectionTimer);
      upstreamAttempts++;
      event('upstream-connect-start');
      // The trusted fixed route maps only to this case's loopback server.
      remote = track(net.createConnection({ host: '127.0.0.1', port: upstream.address().port }));
      remote.once('connect', () => {
        connected = true;
        event('upstream-connected');
        for (const bytes of buffered) remote.write(bytes);
        buffered = [];
        bufferedBytes = 0;
      });
      remote.on('data', bytes => {
        serverBytes += bytes.toString();
        event('server-to-client', { bytes: bytes.toString() });
        downstream.write(bytes);
      });
      remote.on('end', () => downstream.end());
      downstream.once('close', () => remote.destroy());
    };
    downstream.on('data', bytes => {
      clientBytes += bytes.toString();
      event('client-to-proxy', { bytes: bytes.toString() });
      if (connected) {
        remote.write(bytes);
      } else {
        bufferedBytes += bytes.length;
        if (bufferedBytes > policy.inspection_buffer_limit) {
          failures.push('inspection buffer exceeded');
          event('buffer-limit');
          downstream.destroy();
          finish('buffer-limit');
          return;
        }
        buffered.push(bytes);
        connectUpstream();
      }
    });
    if (connectStart === 'immediate') {
      connectUpstream();
    } else {
      // Timeout closure is a test-model choice; the real adapter's failure signal is unspecified.
      inspectionTimer = setTimer(() => {
        inspectionExpired = true;
        event('inspection-expired');
        downstream.destroy();
        finish('inspection-expired');
      }, policy.inspection_timeout_ms);
    }
  });
  const listen = server => new Promise((resolve, reject) => {
    server.once('error', reject);
    server.listen(0, '127.0.0.1', resolve);
  });
  const close = server => new Promise((resolve, reject) => {
    if (!server.listening) return resolve();
    server.close(error => error ? reject(error) : resolve());
  });
  let observed;
  try {
    await listen(upstream);
    await listen(proxy);
    event('listeners-ready', { proxyPort: proxy.address().port, upstreamPort: upstream.address().port });
    setTimer(() => { event('case-deadline'); finish('case-deadline'); }, 3000);
    const client = track(net.createConnection({ host: '127.0.0.1', port: proxy.address().port }));
    client.once('connect', () => event('client-connected'));
    attachClient(client, { first_message: firstMessage }, () => {
      completions++;
      event('handshake-complete');
      finish('complete');
    });
    observed = await outcome;
  } finally {
    for (const timer of timers) clearTimeout(timer);
    for (const socket of sockets) socket.destroy();
    await Promise.all([close(proxy), close(upstream)]);
    event('listeners-closed');
  }
  const result = { name, firstMessage, connectStart, observed, upstreamAttempts, upstreamAccepted, clientBytes, serverBytes, completions, inspectionExpired, failures, events };
  results.push(result);
  assert.deepEqual(failures, []);
  assert.equal(observed, expectComplete ? 'complete' : 'inspection-expired');
  assert.equal(upstreamAttempts, expectComplete ? 1 : 0);
  assert.equal(upstreamAccepted, expectComplete ? 1 : 0);
  assert.equal(completions, expectComplete ? 1 : 0);
  assert.equal(clientBytes, expectComplete ? 'PING\n' : '');
  assert.equal(serverBytes, expectComplete ? (firstMessage === 'server' ? 'HELLO\nPONG\n' : 'PONG\n') : '');
  assert.equal(inspectionExpired, !expectComplete);
  if (expectComplete) {
    const firstBytes = events.findIndex(item => item.kind === 'client-to-proxy');
    const firstConnect = events.findIndex(item => item.kind === 'upstream-connect-start');
    assert.ok(connectStart === 'first_client_data' ? firstBytes < firstConnect : firstConnect < firstBytes);
  }
  console.log(`PASS ${name}: ${observed}; upstream attempts/accepts=${upstreamAttempts}/${upstreamAccepted}; completions=${completions}`);
}

try {
  await runCase('supplied client-first / deferred', protocol.first_message, policy.connect_start, true);
  await runCase('client-first / immediate control', 'client', 'immediate', true);
  await runCase('server-first / deferred control', 'server', 'first_client_data', false);
  await runCase('server-first / immediate control', 'server', 'immediate', true);
  console.log('PASS 4/4 cases; original fixture hashes unchanged; all case sockets and listeners closed.');
} catch (error) {
  console.error(error);
  process.exitCode = 1;
} finally {
  await writeFile(new URL('handshake-results.json', logDirectory), JSON.stringify({ node: process.version, policy, protocol, hashes, modelOnly: true, caseDeadlineMs: 3000, wholeProcessDeadlineMs: 15000, results }, null, 2) + '\n');
  clearTimeout(globalTimer);
}
