import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import net from 'node:net';
import { attachClient } from './client.mjs';
import { attachServer } from './server.mjs';

const protocol = JSON.parse(await readFile(new URL('./protocol.json', import.meta.url), 'utf8'));
const policy = JSON.parse(await readFile(new URL('./proxy-policy.json', import.meta.url), 'utf8'));
assert.equal(protocol.first_message, 'client');
assert.equal(policy.connect_start, 'first_client_data');
assert.equal(policy.route, 'fixed-service-a');
assert.equal(policy.listener, '127.0.0.1:15001');
assert.equal(policy.inspection_timeout_ms, 1000);
assert.equal(policy.inspection_buffer_limit, 4096);

// A local contract model, not the installed adapter or a vendor configuration.
// Both listeners use ephemeral loopback ports. The fixed route maps to our stub.
async function runScenario(name, sendProtocolMessage) {
  const sockets = new Set();
  const timers = new Set();
  const events = [];
  const observed = { upstreamAttempts: 0, upstreamConnections: 0, upstreamReceived: '', clientReceived: '' };
  let resolveCompletion;
  let rejectCompletion;
  const completion = new Promise((resolve, reject) => {
    resolveCompletion = resolve;
    rejectCompletion = reject;
  });
  // Handle a setup-time failure even before awaiting completion below.
  completion.catch(() => {});
  const timer = (fn, delay) => {
    const handle = setTimeout(fn, delay);
    timers.add(handle);
    return handle;
  };
  const track = socket => {
    sockets.add(socket);
    socket.once('close', () => sockets.delete(socket));
    socket.on('error', rejectCompletion);
    return socket;
  };
  const listen = server => new Promise((resolve, reject) => {
    server.once('error', reject);
    server.listen(0, '127.0.0.1', resolve);
  });
  const upstreamServer = net.createServer(socket => {
    track(socket);
    observed.upstreamConnections++;
    events.push('upstream-accepted');
    socket.on('data', bytes => {
      observed.upstreamReceived += bytes.toString();
      events.push('upstream-received-data');
    });
    attachServer(socket, protocol);
  });
  const proxyServer = net.createServer(downstream => {
    track(downstream);
    events.push('downstream-accepted');
    let upstream;
    let connected = false;
    let pendingBytes = 0;
    const pending = [];
    const inspection = timer(() => {
      events.push('inspection-expired');
      downstream.destroy();
    }, policy.inspection_timeout_ms);
    downstream.on('data', bytes => {
      events.push('downstream-received-data');
      if (connected) {
        upstream.write(bytes);
        return;
      }
      pendingBytes += bytes.length;
      if (pendingBytes > policy.inspection_buffer_limit) {
        rejectCompletion(new Error('Inspection buffer limit exceeded'));
        downstream.destroy();
        return;
      }
      pending.push(bytes);
      if (upstream) return;
      clearTimeout(inspection);
      observed.upstreamAttempts++;
      events.push('upstream-connect-start');
      upstream = track(net.createConnection({ host: '127.0.0.1', port: upstreamServer.address().port }));
      upstream.once('connect', () => {
        connected = true;
        for (const chunk of pending) upstream.write(chunk);
        pending.length = 0;
      });
      upstream.on('data', response => downstream.write(response));
      upstream.once('end', () => downstream.end());
    });
    downstream.once('close', () => {
      clearTimeout(inspection);
      upstream?.destroy();
    });
  });

  try {
    await listen(upstreamServer);
    await listen(proxyServer);
    timer(() => rejectCompletion(new Error(`${name}: 2500 ms watchdog expired`)), 2500);
    const client = track(net.createConnection({ host: '127.0.0.1', port: proxyServer.address().port }));
    client.once('connect', () => events.push('client-tcp-connected'));
    client.on('data', bytes => {
      observed.clientReceived += bytes.toString();
      events.push('client-received-data');
    });
    if (sendProtocolMessage) {
      attachClient(client, protocol, () => {
        events.push('protocol-complete');
        resolveCompletion();
      });
      client.once('close', () => {
        if (!events.includes('protocol-complete')) rejectCompletion(new Error('Closed before PONG'));
      });
    } else {
      client.once('close', resolveCompletion);
    }
    await completion;
    assert.ok(events.includes('client-tcp-connected'));
    if (sendProtocolMessage) {
      assert.equal(observed.upstreamAttempts, 1);
      assert.equal(observed.upstreamConnections, 1);
      assert.equal(observed.upstreamReceived, 'PING\n');
      assert.equal(observed.clientReceived, 'PONG\n');
      assert.ok(events.indexOf('downstream-received-data') < events.indexOf('upstream-connect-start'));
      assert.ok(events.includes('protocol-complete'));
      assert.ok(!events.includes('inspection-expired'));
    } else {
      assert.equal(observed.upstreamAttempts, 0);
      assert.equal(observed.upstreamConnections, 0);
      assert.equal(observed.upstreamReceived, '');
      assert.equal(observed.clientReceived, '');
      assert.ok(events.includes('inspection-expired'));
    }
  } finally {
    for (const handle of timers) clearTimeout(handle);
    for (const socket of sockets) socket.destroy();
    await Promise.all([upstreamServer, proxyServer].map(server => new Promise((resolve, reject) => {
      if (!server.listening) return resolve();
      server.close(error => error ? reject(error) : resolve());
    })));
  }
  return { scenario: name, result: 'PASS', ...observed, events, cleanup: 'sockets destroyed; listeners closed; timers cleared' };
}

console.log(JSON.stringify({ node: process.version, kind: 'local adapter-contract model', protocol, policy }));
console.log(JSON.stringify(await runScenario('client-first PING/PONG', true)));
console.log(JSON.stringify(await runScenario('TCP-only idle control', false)));
console.log('PASS: 2 scenarios; 2500 ms watchdog per scenario; no retries.');
