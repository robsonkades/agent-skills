import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL';
const report='C:/git/agent-skills/docs/skill-validation/wave3-2026-10-16';
const temp='C:/Users/robso/AppData/Local/Temp/';
const allowed=new Set(['.log','.txt','.json','.java','.mjs','.js','.sh','.ps1','.xml','.csv','.md','.yaml','.yml','.awk','.out','.err','.properties','.sql','.jfc','.folded','.gradle','.kts','.pom']);
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const records=[];
for(const handoff of fs.readdirSync(report+'/handoffs').filter(f=>f.endsWith('.json'))){
 const files=new Set(),seen=new Set();
 function add(p){p=p.replaceAll('\\','/').replace(/\/$/,'');if(!p.startsWith(temp)||p===root||p===temp.slice(0,-1)||/\/(?:gradle-user-home|\.gradle|daemon|caches)\//.test(p)||seen.has(p)||!fs.existsSync(p))return;seen.add(p);const s=fs.statSync(p);if(s.isDirectory()){for(const e of fs.readdirSync(p,{withFileTypes:true}))if(!e.isSymbolicLink()&&!['.git','node_modules','classes','cache','.cache','plugins','gradle-user-home','.gradle','daemon','caches'].includes(e.name))add(p+'/'+e.name);}else if(s.isFile()&&allowed.has(path.extname(p).toLowerCase())&&s.size<=1048576)files.add(p);}
 function visit(v){if(typeof v==='string'&&/^C:[/\\]/.test(v))add(v);else if(Array.isArray(v))v.forEach(visit);else if(v&&typeof v==='object')Object.values(v).forEach(visit);}
 visit(JSON.parse(fs.readFileSync(report+'/handoffs/'+handoff,'utf8')));
 const entries=[],skill=handoff.slice(0,-5);
 for(const f of [...files].sort().filter(p=>p!==root+'/'+handoff)){
  const b=fs.readFileSync(f),hash=sha(b);
  const thirdParty=/\.(java|js|mjs|sh|ps1|cpp|hpp)$/i.test(f)&&/copyright|licensed under|SPDX-License-Identifier/i.test(b.subarray(0,2500).toString('utf8'));
  if(thirdParty){entries.push({path:f,bytes:b.length,sha256:hash,archived:false,reason:'Third-party source remains referenced, not vendored.'});continue;}
  const target=`evidence/${skill}/${sha(f).slice(0,12)}-${path.basename(f)}.txt`;
  fs.mkdirSync(path.dirname(report+'/'+target),{recursive:true});fs.writeFileSync(report+'/'+target,b);
  entries.push({path:f,bytes:b.length,sha256:hash,archived:true,target});
 }
 let embeddedEvidence;
 if(!entries.length){
  const rawHandoff='handoffs-raw/'+skill+'.json.txt',raw=fs.readFileSync(report+'/'+rawHandoff),h=JSON.parse(raw),executedChecks=[];
  for(const key of ['checks','commands'])for(const [i,c] of (h[key]??[]).entries()){
   const result=c.result??c;
   if(typeof c.command==='string'&&typeof result.chunk_id==='string'&&Number.isInteger(result.exit_code)&&typeof result.output==='string')executedChecks.push({pointer:`/${key}/${i}`,exitCode:result.exit_code,stdoutSha256:sha(result.output)});
  }
  assert.ok(executedChecks.length,skill+' has no archived command evidence');
  assert.ok(Array.isArray(h.sources)&&h.sources.length,skill+' has no primary-source record');
  embeddedEvidence={rawHandoff,sha256:sha(raw),executedChecks,sourceListPointer:'/sources',sourceCount:h.sources.length,classification:'Executed structural command records and primary-source references stored inside the original handoff; not runtime or agent behavioral execution.'};
 }
 records.push({skill,entries,...(embeddedEvidence?{embeddedEvidence}:{})});
}
const manifest={createdAt:new Date().toISOString(),policy:'Byte-preserving copies of explicitly referenced small text evidence. No binaries, recordings, dependencies, large dumps or third-party implementation sources. .txt suffix preserves raw logs. This archive itself is not a test execution.',records};
fs.writeFileSync(report+'/evidence-manifest.json',JSON.stringify(manifest,null,2)+'\n');
console.log(JSON.stringify({skills:records.length,archived:records.reduce((s,r)=>s+r.entries.filter(e=>e.archived).length,0),referenceOnly:records.reduce((s,r)=>s+r.entries.filter(e=>!e.archived).length,0),embeddedEvidenceReviews:records.filter(r=>r.embeddedEvidence).length},null,2));
