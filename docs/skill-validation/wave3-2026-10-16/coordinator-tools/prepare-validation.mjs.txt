import fs from 'node:fs';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL';
const report='C:/git/agent-skills/docs/skill-validation/wave3-2026-10-16';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const queue=JSON.parse(fs.readFileSync(report+'/queue.json','utf8'));
assert.equal(queue.tasks.length,63);
assert.ok(queue.tasks.every(t=>t.status==='implemented'&&t.diffReviewed));
const handoffs=[],helpers=[];
fs.mkdirSync(report+'/handoffs-raw',{recursive:true});fs.mkdirSync(report+'/coordinator-tools',{recursive:true});
for(const t of queue.tasks){const bytes=fs.readFileSync(root+'/'+t.name+'.json');const h=JSON.parse(bytes);assert.equal(h.allWritesStopped,true);assert.deepEqual(h,JSON.parse(fs.readFileSync(report+'/handoffs/'+t.name+'.json','utf8')));const raw='handoffs-raw/'+t.name+'.json.txt';fs.writeFileSync(report+'/'+raw,bytes);handoffs.push({skill:t.name,raw,sha256:sha(bytes),source:root+'/'+t.name+'.json'});}
for(const name of ['coordinator.mjs','audit-wave.mjs','archive-evidence.mjs','prepare-behavior.mjs','grade-behavior.mjs','finalize-behavior.mjs','audit-report.mjs','run-global.mjs','finish-wave.mjs','prepare-validation.mjs']){const bytes=fs.readFileSync(root+'/'+name);const target='coordinator-tools/'+name+'.txt';fs.writeFileSync(report+'/'+target,bytes);helpers.push({source:root+'/'+name,target,sha256:sha(bytes),note:'Coordinator helper snapshot; paths are environment-specific. Archival copy, not shipped runtime code.'});}
fs.writeFileSync(report+'/review-provenance.json',JSON.stringify({capturedAt:new Date().toISOString(),allReviewersStopped:true,handoffs,helpers},null,2)+'\n');
console.log(JSON.stringify({stopped:handoffs.length,helperSnapshots:helpers.length},null,2));
