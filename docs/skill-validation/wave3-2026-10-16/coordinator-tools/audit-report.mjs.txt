import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const report='C:/git/agent-skills/docs/skill-validation/wave3-2026-10-16';
const read=p=>JSON.parse(fs.readFileSync(report+'/'+p,'utf8'));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const queue=read('queue.json');
assert.equal(queue.tasks.length,63);
assert.ok(queue.tasks.every(t=>['implemented','validated'].includes(t.status)&&t.diffReviewed));
const notes=read('review-notes.json');
assert.deepEqual(Object.keys(notes).sort(),queue.tasks.map(t=>t.name).sort());
const evidence=read('evidence-manifest.json');
assert.deepEqual(evidence.records.map(r=>r.skill).sort(),queue.tasks.map(t=>t.name).sort());
let evidenceFiles=0,behaviorFiles=0,bytes=0,embeddedEvidenceReviews=0;
for(const r of evidence.records){
 assert.ok(r.entries.length>0||r.embeddedEvidence,r.skill+' evidence');
 for(const e of r.entries.filter(e=>e.archived)){const b=fs.readFileSync(report+'/'+e.target);assert.equal(sha(b),e.sha256);assert.equal(b.length,e.bytes);evidenceFiles++;bytes+=b.length;}
 if(r.embeddedEvidence){
  const e=r.embeddedEvidence,raw=fs.readFileSync(report+'/'+e.rawHandoff),h=JSON.parse(raw);
  assert.equal(sha(raw),e.sha256);assert.deepEqual(h,read('handoffs/'+r.skill+'.json'));
  assert.ok(e.executedChecks.length>0);
  for(const c of e.executedChecks){const record=c.pointer.split('/').slice(1).reduce((v,k)=>v[k],h),result=record.result??record;assert.equal(typeof record.command,'string');assert.equal(typeof result.chunk_id,'string');assert.equal(result.exit_code,c.exitCode);assert.equal(sha(result.output),c.stdoutSha256);}
  const sources=e.sourceListPointer.split('/').slice(1).reduce((v,k)=>v[k],h);assert.ok(Array.isArray(sources)&&sources.length>0);assert.equal(sources.length,e.sourceCount);
  embeddedEvidenceReviews++;
 }
}
const artifacts=read('behavior-artifact-manifest.json');
for(const a of artifacts.artifacts){const b=fs.readFileSync(report+'/'+a.path);assert.equal(sha(b),a.sha256);assert.equal(b.length,a.bytes);behaviorFiles++;bytes+=b.length;}
const provenance=read('review-provenance.json');
assert.equal(provenance.handoffs.length,63);
for(const h of provenance.handoffs){const b=fs.readFileSync(report+'/'+h.raw);assert.equal(sha(b),h.sha256);assert.deepEqual(JSON.parse(b),read('handoffs/'+h.skill+'.json'));}
for(const h of provenance.helpers)assert.equal(sha(fs.readFileSync(report+'/'+h.target)),h.sha256);
const freeze=read('behavior-freeze.json'),raw=fs.readFileSync(report+'/behavior-manifest.frozen.json.txt');
assert.equal(sha(raw),freeze.manifestSha256);assert.deepEqual(JSON.parse(raw),read('behavior-manifest.json'));
const manifest=JSON.parse(raw),behavior=read('behavior-results.json');
assert.equal(behavior.runs.length,6);
for(const r of manifest.runs){assert.ok(r.criteria.expected.length&&r.criteria.failures.length);assert.equal(sha(JSON.stringify(r.criteria)),r.criteriaSha256);assert.equal(behavior.runs.find(x=>x.runId===r.runId).criteriaSha256,r.criteriaSha256);}
assert.ok(read('behavior-technical-checks.json').records.every(r=>r.exit===0));
for(const t of queue.tasks){const h=read('handoffs/'+t.name+'.json');assert.equal(h.allWritesStopped,true);assert.ok(['READY_FOR_GLOBAL_VALIDATION','NO_CHANGES_NEEDED'].includes(h.status));}
const result={checkedAt:new Date().toISOString(),passed:true,scope:'Report integrity and completeness, not new technical or behavioral execution',skills:63,evidenceFiles,embeddedEvidenceReviews,behaviorFiles,archivedBytes:bytes,frozenBehaviorManifestSha256:sha(raw),behaviorCriteriaHashesVerified:true,allHandoffsStopped:true,notesComplete:true};
fs.writeFileSync(report+'/report-integrity.json',JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result,null,2));
