import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import {spawnSync} from 'node:child_process';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL';
const report='C:/git/agent-skills/docs/skill-validation/wave3-2026-10-16';
const bytes=fs.readFileSync(report+'/behavior-manifest.frozen.json.txt');
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const freeze=JSON.parse(fs.readFileSync(report+'/behavior-freeze.json','utf8'));
assert.equal(sha(bytes),freeze.manifestSha256);
assert.deepEqual(JSON.parse(bytes),JSON.parse(fs.readFileSync(report+'/behavior-manifest.json','utf8')));
const manifest=JSON.parse(bytes), records=[];
const execution=JSON.parse(fs.readFileSync(report+'/behavior-execution.json','utf8'));
for(const run of manifest.runs){
 assert.ok(run.criteria.expected.length&&run.criteria.failures.length);
 assert.equal(sha(JSON.stringify(run.criteria)),run.criteriaSha256);
 const allowed=new Set(['decision.md','proxy-policy.json']);
 const changed=run.files.filter(f=>!fs.existsSync(run.directory+'/'+f.path)||sha(fs.readFileSync(run.directory+'/'+f.path))!==f.sha256).map(f=>f.path);
 assert.ok(changed.includes('decision.md')); assert.deepEqual(changed.filter(p=>!allowed.has(p)),[]);
 const expected={listener:'127.0.0.1:15001',route:'fixed-service-a',connect_start:run.case==='C1-server-greeting'?'immediate':'first_client_data',inspection_timeout_ms:1000,inspection_buffer_limit:4096};
 assert.deepEqual(JSON.parse(fs.readFileSync(run.directory+'/proxy-policy.json','utf8')),expected);
 if(run.case==='C2-client-first')assert.ok(!changed.includes('proxy-policy.json'));
 const script=fs.existsSync(run.directory+'/local-handshake-check.mjs')?'local-handshake-check.mjs':fs.existsSync(run.directory+'/local-check.mjs')?'local-check.mjs':'verify-handshake.mjs';
 assert.ok(fs.existsSync(run.directory+'/REVIEW.md'));
 const dest=root+'/behavior-grader/'+run.runId;
 if(fs.existsSync(dest))throw Error('Preserve previous grading run');
 fs.cpSync(run.directory,dest,{recursive:true});
 const result=spawnSync(process.execPath,[script],{cwd:dest,encoding:'utf8',windowsHide:true,timeout:30000,maxBuffer:4*1024*1024});
 const log=report+'/behavior-grader/'+run.runId+'.log.txt';fs.mkdirSync(path.dirname(log),{recursive:true});fs.writeFileSync(log,(result.stdout||'')+(result.stderr||''));
 const entry={runId:run.runId,case:run.case,condition:run.condition,changedSuppliedFiles:changed,inputAndResourceIntegrityPassed:true,policyContractPassed:true,command:`${process.execPath} ${script}`,cwd:dest,exit:result.status,error:result.error?.message,log:path.relative(report,log).replaceAll('\\','/'),scriptSha256:sha(fs.readFileSync(dest+'/'+script)),classification:'Coordinator rerun of actor-produced bounded loopback check; not a new actor repetition or real proxy integration'};
 records.push(entry);
 Object.assign(execution.runs.find(r=>r.runId===run.runId),{status:'completed-awaiting-grading',allWritesStopped:true});
 console.log(JSON.stringify({run:run.runId,exit:result.status,tail:result.stdout?.split(/\r?\n/).slice(-2).join('\n').slice(0,160)}));
}
const result={checkedAt:new Date().toISOString(),frozenManifestSha256:sha(bytes),criteriaHashesMatch:true,records};
fs.writeFileSync(report+'/behavior-technical-checks.json',JSON.stringify(result,null,2)+'\n');
execution.updatedAt=new Date().toISOString();fs.writeFileSync(report+'/behavior-execution.json',JSON.stringify(execution,null,2)+'\n');
assert.ok(records.every(r=>r.exit===0));
