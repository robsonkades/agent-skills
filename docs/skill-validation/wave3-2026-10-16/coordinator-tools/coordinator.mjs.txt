import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {spawnSync} from 'node:child_process';
const repo='C:/git/agent-skills', root='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL';
const report=path.join(repo,'docs/skill-validation/wave3-2026-10-16');
const queue=JSON.parse(fs.readFileSync(path.join(report,'queue.json'),'utf8'));
const base=JSON.parse(fs.readFileSync(path.join(report,'baseline.json'),'utf8'));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const item=n=>{const x=queue.tasks.find(t=>t.name===n);if(!x)throw Error(n);return x};
const diff=n=>{const r=spawnSync('git',['diff','--no-index','--no-ext-diff','--',path.join(root,'baseline',n),path.join(repo,'skills',n)],{cwd:repo,encoding:'utf8',maxBuffer:16*1024*1024});if(r.status!==0&&r.status!==1)throw Error(r.stderr);return r.stdout};
for(const t of queue.tasks){const p=path.join(root,`${t.name}.json`);if(!fs.existsSync(p))continue;const bytes=fs.readFileSync(p);if(t.ignoredHandoffSha===sha(bytes))continue;const h=JSON.parse(bytes.toString('utf8'));if(h.allWritesStopped!==true)continue;fs.copyFileSync(p,path.join(report,'handoffs',`${t.name}.json`));if(t.status!=='validated')t.status='implemented';t.reviewResult=h.status;t.finalVersion=fs.readFileSync(path.join(repo,'skills',t.name,'skill.yaml'),'utf8').match(/^version:\s*(.+)$/m)[1];}
for(const arg of process.argv.slice(2)){
 if(arg==='snapshot'){for(const p of base.packages){const dest=path.join(root,'baseline',p.name);if(fs.existsSync(dest))throw Error('Already snapshotted');fs.cpSync(path.join(repo,'skills',p.name),dest,{recursive:true});}continue;}
 const [action,value]=arg.split('=');
 if(action==='assign'){for(const pair of value.split(',')){const [owner,n]=pair.split(':');const t=item(n);if(t.status!=='pending')throw Error(`Not pending ${n}`);if(queue.tasks.some(x=>x.owner===owner&&x.status==='running'))throw Error(`Owner busy ${owner}`);Object.assign(t,{owner,status:'running',validation:'local pending; global deferred'});}}
 else if(action==='reopen'){const t=item(value);const bytes=fs.readFileSync(path.join(root,`${t.name}.json`));fs.mkdirSync(path.join(report,'handoff-revisions'),{recursive:true});fs.writeFileSync(path.join(report,'handoff-revisions',`${t.name}-${sha(bytes).slice(0,12)}.json`),bytes);Object.assign(t,{status:'running',diffReviewed:false,ignoredHandoffSha:sha(bytes),validation:'cross-skill finding under review; global deferred'});}
 else if(action==='diff'){for(const n of value.split(',')){const d=diff(n);fs.writeFileSync(path.join(report,'diffs',`${n}.patch`),d);console.log(`SKILL ${n}\n${d||'(no changes)'}`);} }
 else if(action==='review'){for(const n of value.split(',')){const t=item(n);if(t.status!=='implemented')throw Error(`Not handed off ${n}`);const d=diff(n);fs.writeFileSync(path.join(report,'diffs',`${n}.patch`),d);Object.assign(t,{diffReviewed:true,reviewedDiffSha256:sha(d),validation:'local checks and coordinator diff review passed; global deferred'});}}
 else throw Error(`Unknown ${arg}`);
}
queue.updatedAt=new Date().toISOString();fs.writeFileSync(path.join(report,'queue.json'),JSON.stringify(queue,null,2)+'\n');
fs.writeFileSync(path.join(report,'tasks.md'),'# Onda 3 — fila de revisão\n\nEscopo: 63 skills. Conclusão local aguarda validação global.\n\n| Skill | Responsável | Estado | Validação |\n| --- | --- | --- | --- |\n'+queue.tasks.map(t=>`| ${t.name} | ${t.owner||'—'} | ${t.status} | ${t.validation} |`).join('\n')+'\n');
if(!process.argv.some(x=>x.startsWith('diff=')))console.log(JSON.stringify({counts:queue.tasks.reduce((a,t)=>(a[t.status]=(a[t.status]||0)+1,a),{}),ready:queue.tasks.filter(t=>t.status==='implemented'&&!t.diffReviewed).map(t=>t.name),running:queue.tasks.filter(t=>t.status==='running').map(t=>`${t.owner}:${t.name}`),next:queue.tasks.filter(t=>t.status==='pending').slice(0,20).map(t=>t.name)},null,2));
