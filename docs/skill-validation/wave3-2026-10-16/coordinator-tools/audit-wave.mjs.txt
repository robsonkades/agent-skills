import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {execFileSync,spawnSync} from 'node:child_process';
import YAML from 'file:///C:/git/agent-skills/node_modules/yaml/dist/index.js';
const repo='C:/git/agent-skills', root='C:/Users/robso/AppData/Local/Temp/skills-wave3-validation-wyg7TL';
const report=repo+'/docs/skill-validation/wave3-2026-10-16';
const base=JSON.parse(fs.readFileSync(report+'/baseline.json','utf8'));
const queue=JSON.parse(fs.readFileSync(report+'/queue.json','utf8'));
const names=new Set(queue.tasks.map(t=>t.name));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const walk=d=>fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(d,e.name)):[path.join(d,e.name)]);
const allowed=p=>p.startsWith('docs/skill-validation/wave3-2026-10-16/')||p==='registry/skills.yaml'||(p.startsWith('skills/')&&names.has(p.split('/')[1]));
const current=execFileSync('git',['ls-files','-z','--cached','--others','--exclude-standard'],{cwd:repo,encoding:'utf8',maxBuffer:32*1024*1024}).split('\0').filter(Boolean);
const allBase=new Map(base.allFiles.map(f=>[f.path,f.sha256]));
const unexpected=[];
for(const [p,h] of allBase)if(!allowed(p)&&(!fs.existsSync(repo+'/'+p)||sha(fs.readFileSync(repo+'/'+p))!==h))unexpected.push({path:p,reason:'changed or missing baseline file'});
for(const p of current)if(!allowed(p)&&!allBase.has(p))unexpected.push({path:p,reason:'new file outside wave'});
const packages=base.packages.map(p=>{
 const old=new Map(p.files.map(f=>[f.path,f.sha256]));
 const now=walk(repo+'/skills/'+p.name).map(f=>path.relative(repo,f).replaceAll('\\','/'));
 const changedFiles=[...new Set([...now,...old.keys()])].filter(f=>!fs.existsSync(repo+'/'+f)||sha(fs.readFileSync(repo+'/'+f))!==old.get(f));
 const manifest=YAML.parse(fs.readFileSync(repo+'/skills/'+p.name+'/skill.yaml','utf8'));
 const front=YAML.parse(fs.readFileSync(repo+'/skills/'+p.name+'/SKILL.md','utf8').match(/^---\r?\n([\s\S]*?)\r?\n---/)[1]);
 const task=queue.tasks.find(t=>t.name===p.name);
 const d=spawnSync('git',['diff','--no-index','--no-ext-diff','--',path.join(root,'baseline',p.name),path.join(repo,'skills',p.name)],{cwd:repo,encoding:'utf8',maxBuffer:16*1024*1024});if(d.status!==0&&d.status!==1)throw Error(d.stderr);
 const hp=report+'/handoffs/'+p.name+'.json';const h=fs.existsSync(hp)?JSON.parse(fs.readFileSync(hp,'utf8')):null;
 return {name:p.name,initialVersion:p.initialVersion,finalVersion:manifest.version,changedFiles,descriptionMatches:front.description.trim()===manifest.description.trim(),versionCorrect:manifest.version===(changedFiles.length?p.initialVersion.replace(/(\d+)$/,n=>String(+n+1)):p.initialVersion),diffReviewed:task.diffReviewed===true,reviewedDiffStillMatches:sha(d.stdout)===task.reviewedDiffSha256,handoffPresent:!!h,allWritesStopped:h?.allWritesStopped===true,reviewStatus:h?.status,status:task.status};
});
const oldIndex=YAML.parse(base.registryBefore),newIndex=YAML.parse(fs.readFileSync(repo+'/registry/skills.yaml','utf8'));
const oldEntries=new Map(oldIndex.skills.map(x=>[x.name,x]));
const indexChanges=newIndex.skills.filter(x=>JSON.stringify(x)!==JSON.stringify(oldEntries.get(x.name))).map(x=>x.name);
const lostIndexEntries=oldIndex.skills.filter(x=>!newIndex.skills.some(y=>y.name===x.name)).map(x=>x.name);
const final=process.argv.includes('--final');
const expectedIndexChanges=packages.filter(p=>p.changedFiles.length).map(p=>p.name).sort();
const indexMatchesPackageChanges=JSON.stringify([...indexChanges].sort())===JSON.stringify(expectedIndexChanges);
const issues=packages.filter(p=>!p.descriptionMatches||!p.versionCorrect||(final&&(!p.diffReviewed||!p.reviewedDiffStillMatches||!p.handoffPresent||!p.allWritesStopped||p.reviewStatus==='BLOCKED')));
const result={at:new Date().toISOString(),classification:'Coordinator preservation/version audit; not behavioral evaluation',baselineHead:base.head,total:packages.length,implemented:packages.filter(p=>['implemented','validated'].includes(p.status)).length,changed:packages.filter(p=>p.changedFiles.length).length,unchanged:packages.filter(p=>!p.changedFiles.length).length,unexpected,indexChanges,indexMatchesPackageChanges,indexChangesOutsideWave:indexChanges.filter(n=>!names.has(n)),lostIndexEntries,packages};
fs.writeFileSync(report+'/'+(final?'preservation-final.json':'preservation-current.json'),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify({...result,packages:undefined,issues,indexChanges:indexChanges.length},null,2));
if(final&&(result.implemented!==63||!indexMatchesPackageChanges||unexpected.length||result.indexChangesOutsideWave.length||lostIndexEntries.length||issues.length))process.exitCode=1;
