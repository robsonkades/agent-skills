function run(exe, args, cwd = dir) {
  const r = spawnSync(exe, args, { cwd, encoding: 'utf8', timeout: 20000 });
  const record = { command: exe, args, cwd, exitCode: r.status, stdout: r.stdout, stderr: r.stderr, error: r.error?.message };
  commands.push(record);
  write('commands.json', JSON.stringify(commands, null, 2));
  assert.equal(r.status, 0, JSON.stringify(record));
  return r.stdout;
}
