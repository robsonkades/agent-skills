import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { spawnSync } from 'node:child_process';
import assert from 'node:assert/strict';
const repo = 'C:/git/agent-skills';
const skill = 'java-memory-model';
const root = 'C:/Users/robso/AppData/Local/Temp/skills-wave5-validation-ujPBF2';
const dir = `${root}/${skill}`;
const sha = b => crypto.createHash('sha256').update(b).digest('hex');
const commands = [];
function git(args) {
  const r = spawnSync('git', args, { cwd: repo, encoding: 'utf8', timeout: 15000 });
  commands.push({ command: 'git', args, cwd: repo, exitCode: r.status, stdout: r.stdout, stderr: r.stderr });
  assert.equal(r.status, 0);
  return r.stdout;
}
const names = ['SKILL.md', 'skill.yaml', 'references/happens-before.md', 'references/review-checklist.md'];
const files = names.map(name => {
  const bytes = fs.readFileSync(`${repo}/skills/${skill}/${name}`);
  assert.ok(bytes.equals(fs.readFileSync(`${root}/baseline/${skill}/${name}`)), `Coordinator baseline differs: ${name}`);
  return { name, sha256: sha(bytes), content: bytes.toString('utf8'), bytes: bytes.length };
});
const status = git(['status', '--short', '--untracked-files=all', '--', `skills/${skill}`]);
const unstaged = git(['diff', '--', `skills/${skill}`]);
const staged = git(['diff', '--cached', '--', `skills/${skill}`]);
const committed = git(['show', `HEAD:skills/${skill}/skill.yaml`]);
assert.equal(status + unstaged + staged, '');
assert.match(committed, /version: 1\.3\.5/);
fs.mkdirSync(`${dir}/sources`, { recursive: true });
fs.writeFileSync(`${dir}/assignment-baseline.json`, JSON.stringify({ skill, capturedAt: new Date().toISOString(), version: '1.3.5', committed, status, unstaged, staged, files, commands }, null, 2));
const sourceSpecs = [
  ['jls-25-17', 'https://docs.oracle.com/javase/specs/jls/se25/html/jls-17.html', '17.2 wait/notify; 17.4.4/5 lifecycle and synchronization; 17.5/17.5.1 final-field reachability; 17.6/7 atomicity.'],
  ['jls-25-12', 'https://docs.oracle.com/javase/specs/jls/se25/html/jls-12.html', '12.4.2 initialization, including recursive initialization and failure.'],
  ['concurrent-25', 'https://docs.oracle.com/en/java/javase/25/docs/api/java.base/java/util/concurrent/package-summary.html', 'API-level memory consistency and exact handoff clauses.'],
  ['thread-25', 'https://docs.oracle.com/en/java/javase/25/docs/api/java.base/java/lang/Thread.html', 'join/timed join/isAlive/interrupt lifecycle distinctions.'],
  ['object-25', 'https://docs.oracle.com/en/java/javase/25/docs/api/java.base/java/lang/Object.html', 'wait releases only its monitor and reacquires; notification/predicate requirements.'],
  ['atomic-25', 'https://docs.oracle.com/en/java/javase/25/docs/api/java.base/java/util/concurrent/atomic/package-summary.html', 'Individual atomic operations and arrays do not imply multi-element snapshot.']
];
const sources = await Promise.all(sourceSpecs.map(async ([id, url, supports]) => {
  try {
    const r = await fetch(url, { signal: AbortSignal.timeout(25000) });
    const text = await r.text();
    const file = `${dir}/sources/${id}.html`;
    fs.writeFileSync(file, text);
    return { id, url, resolvedUrl: r.url, status: r.status, consultedAt: new Date().toISOString(), version: 'Java SE25', supports, file, sha256: sha(text), bytes: Buffer.byteLength(text), memoryVisibilityAnchor: id === 'concurrent-25' ? /(?:id|name)="MemoryVisibility"/.test(text) : undefined };
  } catch (e) { return { id, url, supports, error: e.message }; }
}));
fs.writeFileSync(`${dir}/sources/source-index.json`, JSON.stringify(sources, null, 2));
assert.ok(sources.every(s => s.status === 200 && !s.error));
console.log(JSON.stringify({ cleanBaseline: true, version: '1.3.5', files: files.length, sources: sources.map(({ id, status, memoryVisibilityAnchor }) => ({ id, status, memoryVisibilityAnchor })) }, null, 2));
