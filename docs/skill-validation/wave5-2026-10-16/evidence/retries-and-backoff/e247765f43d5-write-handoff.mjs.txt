function evidenceFiles(dir) {
  return readdirSync(dir).flatMap((name) => {
    const path = join(dir, name);
    if (statSync(path).isDirectory()) return evidenceFiles(path);
    if (!/\.(?:mjs|java|json)$/.test(name)) return [];
    return [{ path, sha256: hash(path), bytes: statSync(path).size,
      classification: /criteria|negative/.test(path) ? 'evaluator-only' : /\.input\.json$/.test(path) ? 'actor-input' : 'technical-review-evidence' }];
  });
}
