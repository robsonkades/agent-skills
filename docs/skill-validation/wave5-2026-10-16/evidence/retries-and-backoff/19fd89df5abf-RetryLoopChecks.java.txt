import java.time.Duration;
import java.util.ArrayDeque;
import java.util.List;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.ThreadLocalRandom;
import java.util.concurrent.atomic.AtomicReference;

public class RetryLoopChecks {
    static Duration fullJitter(int attempt, Duration base, Duration cap) {
    if (attempt < 0 || base.isNegative() || base.isZero()
            || cap.isNegative() || cap.isZero()) {
        throw new IllegalArgumentException("positive base/cap and non-negative attempt required");
    }
    long baseNanos = base.toNanos(); // ArithmeticException rejects an unrepresentable policy
    long capNanos = cap.toNanos();
    long exponential = attempt >= 63 || baseNanos > (Long.MAX_VALUE >> attempt)
            ? Long.MAX_VALUE : baseNanos << attempt;
    long window = Math.min(capNanos, exponential);
    return Duration.ofNanos(ThreadLocalRandom.current().nextLong(window)); // [0, window)
}

    // Record patterns in the switch below require Java 21; use instanceof/visitor on Java 17.
public sealed interface Outcome<T> {
    record Ok<T>(T value) implements Outcome<T> {}
    record Transient<T>(String code, Duration advisedDelay) implements Outcome<T> {}  // ZERO = none
    record Permanent<T>(String code) implements Outcome<T> {}
    record Ambiguous<T>(String code) implements Outcome<T> {}   // may or may not have been applied
}

    // Conceptual: no metrics, no per-endpoint budget scoping.
<T> T execute(Op<T> op, Policy policy, Deadline deadline, boolean replaySafe)
        throws InterruptedException {
    boolean unresolved = false;
    for (int attempt = 0; ; attempt++) {
        if (Thread.currentThread().isInterrupted()) throw new InterruptedException();
        Duration remaining = deadline.remaining();
        if (remaining.isNegative() || remaining.isZero()
                || remaining.compareTo(policy.expectedCost()) < 0) {
            throw stopped("deadline-exhausted", unresolved);
        }
        Outcome<T> outcome = op.call(deadline);
        Duration advised = Duration.ZERO;
        switch (outcome) {
            case Outcome.Ok<T>(T value) -> { budget.recordSuccess(); return value; }
            case Outcome.Permanent<T> p -> throw stopped(p.code(), unresolved);
            case Outcome.Ambiguous<T> a -> {
                unresolved = true;
                if (!replaySafe) throw stopped(a.code(), true); // pending/unknown, not definite failure
            }
            case Outcome.Transient<T> t -> advised = t.advisedDelay();
        }
        if (attempt + 1 >= policy.maxAttempts()) throw stopped("attempts-exhausted", unresolved);

        Duration local = fullJitter(attempt, policy.base(), policy.cap());
        Duration wait = advised.compareTo(local) > 0 ? advised : local;
        if (deadline.remaining().minus(wait).compareTo(policy.expectedCost()) < 0) {
            throw stopped("deadline-would-be-exceeded", unresolved);
        }
        long waitNanos = wait.toNanos(); // validate conversion before reserving retry budget
        if (!budget.tryAcquire()) throw stopped("retry-budget-exhausted", unresolved);
        TimeUnit.NANOSECONDS.sleep(waitNanos); // Java 17 API; unmounting depends on JDK/context
    }
}


    interface Deadline { Duration remaining(); }
    interface Op<T> { Outcome<T> call(Deadline deadline) throws InterruptedException; }
    record Policy(int maxAttempts, Duration base, Duration cap, Duration expectedCost) {}
    static final class Stopped extends RuntimeException {
        final boolean unresolved;
        Stopped(String reason, boolean unresolved) { super(reason); this.unresolved = unresolved; }
    }
    static Stopped stopped(String reason, boolean unresolved) { return new Stopped(reason, unresolved); }
    static final class Budget {
        int grants;
        int requests;
        int successes;
        final CountDownLatch acquired = new CountDownLatch(1);
        Budget(int grants) { this.grants = grants; }
        boolean tryAcquire() {
            requests++;
            if (grants == 0) return false;
            grants--;
            acquired.countDown();
            return true;
        }
        void recordSuccess() { successes++; }
    }
    Budget budget;
    RetryLoopChecks(int grants) { budget = new Budget(grants); }
    static final class Script implements Op<String> {
        final ArrayDeque<Outcome<String>> outcomes;
        int calls;
        Script(List<Outcome<String>> outcomes) { this.outcomes = new ArrayDeque<>(outcomes); }
        public Outcome<String> call(Deadline deadline) { calls++; return outcomes.removeFirst(); }
    }
    static Policy policy(int attempts) {
        return new Policy(attempts, Duration.ofNanos(1), Duration.ofNanos(1), Duration.ofMillis(1));
    }
    static Deadline enough() { return () -> Duration.ofSeconds(5); }
    static void check(boolean ok, String message) { if (!ok) throw new AssertionError(message); }
    static void expectStop(String name, List<Outcome<String>> outcomes, boolean safe,
                           Policy policy, Deadline deadline, int grants, String reason,
                           boolean unknown, int calls, int budgetRequests) throws Exception {
        var loop = new RetryLoopChecks(grants);
        var op = new Script(outcomes);
        try { loop.execute(op, policy, deadline, safe); throw new AssertionError(name + " unexpectedly succeeded"); }
        catch (Stopped stopped) {
            check(reason.equals(stopped.getMessage()), name + " reason=" + stopped.getMessage());
            check(stopped.unresolved == unknown, name + " unresolved=" + stopped.unresolved);
        }
        check(op.calls == calls, name + " attempts=" + op.calls);
        check(loop.budget.requests == budgetRequests, name + " retry grants requested=" + loop.budget.requests);
        System.out.println("PASS " + name + " attempts=" + op.calls + " unresolved=" + unknown);
    }
    static void preInterrupted() throws Exception {
        var loop = new RetryLoopChecks(3);
        var op = new Script(List.of(new Outcome.Ok<>("receipt")));
        Thread.currentThread().interrupt();
        try {
            loop.execute(op, policy(3), enough(), true);
            throw new AssertionError("pre-interrupted entry executed the operation");
        } catch (InterruptedException expected) {
            check(op.calls == 0 && loop.budget.requests == 0, "pre-interrupted entry must make zero attempts");
        } finally { Thread.interrupted(); }
        System.out.println("PASS pre-interrupted entry attempts=0");
    }
    static void interruptBackoff() throws Exception {
        var loop = new RetryLoopChecks(3);
        var op = new Script(List.of(new Outcome.Transient<>("busy", Duration.ofSeconds(60)), new Outcome.Ok<>("receipt")));
        var failure = new AtomicReference<Throwable>();
        Thread worker = new Thread(() -> {
            try {
                loop.execute(op, policy(3), () -> Duration.ofSeconds(120), true);
                failure.set(new AssertionError("backoff interruption returned success"));
            } catch (InterruptedException expected) { /* propagated */ }
            catch (Throwable caught) { failure.set(caught); }
        }, "retry-check-interruption");
        worker.start();
        try {
            check(loop.budget.acquired.await(5, TimeUnit.SECONDS), "retry budget not reached");
        } finally {
            worker.interrupt();
            worker.join(5000);
        }
        check(!worker.isAlive(), "worker did not terminate");
        if (failure.get() != null) throw new AssertionError("worker failed", failure.get());
        check(op.calls == 1, "interrupted wait started another attempt");
        System.out.println("PASS interrupted backoff attempts=1 workerStopped=true");
    }
    public static void main(String[] args) throws Exception {
        preInterrupted();
        if (args.length > 0) return;
        expectStop("expired entry", List.of(new Outcome.Ok<>("receipt")), true, policy(3),
                () -> Duration.ZERO, 3, "deadline-exhausted", false, 0, 0);
        expectStop("insufficient attempt time", List.of(new Outcome.Ok<>("receipt")), true, policy(3),
                () -> Duration.ofNanos(1), 3, "deadline-exhausted", false, 0, 0);
        expectStop("terminal", List.of(new Outcome.Permanent<>("validation")), true, policy(3),
                enough(), 3, "validation", false, 1, 0);
        expectStop("unsafe ambiguous", List.of(new Outcome.Ambiguous<>("read-timeout")), false, policy(3),
                enough(), 3, "read-timeout", true, 1, 0);
        expectStop("ambiguous then terminal", List.of(new Outcome.Ambiguous<>("read-timeout"), new Outcome.Permanent<>("rejected")), true,
                policy(3), enough(), 3, "rejected", true, 2, 1);
        expectStop("unknown attempts exhausted", List.of(new Outcome.Ambiguous<>("read-timeout")), true,
                policy(1), enough(), 3, "attempts-exhausted", true, 1, 0);
        expectStop("unknown budget exhausted", List.of(new Outcome.Ambiguous<>("read-timeout")), true,
                policy(3), enough(), 0, "retry-budget-exhausted", true, 1, 1);
        expectStop("server delay beyond deadline", List.of(new Outcome.Transient<>("busy", Duration.ofSeconds(10))), true,
                policy(3), enough(), 3, "deadline-would-be-exceeded", false, 1, 0);
        var times = new ArrayDeque<>(List.of(Duration.ofSeconds(5), Duration.ofSeconds(5), Duration.ZERO));
        expectStop("deadline expired before next admission", List.of(new Outcome.Transient<>("busy", Duration.ZERO), new Outcome.Ok<>("receipt")),
                true, policy(3), times::removeFirst, 3, "deadline-exhausted", false, 1, 1);
        for (boolean ambiguous : new boolean[] {false, true}) {
            var loop = new RetryLoopChecks(3);
            Outcome<String> first = ambiguous ? new Outcome.Ambiguous<>("read-timeout") : new Outcome.Transient<>("busy", Duration.ZERO);
            var op = new Script(List.of(first, new Outcome.Ok<>("original-receipt")));
            check("original-receipt".equals(loop.execute(op, policy(3), enough(), true)), "authoritative receipt");
            check(op.calls == 2 && loop.budget.requests == 1 && loop.budget.successes == 1, "success accounting");
            System.out.println("PASS " + (ambiguous ? "ambiguous" : "transient") + " then authoritative success attempts=2");
        }
        interruptBackoff();
        System.out.println("PASS 13 loop scenarios; integration placeholders, no remote operation or durable store");
    }
}
