import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

const evidence = dirname(fileURLToPath(import.meta.url));
const reference = readFileSync('C:/git/agent-skills/skills/retries-and-backoff/references/retry-in-java.md', 'utf8');
const markdownLinks = [...reference.matchAll(/\]\((?:<([^>]+)>|([^\s)]+))\)/g)].map((match) => match[1] ?? match[2]);
const urls = [...new Set(markdownLinks.filter((url) => url.startsWith('https://')))];
urls.push('https://www.rfc-editor.org/rfc/rfc9110.txt',
  'https://raw.githubusercontent.com/resilience4j/resilience4j/v2.3.0/resilience4j-core/src/main/java/io/github/resilience4j/core/predicate/PredicateCreator.java',
  'https://raw.githubusercontent.com/spring-projects/spring-retry/v2.0.12/src/main/java/org/springframework/retry/backoff/ExponentialRandomBackOffPolicy.java');
const sources = await Promise.all(urls.map(async (url) => {
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(15000) });
    const body = await response.text();
    const record = { url, status: response.status, finalUrl: response.url, consultedAt: new Date().toISOString(),
      contentType: response.headers.get('content-type'), bytes: Buffer.byteLength(body),
      sha256: createHash('sha256').update(body).digest('hex') };
    if (url.endsWith('rfc9110.txt')) {
      const position = body.lastIndexOf('10.2.3.  Retry-After');
      record.selectedSectionFound = position >= 0;
      process.stdout.write('RFC 9110 primary text, selected section:\n' + body.slice(position, position + 1250) + '\n');
    }
    return record;
  } catch (error) { return { url, error: error.message, consultedAt: new Date().toISOString() }; }
}));
writeFileSync(join(evidence, 'source-audit.json'), JSON.stringify({ sources,
  note: 'HTTP availability is not source correctness. Material contracts were read separately; only metadata/hashes retained, no downloaded implementation/documents copied into the package.' }, null, 2) + '\n');
process.stdout.write(JSON.stringify(sources.map(({ url, status, error, selectedSectionFound }) => ({ url, status, error, selectedSectionFound })), null, 2) + '\n');
