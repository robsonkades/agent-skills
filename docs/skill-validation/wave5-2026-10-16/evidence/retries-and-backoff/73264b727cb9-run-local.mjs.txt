function run(label, command, args, cwd = repo, expected = 0) {
  const result = spawnSync(command, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = { label, command, args, cwd, exitCode: result.status, expectedExitCode: expected,
    signal: result.signal, error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  record.log = join(evidence, 'logs', label + '.json');
  writeFileSync(record.log, JSON.stringify(record, null, 2) + '\n');
  commands.push(record);
  assert.equal(record.error, null, label + ': ' + record.error);
  assert.equal(record.signal, null, label);
  assert.equal(record.exitCode, expected, label + '\n' + record.stdout + record.stderr);
  return record;
}
