import java.time.Duration;
import java.util.concurrent.ThreadLocalRandom;

public class JitterChecks {
    static Duration fullJitter(int attempt, Duration base, Duration cap) {
    if (attempt < 0 || base.isNegative() || base.isZero()
            || cap.isNegative() || cap.isZero()) {
        throw new IllegalArgumentException("positive base/cap and non-negative attempt required");
    }
    long baseNanos = base.toNanos(); // ArithmeticException rejects an unrepresentable policy
    long capNanos = cap.toNanos();
    long exponential = attempt >= 63 || baseNanos > (Long.MAX_VALUE >> attempt)
            ? Long.MAX_VALUE : baseNanos << attempt;
    long window = Math.min(capNanos, exponential);
    return Duration.ofNanos(ThreadLocalRandom.current().nextLong(window)); // [0, window)
}

    static int cases;
    static void check(boolean condition, String detail) { if (!condition) throw new AssertionError(detail); }
    static void bounded(int attempt, long base, long cap, long window) {
        for (int sample = 0; sample < 1000; sample++) {
            long value = fullJitter(attempt, Duration.ofNanos(base), Duration.ofNanos(cap)).toNanos();
            check(value >= 0 && value < window, "outside [0," + window + "): " + value);
        }
        cases++;
    }
    static void rejects(Runnable task, Class<? extends Throwable> type) {
        try { task.run(); throw new AssertionError("invalid policy accepted"); }
        catch (Throwable caught) { check(type.isInstance(caught), "wrong rejection: " + caught); }
        cases++;
    }
    public static void main(String[] args) {
        bounded(0, 100, 1000, 100);
        bounded(1, 100, 1000, 200);
        bounded(4, 100, 1000, 1000);
        bounded(62, 100, 1000, 1000);
        bounded(63, 1, 1000, 1000);
        bounded(Integer.MAX_VALUE, 1, 1000, 1000);
        bounded(0, 1, 1, 1);
        rejects(() -> fullJitter(-1, Duration.ofNanos(1), Duration.ofNanos(1)), IllegalArgumentException.class);
        rejects(() -> fullJitter(0, Duration.ZERO, Duration.ofNanos(1)), IllegalArgumentException.class);
        rejects(() -> fullJitter(0, Duration.ofNanos(-1), Duration.ofNanos(1)), IllegalArgumentException.class);
        rejects(() -> fullJitter(0, Duration.ofNanos(1), Duration.ZERO), IllegalArgumentException.class);
        rejects(() -> fullJitter(0, Duration.ofSeconds(Long.MAX_VALUE), Duration.ofNanos(1)), ArithmeticException.class);
        System.out.println("PASS " + cases + " jitter policy cases; 7000 bounded samples, no distribution/performance claim");
    }
}
