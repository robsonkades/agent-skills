$ErrorActionPreference = 'Stop'
$evidenceRoot = 'C:/Users/robso/AppData/Local/Temp/skills-wave5-validation-ujPBF2/spring-transactions-and-events'
$repositoryCache = 'C:/Users/robso/.m2/repository'
$artifacts = @('spring-context','spring-aop','spring-beans','spring-core','spring-expression','spring-tx','spring-jdbc') | ForEach-Object { "$repositoryCache/org/springframework/$_/7.0.9/$_-7.0.9.jar" }
$artifacts += @(
  "$repositoryCache/commons-logging/commons-logging/1.3.5/commons-logging-1.3.5.jar",
  "$repositoryCache/org/jspecify/jspecify/1.0.0/jspecify-1.0.0.jar",
  "$repositoryCache/io/micrometer/micrometer-observation/1.16.7/micrometer-observation-1.16.7.jar",
  "$repositoryCache/io/micrometer/micrometer-commons/1.16.7/micrometer-commons-1.16.7.jar",
  "$repositoryCache/com/h2database/h2/2.4.240/h2-2.4.240.jar"
)
foreach ($artifact in $artifacts) { if (-not (Test-Path -LiteralPath $artifact)) { throw "Unavailable existing dependency: $artifact" } }
$artifacts | ForEach-Object { [pscustomobject]@{ path = $_; sha256 = (Get-FileHash -LiteralPath $_ -Algorithm SHA256).Hash.ToLowerInvariant() } } | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath "$evidenceRoot/logs/dependencies.json" -Encoding utf8
$classpath = $artifacts -join [System.IO.Path]::PathSeparator
New-Item -ItemType Directory -Path "$evidenceRoot/controls/classes" -Force | Out-Null
$commands = [System.Collections.Generic.List[object]]::new()
$compileArguments = @('--release','25','-cp',$classpath,'-d',"$evidenceRoot/controls/classes","$evidenceRoot/controls/ListenerContractProbe.java")
$compileOutput = & javac @compileArguments 2>&1 | Out-String
$compileExit = $LASTEXITCODE
$compileOutput | Set-Content -LiteralPath "$evidenceRoot/logs/javac.txt" -Encoding utf8
$commands.Add([pscustomobject]@{ executable = (Get-Command javac).Source; arguments = $compileArguments; exitCode = $compileExit; output = $compileOutput })
if ($compileExit -ne 0) { $commands | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath "$evidenceRoot/logs/commands.json" -Encoding utf8; throw 'Compilation failed' }
$runArguments = @('-cp',"$evidenceRoot/controls/classes$([System.IO.Path]::PathSeparator)$classpath",'ListenerContractProbe')
$runtimeOutput = & java @runArguments 2>&1 | Out-String
$runtimeExit = $LASTEXITCODE
$runtimeOutput | Set-Content -LiteralPath "$evidenceRoot/logs/framework-h2.txt" -Encoding utf8
$commands.Add([pscustomobject]@{ executable = (Get-Command java).Source; arguments = $runArguments; exitCode = $runtimeExit; output = $runtimeOutput })
$commands | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath "$evidenceRoot/logs/commands.json" -Encoding utf8
$runtimeOutput
if ($runtimeExit -ne 0) { throw 'Runtime control failed' }
