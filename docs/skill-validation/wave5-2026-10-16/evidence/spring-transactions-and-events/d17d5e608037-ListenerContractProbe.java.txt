import java.sql.Connection;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;
import org.springframework.context.ApplicationEventPublisher;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.datasource.DataSourceTransactionManager;
import org.springframework.jdbc.datasource.DriverManagerDataSource;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.annotation.EnableTransactionManagement;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.transaction.event.TransactionPhase;
import org.springframework.transaction.event.TransactionalEventListener;
import org.springframework.transaction.support.TransactionTemplate;

public class ListenerContractProbe {
    static int assertions;
    public record ReceiptRequested(int id, String reference) {}
    public static class Observations {
        final List<Integer> sourceRowsDuringCallback = new ArrayList<>();
    }
    @Configuration(proxyBeanMethods = false)
    @EnableTransactionManagement(proxyTargetClass = true)
    static class Infrastructure {
        @Bean DataSource dataSource() {
            return new DriverManagerDataSource(System.getProperty("probe.url"), "sa", "");
        }
        @Bean JdbcTemplate jdbc(DataSource dataSource) { return new JdbcTemplate(dataSource); }
        @Bean PlatformTransactionManager transactionManager(DataSource dataSource) {
            return new DataSourceTransactionManager(dataSource);
        }
        @Bean Observations observations() { return new Observations(); }
        @Bean OrderService orderService(JdbcTemplate jdbc, ApplicationEventPublisher events) {
            return new OrderService(jdbc, events);
        }
    }
    public static class OrderService {
        final JdbcTemplate jdbc;
        final ApplicationEventPublisher events;
        OrderService(JdbcTemplate jdbc, ApplicationEventPublisher events) {
            this.jdbc = jdbc;
            this.events = events;
        }
        @Transactional
        public void place(int id, String reference) {
            jdbc.update("insert into orders values (?, ?)", id, reference);
            events.publishEvent(new ReceiptRequested(id, reference));
        }
    }
    public abstract static class ReceiptWork {
        final JdbcTemplate jdbc;
        final DataSource dataSource;
        final Observations observations;
        ReceiptWork(JdbcTemplate jdbc, DataSource dataSource, Observations observations) {
            this.jdbc = jdbc;
            this.dataSource = dataSource;
            this.observations = observations;
        }
        void write(ReceiptRequested event) {
            try (Connection connection = dataSource.getConnection()) {
                connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                connection.setAutoCommit(false);
                try (var statement = connection.prepareStatement("select count(*) from orders where id = ?")) {
                    statement.setQueryTimeout(2);
                    statement.setInt(1, event.id());
                    try (var rows = statement.executeQuery()) {
                        rows.next();
                        observations.sourceRowsDuringCallback.add(rows.getInt(1));
                    }
                }
                connection.commit();
            } catch (Exception failure) {
                throw new IllegalStateException("Independent observation failed", failure);
            }
            jdbc.update("insert into receipt values (?, ?)", event.id(), event.reference());
        }
    }
    public static class AfterCommitReceipt extends ReceiptWork {
        public AfterCommitReceipt(JdbcTemplate jdbc, DataSource dataSource, Observations observations) {
            super(jdbc, dataSource, observations);
        }
        @TransactionalEventListener
        @Transactional(propagation = Propagation.REQUIRES_NEW)
        public void on(ReceiptRequested event) { write(event); }
    }
    public static class BeforeCommitReceipt extends ReceiptWork {
        public BeforeCommitReceipt(JdbcTemplate jdbc, DataSource dataSource, Observations observations) {
            super(jdbc, dataSource, observations);
        }
        @TransactionalEventListener(phase = TransactionPhase.BEFORE_COMMIT)
        @Transactional
        public void on(ReceiptRequested event) { write(event); }
    }
    public static class InvalidAfterCommitReceipt {
        @TransactionalEventListener
        @Transactional
        public void on(ReceiptRequested event) {}
    }
    static AnnotationConfigApplicationContext open(String label, Class<?> listener) {
        System.setProperty("probe.url", "jdbc:h2:mem:" + label + ";DB_CLOSE_DELAY=-1;LOCK_TIMEOUT=2000");
        var context = new AnnotationConfigApplicationContext();
        context.register(Infrastructure.class, listener);
        try { context.refresh(); }
        catch (RuntimeException failure) { context.close(); throw failure; }
        JdbcTemplate jdbc = context.getBean(JdbcTemplate.class);
        jdbc.execute("create table orders (id integer primary key, reference varchar(100))");
        jdbc.execute("create table receipt (order_id integer primary key, reference varchar(8))");
        return context;
    }
    static int count(JdbcTemplate jdbc, String table, int id) {
        String key = table.equals("orders") ? "id" : "order_id";
        return jdbc.queryForObject("select count(*) from " + table + " where " + key + " = ?", Integer.class, id);
    }
    static void expect(String label, Object actual, Object expected) {
        assertions++;
        if (!java.util.Objects.equals(actual, expected)) {
            throw new AssertionError(label + ": expected " + expected + ", observed " + actual);
        }
        System.out.println("PASS " + label + " = " + actual);
    }
    public static void main(String[] args) {
        try (var context = open("after_commit", AfterCommitReceipt.class)) {
            var service = context.getBean(OrderService.class);
            var jdbc = context.getBean(JdbcTemplate.class);
            var observations = context.getBean(Observations.class);
            service.place(1, "ok");
            expect("after-success-source", count(jdbc, "orders", 1), 1);
            expect("after-success-receipt", count(jdbc, "receipt", 1), 1);
            expect("after-phase-independent-read", observations.sourceRowsDuringCallback.getFirst(), 1);
            service.place(2, "REFERENCE-TOO-LONG");
            expect("after-failure-caller-returned", true, true);
            expect("after-failure-source-still-committed", count(jdbc, "orders", 2), 1);
            expect("after-failure-receipt-absent", count(jdbc, "receipt", 2), 0);
            int callbacksBefore = observations.sourceRowsDuringCallback.size();
            try {
                new TransactionTemplate(context.getBean(PlatformTransactionManager.class)).executeWithoutResult(status -> {
                    service.place(3, "ok");
                    throw new IllegalArgumentException("Caller aborts its outer unit");
                });
                throw new AssertionError("Expected outer rollback");
            } catch (IllegalArgumentException expected) {
                expect("outer-rollback-source-absent", count(jdbc, "orders", 3), 0);
                expect("outer-rollback-receipt-absent", count(jdbc, "receipt", 3), 0);
                expect("outer-rollback-callback-skipped", observations.sourceRowsDuringCallback.size(), callbacksBefore);
            }
            context.publishEvent(new ReceiptRequested(4, "ok"));
            expect("no-transaction-callback-skipped", observations.sourceRowsDuringCallback.size(), callbacksBefore);
            expect("no-transaction-receipt-absent", count(jdbc, "receipt", 4), 0);
        }
        try (var context = open("before_commit", BeforeCommitReceipt.class)) {
            var service = context.getBean(OrderService.class);
            var jdbc = context.getBean(JdbcTemplate.class);
            var observations = context.getBean(Observations.class);
            service.place(1, "ok");
            expect("before-success-source", count(jdbc, "orders", 1), 1);
            expect("before-success-receipt", count(jdbc, "receipt", 1), 1);
            expect("before-phase-independent-read", observations.sourceRowsDuringCallback.getFirst(), 0);
            boolean callerFailed = false;
            try { service.place(2, "REFERENCE-TOO-LONG"); }
            catch (org.springframework.dao.DataIntegrityViolationException expected) { callerFailed = true; }
            expect("before-failure-reaches-caller", callerFailed, true);
            expect("before-failure-source-absent", count(jdbc, "orders", 2), 0);
            expect("before-failure-receipt-absent", count(jdbc, "receipt", 2), 0);
        }
        boolean rejected = false;
        try (var ignored = open("invalid_listener", InvalidAfterCommitReceipt.class)) {
            throw new AssertionError("Expected registration rejection");
        } catch (RuntimeException failure) {
            for (Throwable cause = failure; cause != null; cause = cause.getCause()) {
                if (cause instanceof IllegalStateException &&
                    cause.getMessage().contains("REQUIRES_NEW or NOT_SUPPORTED")) rejected = true;
            }
        }
        expect("after-required-registration-rejected", rejected, true);
        System.out.println("RESULT assertions=" + assertions + " runtime=" + System.getProperty("java.runtime.version"));
    }
}
