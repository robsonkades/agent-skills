final class QuoteService {
    private final Map<String, Quote> catalog;
    private final Map<String, Quote> quotes = new HashMap<>();
    private final boolean cacheEnabled;

    QuoteService(Map<String, Quote> catalog, boolean cacheEnabled) {
        this.catalog = Map.copyOf(catalog);
        this.cacheEnabled = cacheEnabled;
    }

    Quote find(String sku, String currency) {
        if (!cacheEnabled) return catalogQuote(sku, currency);
        return quotes.computeIfAbsent(sku, ignored -> catalogQuote(sku, currency));
