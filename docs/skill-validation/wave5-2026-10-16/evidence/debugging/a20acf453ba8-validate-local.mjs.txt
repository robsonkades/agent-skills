import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { fileURLToPath, pathToFileURL } from 'node:url';

const repo = 'C:/git/agent-skills';
const out = path.dirname(fileURLToPath(import.meta.url));
const packagePath = path.join(repo, 'skills/debugging');
const baseline = JSON.parse(fs.readFileSync(path.join(out, 'baseline.json'), 'utf8'));
const { parseManifest } = await import(pathToFileURL(path.join(repo, 'packages/core/src/domain/manifest.ts')));
const { parseSkillDocument, documentName, documentDescription } = await import(pathToFileURL(path.join(repo, 'packages/core/src/domain/skill-document.ts')));
const manifest = parseManifest(fs.readFileSync(path.join(packagePath, 'skill.yaml'), 'utf8'), { strict: true });
assert.deepEqual(manifest.issues, []);
const entry = fs.readFileSync(path.join(packagePath, 'SKILL.md'), 'utf8');
const doc = parseSkillDocument(entry);
assert.equal(documentName(doc), manifest.manifest.name);
assert.equal(documentDescription(doc), manifest.manifest.description);
assert.equal(manifest.manifest.name, 'debugging');
const files = baseline.files.map(item => item.path);
for (const item of baseline.files) {
  const actual = fs.readFileSync(path.join(repo, item.path));
  assert.equal(crypto.createHash('sha256').update(actual).digest('hex'), item.sha256, item.path);
}
const relativeReferences = [...new Set([...entry.matchAll(/`(references\/[^`]+\.md)`/g)].map(match => match[1]))];
assert.deepEqual(relativeReferences.sort(), ['references/method.md', 'references/production-evidence.md']);
for (const ref of relativeReferences) assert.ok(fs.existsSync(path.join(packagePath, ref)), ref);
const packageText = files.filter(file => file.endsWith('.md')).map(file => fs.readFileSync(path.join(repo, file), 'utf8')).join('\n');
const links = [...packageText.matchAll(/\[[^\]]+\]\(([^)]+)\)/g)].map(match => match[1]);
for (const link of links) assert.ok(/^https:\/\//.test(link), link);
for (const name of manifest.manifest.suggests) {
  assert.ok(fs.existsSync(path.join(repo, 'skills', name, 'SKILL.md')), name);
  assert.ok(packageText.includes(name), name);
}
const sources = JSON.parse(fs.readFileSync(path.join(out, 'source-captures.json'), 'utf8'));
assert.equal(sources.length, 8);
assert.ok(sources.every(source => source.status === 200));
const jfc = fs.readFileSync(path.join(out, 'sources/default-jfc25.xml'), 'utf8');
const eventBlocks = {};
for (const name of ['jdk.JavaExceptionThrow', 'jdk.ObjectAllocationSample', 'jdk.JavaMonitorEnter']) {
  const escaped = name.replaceAll('.', '\\.');
  const match = jfc.match(new RegExp(`<event name="${escaped}">([\\s\\S]*?)<\\/event>`));
  assert.ok(match, name);
  eventBlocks[name] = Object.fromEntries([...match[1].matchAll(/<setting name="([^"]+)"[^>]*>([^<]+)<\/setting>/g)].map(item => [item[1], item[2]]));
}
assert.equal(eventBlocks['jdk.JavaExceptionThrow'].throttle, '100/s');
assert.equal(eventBlocks['jdk.ObjectAllocationSample'].throttle, '150/s');
assert.equal(eventBlocks['jdk.JavaMonitorEnter'].threshold, '20 ms');
const commands = [];
function run(command, args, cwd, logName) {
  const result = spawnSync(command, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = { command, args, cwd, status: result.status, signal: result.signal, error: result.error ? String(result.error) : null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  commands.push(record);
  fs.writeFileSync(path.join(out, logName), JSON.stringify(record, null, 2) + '\n');
  assert.equal(result.error, undefined, `${logName} execution`);
  assert.equal(result.status, 0, `${logName}: ${record.stderr}`);
  return record.stdout.trim();
}
run(process.execPath, ['node_modules/prettier/bin/prettier.cjs', '--check', ...files], repo, 'prettier-check.json');
const javaHome = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
run(path.join(javaHome, 'java.exe'), ['-version'], repo, 'java-version.json');
run(path.join(javaHome, 'javac.exe'), ['-version'], repo, 'javac-version.json');
const caseInputs = JSON.parse(fs.readFileSync(path.join(out, 'case-inputs.json'), 'utf8'));
const classes = path.join(out, 'technical/classes');
fs.mkdirSync(classes, { recursive: true });
const c1 = caseInputs.cases[0].directory;
const c2 = caseInputs.cases[1].directory;
run(path.join(javaHome, 'javac.exe'), ['--release', '17', '-d', classes, ...['Quote.java', 'QuoteService.java', 'Replay.java'].map(file => path.join(c1, 'src', file))], repo, 'fixture-compile.json');
const replayA = run(path.join(javaHome, 'java.exe'), ['-cp', classes, 'Replay', 'config/app.properties', 'data/requests.csv'], c1, 'fixture-c1.json');
const replayB = run(path.join(javaHome, 'java.exe'), ['-cp', classes, 'Replay', 'config/app.properties', 'data/requests.csv'], c2, 'fixture-c2.json');
assert.deepEqual(replayA.split(/\r?\n/), ['W-17\tUSD\tUSD\t1299', 'W-17\tEUR\tUSD\t1299']);
assert.deepEqual(replayB.split(/\r?\n/), ['W-17\tUSD\tUSD\t1299', 'W-17\tEUR\tEUR\t1199']);
const secondOnly = path.join(out, 'technical/second-only.csv');
fs.writeFileSync(secondOnly, 'sku,currency\nW-17,EUR\n');
const resetControl = run(path.join(javaHome, 'java.exe'), ['-cp', classes, 'Replay', 'config/app.properties', secondOnly], c1, 'fixture-reset-control.json');
assert.equal(resetControl, 'W-17\tEUR\tEUR\t1199');
assert.notEqual(replayA.split(/\r?\n/)[1], resetControl);
assert.notEqual(replayA, replayB);
const unchanged = run('git', ['diff', '--exit-code', '--', 'skills/debugging'], repo, 'owned-diff.json');
assert.equal(unchanged, '');
const result = {
  skill: 'debugging', metadata: { strictIssues: manifest.issues.length, nameMatches: true, descriptionMatches: true, assignmentHashesMatched: baseline.files.length, routedReferences: relativeReferences, declaredReferrals: manifest.manifest.suggests.length, externalMarkdownLinks: links.length },
  sources: { http200: sources.length, pinnedJfcSettings: eventBlocks },
  technical: { java: 'Temurin 25.0.3+9', compilerRelease: 17, compilationUnits: 3, actorRuns: 0, replayProcesses: 3, c1: replayA, c2: replayB, freshProcessNegativeControl: resetControl, interpretation: 'Both ordered replay commands exit 0; only C2 satisfies both quote contracts. The fresh-process single-request control hides C1\'s cross-currency reuse. This validates case discriminability, not an agent or a historical incident.' },
  commands, allChildProcessesExited: true, packageUnchanged: true
};
fs.writeFileSync(path.join(out, 'local-validation.json'), JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify({ metadata: result.metadata, technical: result.technical, commandsExecuted: commands.length, packageUnchanged: true, allChildProcessesExited: true }, null, 2));
