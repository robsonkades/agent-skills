import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { fileURLToPath } from 'node:url';

const out = path.dirname(fileURLToPath(import.meta.url));
const directory = path.join(out, 'sources');
fs.mkdirSync(directory, { recursive: true });
const inputs = [
  ['git-bisect', 'https://git-scm.com/docs/git-bisect'],
  ['sre-troubleshooting', 'https://sre.google/sre-book/effective-troubleshooting/'],
  ['jcmd25', 'https://docs.oracle.com/en/java/javase/25/docs/specs/man/jcmd.html'],
  ['jep444', 'https://openjdk.org/jeps/444'],
  ['default-jfc25', 'https://raw.githubusercontent.com/openjdk/jdk/jdk-25-ga/src/jdk.jfr/share/conf/jfr/default.jfc'],
  ['throwable25', 'https://docs.oracle.com/en/java/javase/25/docs/api/java.base/java/lang/Throwable.html'],
  ['postgres17-isolation', 'https://www.postgresql.org/docs/17/transaction-iso.html'],
  ['nist-doe', 'https://www.itl.nist.gov/div898/handbook/pri/section3/pri33.htm']
];
function strip(text) {
  return text.replace(/<script\b[\s\S]*?<\/script>/gi, '').replace(/<style\b[\s\S]*?<\/style>/gi, '')
    .replace(/<\/(?:p|h\d|li|tr|div)>/gi, '\n').replace(/<br\s*\/?\s*>/gi, '\n')
    .replace(/<[^>]+>/g, ' ').replace(/&nbsp;|&#160;/g, ' ').replace(/&quot;/g, '"')
    .replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>')
    .replace(/[ \t]+/g, ' ').replace(/\n\s*\n\s*\n/g, '\n\n').trim();
}
const sources = await Promise.all(inputs.map(async ([id, url]) => {
  const capturedAt = new Date().toISOString();
  try {
    const response = await fetch(url, { signal: AbortSignal.timeout(25000) });
    const raw = await response.text();
    const extension = id === 'default-jfc25' ? 'xml' : 'html';
    const rawPath = path.join(directory, id + '.' + extension);
    fs.writeFileSync(rawPath, raw);
    const textPath = path.join(directory, id + '.txt');
    fs.writeFileSync(textPath, extension === 'xml' ? raw : strip(raw));
    return { id, requestedUrl: url, finalUrl: response.url, status: response.status, capturedAt, bytes: Buffer.byteLength(raw), sha256: crypto.createHash('sha256').update(raw).digest('hex'), rawPath, textPath };
  } catch (error) {
    return { id, requestedUrl: url, capturedAt, error: String(error) };
  }
}));
fs.writeFileSync(path.join(out, 'source-captures.json'), JSON.stringify(sources, null, 2) + '\n');
console.log(JSON.stringify(sources, null, 2));
if (sources.some(source => source.status !== 200)) process.exitCode = 1;
