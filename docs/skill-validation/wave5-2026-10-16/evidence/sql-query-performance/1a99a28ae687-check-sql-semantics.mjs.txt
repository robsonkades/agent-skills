import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { DatabaseSync } from 'node:sqlite';

const artifactRoot = path.dirname(fileURLToPath(import.meta.url));
const repo = 'C:/git/agent-skills';
const checks = [];
let assertions = 0;
const equal = (actual, expected, label) => {
  assert.deepEqual(actual, expected, label);
  assertions++;
};
const differs = (actual, expected, label) => {
  assert.notDeepEqual(actual, expected, label);
  assertions++;
};
const values = (db, sql, ...args) => db.prepare(sql).all(...args).map(row => Object.values(row));
const read = file => fs.readFileSync(file, 'utf8');
const withDb = fn => {
  const db = new DatabaseSync(':memory:');
  try { return fn(db); } finally { db.close(); }
};
let sqliteVersion;
withDb(db => { sqliteVersion = db.prepare('select sqlite_version() as version').get().version; });

// The paired inputs differ in schema nullability only. No agent actor is run.
const inputA = path.join(artifactRoot, 'cases/input-a');
const inputB = path.join(artifactRoot, 'cases/input-b');
const inputNames = fs.readdirSync(inputA).sort();
equal(inputNames, fs.readdirSync(inputB).sort(), 'same input filenames');
const differingFiles = inputNames.filter(name => read(path.join(inputA, name)) !== read(path.join(inputB, name)));
equal(differingFiles, ['schema.sql'], 'single varied file');
equal(read(path.join(inputA, 'schema.sql')).replace('active boolean NOT NULL', 'active boolean'),
  read(path.join(inputB, 'schema.sql')), 'single varied constraint');
for (const [name, dir] of [['a', inputA], ['b', inputB]]) {
  withDb(db => {
    db.exec(read(path.join(dir, 'schema.sql')));
    db.exec(read(path.join(dir, 'seed.sql')));
    const current = read(path.join(dir, 'current.sql'));
    const proposed = read(path.join(dir, 'proposed.sql'));
    const initial = values(db, current);
    equal(initial, [['Acme'], ['Acme'], ['Bravo']], name + ': fixture result and multiplicity');
    equal(values(db, proposed), initial, name + ': seed alone does not expose the difference');
    const nullInsert = "INSERT INTO invoices VALUES (6, 71, 'Nullable', NULL, 120)";
    if (name === 'a') {
      assert.throws(() => db.exec(nullInsert), /NOT NULL constraint failed/);
      assertions++;
      checks.push({ family: 'paired-input-a', result: 'PASS', evidence: 'declared domain rejects null counterexample; seed preserves overlap and repeated labels' });
    } else {
      db.exec(nullInsert);
      differs(values(db, proposed), values(db, current), 'b: allowed null is lost by NOT predicate');
      const corrected = proposed.replace('NOT (active = true)', '(active = true) IS NOT TRUE');
      equal(values(db, corrected), values(db, current), 'b: null-aware disjoint branch preserves rows');
      checks.push({ family: 'paired-input-b', result: 'PASS', negativeControl: 'proposed query loses allowed NULL-active row', correction: 'IS NOT TRUE retains it' });
    }
  });
}

withDb(db => {
  db.exec('CREATE TABLE truth_rows (id integer primary key, p integer, q integer, label text)');
  const insert = db.prepare('INSERT INTO truth_rows VALUES (?, ?, ?, ?)');
  let id = 0;
  for (const p of [null, 0, 1]) for (const q of [null, 0, 1]) {
    insert.run(++id, p, q, 'same label');
  }
  const original = 'SELECT id FROM truth_rows WHERE p OR q ORDER BY id';
  const disjoint = 'SELECT id FROM truth_rows WHERE p UNION ALL SELECT id FROM truth_rows WHERE q AND p IS NOT TRUE ORDER BY id';
  const wrongNull = 'SELECT id FROM truth_rows WHERE p UNION ALL SELECT id FROM truth_rows WHERE q AND NOT p ORDER BY id';
  const wrongOverlap = 'SELECT id FROM truth_rows WHERE p UNION ALL SELECT id FROM truth_rows WHERE q ORDER BY id';
  equal(values(db, disjoint), values(db, original), 'all nine SQL truth combinations preserve selected row identities');
  differs(values(db, wrongNull), values(db, original), 'NOT p loses unknown p / true q');
  differs(values(db, wrongOverlap), values(db, original), 'unguarded UNION ALL repeats true/true row');
  const originalLabels = values(db, 'SELECT label FROM truth_rows WHERE p OR q ORDER BY label');
  const unionLabels = values(db, 'SELECT label FROM truth_rows WHERE p UNION SELECT label FROM truth_rows WHERE q ORDER BY label');
  differs(unionLabels, originalLabels, 'UNION collapses different rows with identical projection');
  checks.push({ family: 'OR-union', result: 'PASS', truthCombinations: 9, negativeControls: ['NOT p', 'unguarded UNION ALL', 'UNION deduplication'] });
});

withDb(db => {
  db.exec('CREATE TABLE excluded (v integer)');
  const insert = db.prepare('INSERT INTO excluded VALUES (?)');
  const results = [];
  for (const [setName, rhs] of [['empty', []], ['null', [null]], ['duplicate', [1, 1]], ['mixed', [1, null]]]) {
    db.exec('DELETE FROM excluded');
    for (const value of rhs) insert.run(value);
    for (const x of [null, 1, 2]) {
      const result = db.prepare('SELECT ? NOT IN (SELECT v FROM excluded) AS not_in, NOT EXISTS (SELECT 1 FROM excluded WHERE v = ?) AS not_exists').get(x, x);
      const expectedNotIn = rhs.length === 0 ? 1 : x === null ? null : rhs.includes(x) ? 0 : rhs.includes(null) ? null : 1;
      const expectedNotExists = x !== null && rhs.includes(x) ? 0 : 1;
      equal(result.not_in, expectedNotIn, setName + '/x=' + x + ': NOT IN');
      equal(result.not_exists, expectedNotExists, setName + '/x=' + x + ': NOT EXISTS');
      results.push({ set: setName, x, notIn: result.not_in, notExists: result.not_exists });
    }
  }
  checks.push({ family: 'anti-membership', result: 'PASS', combinations: results.length, rows: results });
});

withDb(db => {
  db.exec("CREATE TABLE orders (tenant_id integer, id integer, label text, primary key(tenant_id,id)); CREATE TABLE payments (tenant_id integer, order_id integer, status text)");
  db.exec("INSERT INTO orders VALUES (71,1,'Same'),(71,2,'Same'),(71,3,'No paid'),(71,4,'Pending'),(72,3,'Other')");
  db.exec("INSERT INTO payments VALUES (71,1,'settled'),(71,1,'settled'),(71,2,'settled'),(71,4,'pending'),(72,3,'settled')");
  const source = read(path.join(repo, 'skills/sql-query-performance/references/query-shapes.md'));
  const fenced = [...source.matchAll(/```sql\r?\n([\s\S]*?)```/g)].map(m => m[1]);
  const existsSql = fenced.find(text => text.includes('SELECT o.id FROM orders o'));
  assert.ok(existsSql, 'exact EXISTS fence found');
  assertions++;
  equal(values(db, existsSql, 71), [[1], [2]], 'exact membership fence: zero, one, many, pending and tenant isolation');
  const projected = existsSql.replace('SELECT o.id', 'SELECT o.label');
  equal(values(db, projected, 71), [['Same'], ['Same']], 'EXISTS preserves repeated outer projection');
  const distinctJoin = "SELECT DISTINCT o.label FROM orders o JOIN payments p ON p.tenant_id=o.tenant_id AND p.order_id=o.id WHERE o.tenant_id=? AND p.status='settled'";
  differs(values(db, projected, 71), values(db, distinctJoin, 71), 'dropping required DISTINCT changes label multiplicity');
  const wrongTenant = existsSql.replace('p.tenant_id = o.tenant_id AND ', '');
  differs(values(db, wrongTenant, 71), values(db, existsSql, 71), 'missing tenant correlation falsely qualifies order 3');
  checks.push({ family: 'EXISTS exact example', result: 'PASS', negativeControls: ['drop projected deduplication', 'drop tenant correlation'] });
});

  databases: 'All SQLite databases were in-memory and closed before output.',
  assertions, checks, behavioralActorsExecuted: 0
}, null, 2));
