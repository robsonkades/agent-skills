import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { pathToFileURL } from 'node:url';
const repo = 'C:/git/agent-skills';
const name = 'sql-query-performance';
const relativeRoot = 'skills/' + name;
const skillRoot = path.join(repo, relativeRoot);
const { loadPackageFromDirectory, validatePackage } = await import(pathToFileURL(path.join(repo, 'packages/core/dist/index.js')));
const { NodeFileSystem } = await import(pathToFileURL(path.join(repo, 'packages/node/dist/index.js')));
const loaded = await loadPackageFromDirectory(new NodeFileSystem(), skillRoot, { strict: true });
const validated = validatePackage(loaded.pkg, { directoryName: name });
const issues = [...loaded.issues, ...validated.issues];
assert.deepEqual(issues, [], 'repository package validation');
const baseline = JSON.parse(fs.readFileSync(path.join(repo, 'docs/skill-validation/wave5-2026-10-16/baseline.json'), 'utf8')).packages.find(p => p.name === name);
assert.ok(baseline, 'assignment baseline exists');
const listFiles = root => fs.readdirSync(root, { withFileTypes: true }).flatMap(entry =>
  entry.isDirectory() ? listFiles(path.join(root, entry.name)) : [path.join(root, entry.name)]);
const files = listFiles(skillRoot).map(file => ({
  path: path.relative(repo, file).replaceAll('\\', '/'),
  sha256: crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex'),
  bytes: fs.statSync(file).size
})).sort((a, b) => a.path.localeCompare(b.path));
assert.deepEqual(files, [...baseline.files].sort((a, b) => a.path.localeCompare(b.path)), 'all files identical to assignment baseline');
assert.equal(loaded.pkg.manifest.version, baseline.initialVersion, 'no version drift');
const links = [];
for (const entry of files.filter(file => file.path.endsWith('.md'))) {
  const content = fs.readFileSync(path.join(repo, entry.path), 'utf8');
  for (const match of content.matchAll(/\[[^\]]+\]\(([^)]+)\)/g)) {
    const target = match[1].replace(/^<|>$/g, '').split('#')[0];
    if (!target || /^[a-z]+:/i.test(target)) continue;
    const resolved = path.resolve(path.dirname(path.join(repo, entry.path)), target);
    assert.ok(fs.existsSync(resolved), entry.path + ' -> ' + target);
    links.push({ from: entry.path, to: target });
  }
}
const routedReferences = files.filter(file => file.path.includes('/references/')).map(file => path.relative(skillRoot, path.join(repo, file.path)).replaceAll('\\', '/'));
const entrypoint = fs.readFileSync(path.join(skillRoot, 'SKILL.md'), 'utf8');
for (const reference of routedReferences) assert.ok(entrypoint.includes('](' + reference + ')'), 'reference routed: ' + reference);
for (const suggested of loaded.pkg.manifest.suggests) assert.ok(fs.existsSync(path.join(repo, 'skills', suggested, 'SKILL.md')), 'suggested skill exists: ' + suggested);
console.log(JSON.stringify({
  status: 'PASS', skill: name, issueCount: issues.length, filesChecked: files.length,
  markdownFiles: files.filter(file => file.path.endsWith('.md')).length,
  localLinks: links, referencesRouted: routedReferences,
  optionalRoutes: loaded.pkg.manifest.suggests,
  versions: { committed: baseline.committedVersion, assignment: baseline.initialVersion, final: loaded.pkg.manifest.version },
  baselineIdentical: true, files
}, null, 2));
