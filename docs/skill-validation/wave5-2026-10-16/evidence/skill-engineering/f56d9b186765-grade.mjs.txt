import { readFileSync } from "node:fs";

const read = (name) => JSON.parse(readFileSync(new URL(name, import.meta.url), "utf8"));
const config = read("./config.json");
if (!["narrative", "contract"].includes(config.completionMode)) {
  throw new Error("Unsupported completionMode");
}
const input = process.argv[2] ?? "archive";
if (!["archive", "samples"].includes(input)) throw new Error("Expected archive or samples");
const tasks = new Map(read("./tasks.json").map((task) => [task.id, task]));
const grades = read("./" + input + ".json").map((item) => {
  const task = tasks.get(item.task);
  if (!task) throw new Error("Unknown task " + item.task);
  const output = item.output;
  const object = output !== null && typeof output === "object" && !Array.isArray(output);
  const checks = {
    object,
    count: object && Number.isInteger(output.count) && output.count >= 0,
    totalCents: object && Number.isInteger(output.totalCents) && output.totalCents >= 0,
    completion: config.completionMode === "narrative"
      ? item.report.includes("Orders processed")
      : object && output.count === task.orders.length &&
        output.totalCents === task.orders.reduce((sum, order) => sum + order.amountCents, 0)
  };
  return {
    id: item.id ?? item.condition + "/" + item.task + "/" + item.trial,
    condition: item.condition ?? null,
    task: item.task,
    trial: item.trial ?? null,
    checks,
    passedChecks: Object.values(checks).filter(Boolean).length,
    totalChecks: Object.keys(checks).length,
    passed: Object.values(checks).every(Boolean)
  };
});
console.log(JSON.stringify({ completionMode: config.completionMode, input, grades }, null, 2));
