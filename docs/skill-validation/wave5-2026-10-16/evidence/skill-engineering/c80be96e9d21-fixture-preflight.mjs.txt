import { readFileSync, readdirSync } from "node:fs";
import { resolve, relative } from "node:path";
import { fileURLToPath } from "node:url";
import { spawnSync } from "node:child_process";
import assert from "node:assert/strict";

const root = fileURLToPath(new URL(".", import.meta.url));
const paths = (dir) => readdirSync(dir, { recursive: true, withFileTypes: true })
  .filter((entry) => entry.isFile())
  .map((entry) => relative(dir, resolve(entry.parentPath, entry.name)).replaceAll("\\", "/")).sort();
const a = resolve(root, "case-a"), b = resolve(root, "case-b");
assert.deepEqual(paths(a), paths(b));
const changed = paths(a).filter((path) => !readFileSync(resolve(a, path)).equals(readFileSync(resolve(b, path))));
assert.deepEqual(changed, ["evaluation/config.json"]);
const commands = [];
for (const [name, dir, expectedMode] of [["A", a, "narrative"], ["B", b, "contract"]]) {
  for (const input of ["archive", "samples"]) {
    const run = spawnSync(process.execPath, ["evaluation/grade.mjs", input], { cwd: dir, encoding: "utf8" });
    assert.equal(run.status, 0, run.stderr);
    const result = JSON.parse(run.stdout);
    assert.equal(result.completionMode, expectedMode);
    assert.ok(result.grades.every((g) => g.totalChecks === 4));
    if (input === "archive") {
      assert.equal(result.grades.length, 4);
      assert.ok(result.grades.every((g) => g.passed));
      for (const condition of ["original", "revised"]) {
        const grades = result.grades.filter((g) => g.condition === condition);
        assert.equal(grades.length, 2);
        assert.equal(grades.reduce((sum, g) => sum + g.passedChecks, 0), 8);
      }
    } else {
      assert.deepEqual(result.grades.map((g) => [g.id, g.passed]),
        name === "A" ? [["alpha", true], ["beta", true], ["gamma", false]] :
          [["alpha", true], ["beta", false], ["gamma", true]]);
    }
    commands.push({ case: name, command: "node evaluation/grade.mjs " + input, cwd: dir, exitCode: run.status, result });
  }
}
console.log(JSON.stringify({
  status: "PASS", kind: "deterministic fixture preflight, not agent evaluation",
  node: process.version, pairDifference: changed, commands,
  processes: "four synchronous child commands exited; no process kept running"
}, null, 2));
