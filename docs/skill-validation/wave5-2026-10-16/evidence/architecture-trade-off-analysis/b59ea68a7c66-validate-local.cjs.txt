'use strict';
// Local audit for one unchanged skill. Does not generate the registry or verify the repository.
const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const assert = require('node:assert/strict');
const { spawnSync } = require('node:child_process');
const { createRequire } = require('node:module');
const repo = 'C:/git/agent-skills';
const name = 'architecture-trade-off-analysis';
const relative = 'skills/' + name;
const directory = path.join(repo, relative);
const evidence = __dirname;
const yaml = createRequire(path.join(repo, 'package.json'))('yaml');
const read = p => fs.readFileSync(path.join(repo, p), 'utf8');
const hash = p => crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');

function numericExample(text) {
  const values = text.match(/X has values (\d+) and (\d+), Y has (\d+) and (\d+)\./);
  const expected = text.match(/At w=([\d.]+), X=(\d+) and Y=(\d+)\. The tie is at w=([\d.]+);/);
  assert.ok(values && expected, 'weighted example and its stated result must be found');
  const [x1,x2,y1,y2] = values.slice(1).map(Number);
  const [w,wantX,wantY,wantTie] = expected.slice(1).map(Number);
  const x = w*x1+(1-w)*x2;
  const y = w*y1+(1-w)*y2;
  const tie = (y2-x2)/(x1-x2-y1+y2);
  assert.ok(Math.abs(x-wantX)<1e-10, 'X total differs from published result');
  assert.ok(Math.abs(y-wantY)<1e-10, 'Y total differs from published result');
  assert.ok(Math.abs(tie-wantTie)<1e-10, 'tie weight differs from published result');
  const range = [0.55,0.70].map(weight => ({weight,x:weight*x1+(1-weight)*x2,y:weight*y1+(1-weight)*y2}));
  assert.ok(range[0].x<range[0].y && range[1].x>range[1].y, 'published range must contain reversal');
  return {weight:w,x,y,tie,range};
}

let numericText = read(relative+'/references/validation-cases.md');
if (process.argv.includes('--negative-model')) {
  const changed = numericText.replace('X=64 and Y=66.', 'X=64 and Y=65.');
  assert.notEqual(changed,numericText,'negative control must mutate the intended result');
  try {
    numericExample(changed);
    console.error('NEGATIVE CONTROL NOT REJECTED');
    process.exitCode=2;
  } catch(e) {
    console.error('REJECTED: '+e.message);
    process.exitCode=1;
  }
} else {
  const commands=[];
  function command(label,executable,args,expectedExit=0,extra={}) {
    const r=spawnSync(executable,args,{cwd:repo,encoding:'utf8',timeout:30000,...extra});
    const record={label,executable,args,cwd:repo,exitCode:r.status,expectedExit,stdout:r.stdout||'',stderr:r.stderr||'',error:r.error?String(r.error):null};
    commands.push(record);
    assert.equal(r.status,expectedExit,label);
    return r;
  }
  const isolated=path.join(evidence,'isolated-config');
  const env={...process.env,HOME:isolated,USERPROFILE:isolated,AGENT_SKILLS_HOME:path.join(isolated,'.agent-skills'),CLAUDE_CONFIG_DIR:path.join(isolated,'.claude'),CODEX_HOME:path.join(isolated,'.codex')};
  command('strict local manifest validation',process.execPath,[path.join(repo,'packages/cli/bin/agent-skills.mjs'),'validate',directory,'--strict','--json'],0,{env});
  const files=[];
  function walk(dir) {
    for(const item of fs.readdirSync(dir,{withFileTypes:true})) {
      const full=path.join(dir,item.name);
      if(item.isDirectory()) walk(full); else files.push(path.relative(repo,full).replaceAll('\\','/'));
    }
  }
  walk(directory); files.sort();
  command('format check: explicit six owned files',process.execPath,[path.join(repo,'node_modules/prettier/bin/prettier.cjs'),'--check',...files]);
  command('owned diff whitespace check','git',['diff','--check','--',relative]);
  const status=command('owned staged/unstaged/untracked status','git',['status','--short','--untracked-files=all','--',relative]).stdout.trim();
  const unstaged=command('owned unstaged diff','git',['diff','--',relative]).stdout;
  const staged=command('owned staged diff','git',['diff','--cached','--',relative]).stdout;
  const baseline=JSON.parse(read('docs/skill-validation/wave5-2026-10-16/baseline.json')).packages.find(x=>x.name===name);
  const manifest=yaml.parse(read(relative+'/skill.yaml'));
  const front=yaml.parse(read(relative+'/SKILL.md').match(/^---\r?\n([\s\S]*?)\r?\n---/)[1]);
  const committed=yaml.parse(command('committed skill version','git',['show','HEAD:'+relative+'/skill.yaml']).stdout).version;
  const hashes=files.map(file=>({path:file,sha256:hash(path.join(repo,file)),bytes:fs.statSync(path.join(repo,file)).size}));
  assert.equal(files.length,baseline.files.length,'file inventory unchanged');
  for(const f of hashes) assert.deepEqual(f,baseline.files.find(b=>b.path===f.path),'assignment hash unchanged: '+f.path);
  assert.equal(status,''); assert.equal(unstaged,''); assert.equal(staged,'');
  assert.equal(front.name,name); assert.equal(manifest.name,name);
  assert.equal(front.description.trim(),manifest.description.trim());
  assert.equal(manifest.version,baseline.initialVersion);
  assert.equal(committed,baseline.committedVersion);
  const localLinks=[];
  for(const f of files.filter(f=>f.endsWith('.md'))) {
    for(const m of read(f).matchAll(/\[[^\]]*\]\((?:<([^>]+)>|([^\s)]+))\)/g)) {
      const target=m[1]||m[2];
      if(/^(?:https?:|mailto:|#)/.test(target)) continue;
      const full=path.resolve(repo,path.dirname(f),decodeURIComponent(target.split('#')[0]));
      localLinks.push({file:f,target,exists:fs.existsSync(full)});
    }
  }
  assert.ok(localLinks.every(x=>x.exists),'all local Markdown targets exist');
  const referrals=[...(manifest.dependencies||[]),...(manifest.suggests||[])].map(x=>typeof x==='string'?x:x.name).map(n=>({name:n,exists:fs.existsSync(path.join(repo,'skills',n,'SKILL.md'))}));
  assert.ok(referrals.every(x=>x.exists),'all declared referrals resolve');
  const arithmetic=numericExample(numericText);
  const control=command('in-memory wrong Y total rejected',process.execPath,[__filename,'--negative-model'],1);
  assert.match(control.stderr,/REJECTED: Y total differs from published result/);
  const result={skill:name,scope:'local metadata, hashes, links and numeric example only',node:process.version,versions:{committed,assignment:baseline.initialVersion,final:manifest.version},initialStatus:baseline.status,finalStatus:status,filesUnchanged:true,hashes,localLinks,referrals,arithmetic,negativeControl:{mutation:'in-memory Y=66 changed to Y=65; original skill remains untouched',exitCode:control.status,diagnostic:control.stderr.trim()},commands,behavioralActorsExecuted:0,allChildProcessesExited:true};
  fs.writeFileSync(path.join(evidence,'local-validation.json'),JSON.stringify(result,null,2)+'\n');
  console.log(JSON.stringify(result,null,2));
}
