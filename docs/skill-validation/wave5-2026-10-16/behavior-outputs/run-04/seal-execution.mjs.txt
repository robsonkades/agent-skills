import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';

const runRoot = path.resolve(import.meta.dirname, '..');
const outputRoot = path.join(runRoot, 'output');
const journalPath = path.join(outputRoot, 'execution.json');
const execution = JSON.parse(fs.readFileSync(journalPath, 'utf8'));
const digest = content => crypto.createHash('sha256').update(content).digest('hex');
function read(relative, purpose) {
  const content = fs.readFileSync(path.join(runRoot, relative), 'utf8');
  execution.filesRead.push({file: relative, purpose, fullFile: true, truncated: false,
    bytes: Buffer.byteLength(content), sha256: digest(content), content});
  return content;
}

const wrapper = (script, label) =>
  `$verificationProcess = Start-Process -FilePath node -ArgumentList @('output/${script}') -WorkingDirectory 'C:/Users/robso/AppData/Local/Temp/skills-wave5-actors-4tJ3OK/run-04' -WindowStyle Hidden -RedirectStandardOutput 'output/${label}-runner.stdout.log' -RedirectStandardError 'output/${label}-runner.stderr.log' -PassThru\n` +
  `if (-not $verificationProcess.WaitForExit(30000)) { $verificationProcess.Kill($true); throw 'Verification exceeded 30 seconds' }\n` +
  `Get-Content -LiteralPath 'output/${label}-runner.stdout.log' -Raw\n` +
  `Get-Content -LiteralPath 'output/${label}-runner.stderr.log' -Raw\n` +
  'exit $verificationProcess.ExitCode';

for (const [script, label] of [['run-verification.mjs', 'baseline'], ['final-verification.mjs', 'final']]) {
  const stdout = read(`output/${label}-runner.stdout.log`, 'Full observed output of the bounded script process');
  const stderr = read(`output/${label}-runner.stderr.log`, 'Full observed error output of the bounded script process');
  execution.commands.push({phase: `${label}-runner`, kind: 'shell', shell: 'PowerShell',

    stdout, stderr, executed: true, truncated: false,
    resultEvidence: 'The tool returned exit 0 after WaitForExit; stdout and stderr are retained.'});
}

execution.inputPreservation = [];
for (const original of execution.filesRead.filter(item => item.file.startsWith('input/project/'))) {
  const actual = fs.readFileSync(path.join(runRoot, original.file), 'utf8');
  const same = digest(actual) === original.sha256;
  execution.inputPreservation.push({file: original.file, originalSha256: original.sha256,
    finalSha256: digest(actual), identical: same, fullFileRead: true, truncated: false,
    contentReference: 'Same content as the earlier filesRead entry for this path'});
  if (!same) throw new Error(`Original input changed: ${original.file}`);
}
