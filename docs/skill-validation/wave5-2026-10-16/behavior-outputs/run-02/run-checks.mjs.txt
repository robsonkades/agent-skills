function readEvidence(relative, purpose, modelRead = true) {
  const absolute = path.join(root, relative);
  const bytes = fs.readFileSync(absolute);
  const text = bytes.toString('utf8');
  const evidence = {
    path: relative.replaceAll('\\', '/'),
    purpose,
    complete: true,
    truncation: false,
    modelRead,
    bytes: bytes.length,
    sha256: crypto.createHash('sha256').update(bytes).digest('hex'),
    content: text
  };
  history.filesRead.push(evidence);
  return text;
}

function execute(id, executable, args, expectedExit = 0, cwd = project) {
  const started = new Date().toISOString();
  const result = spawnSync(executable, args, {
    cwd, encoding: 'utf8', timeout: 30000, windowsHide: true, maxBuffer: 4 * 1024 * 1024
  });
  const command = {
    id, executable, args, cwd, timeoutMs: 30000, started,
    executed: true, exit: result.status, signal: result.signal,
    error: result.error?.message ?? null,
    stdout: result.stdout ?? '', stderr: result.stderr ?? '',
    expectedExit, expectationMet: result.status === expectedExit
  };
  const log = path.join(logs, `${id}.json`);
  fs.writeFileSync(log, JSON.stringify(command, null, 2) + '\n');
  history.commands.push({ ...command, log: path.relative(out, log).replaceAll('\\', '/') });
  persist();
  process.stdout.write(`${id}: exit=${command.exit}, expected=${expectedExit}\n`);
  process.stdout.write(command.stdout);
  process.stderr.write(command.stderr);
  if (!command.expectationMet || command.error) process.exitCode = 1;
}
