import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { fileURLToPath } from 'node:url';

const output = path.dirname(fileURLToPath(import.meta.url));
const root = path.dirname(output);
const project = path.join(output, 'project');
const executionPath = path.join(output, 'execution.json');
const audit = JSON.parse(fs.readFileSync(executionPath, 'utf8'));
const relative = file => path.relative(root, file).replaceAll('\\', '/');
const digest = data => crypto.createHash('sha256').update(data).digest('hex');
const initial = name => audit.fileReads.find(entry => entry.path === name && entry.phase === 'initial');
const log = (name, content) => {
  const file = path.join(output, 'logs', name);
  fs.writeFileSync(file, content);
  return relative(file);
};

const helpers = [

  { cmd: 'node output/audit.mjs read baseline output/baseline-export.txt', exit: 0, outputRead: ['output/baseline-export.txt'] },

audit.helperCommands = helpers.map(entry => ({ ...entry, cwd: root, tool: 'exec_command', executed: true, outputTruncated: false }));
const quote = value => /[\s;|]/.test(value) || value.startsWith('-Xlog:') ? `'${value.replaceAll("'", "''")}'` : value;
audit.auditedShellCommands = audit.commands.map(command => ({
  cmd: ['node', 'output/audit.mjs', 'run', command.id, command.executable, ...command.args].map(quote).join(' '),
  cwd: root,
  exit: command.exit,
  executed: true,
  stdout: command.stdout,
  stderr: command.stderr,
  note: 'The helper also printed the recorded exit/status footer. Exact executable argument arrays and actual child cwd are in commands.'
}));

assert.ok(audit.integrity?.inputFilesUnchanged);

for (const command of audit.commands) {

  assert.equal(command.error, null);
  if (command.id !== 'regression-before-fix') assert.equal(command.exit, 0, command.id);
}

  const data = fs.readFileSync(path.join(project, name));
  assert.equal(data.readUInt32BE(0), 0xcafebabe);
  assert.equal(data.readUInt16BE(6), 61);
  const entry = { path: relative(path.join(project, name)), bytes: data.length, sha256: digest(data), majorVersion: data.readUInt16BE(6) };
  audit.fileReads.push({ ...entry, phase: 'final-bytecode-header-verification', contentRead: 'Full binary bytes for hash; class magic and version fields parsed. Normalizer instructions inspected separately with javap.', truncated: false });
  return entry;
});
