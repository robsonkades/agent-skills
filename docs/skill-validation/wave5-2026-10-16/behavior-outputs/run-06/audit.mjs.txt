import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const output = path.dirname(fileURLToPath(import.meta.url));
const root = path.dirname(output);
const project = path.join(output, 'project');
const recordPath = path.join(output, 'execution.json');
const relative = file => path.relative(root, file).replaceAll('\\', '/');
const sha256 = data => crypto.createHash('sha256').update(data).digest('hex');
const audit = fs.existsSync(recordPath) ? JSON.parse(fs.readFileSync(recordPath, 'utf8')) : {
  scope: root,
  writesRestrictedTo: output,
  javaTarget: 17,
  runtimeRequired: 'Temurin 25.0.3+9',
  timeoutMs: 30000,
  commands: [],
  fileReads: [],
  observations: [],
  limitations: [],
  accidentalAccessOutsideScope: [],
  allWritesStopped: false
};
const save = () => fs.writeFileSync(recordPath, JSON.stringify(audit, null, 2) + '\n');
const allowed = file => {
  const full = path.resolve(root, file);
  if (full !== root && !full.startsWith(root + path.sep)) throw Error('Outside run: ' + full);
  return full;
};
const read = (name, phase, print = false) => {
  const file = allowed(name);
  const data = fs.readFileSync(file);
  const binary = /\.(jar|class)$/.test(file);
  const entry = { path: relative(file), phase, bytes: data.length, sha256: sha256(data), truncated: false };
  if (binary) entry.contentRead = 'Binary bytes for SHA-256; jar/class semantics recorded by jar/javap command logs.';
  else entry.content = data.toString('utf8');
  if (file.endsWith('.txt')) entry.hex = data.toString('hex');
  audit.fileReads.push(entry);
  if (print) process.stdout.write(JSON.stringify(entry, null, 2) + '\n');
};
const files = dir => fs.readdirSync(dir, { withFileTypes: true }).flatMap(item => {
  const name = path.join(dir, item.name);
  return item.isDirectory() ? files(name) : [name];
});
const [action, ...args] = process.argv.slice(2);
if (action === 'init') {
  for (const name of ['TASK.md', 'skill/SKILL.md', 'skill/references/method.md', 'output/audit.mjs']) read(name, 'initial');
  for (const prefix of ['input/project', 'output/project']) {
    for (const file of files(allowed(prefix))) read(relative(file), 'initial');
  }
  audit.environment = Object.fromEntries(['JAVA_TOOL_OPTIONS', 'JDK_JAVA_OPTIONS', '_JAVA_OPTIONS', 'CLASSPATH'].map(key => [key, process.env[key] ?? null]));
  audit.referenceSelection = {
    read: ['skill/SKILL.md (full)', 'skill/references/method.md (full; differential diagnosis and controlled experiments)'],
    notRead: ['skill/references/production-evidence.md: no access to or intervention in production']
  };
  audit.observations.push({ phase: 'before execution', expected: 'Accepted export replaces CRLF and isolated CR with LF, preserves all other text and LegacyBanner; --release 17 and existing runtime/library retained.', hypothesis: 'The export classpath selects a stale work.Normalizer from compat.jar before .build/app, whereas SelfTest selects .build/app. Prediction: loading provenance or bytecode differs and swapping only classpath order corrects the same export bytes. Refutation: both runs load the same Normalizer implementation, or only changing classpath order leaves the same violation.' });
  process.stdout.write('Audit initialized; initial input and output snapshots retained.\n');
} else if (action === 'run') {
  const [id, executable, ...commandArgs] = args;
  if (!/^[a-z0-9-]+$/.test(id)) throw Error('Invalid command ID');
  const started = new Date().toISOString();
  const result = spawnSync(executable, commandArgs, { cwd: project, encoding: 'utf8', timeout: 30000, windowsHide: true, maxBuffer: 10 * 1024 * 1024 });
  fs.mkdirSync(path.join(output, 'logs'), { recursive: true });
  const stdoutFile = path.join(output, 'logs', `${id}.stdout.log`);
  const stderrFile = path.join(output, 'logs', `${id}.stderr.log`);
  fs.writeFileSync(stdoutFile, result.stdout || '');
  fs.writeFileSync(stderrFile, result.stderr || '');
  audit.commands.push({ id, executed: true, executable, args: commandArgs, cwd: project, timeoutMs: 30000, started, ended: new Date().toISOString(), exit: result.status, signal: result.signal, error: result.error ? String(result.error) : null, stdout: relative(stdoutFile), stderr: relative(stderrFile), outputTruncated: false });
  save();
  process.stdout.write(result.stdout || '');
  process.stderr.write(result.stderr || '');
  process.stdout.write(`Recorded ${id}: exit=${result.status}, signal=${result.signal}, error=${result.error || 'none'}\n`);
  process.exitCode = result.status ?? 1;
} else if (action === 'read') {
  const [phase, ...names] = args;
  for (const name of names) read(name, phase, true);
} else if (action === 'snapshot') {
  const [phase, ...prefixes] = args;
  for (const prefix of prefixes) {
    const resolved = allowed(prefix);
    for (const file of fs.statSync(resolved).isDirectory() ? files(resolved) : [resolved]) read(relative(file), phase);
  }
  process.stdout.write('Snapshot recorded: ' + phase + '\n');
} else if (action === 'clean-build') {
  const build = path.join(project, '.build');
  if (fs.existsSync(build)) {
    const resolved = fs.realpathSync(build);
    if (path.dirname(resolved) !== fs.realpathSync(project) || path.basename(resolved) !== '.build') throw Error('Unexpected build target: ' + resolved);
    audit.observations.push({ phase: 'before final verification', action: 'Removed disposable .build after resolving and checking its path inside output/project.', target: resolved });
    fs.rmSync(resolved, { recursive: true, force: true });
  }
  process.stdout.write('Disposable build removed inside output/project.\n');
} else if (action === 'verify-integrity') {
  const changes = [];
  const initial = audit.fileReads.filter(entry => entry.phase === 'initial');
  for (const before of initial) {
    if (before.path === 'output/audit.mjs' || before.path.includes('/.build/')) continue;
    read(before.path, 'integrity-verification');
    const after = audit.fileReads.at(-1);
    const changed = before.sha256 !== after.sha256;
    if (changed && !['output/project/run.mjs', 'output/project/launch.json'].includes(before.path)) throw Error('Unexpected change: ' + before.path);
    if (changed) changes.push({ path: before.path, before: before.sha256, after: after.sha256 });
  }
  read('output/project/test/export.test.mjs', 'integrity-verification');
  const jar = audit.fileReads.find(entry => entry.path === 'output/project/lib/compat.jar' && entry.phase === 'initial');
  const summary = { inputFilesUnchanged: true, contractUnchanged: true, fixtureUnchanged: true, javaSourcesAndExistingTestUnchanged: true, jarUnchanged: true, jarSha256: jar.sha256, changedOriginalFiles: changes, addedProjectFiles: ['test/export.test.mjs'] };
  audit.integrity = summary;
  fs.writeFileSync(path.join(output, 'logs', 'integrity.json'), JSON.stringify(summary, null, 2) + '\n');
  process.stdout.write(JSON.stringify(summary, null, 2) + '\n');
} else {
  throw Error('Usage: audit.mjs init | run id executable args... | read phase paths... | snapshot phase paths... | clean-build | verify-integrity');
}
save();
