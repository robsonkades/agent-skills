import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import test from 'node:test';

const project = fileURLToPath(new URL('..', import.meta.url));
const unchangedText = '\uFEFFOlá 🌍\t \u0000\\r\u2028fim';

test('accepted export command honors the text and compatibility contract', async t => {
  const build = path.join(project, '.build');
  fs.mkdirSync(build, { recursive: true });
  const temporary = fs.mkdtempSync(path.join(build, 'export-regression-'));
  const cases = [
    { name: 'original fixture', fixture: 'fixture/input.txt', expected: 'a\nb\nc\n' },
    { name: 'one isolated CR', input: '\r', expected: '\n' },
    { name: 'CRLF is one LF', input: '\r\n', expected: '\n' },
    { name: 'adjacent separators and trailing CR', input: '\r\r\n\nx\r', expected: '\n\n\nx\n' },
    { name: 'empty input', input: '', expected: '' },
    { name: 'LF remains unchanged', input: 'unchanged\nline\n', expected: 'unchanged\nline\n' },
    { name: 'other UTF-8 text remains byte-identical', input: unchangedText, expected: unchangedText },
    { name: 'mixed UTF-8 separators', input: 'ação\r\n東京\r🌍\n', expected: 'ação\n東京\n🌍\n' }
  ];
  try {
    for (const [index, entry] of cases.entries()) {
      await t.test(entry.name, () => {
        const input = entry.fixture ? path.join(project, entry.fixture) : path.join(temporary, `${index}.input`);
        const output = path.join(temporary, `${index}.output`);
        if (!entry.fixture) fs.writeFileSync(input, entry.input, 'utf8');
        const original = fs.readFileSync(input);
        const result = spawnSync(process.execPath, ['run.mjs', 'export', input, output], {
          cwd: project,
          encoding: 'utf8',
          timeout: 30000,
          windowsHide: true
        });
        assert.equal(result.error, undefined, String(result.error));
        assert.equal(result.status, 0, result.stderr);
        assert.equal(result.signal, null);
        assert.equal(result.stdout.trim(), 'Exporter compatible mode', 'LegacyBanner must remain available');
        assert.deepEqual(fs.readFileSync(input), original, 'export must preserve the input');
        assert.deepEqual(fs.readFileSync(output), Buffer.from(entry.expected, 'utf8'), entry.name);
      });
    }
  } finally {
    const resolvedTemporary = fs.realpathSync(temporary);
    const resolvedBuild = fs.realpathSync(build);
    assert.equal(path.dirname(resolvedTemporary), resolvedBuild);
    assert.ok(path.basename(resolvedTemporary).startsWith('export-regression-'));
    fs.rmSync(resolvedTemporary, { recursive: true, force: true });
  }
});
