import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

// Exercise the supported command and launch.json, including the retained JAR.
// Generated inputs and outputs stay in the disposable project build directory.
const directory = fs.mkdtempSync(path.join('.build', 'export-test-'));
const preserved = '\uFEFFação\t東京 😀\nNUL:\0\u2028\u2029 no final newline';
const cases = [
  { name: 'original-fixture', file: 'fixture/input.txt', expected: 'a\nb\nc\n' },
  { name: 'mixed-endings', input: 'alpha\rbravo\r\ncharlie\n\rdelta\r\recho\r\n\r', expected: 'alpha\nbravo\ncharlie\n\ndelta\n\necho\n\n' },
  { name: 'preserve-text', input: preserved, expected: preserved },
  { name: 'only-cr', input: '\r', expected: '\n' },
  { name: 'only-crlf', input: '\r\n', expected: '\n' },
  { name: 'empty', input: '', expected: '' },
];

for (const example of cases) {
  const input = example.file ?? path.join(directory, `${example.name}.input.txt`);
  const output = path.join(directory, `${example.name}.output.txt`);
  if (!example.file) fs.writeFileSync(input, example.input, 'utf8');
  const args = ['run.mjs', 'export', input, output];
  const result = spawnSync(process.execPath, args, {
    encoding: 'utf8', timeout: 10000, windowsHide: true,
  });
  fs.writeFileSync(path.join(directory, `${example.name}.execution.json`), JSON.stringify({
    command: process.execPath, args, cwd: process.cwd(), timeoutMs: 10000,
    exit: result.status, signal: result.signal, stdout: result.stdout, stderr: result.stderr,
    error: result.error?.message ?? null,
  }, null, 2) + '\n');
  assert.ifError(result.error);
  assert.equal(result.status, 0, `${example.name}: export failed: ${result.stderr}`);
  assert.equal(result.stdout.trim(), 'Exporter compatible mode', `${example.name}: LegacyBanner`);
  assert.deepEqual(fs.readFileSync(output), Buffer.from(example.expected, 'utf8'), `${example.name}: exact UTF-8 output`);
  console.log(`Export PASS: ${example.name}`);
}
console.log(`Export integration PASS (${cases.length} cases)`);
