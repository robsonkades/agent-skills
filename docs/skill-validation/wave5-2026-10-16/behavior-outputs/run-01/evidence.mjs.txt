import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';

const out = path.dirname(fileURLToPath(import.meta.url));
const root = path.dirname(out);
const project = path.join(out, 'project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const auditPath = path.join(out, 'execution.json');
const audit = JSON.parse(fs.readFileSync(auditPath, 'utf8'));
const phase = process.argv[2];
const started = Date.now();
fs.mkdirSync(path.join(out, 'logs'), { recursive: true });
const relative = p => path.relative(root, p).replaceAll('\\', '/');
const hash = content => createHash('sha256').update(content).digest('hex');
const save = () => fs.writeFileSync(auditPath, JSON.stringify(audit, null, 2) + '\n');

function read(file) {
  const bytes = fs.readFileSync(file);
  const entry = { path: relative(file), phase, sha256: hash(bytes), content: bytes.toString('utf8'), truncated: false };
  audit.reads.push(entry);
  return entry;
}

function files(dir) {
  return fs.readdirSync(dir, { withFileTypes: true }).flatMap(item => {
    const file = path.join(dir, item.name);
    return item.isDirectory() ? files(file) : [file];
  }).sort();
}

function run(name, executable, args, expectedExit = 0, expectedOutput) {
  const timeoutMs = Math.min(30000, 29000 - (Date.now() - started));
  if (timeoutMs <= 0) throw new Error('Evidence process deadline exceeded');
  const result = spawnSync(executable, args, { cwd: project, encoding: 'utf8', timeout: timeoutMs, windowsHide: true });
  const log = path.join(out, 'logs', `${phase}-${name}.log`);
  const record = {
    phase, executable, args, cwd: project, timeoutMs,
    exit: result.status, signal: result.signal, error: result.error?.message ?? null,
    stdout: result.stdout ?? '', stderr: result.stderr ?? '', log: relative(log), expectedExit
  };
  if (name === 'check') {
    const walk = dir => fs.readdirSync(path.join(project, dir), { withFileTypes: true }).flatMap(item => {
      const file = path.join(dir, item.name);
      return item.isDirectory() ? walk(file) : item.name.endsWith('.java') ? [file] : [];
    });
    record.nestedCommands = [
      { executable: path.join(jdk, 'javac.exe'), args: ['--release', '17', '-d', '.build', ...walk('src'), ...walk('test')], cwd: project, timeoutMs: 30000, exit: 0 },
      { executable: path.join(jdk, 'java.exe'), args: ['-cp', '.build', 'work.SelfTest'], cwd: project, timeoutMs: 30000, exit: result.status }
    ];
    record.nestedCommandEvidence = 'Exact arguments reconstructed from unchanged check.mjs and its directory traversal. Compiler exit 0 follows from execution reaching SelfTest; SelfTest exit is propagated by check.mjs. Combined stdout/stderr is retained above.';
  }
  fs.writeFileSync(log, JSON.stringify(record, null, 2) + '\n');
  audit.commands.push(record);
  save();
  console.log(JSON.stringify(record));
  if (result.error || result.status !== expectedExit) throw new Error(`${name}: unexpected command result`);
  if (expectedOutput && !expectedOutput.test(result.stdout)) throw new Error(`${name}: unexpected output`);
  return record;
}

try {
  read(fileURLToPath(import.meta.url));
  audit.internalReads ??= [];
  audit.internalReads.push({ phase, path: 'output/execution.json', purpose: 'Load prior audit state; contents retained as this cumulative JSON, not recursively embedded' });
  if (phase === 'baseline') {
    read(path.join(root, 'TASK.md'));
    const input = files(path.join(root, 'input', 'project'));
    audit.initialHashes = {};
    for (const original of input) {
      const source = read(original);
      const destination = path.join(project, path.relative(path.join(root, 'input', 'project'), original));
      const copy = read(destination);
      if (source.sha256 !== copy.sha256) throw new Error(`Initial copy differs: ${copy.path}`);
      audit.initialHashes[source.path] = source.sha256;
    }
    run('runtime', path.join(jdk, 'java.exe'), ['-version']);
    run('check', process.execPath, ['check.mjs'], 0, /SelfTest PASS/);
    run('demo', path.join(jdk, 'java.exe'), ['-cp', '.build', 'work.Demo'], 0, /first=1000\r?\nfirst-after-reload=1200\r?\nlatest=1200/);
  } else if (phase === 'regression') {
    read(path.join(project, 'test', 'work', 'SelfTest.java'));
    run('check', process.execPath, ['check.mjs'], 1);
  } else if (phase === 'final') {
    for (const original of files(path.join(root, 'input', 'project'))) {
      const source = read(original);
      if (source.sha256 !== audit.initialHashes[source.path]) throw new Error(`Input changed: ${source.path}`);
      const final = read(path.join(project, path.relative(path.join(root, 'input', 'project'), original)));
      if (['contract.md', 'support.md', 'check.mjs'].includes(path.basename(original)) && source.sha256 !== final.sha256) {
        throw new Error(`Protected or baseline configuration changed: ${final.path}`);
      }
    }
    run('check', process.execPath, ['check.mjs'], 0, /SelfTest PASS/);
    run('demo', path.join(jdk, 'java.exe'), ['-cp', '.build', 'work.Demo'], 0, /first=1000\r?\nfirst-after-reload=1000\r?\nlatest=1200/);
    run('bytecode', path.join(jdk, 'javap.exe'), ['-verbose', '.build/work/Report.class'], 0, /major version: 61/);
    audit.inputUnchanged = true;
  } else {
    throw new Error(`Unknown phase: ${phase}`);
  }
  audit.phaseResults ??= {};
  audit.phaseResults[phase] = { success: true, elapsedMs: Date.now() - started };
} catch (error) {
  audit.phaseResults ??= {};
  audit.phaseResults[phase] = { success: false, message: error.message, elapsedMs: Date.now() - started };
  console.error(error);
  process.exitCode = 1;
} finally {
  audit.commands.push({ executable: process.execPath, args: process.argv.slice(1), cwd: process.cwd(), phase, exit: process.exitCode ?? 0, output: 'JSON command results printed by this script; full child output in the phase logs and commands entries.', elapsedMs: Date.now() - started, timeoutStrategy: 'Synchronous child execution with a shared 29 s deadline; no background processes.' });
  save();
}
