package work;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public final class SelfTest {
    public static void main(String[] args) {
        Catalog c = new Catalog();
        c.load(new Entry("A", 450), new Entry("B", 550));
        Report first = c.open();
        if (first.total() != 1000 || first.entries().size() != 2) throw new AssertionError("initial report");
        try { first.entries().clear(); throw new AssertionError("writable report"); }
        catch (UnsupportedOperationException expected) { }
        reloadKeepsEarlierReports();
        entriesStayReadOnly();
        constructorOwnsItsSnapshot();
        System.out.println("SelfTest PASS");
    }

    private static void reloadKeepsEarlierReports() {
        Catalog catalog = new Catalog();
        Report initiallyEmpty = catalog.open();
        List<Entry> original = List.of(new Entry("A", 450), new Entry("B", 550));
        catalog.load(original.toArray(Entry[]::new));
        Report first = catalog.open();
        List<Entry> retainedEntries = first.entries();
        catalog.load(new Entry("C", 1200));
        check(first, original, 1000, "first after reload");
        if (!retainedEntries.equals(original)) throw new AssertionError("retained entries changed");
        Report latest = catalog.open();
        check(latest, List.of(new Entry("C", 1200)), 1200, "latest after reload");

        Entry duplicate = new Entry("D", 450);
        List<Entry> signed = List.of(duplicate, new Entry("credit", -550), duplicate, new Entry("zero", 0));
        catalog.load(signed.toArray(Entry[]::new));
        Report signedReport = catalog.open();
        check(signedReport, signed, 350, "order, duplicates and signed cents");
        catalog.load();
        check(catalog.open(), List.of(), 0, "latest empty report");
        check(initiallyEmpty, List.of(), 0, "initial empty report after reloads");
        check(first, original, 1000, "first after multiple reloads");
        check(latest, List.of(new Entry("C", 1200)), 1200, "second after multiple reloads");
        check(signedReport, signed, 350, "signed report after empty load");
    }

    private static void entriesStayReadOnly() {
        Catalog catalog = new Catalog();
        List<Entry> expected = List.of(new Entry("A", 450), new Entry("B", 550));
        catalog.load(expected.toArray(Entry[]::new));
        Report report = catalog.open();
        expectReadOnly(() -> report.entries().add(new Entry("C", 1200)), "add");
        expectReadOnly(() -> report.entries().remove(0), "remove");
        expectReadOnly(() -> report.entries().set(0, new Entry("C", 1200)), "replace");
        expectReadOnly(() -> Collections.swap(report.entries(), 0, 1), "reorder");
        expectReadOnly(() -> report.entries().clear(), "clear");
        check(report, expected, 1000, "after attempted modifications");
    }

    private static void constructorOwnsItsSnapshot() {
        List<Entry> expected = List.of(new Entry("A", 450), new Entry("B", 550));
        List<Entry> source = new ArrayList<>(expected);
        Report report = new Report(source);
        source.clear();
        check(report, expected, 1000, "external constructor input mutation");
        expectReadOnly(() -> report.entries().add(new Entry("C", 1200)), "direct report add");
    }

    private static void check(Report report, List<Entry> entries, long total, String context) {
        if (report.total() != total || !report.entries().equals(entries)) {
            throw new AssertionError(context + ": expected " + entries + " / " + total
                    + ", got " + report.entries() + " / " + report.total());
        }
    }

    private static void expectReadOnly(Runnable operation, String context) {
        try {
            operation.run();
        } catch (UnsupportedOperationException expected) {
            return;
        }
        throw new AssertionError("writable report: " + context);
    }
}
