import fs from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
import {spawnSync} from 'node:child_process';
import {createHash} from 'node:crypto';

const output = path.dirname(fileURLToPath(import.meta.url));
const run = path.dirname(output);
const project = path.join(output, 'project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const phase = process.argv[2];
if (!['baseline', 'final'].includes(phase)) throw new Error('Use baseline or final');
const manifestPath = path.join(output, 'execution.json');
const manifest = fs.existsSync(manifestPath)
  ? JSON.parse(fs.readFileSync(manifestPath, 'utf8'))
  : {
      scope: run,
      targetJava: 17,
      processTimeoutMs: 30000,
      networkUsed: false,
      dependenciesChanged: false,
      productionSourcesChanged: false,
      delegated: false,
      commands: [],
      reads: [],
      limitations: [],
      outOfScopeAccess: [],
      allWritesStopped: false
    };
manifest.allWritesStopped = false;
const logDir = path.join(output, 'logs');
fs.mkdirSync(logDir, {recursive: true});

for (const relative of [
  'TASK.md',
  'output/project/support.md',
  'output/project/contract.md',
  'output/project/check.mjs',
  'output/project/src/work/Catalog.java',
  'output/project/src/work/Entry.java',
  'output/project/src/work/Report.java',
  'output/project/src/work/Demo.java',
  'output/project/test/work/SelfTest.java'
]) {
  const bytes = fs.readFileSync(path.join(run, relative));
  manifest.reads.push({
    phase,
    path: relative,
    sha256: createHash('sha256').update(bytes).digest('hex'),
    bytes: bytes.length,
    truncated: false,
    content: bytes.toString('utf8')
  });
}

function execute(label, executable, args) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(executable, args, {
    cwd: project,
    encoding: 'utf8',
    windowsHide: true,
    timeout: 30000,
    maxBuffer: 1024 * 1024
  });
  const log = {
    phase, label, executed: true,
    executable, args, cwd: project, timeoutMs: 30000,
    startedAt, finishedAt: new Date().toISOString(),
    exit: result.status, signal: result.signal,
    error: result.error?.message ?? null,
    stdout: result.stdout ?? '', stderr: result.stderr ?? '',
    outputTruncated: false
  };
  const relativeLog = `output/logs/${phase}-${label}.json`;
  fs.writeFileSync(path.join(run, relativeLog), JSON.stringify(log, null, 2) + '\n');
  manifest.commands.push({...log, log: relativeLog});
  fs.writeFileSync(manifestPath, JSON.stringify(manifest, null, 2) + '\n');
  process.stdout.write(`${phase} ${label}: exit=${result.status}\n${log.stdout}${log.stderr}`);
  if (result.status !== 0 || result.error) process.exit(result.status || 1);
}

execute('java-version', path.join(jdk, 'java.exe'), ['-version']);
execute('existing-check', process.execPath, ['check.mjs']);
execute('demo', path.join(jdk, 'java.exe'), ['-cp', '.build', 'work.Demo']);
