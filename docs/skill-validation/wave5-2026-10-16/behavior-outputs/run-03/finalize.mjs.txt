const run = path.dirname(output);
const manifestPath = path.join(output, 'execution.json');
const manifest = JSON.parse(fs.readFileSync(manifestPath, 'utf8'));
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const projectFiles = [
  'support.md', 'contract.md', 'check.mjs',
  'src/work/Report.java', 'src/work/Entry.java',

    },
    {

        '.\\input\\project\\src\\work\\Demo.java',
        '.\\input\\project\\src\\work\\Catalog.java'
      ].join('\n') + '\n',
      truncated: false
    },
    ...projectFiles.map(relative => ({
      commandText: `Get-Content -LiteralPath 'output/project/${relative}'`,
      cwd: run, exit: 0,
      output: `Complete initial content recorded under reads for output/project/${relative}, phase baseline; no truncation.`
    }))
  ]
};
manifest.supervisedInvocations = ['baseline', 'final'].map(phase => ({
  executed: true,
  executable: 'node',
  args: ['-e', `const {spawnSync}=require('node:child_process');const r=spawnSync(process.execPath,['output/verify.mjs','${phase}'],{cwd:process.cwd(),encoding:'utf8',timeout:30000,windowsHide:true});process.stdout.write(r.stdout||'');process.stderr.write(r.stderr||'');if(r.error)process.stderr.write(r.error.message);process.exit(r.status??1)`],
  cwd: run, exit: 0, timeoutMs: 30000,
  output: manifest.commands.filter(command => command.phase === phase)
    .map(command => `${phase} ${command.label}: exit=${command.exit}\n${command.stdout}${command.stderr}`).join(''),
  truncated: false
}));

manifest.fileComparison = [];
for (const relative of projectFiles) {
  const inputRelative = `input/project/${relative}`;
  const inputBytes = fs.readFileSync(path.join(run, inputRelative));
  const finalRelative = `output/project/${relative}`;
  const finalBytes = fs.readFileSync(path.join(run, finalRelative));
  const baseline = manifest.reads.find(read => read.path === finalRelative && read.phase === 'baseline');
  manifest.reads.push({
    phase: 'final-comparison', path: inputRelative,
    bytes: inputBytes.length, sha256: hash(inputBytes),
    truncated: false, content: inputBytes.toString('utf8')
  });
  manifest.reads.push({
    phase: 'final-comparison', path: finalRelative,
    bytes: finalBytes.length, sha256: hash(finalBytes),
    truncated: false, content: finalBytes.toString('utf8')
  });
  const comparison = {
    file: relative,
    inputSha256: hash(inputBytes), finalSha256: hash(finalBytes),
    inputMatchesInitialWorkingCopy: baseline?.sha256 === hash(inputBytes),
    finalMatchesInput: inputBytes.equals(finalBytes)
  };
  manifest.fileComparison.push(comparison);
  if (!comparison.inputMatchesInitialWorkingCopy) throw new Error(`Input diverged from original working copy: ${relative}`);
  if (relative !== 'test/work/SelfTest.java' && !comparison.finalMatchesInput) throw new Error(`Unexpected project change: ${relative}`);
}

for (const relative of ['output/verify.mjs', 'output/finalize.mjs', 'output/REVIEW.md']) {
  const bytes = fs.readFileSync(path.join(run, relative));
  manifest.reads.push({
    phase: 'artifact-finalization', path: relative,
    bytes: bytes.length, sha256: hash(bytes),
    truncated: false, content: bytes.toString('utf8')
  });
}
