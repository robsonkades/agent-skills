import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';

 return {name:p.name,initialVersion:p.initialVersion,finalVersion:manifest.version,changedFiles,descriptionMatches:front.description.trim()===manifest.description.trim(),versionCorrect:manifest.version===(changedFiles.length?p.initialVersion.replace(/(\d+)$/,n=>String(+n+1)):p.initialVersion),diffReviewed:t.diffReviewed===true,reviewedDiffStillMatches:sha(d.stdout)===t.reviewedDiffSha256,allWritesStopped:h?.allWritesStopped===true,rawHandoffMatches:fs.existsSync(raw)&&sha(fs.readFileSync(raw))===t.capturedHandoffSha,canonicalMatchesRaw:!!h&&fs.existsSync(raw)&&JSON.stringify(h)===JSON.stringify(JSON.parse(fs.readFileSync(raw,'utf8'))),status:t.status};
});

const indexChanges=newIndex.skills.filter(x=>JSON.stringify(x)!==JSON.stringify(oldEntries.get(x.name))).map(x=>x.name),lostIndexEntries=oldIndex.skills.filter(x=>!newIndex.skills.some(y=>y.name===x.name)).map(x=>x.name);

if(final&&(result.total!==7||issues.length||unexpected.length||lostIndexEntries.length||!indexMatchesPackageChanges))process.exitCode=1;
