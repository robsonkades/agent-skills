 const marked=new Set(),paths=new Set(),seen=new Set();
 function mark(v,source=false){if(typeof v==='string'&&source&&/^C:[/\\]/.test(v))marked.add(v.replaceAll('\\','/').replace(/\/$/,''));else if(Array.isArray(v))v.forEach(x=>mark(x,source));else if(v&&typeof v==='object'){const primary=source||Object.entries(v).some(([k,x])=>/^(url|sourceUrl|uri)$/.test(k)&&typeof x==='string'&&/^https?:/.test(x));for(const[k,x]of Object.entries(v))mark(x,primary||/^(sources|primarySources|sourceDownloads|sourceSnapshots|verifiedSources|sourceEvidence)$/.test(k));}}
 mark(h);
 function add(p){p=p.replaceAll('\\','/').replace(/\/$/,'');if(!p.startsWith(root+'/')||seen.has(p)||!fs.existsSync(p))return;seen.add(p);const st=fs.lstatSync(p);if(st.isSymbolicLink())return;if(st.isDirectory()){if(skip.has(path.basename(p)))return;for(const e of fs.readdirSync(p))add(p+'/'+e);}else if(st.isFile())paths.add(p);}
 function visit(v){if(typeof v==='string'&&/^C:[/\\]/.test(v))add(v);else if(Array.isArray(v))v.forEach(visit);else if(v&&typeof v==='object')Object.values(v).forEach(visit);}
 if(h.evidenceFiles)visit(h.evidenceFiles);else visit(h);
 // Include referenced authored scripts/logs even when a reviewer supplied only a directory manifest.
 visit(h);
 const entries=[];
 for(const p of [...paths].sort()){
  if(p===root+'/'+t.name+'.json')continue;
  const bytes=fs.statSync(p).size;if(bytes>8*1024*1024){entries.push({path:p,bytes,archived:false,reason:'Large artifact remains at original location.'});continue;}
  const b=fs.readFileSync(p),start=b.subarray(0,6000).toString('utf8'),ext=path.extname(p).toLowerCase();
  const thirdParty=[...marked].some(m=>p===m||p.startsWith(m+'/'))||/<!doctype\s+html|<html(?:\s|>)/i.test(start)||/\[wordlim:/.test(start)||(/\.(java|js|mjs|txt|md|adoc)$/i.test(p)&&/copyright|SPDX-License-Identifier|licensed under/i.test(start))||(/source.*\.json$/i.test(p)&&bytes>16000);
  const entry={path:p,bytes,sha256:sha(b),archived:!thirdParty&&allowed.has(ext)&&bytes<=1048576};
  if(entry.archived){entry.target='evidence/'+t.name+'/'+sha(p).slice(0,12)+'-'+path.basename(p)+'.txt';put(report+'/'+entry.target,b);}else entry.reason=thirdParty?'Source reference retained, third-party content not vendored.':'Binary/large evidence retained by path and hash.';
  entries.push(entry);
 }
 records.push({skill:t.name,rawHandoff:'handoffs-raw/'+t.name+'.json.txt',handoffSha256:sha(raw),entries});
}

 put(report+'/behavior-artifact-manifest.json',JSON.stringify({createdAt:new Date().toISOString(),policy:'Byte-preserving local fixture/actor output copies; all binary fixtures here were authored locally, no external dependencies vendored.',artifacts},null,2)+'\n');
 console.log(JSON.stringify({actorAndCaseArtifacts:artifacts.length}));
}
console.log(JSON.stringify({reviewed:records.length,archived:records.reduce((s,r)=>s+r.entries.filter(e=>e.archived).length,0),referenceOnly:records.reduce((s,r)=>s+r.entries.filter(e=>!e.archived).length,0),helpers:helpers.length}));
