const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
function walk(dir,sub=''){return fs.readdirSync(dir+'/'+sub,{withFileTypes:true}).flatMap(e=>{const p=sub?sub+'/'+e.name:e.name;return e.isDirectory()?(e.name==='.build'?[]:walk(dir,p)):[p];});}
function compact(value,key=''){
 if(Array.isArray(value))return value.map(v=>compact(v,key));
 if(value&&typeof value==='object')return Object.fromEntries(Object.entries(value).map(([k,v])=>[k,compact(v,k)]));
 if(typeof value==='string'&&(/content|stdout|stderr|text|script/i.test(key)||value.length>2500))return {omittedCharacters:value.length,sha256:sha(value)};
 return value;
}
