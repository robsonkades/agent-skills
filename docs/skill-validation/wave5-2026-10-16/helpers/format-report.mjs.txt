import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const repo='C:/git/agent-skills',report=repo+'/docs/skill-validation/wave5-2026-10-16';
const label=process.argv[2];if(!label||!/^[a-z0-9-]+$/.test(label))throw Error('Unique label required');
const log=report+'/format-'+label+'.log.txt';if(fs.existsSync(log))throw Error('Preserve earlier logs');
const walk=d=>fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(d,e.name)):[path.join(d,e.name)]);
const files=walk(report).filter(p=>/\.(json|md|ya?ml)$/.test(p));
const args=['node_modules/prettier/bin/prettier.cjs',process.argv.includes('--check')?'--check':'--write',...files.map(f=>path.relative(repo,f))];
const r=spawnSync(process.execPath,args,{cwd:repo,encoding:'utf8',timeout:60000,windowsHide:true,maxBuffer:16*1024*1024});
fs.writeFileSync(log,(r.stdout||'')+(r.stderr||''));console.log(JSON.stringify({files:files.length,exit:r.status,error:r.error?.message,log}));process.exitCode=r.status??1;
