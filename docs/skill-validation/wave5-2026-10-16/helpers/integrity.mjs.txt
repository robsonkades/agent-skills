import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const report='C:/git/agent-skills/docs/skill-validation/wave5-2026-10-16';
const read=p=>JSON.parse(fs.readFileSync(report+'/'+p,'utf8'));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const same=(p,h)=>assert.equal(sha(fs.readFileSync(p)),h,p);
const evidence=read('evidence-manifest.json'),artifacts=read('behavior-artifact-manifest.json'),manifest=read('behavior-manifest.json'),queue=read('queue.json');
const coverage=read('coverage.json'),repo='C:/git/agent-skills';
for(const prior of coverage.prior)same(repo+'/'+prior.source,prior.sha256);
const covered=[...coverage.prior.flatMap(p=>p.names),...queue.tasks.map(t=>t.name)];
const catalog=fs.readdirSync(repo+'/skills',{withFileTypes:true}).filter(e=>e.isDirectory()&&fs.existsSync(repo+'/skills/'+e.name+'/skill.yaml')&&fs.existsSync(repo+'/skills/'+e.name+'/SKILL.md')).map(e=>e.name).sort();
const inventory=[...fs.readFileSync(repo+'/docs/skills-evolution-inventory.md','utf8').matchAll(/\]\(\.\.\/skills\/([^/]+)\/SKILL\.md\)/g)].map(m=>m[1]).sort();
assert.equal(covered.length,305);assert.equal(new Set(covered).size,305);assert.deepEqual([...covered].sort(),catalog);assert.deepEqual(catalog,inventory);
assert.deepEqual(coverage.selected.map(s=>s.name).sort(),queue.tasks.map(t=>t.name).sort());
let archived=0,helpers=0;
for(const group of evidence.records){same(report+'/'+group.rawHandoff,group.handoffSha256);assert.deepEqual(read('handoffs/'+group.skill+'.json'),read(group.rawHandoff));for(const entry of group.entries.filter(e=>e.archived)){same(report+'/'+entry.target,entry.sha256);same(entry.path,entry.sha256);archived++;}}
for(const helper of evidence.helpers){same(helper.path,helper.sha256);same(report+'/'+helper.target,helper.sha256);helpers++;}
for(const artifact of artifacts.artifacts){same(artifact.path,artifact.sha256);same(report+'/'+artifact.target,artifact.sha256);}
same(report+'/behavior-manifest.frozen.json.txt',read('behavior-freeze.json').manifestSha256);
assert.deepEqual(read('behavior-manifest.json'),read('behavior-manifest.frozen.json.txt'));
assert.deepEqual(read('reserved-cases.json'),read('reserved-cases.frozen.json.txt'));
same(report+'/reserved-cases.frozen.json.txt',manifest.reservedCasesHash);
for(const run of manifest.runs){
 for(const input of run.inputs.filter(f=>!f.path.startsWith('output/')))same(run.directory+'/'+input.path,input.sha256);
 const result=read('behavior-results.json').runs.find(r=>r.id===run.id);same(run.directory+'/output/REVIEW.md',result.evidence.reviewSha256);same(run.directory+'/output/execution.json',result.evidence.journalSha256);
 assert.equal(JSON.parse(fs.readFileSync(run.directory+'/output/execution.json','utf8')).allWritesStopped,true);
}
assert.ok(queue.tasks.every(t=>t.status==='validated'&&t.diffReviewed));
assert.equal(read('completion.json').globalChecks.verify.exit,0);
assert.equal(read('preservation-final.json').unexpected.length,0);
const links=[];
for(const file of ['README.md','tasks.md','review-brief.md']){
 const text=fs.readFileSync(report+'/'+file,'utf8');
 for(const match of text.matchAll(/\[[^\]]+\]\(([^)]+)\)/g)){
  const href=match[1];if(/^[a-z][a-z0-9+.-]*:/i.test(href)||href.startsWith('#'))continue;
  const target=path.resolve(report,href.split('#')[0]);
  if(path.basename(target)==='integrity-final.json')continue;
  assert.ok(fs.existsSync(target),'Broken report link: '+href);links.push({from:file,target:href});
 }
}
const result={at:new Date().toISOString(),status:'PASS',checks:{rawHandoffs:evidence.records.length,reviewerEvidenceFiles:archived,helpers,actorAndCaseArtifacts:artifacts.artifacts.length,frozenManifests:2,actorRuns:manifest.runs.length,localReportLinks:links.length,validatedQueueTasks:queue.tasks.length,uniqueCatalogNamesCovered:covered.length,inventoryNames:inventory.length,priorCoverageSourceHashes:coverage.prior.length},limitations:['Hash and provenance checks establish preservation, not general behavioral effectiveness.','Actor output archives contain locally authored binary fixtures as byte-preserving .txt files.'],links};
fs.writeFileSync(report+'/integrity-final.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result.checks));
