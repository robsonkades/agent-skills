import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const jdk="C:/Users/robso/.jdks/temurin-25.0.3/bin";
const walk=p=>fs.readdirSync(p,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(p,e.name)):e.name.endsWith('.java')?[path.join(p,e.name)]:[]);
fs.mkdirSync('.build',{recursive:true});
const commands=[['javac.exe',['--release','17','-d','.build',...walk('src'),...walk('test')]],['java.exe',['-cp','.build','work.SelfTest']]];
for(const [tool,args] of commands){const r=spawnSync(path.join(jdk,tool),args,{encoding:'utf8',timeout:30000,windowsHide:true});process.stdout.write(r.stdout||'');process.stderr.write(r.stderr||'');if(r.status!==0)process.exit(r.status??1);}
