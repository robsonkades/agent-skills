package work;
import java.util.UUID;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.annotation.Transactional;
public class Dispatcher {
 private final JdbcTemplate jdbc;private final ReceiverClient receiver;private final Checkpoint checkpoint;
 public Dispatcher(JdbcTemplate jdbc,ReceiverClient receiver,Checkpoint checkpoint){this.jdbc=jdbc;this.receiver=receiver;this.checkpoint=checkpoint;}
 @Transactional public boolean replayOne(){
  var rows=jdbc.queryForList("select event_id,payload from outbox where done=false order by event_id limit 1");
  if(rows.isEmpty())return false;
  String eventId=(String)rows.getFirst().get("EVENT_ID"),payload=(String)rows.getFirst().get("PAYLOAD");
  receiver.accept(eventId,payload);
  jdbc.update("update outbox set done=true where event_id=?",eventId);
  checkpoint.afterAccepted(eventId);
  return true;
 }
}
