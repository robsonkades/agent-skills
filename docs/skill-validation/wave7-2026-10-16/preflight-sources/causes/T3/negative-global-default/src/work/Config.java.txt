package work;
import javax.sql.DataSource;
import org.springframework.context.annotation.*;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.jdbc.datasource.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.annotation.EnableTransactionManagement;
@Configuration @EnableTransactionManagement(proxyTargetClass=true)
public class Config {
 @Bean("ordersDataSource") public DataSource ordersDataSource(){return new DriverManagerDataSource("jdbc:h2:mem:orders;DB_CLOSE_DELAY=-1;LOCK_TIMEOUT=1000","sa","");}
 @Bean("metricsDataSource") public DataSource metricsDataSource(){return new DriverManagerDataSource("jdbc:h2:mem:metrics;DB_CLOSE_DELAY=-1;LOCK_TIMEOUT=1000","sa","");}
 @Bean("ordersJdbc") public JdbcTemplate ordersJdbc(@Qualifier("ordersDataSource") DataSource ds){return new JdbcTemplate(ds);}
 @Bean("metricsJdbc") public JdbcTemplate metricsJdbc(@Qualifier("metricsDataSource") DataSource ds){return new JdbcTemplate(ds);}
 @Bean("ordersTx") @Primary public PlatformTransactionManager ordersTx(@Qualifier("ordersDataSource") DataSource ds){return new DataSourceTransactionManager(ds);}
 @Bean("metricsTx") public PlatformTransactionManager metricsTx(@Qualifier("metricsDataSource") DataSource ds){return new DataSourceTransactionManager(ds);}
 @Bean public OrderNotifier orderNotifier(){return id->{};}
 @Bean public OrderService orderService(@Qualifier("ordersJdbc") JdbcTemplate jdbc,OrderNotifier notifier){return new OrderService(jdbc,notifier);}
 @Bean public MetricsService metricsService(@Qualifier("metricsJdbc") JdbcTemplate jdbc){return new MetricsService(jdbc);}
}
