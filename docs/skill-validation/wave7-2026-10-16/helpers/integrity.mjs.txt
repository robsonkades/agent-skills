import fs from 'node:fs';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';

const freeze=json(report+'/freeze.json'),base=json(report+'/baseline.json'),actors=json(report+'/actors.frozen.json.txt');

assert.equal(sha(fs.readFileSync(root+'/review/criteria-resolution.json')),freeze.criteriaResolutionSha256);
assert.equal(sha(fs.readFileSync(root+'/review/calibration-grades.json')),freeze.calibrationGradesSha256);
for(const f of freeze.caseFiles)assert.equal(sha(fs.readFileSync(root+'/cases/'+f.caseId+'/input/'+f.path)),f.sha256);
for(const f of freeze.probes)assert.equal(sha(fs.readFileSync(root+'/'+f.path)),f.sha256);

let protectedFiles=0;
