import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {execFileSync} from 'node:child_process';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave7-a765972c5d',repo='C:/git/agent-skills',report=repo+'/docs/skill-validation/wave7-2026-10-16';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const git=args=>execFileSync('git',args,{cwd:repo,encoding:'utf8',maxBuffer:64*1024*1024});
if(fs.existsSync(report))throw Error('Preserve existing wave');
const files=[...new Set(git(['ls-files','-z','--cached','--others','--exclude-standard']).split('\0').filter(Boolean))];
const allFiles=files.filter(f=>fs.existsSync(repo+'/'+f)).map(p=>({path:p,sha256:sha(fs.readFileSync(repo+'/'+p))}));
const walk=(d,s='')=>fs.readdirSync(d+'/'+s,{withFileTypes:true}).flatMap(e=>{const p=s?s+'/'+e.name:e.name;return e.isDirectory()?walk(d,p):[p];});
fs.cpSync(repo+'/skills/spring-transactions-and-events',root+'/snapshot',{recursive:true});
const skillFiles=walk(root+'/snapshot').map(p=>({path:p,sha256:sha(fs.readFileSync(root+'/snapshot/'+p))}));
fs.mkdirSync(report,{recursive:true});fs.mkdirSync(root+'/review',{recursive:true});
const baseline={at:new Date().toISOString(),root,head:git(['rev-parse','HEAD']).trim(),allFiles,gitStatus:git(['status','--short']),registryBefore:fs.readFileSync(repo+'/registry/skills.yaml','utf8'),ownedDiff:git(['diff','--','skills/spring-transactions-and-events']),skill:{name:'spring-transactions-and-events',version:'1.1.0',files:skillFiles}};
fs.writeFileSync(report+'/baseline.json',JSON.stringify(baseline,null,2)+'\n');
const prior=JSON.parse(fs.readFileSync(repo+'/docs/skill-validation/wave5-2026-10-16/handoffs/spring-transactions-and-events.json','utf8'));
for(const f of prior.finalOwnedHashes){const rel=f.path.replace('skills/spring-transactions-and-events/','');if(sha(fs.readFileSync(root+'/snapshot/'+rel))!==f.sha256)throw Error('Version changed');}
console.log(JSON.stringify({root,report,baselineFiles:allFiles.length,skill:baseline.skill,priorWaveIdentical:true,ownedDiff:baseline.ownedDiff}));
