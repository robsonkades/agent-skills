import fs from 'node:fs';
import path from 'node:path';
import {spawn} from 'node:child_process';
const repo='C:/git/agent-skills',report=repo+'/docs/skill-validation/wave7-2026-10-16';
const task=process.argv[2],attempt=Number(process.argv[3]??1);
if(!['registry:build','verify'].includes(task)||!Number.isSafeInteger(attempt)||attempt<1)throw Error('Invalid task');
const label=task.replace(':','-')+(attempt===1?'':'-attempt'+attempt),file=report+'/'+label+'.log.txt';
if(fs.existsSync(file))throw Error('Preserve earlier log');
const stream=fs.createWriteStream(file),startedAt=new Date().toISOString();
console.log('Starting npm run '+task+'; log '+file);
const child=spawn(process.env.ComSpec||'cmd.exe',['/d','/s','/c','npm run '+task],{cwd:repo,windowsHide:true});
let output='';for(const ch of [child.stdout,child.stderr])ch.on('data',b=>{stream.write(b);output+=b.toString();});
const exit=await new Promise((resolve,reject)=>{child.on('error',reject);child.on('close',resolve);});
await new Promise(resolve=>stream.end(resolve));
const record={command:'npm run '+task,cwd:repo,attempt,startedAt,finishedAt:new Date().toISOString(),exit,log:path.basename(file),node:process.version};
fs.writeFileSync(report+'/'+label+'-result.json',JSON.stringify(record,null,2)+'\n');
console.log(JSON.stringify(record));console.log(output.split(/\r?\n/).slice(-35).join('\n'));process.exitCode=exit??1;
