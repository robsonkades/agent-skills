import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave7-a765972c5d',report='C:/git/agent-skills/docs/skill-validation/wave7-2026-10-16';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const put=(p,b)=>{fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,b);};
const walk=(d,s='')=>fs.readdirSync(d+'/'+s,{withFileTypes:true}).flatMap(e=>{const p=s?s+'/'+e.name:e.name;return e.isDirectory()?walk(d,p):[p];});
const entries=[];
function archive(file,target,referenceOnly=false){
 const b=fs.readFileSync(file),externalBody=file.startsWith(root+'/actors/')&&/copyright|SPDX-License-Identifier|\[wordlim:|<!doctype\s+html|<html(?:\s|>)/i.test(b.toString('utf8'));
 const denied=referenceOnly||externalBody||/\.(class|jar|exe|dll|zip|html|pdf|db)$/i.test(file)||b.length>1048576;
 const e={path:file,bytes:b.length,sha256:sha(b),archived:!denied};
 if(!denied){e.target=target;put(report+'/'+target,b);}else e.reason='Binary or full external source retained by metadata/hash; not vendored.';
 entries.push(e);
}
const m=JSON.parse(fs.readFileSync(report+'/actors.frozen.json.txt','utf8'));
for(const r of m.runs){
 assert.equal(JSON.parse(fs.readFileSync(r.directory+'/output/execution.json','utf8')).allWritesStopped,true);
 for(const p of walk(r.directory+'/output'))archive(r.directory+'/output/'+p,'actor-outputs/'+r.id+'/'+p+'.txt');
}
for(const name of fs.readdirSync(root).filter(n=>/\.(mjs|cjs|ps1)$/.test(n)))archive(root+'/'+name,'helpers/'+name+'.txt');
for(const name of walk(root+'/review'))archive(root+'/review/'+name,'review/'+name+'.txt',name.startsWith('sources/')&&name!=='sources/retrieval.json');
for(const name of ['criteria-draft.json','calibration-expected.json','dependencies.json','root-grades-part1.json','root-grades-part2.json'])archive(root+'/'+name,'review/'+name+'.txt');
for(const rel of walk(root+'/snapshot'))archive(root+'/snapshot/'+rel,'snapshot/'+rel+'.txt');
for(const rel of walk(root+'/evaluator'))archive(root+'/evaluator/'+rel,'evaluator/'+rel+'.txt');
for(const rel of walk(root+'/preflight').filter(p=>!p.split('/').includes('.build')&&!p.split('/').includes('lib')))archive(root+'/preflight/'+rel,'preflight-sources/'+rel+'.txt');
put(report+'/artifact-manifest.json',JSON.stringify({at:new Date().toISOString(),policy:'Raw authored text copied without reformatting. Libraries, compilation output and full external source documents retained by reference/hash only. Actor execution journals are not independently captured complete tool traces.',entries},null,2)+'\n');
console.log(JSON.stringify({artifacts:entries.length,archived:entries.filter(e=>e.archived).length,referencesOnly:entries.filter(e=>!e.archived).length}));
