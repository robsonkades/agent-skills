package work;

import java.sql.Connection;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import org.springframework.transaction.support.TransactionTemplate;

public class Regression {
    record Observation(String id, int localBooking, int independentBooking, int headerBeforeWrite) {}

    static class Probe implements ReceiptObserver {
        final JdbcTemplate jdbc;
        final DataSource ds;
        final List<Observation> observations = new ArrayList<>();
        Probe(JdbcTemplate jdbc, DataSource ds) { this.jdbc = jdbc; this.ds = ds; }

        public void beforeWrite(String id) {
            int local = jdbc.queryForObject("select count(*) from booking where id=?", Integer.class, id);
            int header = jdbc.queryForObject("select count(*) from receipt where id=?", Integer.class, id);
            observations.add(new Observation(id, local, independentCount(ds, id), header));
        }
    }

    @Configuration
    static class ProbeConfig {
        @Bean @Primary
        Probe probe(JdbcTemplate jdbc, DataSource ds) { return new Probe(jdbc, ds); }
    }

    static class OwnerFailure extends RuntimeException {}
    static int passed;

    public static void main(String[] args) {
        try (var ctx = new AnnotationConfigApplicationContext(Config.class, ProbeConfig.class)) {
            var jdbc = ctx.getBean(JdbcTemplate.class);
            var service = ctx.getBean(BookingService.class);
            var probe = ctx.getBean(Probe.class);
            var owner = new TransactionTemplate(ctx.getBean("businessTx", PlatformTransactionManager.class));

            reset(jdbc, probe);
            service.place("normal", "ready", "abcdefghijkl");
            state(jdbc, 1, 1, 1, 19);
            equal("abcdefghijkl", jdbc.queryForObject("select body from receipt_detail", String.class), "exact 12-character body");
            observations(probe, 1);
            pass("normal: booking/header/body=1/1/1, stock=19; writer sees local booking=1, independent=0, header=0");

            reset(jdbc, probe);
            expect(DataIntegrityViolationException.class, () -> service.place("invalid", "ready", "abcdefghijklm"));
            state(jdbc, 0, 0, 0, 20);
            observations(probe, 1);
            pass("13-character rejection: booking/header/body=0/0/0, stock=20");

            reset(jdbc, probe);
            expect(IllegalStateException.class, () -> service.place("no-stock", "sold-out", "valid"));
            state(jdbc, 0, 0, 0, 20);
            equal(0, jdbc.queryForObject("select available from stock where sku='sold-out'", Integer.class), "sold-out stock");
            equal(0, probe.observations.size(), "no callback on stock failure");
            pass("stock failure: booking/header/body=0/0/0, sold-out stock=0, no callback");

            reset(jdbc, probe);
            ctx.publishEvent(new BookingRecorded("outside", "valid"));
            state(jdbc, 0, 0, 0, 20);
            equal(0, probe.observations.size(), "no nontransactional callback");
            pass("nontransactional publication: no persisted state and no callback");

            reset(jdbc, probe);
            owner.executeWithoutResult(status -> {
                service.place("joined", "ready", "valid");
                equal(1, count(jdbc, "booking"), "owner sees pending booking");
                equal(0, count(jdbc, "receipt"), "receipt deferred until owner completion");
                equal(0, independentCount(probe.ds, "joined"), "owner booking remains uncommitted");
            });
            state(jdbc, 1, 1, 1, 19);
            observations(probe, 1);
            pass("joined success: owner controls completion; all three rows commit together");

            reset(jdbc, probe);
            expect(OwnerFailure.class, () -> owner.executeWithoutResult(status -> {
                service.place("owner-failure", "ready", "valid");
                throw new OwnerFailure();
            }));
            state(jdbc, 0, 0, 0, 20);
            equal(0, probe.observations.size(), "owner failure before completion skips callback");
            pass("owner failure after place returns: booking/header/body=0/0/0, stock=20");

            reset(jdbc, probe);
            boolean[] receiptWrittenBeforeFailure = {false};
            expect(OwnerFailure.class, () -> owner.executeWithoutResult(status -> {
                service.place("late-failure", "ready", "valid");
                TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                    @Override public void beforeCommit(boolean readOnly) {
                        equal(1, count(jdbc, "receipt"), "receipt exists before late owner failure");
                        equal(1, count(jdbc, "receipt_detail"), "body exists before late owner failure");
                        receiptWrittenBeforeFailure[0] = true;
                        throw new OwnerFailure();
                    }
                });
            }));
            equal(true, receiptWrittenBeforeFailure[0], "late failure was reached after both receipt writes");
            state(jdbc, 0, 0, 0, 20);
            observations(probe, 1);
            pass("failure after receipt writes: booking/header/body=0/0/0, stock=20; no independent receipt survives");

            reset(jdbc, probe);
            expect(DataIntegrityViolationException.class, () -> owner.executeWithoutResult(status -> {
                service.place("first", "ready", "valid");
                service.place("second", "ready", "abcdefghijklm");
            }));
            state(jdbc, 0, 0, 0, 20);
            observations(probe, 2);
            pass("second receipt body failure: both bookings and both receipt units roll back, stock=20");

            System.out.println("PASS " + passed + " regression scenarios");
        }
    }

    static void reset(JdbcTemplate jdbc, Probe probe) {
        probe.observations.clear();
        jdbc.execute("drop all objects");
        jdbc.execute("create table booking(id varchar(80) primary key,sku varchar(80))");
        jdbc.execute("create table receipt(id varchar(80) primary key)");
        jdbc.execute("create table receipt_detail(id varchar(80) primary key,body varchar(12))");
        jdbc.execute("create table stock(sku varchar(80) primary key,available int)");
        jdbc.update("insert into stock values ('ready',20),('sold-out',0)");
    }

    static int independentCount(DataSource ds, String id) {
        try (var connection = ds.getConnection()) {
            connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
            connection.setAutoCommit(false);
            try (var query = connection.prepareStatement("select count(*) from booking where id=?")) {
                query.setQueryTimeout(2);
                query.setString(1, id);
                try (var result = query.executeQuery()) {
                    result.next();
                    int count = result.getInt(1);
                    connection.rollback();
                    return count;
                }
            }
        } catch (Exception failure) {
            throw new AssertionError("independent READ_COMMITTED observation failed", failure);
        }
    }

    static int count(JdbcTemplate jdbc, String table) {
        return jdbc.queryForObject("select count(*) from " + table, Integer.class);
    }

    static void state(JdbcTemplate jdbc, int booking, int header, int body, int available) {
        equal(booking, count(jdbc, "booking"), "committed bookings");
        equal(header, count(jdbc, "receipt"), "committed headers");
        equal(body, count(jdbc, "receipt_detail"), "committed bodies");
        equal(available, jdbc.queryForObject("select available from stock where sku='ready'", Integer.class), "committed stock");
    }

    static void observations(Probe probe, int expected) {
        equal(expected, probe.observations.size(), "writer observation count");
        for (var observation : probe.observations) {
            equal(1, observation.localBooking(), "writer sees publisher state: " + observation.id());
            equal(0, observation.independentBooking(), "publisher state is uncommitted: " + observation.id());
            equal(0, observation.headerBeforeWrite(), "observer runs before header insert: " + observation.id());
        }
    }

    static void expect(Class<? extends Throwable> type, Runnable action) {
        try { action.run(); }
        catch (Throwable failure) {
            if (type.isInstance(failure)) return;
            throw new AssertionError("expected " + type.getSimpleName() + " but got " + failure, failure);
        }
        throw new AssertionError("expected " + type.getSimpleName());
    }

    static void equal(Object expected, Object actual, String message) {
        if (!expected.equals(actual)) throw new AssertionError(message + ": expected=" + expected + ", actual=" + actual);
    }

    static void pass(String message) { passed++; System.out.println("PASS " + message); }
}
