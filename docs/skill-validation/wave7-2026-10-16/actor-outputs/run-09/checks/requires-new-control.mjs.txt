import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';

const root=process.cwd();
const mutant=path.resolve('../checks/requires-new');
fs.mkdirSync(mutant,{recursive:true});
for(const name of ['src','test','lib','check.mjs','dependencies.json','toolchain.json']) {
  fs.cpSync(path.join(root,name),path.join(mutant,name),{recursive:true});
}
const target=path.join(mutant,'src/work/ReceiptWriter.java');
const source=fs.readFileSync(target,'utf8');
const marker=' public void record(BookingRecorded event)';
if(!source.includes(marker)) throw new Error('Expected original writer method missing');
fs.writeFileSync(target,source.replace(marker,' @org.springframework.transaction.annotation.Transactional(transactionManager="businessTx", propagation=org.springframework.transaction.annotation.Propagation.REQUIRES_NEW)\n'+marker));
const result=spawnSync(process.execPath,['check.mjs','work.Regression'],{cwd:mutant,encoding:'utf8',timeout:30000,windowsHide:true});
fs.writeFileSync(path.resolve('../checks/requires-new.stdout.log'),result.stdout??'');
fs.writeFileSync(path.resolve('../checks/requires-new.stderr.log'),result.stderr??'');
console.log('REQUIRES_NEW candidate exit='+result.status);
process.stdout.write(result.stdout??'');
process.stderr.write(result.stderr??'');
if(result.error) throw result.error;
if(result.status===0) throw new Error('Regression did not reject REQUIRES_NEW');
if(!(result.stderr??'').includes('writer sees publisher state: normal: expected=1, actual=0')) throw new Error('Unexpected candidate failure');
console.log('PASS regression rejects independent writer transaction by persisted-state visibility');
