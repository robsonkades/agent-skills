import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';

const input = path.resolve('../../input/project');
const output = process.cwd();
const frozen = new Set(['CONTRACT.md', 'REQUEST.md', 'README.md', 'check.mjs', 'dependencies.json', 'toolchain.json']);
const walk = root => fs.readdirSync(root, { withFileTypes: true }).flatMap(entry => {
    const full = path.join(root, entry.name);
    return entry.isDirectory() ? walk(full) : [full];
});
const inputs = walk(input);
const checked = inputs.filter(file => {
    const rel = path.relative(input, file).replaceAll('\\', '/');
    return frozen.has(rel) || rel.startsWith('lib/') || rel.startsWith('support/');
}).map(file => {
    const relative = path.relative(input, file);
    const digest = value => crypto.createHash('sha256').update(fs.readFileSync(value)).digest('hex');
    return { path: relative.replaceAll('\\', '/'), equal: digest(file) === digest(path.join(output, relative)) };
});
console.log(JSON.stringify({ frozenFilesChecked: checked.length, checked, inputFiles: inputs.map(file => path.relative(input, file).replaceAll('\\', '/')) }, null, 2));
if (checked.some(file => !file.equal)) process.exitCode = 1;
