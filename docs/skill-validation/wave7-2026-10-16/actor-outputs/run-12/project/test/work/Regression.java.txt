package work;

import java.sql.Connection;
import java.util.HashMap;
import java.util.Map;
import javax.sql.DataSource;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;

public class Regression {
 private static int failures;
 static final class Observation {
  final int visible; final Thread thread;
  Observation(int visible){this.visible=visible;this.thread=Thread.currentThread();}
 }
 public static class Probe implements ReceiptObserver {
  private final DataSource ds;
  final Map<String,Observation> observations=new HashMap<>();
  Probe(DataSource ds){this.ds=ds;}
  public void beforeWrite(String id){observations.put(id,new Observation(count(ds,"booking",id)));}
 }
 @Configuration
 public static class Diagnostics {
  @Bean @Primary public Probe diagnosticObserver(DataSource ds){return new Probe(ds);}
 }
 private static int count(DataSource ds,String table,String id){
  try(var c=ds.getConnection()){
   c.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
   c.setAutoCommit(false);
   try(var q=c.prepareStatement("select count(*) from "+table+" where id=?")){
    q.setQueryTimeout(1);q.setString(1,id);
    try(var r=q.executeQuery()){r.next();int value=r.getInt(1);c.rollback();return value;}
   }
  }catch(Exception e){throw new IllegalStateException(e);}
 }
 private static void expect(boolean condition,String message){
  if(!condition){failures++;System.out.println("FAIL "+message);}
 }
 private static void state(DataSource ds,JdbcTemplate jdbc,String id,String sku,int booking,int header,int body,int stock){
  int b=count(ds,"booking",id),h=count(ds,"receipt",id),d=count(ds,"receipt_detail",id);
  int s=jdbc.queryForObject("select available from stock where sku=?",Integer.class,sku);
  System.out.println(id+": booking="+b+", header="+h+", body="+d+", stock="+s);
  expect(b==booking&&h==header&&d==body&&s==stock,id+" committed state");
 }
 private static void reset(JdbcTemplate jdbc,Probe probe){
  jdbc.execute("drop all objects");
  jdbc.execute("create table booking(id varchar(80) primary key,sku varchar(80))");
  jdbc.execute("create table receipt(id varchar(80) primary key)");
  jdbc.execute("create table receipt_detail(id varchar(80) primary key,body varchar(12))");
  jdbc.execute("create table stock(sku varchar(80) primary key,available int)");
  jdbc.update("insert into stock values ('ready',20),('sold-out',0)");
  probe.observations.clear();
 }
 private static RuntimeException place(BookingService service,String id,String sku,String text){
  try{service.place(id,sku,text);return null;}catch(RuntimeException e){
   System.out.println(id+": command threw "+e.getClass().getSimpleName());return e;
  }
 }
 private static void observed(Probe probe,String id){
  Observation observation=probe.observations.get(id);
  expect(observation!=null,id+" writer called");
  if(observation!=null){
   System.out.println(id+": booking visible inside writer="+observation.visible);
   expect(observation.visible==1,id+" source committed before actual write");
   expect(observation.thread==Thread.currentThread(),id+" synchronous receipt");
  }
 }
 public static void main(String[] args){
  try(var ctx=new AnnotationConfigApplicationContext(Config.class,Diagnostics.class)){
   var jdbc=ctx.getBean(JdbcTemplate.class);var ds=ctx.getBean(DataSource.class);
   var service=ctx.getBean(BookingService.class);var probe=ctx.getBean(Probe.class);
   var owner=new TransactionTemplate(ctx.getBean("businessTx",PlatformTransactionManager.class));

   reset(jdbc,probe);
   expect(place(service,"valid","ready","123456789012")==null,"valid command succeeds");
   state(ds,jdbc,"valid","ready",1,1,1,19);observed(probe,"valid");
   expect("123456789012".equals(jdbc.queryForObject("select body from receipt_detail where id='valid'",String.class)),"valid body preserved");

   reset(jdbc,probe);
   expect(place(service,"invalid","ready","1234567890123")==null,"invalid receipt does not fail booking");
   state(ds,jdbc,"invalid","ready",1,0,0,19);observed(probe,"invalid");

   reset(jdbc,probe);
   expect(place(service,"unavailable","sold-out","normal") instanceof IllegalStateException,"unavailable stock fails");
   state(ds,jdbc,"unavailable","sold-out",0,0,0,0);
   expect(probe.observations.isEmpty(),"rollback never invokes writer");

   reset(jdbc,probe);
   ctx.publishEvent(new BookingRecorded("outside","normal"));
   state(ds,jdbc,"outside","ready",0,0,0,20);
   expect(probe.observations.isEmpty(),"nontransactional event ignored");

   reset(jdbc,probe);
   try{
    owner.executeWithoutResult(status->{
     service.place("owner-rollback","ready","normal");
     expect(count(ds,"booking","owner-rollback")==0,"service joins owner's transaction");
     expect(count(ds,"receipt","owner-rollback")==0,"receipt waits for owner completion");
     expect(probe.observations.isEmpty(),"no callback before owner rollback");
     throw new IllegalArgumentException("owner aborts");
    });
    expect(false,"owner failure escaped");
   }catch(IllegalArgumentException expected){expect("owner aborts".equals(expected.getMessage()),"expected owner failure");}
   state(ds,jdbc,"owner-rollback","ready",0,0,0,20);
   expect(probe.observations.isEmpty(),"no receipt survives owner rollback");

   reset(jdbc,probe);
   owner.executeWithoutResult(status->{
    service.place("owner-commit","ready","normal");
    expect(count(ds,"booking","owner-commit")==0,"owner retains mandatory commit");
    expect(probe.observations.isEmpty(),"receipt deferred until owner commit");
   });
   state(ds,jdbc,"owner-commit","ready",1,1,1,19);observed(probe,"owner-commit");

   reset(jdbc,probe);
   try{owner.executeWithoutResult(status->service.place("owner-invalid","ready","1234567890123"));}
   catch(RuntimeException e){expect(false,"invalid receipt must not fail owner completion: "+e.getClass().getSimpleName());}
   state(ds,jdbc,"owner-invalid","ready",1,0,0,19);observed(probe,"owner-invalid");
  }
  if(failures!=0)throw new AssertionError(failures+" contract checks failed");
  System.out.println("PASS 7 receipt contract scenarios");
 }
}
