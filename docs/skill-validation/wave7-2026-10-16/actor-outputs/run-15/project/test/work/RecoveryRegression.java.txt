package work;

import java.nio.file.Files;
import java.nio.file.Path;
import java.sql.DriverManager;
import java.util.concurrent.TimeUnit;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.jdbc.core.JdbcTemplate;

public class RecoveryRegression {
    public static void main(String[] args) throws Exception {
        if (args.length > 0) {
            System.setProperty("work.businessUrl", "jdbc:h2:file:" + args[1] + "/business");
            System.setProperty("work.receiverUrl", "jdbc:h2:file:" + args[1] + "/receiver");
            if (args[0].equals("fail")) failAfterAcknowledgment();
            else if (args[0].equals("replay")) replayRetainedWork();
            else throw new IllegalArgumentException("Unknown phase");
            return;
        }
        Path database = Files.createTempDirectory(Path.of(".build").toAbsolutePath(), "recovery-");
        child("fail", database);
        child("replay", database);
        controls();
        System.out.println("PASS restart recovery and transaction controls");
    }

    private static void child(String phase, Path database) throws Exception {
        var process = new ProcessBuilder(
            Path.of(System.getProperty("java.home"), "bin", "java").toString(),
            "-cp", System.getProperty("java.class.path"), RecoveryRegression.class.getName(),
            phase, database.toString()).inheritIO().start();
        if (!process.waitFor(25, TimeUnit.SECONDS)) {
            process.destroyForcibly();
            process.waitFor();
            throw new AssertionError("Child timed out: " + phase);
        }
        require(process.exitValue() == 0, "Child failed: " + phase);
    }

    private static AnnotationConfigApplicationContext context(Checkpoint checkpoint) {
        var context = new AnnotationConfigApplicationContext();
        context.register(Config.class);
        context.addBeanFactoryPostProcessor(factory -> factory.registerSingleton("checkpoint", checkpoint));
        context.refresh();
        return context;
    }

    private static void schema(JdbcTemplate jdbc) {
        jdbc.execute("drop all objects");
        jdbc.execute("create table orders(id varchar primary key,payload varchar)");
        jdbc.execute("create table outbox(event_id varchar primary key,payload varchar,done boolean)");
    }

    private static void failAfterAcknowledgment() throws Exception {
        RuntimeException checkpointFailure = new IllegalStateException("checkpoint failed after acknowledgment");
        try (var context = context(id -> {
            require(id.equals("event:restart"), "Checkpoint event identity");
            require(receiverCount() == 1, "Receiver commit precedes checkpoint");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=false") == 1,
                "Progress must not precede checkpoint");
            throw checkpointFailure;
        })) {
            var jdbc = context.getBean(JdbcTemplate.class);
            schema(jdbc);
            context.getBean(Publisher.class).create("restart", "immutable fact");
            try {
                context.getBean(Dispatcher.class).replayOne();
                throw new AssertionError("Checkpoint failure must propagate");
            } catch (RuntimeException actual) {
                require(actual == checkpointFailure, "Original checkpoint exception propagates");
            }
            require(scalar("work.businessUrl", "select count(*) from orders") == 1, "Committed order retained");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=false and event_id='event:restart' and payload='immutable fact'") == 1,
                "Original pending work retained after failure");
            require(receiverCount() == 1, "Acknowledged effect survives local rollback");
            System.out.println("PASS failure window: receiver effect=1, pending=1, order=1");
        }
    }

    private static void replayRetainedWork() throws Exception {
        int[] checkpointCalls = {0};
        try (var context = context(id -> {
            require(id.equals("event:restart"), "Replay checkpoint identity");
            checkpointCalls[0]++;
        })) {
            require(scalar("work.businessUrl", "select count(*) from outbox where done=false") == 1,
                "New JVM opens retained work");
            require(receiverCount() == 1, "New JVM opens receiver acceptance");
            var dispatcher = context.getBean(Dispatcher.class);
            require(dispatcher.replayOne(), "Retained event replayed");
            require(checkpointCalls[0] == 1, "Checkpoint still called on replay");
            require(receiverCount() == 1, "Replay must preserve one receiver effect");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=true and event_id='event:restart' and payload='immutable fact'") == 1,
                "Progress committed against original row");
            require(scalar("work.businessUrl", "select count(*) from orders") == 1, "No publisher rerun");
            require(!dispatcher.replayOne(), "Completed work is not replayed");
            require(checkpointCalls[0] == 1, "Empty replay skips checkpoint");
            System.out.println("PASS new JVM replay: receiver effect=1, pending=0, order=1");
        }
    }

    private static void controls() throws Exception {
        System.setProperty("work.businessUrl", "jdbc:h2:mem:recoveryControls;DB_CLOSE_DELAY=-1");
        System.setProperty("work.receiverUrl", "jdbc:h2:mem:receiverControls;DB_CLOSE_DELAY=-1");
        int[] checkpointCalls = {0};
        try (var context = context(id -> checkpointCalls[0]++)) {
            var jdbc = context.getBean(JdbcTemplate.class);
            var publisher = context.getBean(Publisher.class);
            var dispatcher = context.getBean(Dispatcher.class);
            var receiver = context.getBean(ReceiverClient.class);
            schema(jdbc);
            jdbc.update("insert into outbox values ('event:collision','original',false)");
            expectRuntime(() -> publisher.create("collision", "replacement"));
            require(scalar("work.businessUrl", "select count(*) from orders") == 0, "Outbox failure rolls back order");
            require(scalar("work.businessUrl", "select count(*) from outbox where payload='original'") == 1, "Existing outbox preserved");

            schema(jdbc);
            publisher.create("reject", "reject");
            expectRuntime(() -> dispatcher.replayOne());
            expectRuntime(() -> dispatcher.replayOne());
            require(receiver.count() == 0, "Receiver rejection has no effect");
            require(checkpointCalls[0] == 0, "Rejection skips checkpoint");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=false") == 1, "Rejected work remains pending");

            schema(jdbc);
            receiver.accept("event:mismatch", "original accepted fact");
            publisher.create("mismatch", "different fact");
            expectRuntime(() -> dispatcher.replayOne());
            require(receiver.count() == 1, "Identity conflict creates no additional effect");
            require(checkpointCalls[0] == 0, "Identity conflict skips checkpoint");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=false") == 1, "Identity conflict stays pending");

            schema(jdbc);
            publisher.create("a", "same payload");
            publisher.create("b", "same payload");
            require(dispatcher.replayOne(), "First distinct event delivered");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=true") == 1, "One row per replay");
            require(scalar("work.businessUrl", "select count(*) from outbox where done=false") == 1, "Second row still pending");
            require(dispatcher.replayOne(), "Second distinct event delivered");
            require(receiver.count() == 3, "Equal payloads with distinct identities both have effects");
            require(checkpointCalls[0] == 2, "Checkpoint for both accepted events");
            require(!dispatcher.replayOne(), "Queue drained");
            System.out.println("PASS controls: atomicity, rejection, identity conflict, distinct events, one-row replay");
        }
    }

    private static int receiverCount() {
        return scalar("work.receiverUrl", "select count(*) from accepted");
    }

    private static int scalar(String property, String sql) {
        try (var connection = DriverManager.getConnection(System.getProperty(property), "sa", "");
             var statement = connection.createStatement(); var result = statement.executeQuery(sql)) {
            require(result.next(), "Expected persisted result");
            return result.getInt(1);
        } catch (Exception failure) {
            throw new AssertionError("Cannot inspect committed state", failure);
        }
    }

    private static void expectRuntime(Runnable action) {
        try { action.run(); }
        catch (RuntimeException expected) { return; }
        throw new AssertionError("Expected failure to propagate");
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
}
