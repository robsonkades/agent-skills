import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const cwd=process.cwd(),tc=JSON.parse(fs.readFileSync('toolchain.json','utf8'));
const walk=d=>fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(d,e.name)):[path.join(d,e.name)]);
const cp=fs.readdirSync('lib').filter(f=>f.endsWith('.jar')).map(f=>path.resolve('lib',f)).join(path.delimiter);
fs.mkdirSync('.build',{recursive:true});
const commands=[[tc.javac,['--release',String(tc.release),'-cp',cp,'-d','.build',...['src','test','support'].flatMap(d=>fs.existsSync(d)?walk(d).filter(p=>p.endsWith('.java')):[])]],[tc.java,['-cp',path.resolve('.build')+path.delimiter+cp,process.argv[2]??'work.Smoke',...process.argv.slice(3)]]];
for(const [exe,args] of commands){const r=spawnSync(exe,args,{cwd,encoding:'utf8',timeout:30000,windowsHide:true});process.stdout.write(r.stdout??'');process.stderr.write(r.stderr??'');if(r.error)process.stderr.write(r.error.message+'\n');if(r.status!==0){process.exitCode=r.status??1;break;}}
