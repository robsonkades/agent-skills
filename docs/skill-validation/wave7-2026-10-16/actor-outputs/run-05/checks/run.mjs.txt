import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';

const [label, ...args] = process.argv.slice(2);
const startedAt = new Date().toISOString();
const result = spawnSync(process.execPath, args, {
  cwd: process.cwd(), encoding: 'utf8', timeout: 30000, windowsHide: true,
});
const record = {
  label, cwd: process.cwd(), argv: ['node', ...args], startedAt,
  finishedAt: new Date().toISOString(), timeoutMs: 30000,
  exit: result.status, signal: result.signal,
  stdout: result.stdout ?? '', stderr: result.stderr ?? '',
  error: result.error?.message ?? null,
};
fs.writeFileSync(path.resolve('../checks', `${label}.json`), JSON.stringify(record, null, 2) + '\n');
process.stdout.write(record.stdout);
process.stderr.write(record.stderr);
if (record.error) process.stderr.write(record.error + '\n');
process.exitCode = result.status ?? 1;
