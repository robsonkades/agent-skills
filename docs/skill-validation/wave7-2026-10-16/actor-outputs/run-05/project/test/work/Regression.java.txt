package work;

import java.util.ArrayList;
import java.util.List;
import java.util.function.Consumer;
import org.springframework.aop.support.AopUtils;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.jdbc.core.ConnectionCallback;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionSynchronizationManager;

public class Regression {
 private static final List<String> failures = new ArrayList<>();
 private static int passed;

 public static void main(String[] args) {
  var notifier = new RecordingNotifier();
  try (var ctx = new AnnotationConfigApplicationContext()) {
   ctx.registerBean("regressionNotifier", OrderNotifier.class, () -> notifier,
     definition -> definition.setPrimary(true));
   ctx.register(Config.class);
   ctx.refresh();
   var orders = ctx.getBean("ordersJdbc", JdbcTemplate.class);
   var metrics = ctx.getBean("metricsJdbc", JdbcTemplate.class);
   var service = ctx.getBean(OrderService.class);
   var metricsService = ctx.getBean(MetricsService.class);

   run("configured proxies and unchanged default", () -> {
    require(AopUtils.isAopProxy(service), "OrderService must be called through its proxy");
    require(AopUtils.isAopProxy(metricsService), "MetricsService must be called through its proxy");
    require(ctx.getBean(PlatformTransactionManager.class) == ctx.getBean("metricsTx"),
      "metricsTx must remain the primary manager");
   });

   run("successful order reserves exactly one unit", () -> {
    reset(orders, metrics, notifier);
    service.place("success", "available");
    outsideTransaction();
    equal(1, count(orders, "orders"), "committed orders");
    equal(2, stock(orders), "remaining stock");
    require(notifier.calls.equals(List.of("success")), "notifier must receive the order once");
   });

   run("unavailable stock rolls back the inserted order", () -> {
    reset(orders, metrics, notifier);
    var failure = expect(IllegalStateException.class, () -> service.place("sold", "sold-out"));
    require("Unavailable stock".equals(failure.getMessage()), "stock rejection cause is preserved");
    outsideTransaction();
    equal(0, count(orders, "orders"), "orders after stock rejection");
    equal(0, orders.queryForObject("select available from stock where sku='sold-out'", Integer.class),
      "sold-out stock");
    equal(3, stock(orders), "other stock");
    require(notifier.calls.isEmpty(), "notifier must not run after a stock rejection");
   });

   run("notifier rejection rolls back both writes and preserves exception", () -> {
    reset(orders, metrics, notifier);
    var rejection = new IllegalArgumentException("local validation rejected order");
    notifier.behavior = id -> {
     System.out.println("notifier diagnostic: active="
       + TransactionSynchronizationManager.isActualTransactionActive()
       + " ordersAutoCommit=" + autoCommit(orders)
       + " metricsAutoCommit=" + autoCommit(metrics));
     equal(1, count(orders, "orders"), "order visible inside notifier");
     equal(2, stock(orders), "stock reserved before notifier");
     throw rejection;
    };
    require(expect(IllegalArgumentException.class, () -> service.place("rejected", "available")) == rejection,
      "caller must receive the original collaborator exception");
    outsideTransaction();
    System.out.println("after notifier rejection: orders=" + count(orders, "orders") + " stock=" + stock(orders));
    require(count(orders, "orders") == 0 && stock(orders) == 3,
      "rejected order and stock decrement must both roll back");
    require(notifier.calls.equals(List.of("rejected")), "notifier must be called once");
   });

   run("insert failure preserves existing data and skips later steps", () -> {
    reset(orders, metrics, notifier);
    orders.update("insert into orders values ('existing','available')");
    var failure = expect(DuplicateKeyException.class, () -> service.place("existing", "available"));
    require(failure.getCause() != null, "JDBC failure retains its meaningful cause");
    outsideTransaction();
    equal(1, count(orders, "orders"), "pre-existing order");
    equal(3, stock(orders), "stock unchanged after insert failure");
    require(notifier.calls.isEmpty(), "notifier must not run after insert failure");
   });

   run("metrics success and rollback preserve committed orders", () -> {
    reset(orders, metrics, notifier);
    service.place("kept", "available");
    metricsService.sample("kept-metric");
    var failure = expect(IllegalArgumentException.class, () -> metricsService.sample(""));
    require("blank sample".equals(failure.getMessage()), "metric rejection cause is preserved");
    outsideTransaction();
    equal(1, count(metrics, "sample"), "successful metric retained and failed sample rolled back");
    equal(1, count(orders, "orders"), "committed order retained after metrics failure");
    equal(2, stock(orders), "committed stock retained after metrics failure");
   });

   run("caught metrics failure does not poison an orders transaction", () -> {
    reset(orders, metrics, notifier);
    notifier.behavior = id -> expect(IllegalArgumentException.class, () -> metricsService.sample(""));
    service.place("independent", "available");
    outsideTransaction();
    equal(0, count(metrics, "sample"), "failed metrics sample rolled back");
    equal(1, count(orders, "orders"), "independent order committed");
    equal(2, stock(orders), "independent reservation committed");
   });
  }
  System.out.println("Regression: " + passed + " passed, " + failures.size() + " failed");
  if (!failures.isEmpty()) throw new AssertionError(String.join("; ", failures));
 }

 private static void reset(JdbcTemplate orders, JdbcTemplate metrics, RecordingNotifier notifier) {
  outsideTransaction();
  orders.execute("drop all objects");
  orders.execute("create table orders(id varchar primary key, sku varchar)");
  orders.execute("create table stock(sku varchar primary key, available int)");
  orders.update("insert into stock values ('available',3),('sold-out',0)");
  metrics.execute("drop all objects");
  metrics.execute("create table sample(id varchar primary key)");
  notifier.calls.clear();
  notifier.behavior = id -> {};
 }

 private static boolean autoCommit(JdbcTemplate jdbc) {
  return jdbc.execute((ConnectionCallback<Boolean>) connection -> connection.getAutoCommit());
 }

 private static int count(JdbcTemplate jdbc, String table) {
  return jdbc.queryForObject("select count(*) from " + table, Integer.class);
 }

 private static int stock(JdbcTemplate jdbc) {
  return jdbc.queryForObject("select available from stock where sku='available'", Integer.class);
 }

 private static void outsideTransaction() {
  require(!TransactionSynchronizationManager.isActualTransactionActive(), "verification must run after transaction completion");
 }

 private static <T extends Throwable> T expect(Class<T> type, Runnable operation) {
  try { operation.run(); }
  catch (Throwable failure) {
   if (type.isInstance(failure)) return type.cast(failure);
   throw new AssertionError("Expected " + type.getName() + ", got " + failure, failure);
  }
  throw new AssertionError("Expected " + type.getName());
 }

 private static void equal(int expected, int actual, String label) {
  require(expected == actual, label + ": expected " + expected + ", got " + actual);
 }

 private static void require(boolean condition, String message) {
  if (!condition) throw new AssertionError(message);
 }

 private static void run(String name, Runnable test) {
  try { test.run(); passed++; System.out.println("PASS " + name); }
  catch (Throwable failure) {
   failures.add(name + ": " + failure);
   System.out.println("FAIL " + name + ": " + failure);
  }
 }

 private static final class RecordingNotifier implements OrderNotifier {
  private final List<String> calls = new ArrayList<>();
  private Consumer<String> behavior = id -> {};
  @Override public void accepted(String id) { calls.add(id); behavior.accept(id); }
 }
}
