import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
import {fileURLToPath} from 'node:url';

const checks = path.dirname(fileURLToPath(import.meta.url));
const project = path.resolve(checks, '../project');
const target = path.join(checks, 'requires-new');
const source = path.join(target, 'src/work/ReceiptWriter.java');
const classes = path.join(target, '.build');
fs.mkdirSync(path.dirname(source), {recursive: true});
fs.mkdirSync(classes, {recursive: true});
const original = fs.readFileSync(path.join(project, 'src/work/ReceiptWriter.java'), 'utf8');
const mutant = original.replace(' public void record(',
    ' @org.springframework.transaction.annotation.Transactional(value="businessTx", propagation=org.springframework.transaction.annotation.Propagation.REQUIRES_NEW)\n public void record(');
if (mutant === original) throw new Error('Mutation was not applied');
fs.writeFileSync(source, mutant);
const tc = JSON.parse(fs.readFileSync(path.join(project, 'toolchain.json'), 'utf8'));
const dependencies = fs.readdirSync(path.join(project, 'lib')).filter(f => f.endsWith('.jar'))
    .map(f => path.join(project, 'lib', f));
const base = [path.join(project, '.build'), ...dependencies].join(path.delimiter);
const records = [];
function run(argv) {
    const result = spawnSync(argv[0], argv.slice(1), {cwd: project, encoding: 'utf8', timeout: 30000, windowsHide: true});
    records.push({cwd: project, argv, exit: result.status, stdout: result.stdout ?? '', stderr: result.stderr ?? '', error: result.error?.message});
    return result;
}
const compile = run([tc.javac, '--release', String(tc.release), '-cp', base, '-d', classes, source]);
let rejected = false;
if (compile.status === 0) {
    const regression = run([tc.java, '-cp', classes + path.delimiter + base, 'work.ReceiptContractRegression']);
    rejected = regression.status !== 0 && regression.stderr.includes('writer sees uncommitted publisher booking: expected=1, actual=0');
}
fs.writeFileSync(path.join(checks, 'requires-new-control.json'), JSON.stringify({records, rejected}, null, 2) + '\n');
if (!rejected) throw new Error('Regression did not reject REQUIRES_NEW as expected; inspect requires-new-control.json');
console.log('PASS negative control: separate REQUIRES_NEW writer rejected because it sees booking=0 instead of uncommitted booking=1');
