import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const [name, ...args] = process.argv.slice(2);
const startedAt = new Date().toISOString();
const result = spawnSync(process.execPath, ['check.mjs', ...args], {
  cwd: process.cwd(), encoding: 'utf8', timeout: 30000, windowsHide: true,
});
const record = {
  cwd: process.cwd(), argv: ['node', 'check.mjs', ...args], startedAt,
  finishedAt: new Date().toISOString(), exit: result.status,
  stdout: result.stdout ?? '', stderr: result.stderr ?? '', error: result.error?.message ?? null,
};
fs.writeFileSync(path.resolve('../checks', name + '.json'), JSON.stringify(record, null, 2) + '\n');
process.stdout.write(record.stdout);
process.stderr.write(record.stderr);
if (record.error) process.stderr.write(record.error + '\n');
process.exitCode = result.status ?? 1;
