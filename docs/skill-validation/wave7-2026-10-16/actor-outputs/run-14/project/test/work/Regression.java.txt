package work;

import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;
import org.springframework.aop.support.AopUtils;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.context.annotation.*;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.datasource.ConnectionHolder;
import org.springframework.transaction.support.TransactionSynchronizationManager;

public class Regression {
 private static final List<String> failures = new ArrayList<>();

 @Configuration
 static class CollaboratorConfig {
  @Bean @Primary
  ProbeNotifier probeNotifier(@Qualifier("ordersDataSource") DataSource orders,
                              @Qualifier("metricsDataSource") DataSource metrics) {
   return new ProbeNotifier(orders, metrics);
  }
 }

 static class ProbeNotifier implements OrderNotifier {
  final DataSource orders, metrics;
  RuntimeException rejection;
  int calls;
  ProbeNotifier(DataSource orders, DataSource metrics) {this.orders=orders;this.metrics=metrics;}
  public void accepted(String id) {
   calls++;
   System.out.println("notifier: active="+TransactionSynchronizationManager.isActualTransactionActive()
       +" ordersAutoCommit="+autoCommit(orders)+" metricsAutoCommit="+autoCommit(metrics));
   if(rejection!=null) throw rejection;
  }
  private String autoCommit(DataSource ds) {
   Object resource=TransactionSynchronizationManager.getResource(ds);
   if(!(resource instanceof ConnectionHolder holder)) return "unbound";
   try {return Boolean.toString(holder.getConnection().getAutoCommit());}
   catch(SQLException e) {throw new IllegalStateException(e);}
  }
 }

 public static void main(String[] args) {
  try(var ctx=new AnnotationConfigApplicationContext(Config.class, CollaboratorConfig.class)) {
   var orders=ctx.getBean("ordersJdbc",JdbcTemplate.class);
   var metrics=ctx.getBean("metricsJdbc",JdbcTemplate.class);
   var service=ctx.getBean(OrderService.class);
   var metricsService=ctx.getBean(MetricsService.class);
   var notifier=ctx.getBean(ProbeNotifier.class);
   check(AopUtils.isAopProxy(service), "OrderService is invoked through a Spring proxy");
   check(AopUtils.isAopProxy(metricsService), "MetricsService is invoked through a Spring proxy");

   reset(orders,metrics,notifier);
   service.place("success","book");
   outsideCompletion();
   check(count(orders,"orders")==1,"success persists exactly one order");
   check(stock(orders,"book")==2,"success reserves exactly one unit");
   check(notifier.calls==1,"success calls notifier once");
   check(service.lastObservedActive(),"diagnostic remains the real framework probe");
   metricsService.sample("metric-success");
   outsideCompletion();
   check(count(metrics,"sample")==1,"successful metrics sample commits");
   check(count(orders,"orders")==1 && stock(orders,"book")==2,"metrics success preserves orders");

   reset(orders,metrics,notifier);
   var unavailable=expect(IllegalStateException.class,()->service.place("unavailable","sold-out"));
   outsideCompletion();
   check(unavailable!=null && "Unavailable stock".equals(unavailable.getMessage()),"unavailable cause survives");
   check(count(orders,"orders")==0,"unavailable stock rolls back the inserted order");
   check(stock(orders,"sold-out")==0 && stock(orders,"book")==3,"unavailable stock leaves stock unchanged");
   check(notifier.calls==0,"unavailable stock does not call notifier");

   reset(orders,metrics,notifier);
   metricsService.sample("keep-metric");
   var rootCause=new IllegalArgumentException("local validation rejected");
   notifier.rejection=new IllegalStateException("notifier rejected order",rootCause);
   var rejected=expect(IllegalStateException.class,()->service.place("rejected","book"));
   outsideCompletion();
   check(rejected==notifier.rejection && rejected.getCause()==rootCause,"notifier exception and cause survive unchanged");
   check(notifier.calls==1,"notifier failure occurs after database writes");
   check(count(orders,"orders")==0,"notifier failure rolls back order");
   check(stock(orders,"book")==3,"notifier failure restores stock");
   check(count(metrics,"sample")==1,"order rejection preserves committed metrics");

   reset(orders,metrics,notifier);
   orders.update("insert into orders values ('existing','book')");
   expect(DuplicateKeyException.class,()->service.place("existing","book"));
   outsideCompletion();
   check(count(orders,"orders")==1,"insert failure preserves only the existing order");
   check(stock(orders,"book")==3,"insert failure leaves stock unchanged");
   check(notifier.calls==0,"insert failure does not call notifier");

   reset(orders,metrics,notifier);
   service.place("keep-order","book");
   metricsService.sample("keep-metric");
   var metricsFailure=expect(IllegalArgumentException.class,()->metricsService.sample(""));
   outsideCompletion();
   check(metricsFailure!=null && "blank sample".equals(metricsFailure.getMessage()),"metrics rejection survives");
   check(count(metrics,"sample")==1,"failed metrics sample rolls back while preserving earlier sample");
   check(metrics.queryForObject("select count(*) from sample where id=''",Integer.class)==0,"failed metrics row is absent");
   check(count(orders,"orders")==1 && stock(orders,"book")==2,"metrics rollback preserves committed order and stock");
  }
  if(!failures.isEmpty()) throw new AssertionError(String.join("; ",failures));
  System.out.println("PASS: success, stock rejection, notifier rollback, insert rejection, metrics independence");
 }

 private static void reset(JdbcTemplate orders,JdbcTemplate metrics,ProbeNotifier notifier) {
  outsideCompletion();
  orders.execute("drop all objects");
  orders.execute("create table orders(id varchar primary key,sku varchar)");
  orders.execute("create table stock(sku varchar primary key,available int)");
  orders.update("insert into stock values ('book',3),('sold-out',0)");
  metrics.execute("drop all objects");
  metrics.execute("create table sample(id varchar primary key)");
  notifier.rejection=null;
  notifier.calls=0;
 }
 private static int count(JdbcTemplate jdbc,String table) {return jdbc.queryForObject("select count(*) from "+table,Integer.class);}
 private static int stock(JdbcTemplate jdbc,String sku) {return jdbc.queryForObject("select available from stock where sku=?",Integer.class,sku);}
 private static RuntimeException expect(Class<? extends RuntimeException> type,Runnable action) {
  try {action.run();check(false,"expected "+type.getSimpleName());return null;}
  catch(RuntimeException e) {check(type.isInstance(e),"expected "+type.getSimpleName()+", got "+e);return e;}
 }
 private static void outsideCompletion() {
  check(!TransactionSynchronizationManager.isActualTransactionActive(),"persisted state is inspected outside transaction completion");
  check(TransactionSynchronizationManager.getResourceMap().isEmpty(),"transaction resources are released");
 }
 private static void check(boolean condition,String message) {
  if(!condition) {failures.add(message);System.out.println("FAIL: "+message);}
 }
}
