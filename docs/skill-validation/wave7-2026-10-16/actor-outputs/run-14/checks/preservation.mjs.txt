import fs from 'node:fs';
import path from 'node:path';
import {createHash} from 'node:crypto';
import {fileURLToPath} from 'node:url';

const run = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '../..');
const input = path.join(run, 'input/project');
const output = path.join(run, 'output/project');
const changed = new Set(['src/work/OrderService.java', 'RUNBOOK.md']);
const files = dir => fs.readdirSync(dir, {withFileTypes:true}).flatMap(entry =>
  entry.isDirectory() ? files(path.join(dir,entry.name)) : [path.join(dir,entry.name)]);
const digest = file => createHash('sha256').update(fs.readFileSync(file)).digest('hex');
let preserved=0;
for (const file of files(input)) {
  const relative=path.relative(input,file).split(path.sep).join('/');
  const same=digest(file)===digest(path.join(output,relative));
  if (changed.has(relative)) {
    if (same) throw new Error(`Expected change missing: ${relative}`);
    console.log(`Expected modification: ${relative}`);
  } else {
    if (!same) throw new Error(`Unexpected modification: ${relative}`);
    preserved++;
  }
}
console.log(`PASS: ${preserved} input files remain byte-identical, including frozen project files and dependencies.`);
