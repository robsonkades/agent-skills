package work;

import javax.sql.DataSource;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import org.springframework.transaction.support.TransactionTemplate;

/** Test-only implementation of the teammate's proposal, never registered by production Config. */
public class RequiresNewCounterexample {
    public static class IndependentWriter extends ReceiptWriter {
        public IndependentWriter(JdbcTemplate jdbc, ReceiptObserver observer) { super(jdbc, observer); }

        @Override @Transactional(value = "businessTx", propagation = Propagation.REQUIRES_NEW)
        public void record(BookingRecorded event) { super.record(event); }
    }

    @Configuration
    static class ProposedChange {
        @Bean @Primary
        public ReceiptWriter independentWriter(JdbcTemplate jdbc, ReceiptObserver observer) {
            return new IndependentWriter(jdbc, observer);
        }
    }

    public static void main(String[] args) throws Exception {
        try (var ctx = new AnnotationConfigApplicationContext(Config.class,
                ReceiptContractRegression.Diagnostics.class, ProposedChange.class)) {
            var jdbc = ctx.getBean(JdbcTemplate.class);
            var ds = ctx.getBean(DataSource.class);
            var probe = ctx.getBean(ReceiptContractRegression.Probe.class);
            var service = ctx.getBean(BookingService.class);
            var owner = new TransactionTemplate(ctx.getBean("businessTx", PlatformTransactionManager.class));
            ReceiptContractRegression.reset(jdbc, probe);
            boolean[] receiptWritten = {false};
            ReceiptContractRegression.fails(ReceiptContractRegression.OwnerFailure.class,
                () -> owner.executeWithoutResult(status -> {
                    service.place("independent", "ready", "normal");
                    TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                        @Override public void beforeCommit(boolean readOnly) {
                            ReceiptContractRegression.equal(1,
                                ReceiptContractRegression.count(jdbc, "receipt_detail", "independent"),
                                "independent receipt already committed");
                            receiptWritten[0] = true;
                            throw new ReceiptContractRegression.OwnerFailure();
                        }
                    });
                }));
            ReceiptContractRegression.equal(true, receiptWritten[0], "failure after receipt writes");
            ReceiptContractRegression.state(ds, "independent", "ready", 0, 1, 1, 20,
                "REQUIRES_NEW counterexample: orphan receipt survives owner rollback");
            ReceiptContractRegression.equal(1, probe.calls.size(), "one proposed writer invocation");
            ReceiptContractRegression.equal(0, probe.calls.getFirst().booking(),
                "independent writer cannot observe uncommitted publisher booking");
            System.out.println("PASS proposal rejected: writer booking=0; header/detail remain after booking rollback");
        }
    }
}
