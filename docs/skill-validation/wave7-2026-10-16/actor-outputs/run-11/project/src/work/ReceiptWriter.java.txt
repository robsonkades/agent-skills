package work;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
public class ReceiptWriter {
 private final JdbcTemplate jdbc; private final ReceiptObserver observer;
 public ReceiptWriter(JdbcTemplate jdbc,ReceiptObserver observer){this.jdbc=jdbc;this.observer=observer;}
 // The source is already committed; header and body need their own atomic commit/rollback.
 @Transactional(transactionManager="businessTx",propagation=Propagation.REQUIRES_NEW)
 public void record(BookingRecorded event){observer.beforeWrite(event.id());jdbc.update("insert into receipt(id) values (?)",event.id());jdbc.update("insert into receipt_detail(id,body) values (?,?)",event.id(),event.receiptText());}
}
