package work;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import org.springframework.transaction.support.TransactionTemplate;

public class Regression {
    record State(int booking, int header, int detail, int stock) {}
    record Observation(int committedBooking, int committedStock, long thread) {}

    public static class Probe implements ReceiptObserver {
        private final DataSource dataSource;
        final List<Observation> observations = new ArrayList<>();
        Probe(DataSource dataSource) { this.dataSource = dataSource; }
        public void beforeWrite(String id) {
            // A raw connection is deliberately independent of Spring's bound writer connection.
            try (var connection = dataSource.getConnection()) {
                connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                try (var query = connection.prepareStatement("select count(*) from booking where id=?")) {
                    query.setString(1, id);
                    try (var rows = query.executeQuery()) {
                        rows.next();
                        observations.add(new Observation(rows.getInt(1),
                                scalar(connection, "select available from stock where sku='ready'"),
                                Thread.currentThread().threadId()));
                    }
                }
            } catch (SQLException failure) { throw new IllegalStateException(failure); }
        }
    }

    @Configuration
    static class ProbeConfig {
        @Bean @Primary Probe probe(DataSource dataSource) { return new Probe(dataSource); }
    }

    static class Fixture implements AutoCloseable {
        final AnnotationConfigApplicationContext context =
                new AnnotationConfigApplicationContext(Config.class, ProbeConfig.class);
        final JdbcTemplate jdbc = context.getBean(JdbcTemplate.class);
        final DataSource dataSource = context.getBean(DataSource.class);
        final BookingService service = context.getBean(BookingService.class);
        final Probe probe = context.getBean(Probe.class);
        final TransactionTemplate owner = new TransactionTemplate(
                context.getBean("businessTx", PlatformTransactionManager.class));
        Fixture() {
            jdbc.execute("drop all objects");
            jdbc.execute("create table booking(id varchar(80) primary key,sku varchar(80))");
            jdbc.execute("create table receipt(id varchar(80) primary key)");
            jdbc.execute("create table receipt_detail(id varchar(80) primary key,body varchar(12))");
            jdbc.execute("create table stock(sku varchar(80) primary key,available int)");
            jdbc.update("insert into stock values ('ready',20),('sold-out',0)");
        }
        State state() {
            try (var connection = dataSource.getConnection()) {
                connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                return new State(scalar(connection, "select count(*) from booking"),
                        scalar(connection, "select count(*) from receipt"),
                        scalar(connection, "select count(*) from receipt_detail"),
                        scalar(connection, "select available from stock where sku='ready'"));
            } catch (SQLException failure) { throw new IllegalStateException(failure); }
        }
        void observedCommit() {
            equal(List.of(new Observation(1, 19, Thread.currentThread().threadId())),
                    probe.observations, "writer must run synchronously after visible source commit");
        }
        public void close() { context.close(); }
    }

    static int scalar(Connection connection, String sql) throws SQLException {
        try (var statement = connection.createStatement(); var rows = statement.executeQuery(sql)) {
            rows.next();
            return rows.getInt(1);
        }
    }
    static void equal(Object expected, Object actual, String label) {
        if (!expected.equals(actual)) throw new AssertionError(label + ": expected=" + expected + " actual=" + actual);
    }
    static void print(String name, Fixture fixture, RuntimeException failure) {
        System.out.println(name + " state=" + fixture.state() + " observer=" + fixture.probe.observations
                + " callerFailure=" + (failure == null ? "none" : failure.getClass().getSimpleName()));
    }
    static void run(String name) {
        try (var fixture = new Fixture()) {
            switch (name) {
                case "valid" -> {
                    fixture.service.place("valid", "ready", "123456789012");
                    print(name, fixture, null);
                    equal(new State(1, 1, 1, 19), fixture.state(), "valid receipt state");
                    equal("123456789012", fixture.jdbc.queryForObject("select body from receipt_detail", String.class), "body unchanged");
                    fixture.observedCommit();
                }
                case "invalid" -> {
                    RuntimeException failure = null;
                    try { fixture.service.place("invalid", "ready", "1234567890123"); }
                    catch (RuntimeException problem) { failure = problem; }
                    print(name, fixture, failure);
                    equal(new State(1, 0, 0, 19), fixture.state(), "invalid body must roll back entire receipt only");
                    if (failure != null) throw new AssertionError("optional receipt failed business caller", failure);
                    fixture.observedCommit();
                }
                case "unavailable" -> {
                    RuntimeException failure = null;
                    try { fixture.service.place("unavailable", "sold-out", "normal"); }
                    catch (RuntimeException problem) { failure = problem; }
                    print(name, fixture, failure);
                    if (!(failure instanceof IllegalStateException)) throw new AssertionError("stock must reject command");
                    equal(new State(0, 0, 0, 20), fixture.state(), "unavailable stock leaves no writes");
                    equal(0, fixture.jdbc.queryForObject("select available from stock where sku='sold-out'", Integer.class), "sold-out stock unchanged");
                    equal(0, fixture.probe.observations.size(), "no writer on rollback");
                }
                case "nontransactional" -> {
                    fixture.context.publishEvent(new BookingRecorded("outside", "normal"));
                    print(name, fixture, null);
                    equal(new State(0, 0, 0, 20), fixture.state(), "nontransactional event discarded");
                    equal(0, fixture.probe.observations.size(), "no writer without source transaction");
                }
                case "ownerCommit", "ownerRollback", "lateOwnerFailure" -> {
                    boolean shouldFail = !name.equals("ownerCommit");
                    RuntimeException marker = new IllegalStateException("owner failure");
                    RuntimeException failure = null;
                    try {
                        fixture.owner.executeWithoutResult(status -> {
                            fixture.service.place(name, "ready", "normal");
                            equal(new State(0, 0, 0, 20), fixture.state(), "caller owns completion; source still uncommitted");
                            equal(0, fixture.probe.observations.size(), "writer waits for owner commit");
                            if (name.equals("ownerRollback")) throw marker;
                            if (name.equals("lateOwnerFailure")) {
                                // Registered after the event: a BEFORE_COMMIT independent receipt would escape.
                                TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                                    public void beforeCommit(boolean readOnly) { throw marker; }
                                });
                            }
                        });
                    } catch (RuntimeException problem) { failure = problem; }
                    print(name, fixture, failure);
                    if (shouldFail && failure != marker) throw new AssertionError("owner failure must propagate", failure);
                    if (!shouldFail && failure != null) throw new AssertionError("owner commit failed", failure);
                    equal(shouldFail ? new State(0, 0, 0, 20) : new State(1, 1, 1, 19), fixture.state(), "owner completion state");
                    if (shouldFail) equal(0, fixture.probe.observations.size(), "no receipt on owner rollback");
                    else fixture.observedCommit();
                }
                default -> throw new IllegalArgumentException(name);
            }
            System.out.println("PASS " + name);
        }
    }

    public static void main(String[] args) {
        String[] names = args.length == 0
                ? new String[]{"valid", "invalid", "unavailable", "nontransactional", "ownerCommit", "ownerRollback", "lateOwnerFailure"}
                : args;
        int failures = 0;
        for (String name : names) {
            try { run(name); }
            catch (Throwable failure) {
                failures++;
                System.err.println("FAIL " + name + ": " + failure);
            }
        }
        if (failures != 0) throw new AssertionError(failures + " regression case(s) failed");
        System.out.println("PASS all " + names.length + " regression cases");
    }
}
