package work;

import java.sql.Connection;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import org.springframework.transaction.support.TransactionTemplate;

public class Regression {
    private static final List<String> failures = new ArrayList<>();
    private record Observation(String id, int visibleBooking, int headerBeforeWrite,
                               Thread thread, String error) {}

    public static void main(String[] args) {
        var observations = new ArrayList<Observation>();
        try (var ctx = new AnnotationConfigApplicationContext()) {
            ctx.register(Config.class);
            ctx.registerBean("probe", ReceiptObserver.class, () -> id -> {
                int visible = -1;
                int header = -1;
                String error = null;
                try {
                    visible = independentCount(ctx.getBean(DataSource.class), "booking", id);
                    header = count(ctx.getBean(JdbcTemplate.class), "receipt", id);
                } catch (Exception ex) {
                    error = ex.toString();
                }
                observations.add(new Observation(id, visible, header, Thread.currentThread(), error));
            }, definition -> definition.setPrimary(true));
            ctx.refresh();
            var jdbc = ctx.getBean(JdbcTemplate.class);
            var source = ctx.getBean(DataSource.class);
            var service = ctx.getBean(BookingService.class);
            var owner = new TransactionTemplate(ctx.getBean("businessTx", PlatformTransactionManager.class));
            jdbc.execute("drop all objects");
            jdbc.execute("create table booking(id varchar(80) primary key,sku varchar(80))");
            jdbc.execute("create table receipt(id varchar(80) primary key)");
            jdbc.execute("create table receipt_detail(id varchar(80) primary key,body varchar(12))");
            jdbc.execute("create table stock(sku varchar(80) primary key,available int)");
            jdbc.update("insert into stock values ('ready',20),('sold-out',0)");

            var caller = Thread.currentThread();
            check(capture(() -> service.place("valid", "ready", "123456789012")) == null,
                    "valid command succeeds");
            state(jdbc, "valid", 1, 1, 1);
            check("123456789012".equals(jdbc.queryForObject(
                    "select body from receipt_detail where id='valid'", String.class)), "12-character body unchanged");
            check(stock(jdbc) == 19, "valid command reserves one item");

            var invalidFailure = capture(() -> service.place("invalid", "ready", "1234567890123"));
            System.out.println("invalid caller failure=" + invalidFailure);
            check(invalidFailure == null, "invalid optional receipt must not fail business command");
            state(jdbc, "invalid", 1, 0, 0);
            check(stock(jdbc) == 18, "invalid receipt retains mandatory reservation");

            int beforeSoldOut = observations.size();
            check(capture(() -> service.place("unavailable", "sold-out", "normal")) instanceof IllegalStateException,
                    "unavailable stock fails business command");
            state(jdbc, "unavailable", 0, 0, 0);
            check(observations.size() == beforeSoldOut, "rollback invokes no receipt writer");

            ctx.publishEvent(new BookingRecorded("outside", "normal"));
            state(jdbc, "outside", 0, 0, 0);
            check(observations.size() == beforeSoldOut, "nontransactional publication is skipped");

            var ownerFailure = new IllegalStateException("owner fails after service returns");
            int beforeOwner = observations.size();
            check(capture(() -> owner.executeWithoutResult(status -> {
                service.place("owner-rollback", "ready", "normal");
                try {
                    check(independentCount(source, "booking", "owner-rollback") == 0,
                            "REQUIRED service has not committed within owner transaction");
                } catch (Exception ex) { throw new AssertionError(ex); }
                check(observations.size() == beforeOwner, "receipt waits for completion owner");
                throw ownerFailure;
            })) == ownerFailure, "owner failure is preserved");
            state(jdbc, "owner-rollback", 0, 0, 0);
            check(observations.size() == beforeOwner, "owner rollback has no callback");
            check(stock(jdbc) == 18, "owner rollback restores stock");

            var lateFailure = new IllegalStateException("owner fails during completion");
            check(capture(() -> owner.executeWithoutResult(status -> {
                service.place("late-rollback", "ready", "normal");
                TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                    @Override public void beforeCommit(boolean readOnly) { throw lateFailure; }
                });
            })) == lateFailure, "late owner failure is preserved");
            state(jdbc, "late-rollback", 0, 0, 0);
            check(stock(jdbc) == 18, "completion failure restores stock");
            check(observations.size() == beforeOwner, "no receipt before failing owner commit");

            int beforeCommit = observations.size();
            check(capture(() -> owner.executeWithoutResult(status -> {
                service.place("owner-valid", "ready", "normal");
                service.place("owner-invalid", "ready", "1234567890123");
                check(observations.size() == beforeCommit, "all receipts wait for owner commit");
            })) == null, "invalid receipt does not fail owner completion");
            state(jdbc, "owner-valid", 1, 1, 1);
            state(jdbc, "owner-invalid", 1, 0, 0);
            check(stock(jdbc) == 16, "successful owner commits two reservations");

            check(observations.size() == 4, "writer invoked exactly for four committed bookings");
            for (var observation : observations) {
                System.out.println("observer=" + observation);
                check(observation.error() == null, "observer query succeeds for " + observation.id());
                check(observation.visibleBooking() == 1, "source committed before writer for " + observation.id());
                check(observation.headerBeforeWrite() == 0, "observer remains immediately before first write");
                check(observation.thread() == caller, "receipt keeps synchronous scheduling");
            }
            System.out.println("final ready stock=" + stock(jdbc));
        }
        if (!failures.isEmpty()) throw new AssertionError(String.join("; ", failures));
        System.out.println("PASS post-commit receipt, atomic rejection, REQUIRED ownership and rollback controls");
    }

    private static RuntimeException capture(Runnable action) {
        try { action.run(); return null; } catch (RuntimeException ex) { return ex; }
    }

    private static void check(boolean satisfied, String description) {
        if (!satisfied) failures.add(description);
    }

    private static int count(JdbcTemplate jdbc, String table, String id) {
        return jdbc.queryForObject("select count(*) from " + table + " where id=?", Integer.class, id);
    }

    private static int independentCount(DataSource source, String table, String id) throws Exception {
        try (var connection = source.getConnection()) {
            connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
            connection.setAutoCommit(false);
            try (var statement = connection.prepareStatement("select count(*) from " + table + " where id=?")) {
                statement.setQueryTimeout(2);
                statement.setString(1, id);
                try (var result = statement.executeQuery()) {
                    result.next();
                    int count = result.getInt(1);
                    connection.rollback();
                    return count;
                }
            }
        }
    }

    private static int stock(JdbcTemplate jdbc) {
        return jdbc.queryForObject("select available from stock where sku='ready'", Integer.class);
    }

    private static void state(JdbcTemplate jdbc, String id, int booking, int receipt, int detail) {
        int actualBooking = count(jdbc, "booking", id);
        int actualReceipt = count(jdbc, "receipt", id);
        int actualDetail = count(jdbc, "receipt_detail", id);
        System.out.println(id + ": booking=" + actualBooking + ", receipt=" + actualReceipt + ", detail=" + actualDetail);
        check(actualBooking == booking && actualReceipt == receipt && actualDetail == detail,
                "persisted state for " + id);
    }
}
