import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const [label,...argv]=process.argv.slice(2);
const startedAt=new Date().toISOString();
const result=spawnSync(argv[0],argv.slice(1),{cwd:process.cwd(),encoding:'utf8',timeout:30000,windowsHide:true});
const log={cwd:process.cwd(),argv,startedAt,finishedAt:new Date().toISOString(),exit:result.status,signal:result.signal,error:result.error?.message??null,stdout:result.stdout??'',stderr:result.stderr??''};
fs.writeFileSync(path.join(import.meta.dirname,label+'.json'),JSON.stringify(log,null,2)+'\n');
process.stdout.write(log.stdout);
process.stderr.write(log.stderr);
process.exitCode=result.status??1;
