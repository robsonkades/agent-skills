import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
const input=path.resolve(import.meta.dirname,'../../input/project');
const output=path.resolve(import.meta.dirname,'../project');
const walk=dir=>fs.readdirSync(dir,{withFileTypes:true}).flatMap(entry=>entry.isDirectory()?walk(path.join(dir,entry.name)):[path.join(dir,entry.name)]);
const mutable=new Set(['src/work/BookingListener.java','src/work/ReceiptWriter.java','RUNBOOK.md']);
const hash=file=>crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
const files=walk(input).map(file=>{
 const relative=path.relative(input,file).replaceAll('\\','/');
 const inputHash=hash(file),outputHash=hash(path.join(output,relative));
 return {relative,inputHash,outputHash,equal:inputHash===outputHash,allowedChange:mutable.has(relative)};
});
const unexpected=files.filter(file=>!file.equal&&!file.allowedChange);
const result={files,unexpected,allFrozenOutputMatchesInput:unexpected.length===0};
fs.writeFileSync(path.join(import.meta.dirname,'preserved-files.json'),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify({compared:files.length,changed:files.filter(file=>!file.equal).map(file=>file.relative),unexpected},null,2));
if(unexpected.length)process.exitCode=1;
