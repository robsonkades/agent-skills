package work;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import org.springframework.transaction.support.TransactionTemplate;

public class Regression {
    record Seen(String id, int booking, int header, boolean transaction, Thread thread) {}

    static class Probe implements ReceiptObserver {
        private final DataSource dataSource;
        final List<Seen> seen = new ArrayList<>();
        Probe(DataSource dataSource) { this.dataSource = dataSource; }
        public void beforeWrite(String id) {
            try (var connection = dataSource.getConnection()) {
                connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                seen.add(new Seen(id, count(connection, "booking", id), count(connection, "receipt", id),
                        TransactionSynchronizationManager.isActualTransactionActive(), Thread.currentThread()));
            } catch (SQLException e) { throw new IllegalStateException(e); }
        }
    }

    @Configuration
    static class Diagnostics {
        @Bean @Primary Probe diagnosticObserver(DataSource ds) { return new Probe(ds); }
    }

    static int count(Connection connection, String table, String id) throws SQLException {
        try (var statement = connection.prepareStatement("select count(*) from " + table + " where id=?")) {
            statement.setString(1, id);
            try (var result = statement.executeQuery()) { result.next(); return result.getInt(1); }
        }
    }

    static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    static void scenario(String name) throws Exception {
        try (var ctx = new AnnotationConfigApplicationContext(Config.class, Diagnostics.class)) {
            var jdbc = ctx.getBean(JdbcTemplate.class);
            var ds = ctx.getBean(DataSource.class);
            var probe = ctx.getBean(Probe.class);
            jdbc.execute("drop all objects");
            jdbc.execute("create table booking(id varchar(80) primary key,sku varchar(80))");
            jdbc.execute("create table receipt(id varchar(80) primary key)");
            jdbc.execute("create table receipt_detail(id varchar(80) primary key,body varchar(12))");
            jdbc.execute("create table stock(sku varchar(80) primary key,available int)");
            jdbc.update("insert into stock values ('ready',20),('sold-out',0)");
            var service = ctx.getBean(BookingService.class);
            var owner = new TransactionTemplate(ctx.getBean("businessTx", PlatformTransactionManager.class));
            var thread = Thread.currentThread();
            String id = name;
            Throwable failure = null;
            boolean[] beforeOwnerCompletion = {true};
            try {
                switch (name) {
                    case "normal" -> service.place(id, "ready", "123456789012");
                    case "invalid" -> service.place(id, "ready", "1234567890123");
                    case "sold-out" -> service.place(id, "sold-out", "normal");
                    case "nontransactional" -> ctx.publishEvent(new BookingRecorded(id, "normal"));
                    case "owner-commit", "owner-rollback", "late-rollback" -> owner.executeWithoutResult(status -> {
                        service.place(id, "ready", "normal");
                        try (var connection = ds.getConnection()) {
                            connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                            beforeOwnerCompletion[0] = count(connection, "booking", id) == 0
                                    && count(connection, "receipt", id) == 0 && probe.seen.isEmpty();
                        } catch (SQLException e) { throw new IllegalStateException(e); }
                        if (name.equals("owner-rollback")) throw new IllegalStateException("Owner rejected command");
                        if (name.equals("late-rollback")) {
                            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                                @Override public void beforeCommit(boolean readOnly) {
                                    throw new IllegalStateException("Owner rejected commit");
                                }
                            });
                        }
                    });
                    default -> throw new IllegalArgumentException(name);
                }
            } catch (Throwable e) { failure = e; }

            int booking, header, detail, stock;
            String body = null;
            try (var connection = ds.getConnection()) {
                connection.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);
                booking = count(connection, "booking", id);
                header = count(connection, "receipt", id);
                detail = count(connection, "receipt_detail", id);
                try (var statement = connection.createStatement();
                     var result = statement.executeQuery("select available from stock where sku='"
                             + (name.equals("sold-out") ? "sold-out" : "ready") + "'")) {
                    result.next(); stock = result.getInt(1);
                }
                if (detail == 1) {
                    try (var statement = connection.prepareStatement("select body from receipt_detail where id=?")) {
                        statement.setString(1, id);
                        try (var result = statement.executeQuery()) { result.next(); body = result.getString(1); }
                    }
                }
            }
            System.out.println(name + ": failure=" + (failure == null ? "none" : failure.getClass().getSimpleName())
                    + " booking=" + booking + " header=" + header + " detail=" + detail + " stock=" + stock
                    + " observer=" + probe.seen + " beforeOwnerCompletion=" + beforeOwnerCompletion[0]);
            boolean committed = name.equals("normal") || name.equals("invalid") || name.equals("owner-commit");
            boolean receipt = name.equals("normal") || name.equals("owner-commit");
            boolean rejected = name.equals("sold-out") || name.endsWith("rollback");
            require((failure != null) == rejected, "Business failure policy: " + name);
            require(booking == (committed ? 1 : 0), "Booking atomicity: " + name);
            require(header == (receipt ? 1 : 0) && detail == (receipt ? 1 : 0), "Receipt atomicity: " + name);
            require(stock == (name.equals("sold-out") ? 0 : committed ? 19 : 20), "Stock atomicity: " + name);
            require(beforeOwnerCompletion[0], "Service must join owner transaction: " + name);
            require(probe.seen.size() == (committed ? 1 : 0), "Receipt must only run after source commit: " + name);
            for (var seen : probe.seen) {
                require(seen.id().equals(id) && seen.booking() == 1 && seen.header() == 0,
                        "Independent READ_COMMITTED visibility immediately before write: " + name);
                require(seen.transaction() && seen.thread() == thread, "Synchronous transactional writer: " + name);
            }
            if (receipt) require(body.equals(name.equals("normal") ? "123456789012" : "normal"), "Body unchanged");
            System.out.println("PASS " + name);
        }
    }

    public static void main(String[] args) throws Exception {
        var failures = new ArrayList<String>();
        String[] scenarios = args.length == 0
                ? new String[]{"normal", "invalid", "sold-out", "nontransactional", "owner-commit", "owner-rollback", "late-rollback"}
                : args;
        for (var name : scenarios) {
            try { scenario(name); }
            catch (Throwable e) { failures.add(name + ": " + e); System.out.println("FAIL " + name + ": " + e); }
        }
        if (!failures.isEmpty()) throw new AssertionError(failures.toString());
        System.out.println("PASS all " + scenarios.length + " regression scenarios");
    }
}
