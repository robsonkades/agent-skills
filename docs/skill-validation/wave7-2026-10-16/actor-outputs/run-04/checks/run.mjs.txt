import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const [label, ...tests] = process.argv.slice(2);
const argv = ['node', 'check.mjs', ...tests];
const startedAt = new Date().toISOString();
const result = spawnSync(argv[0], argv.slice(1), {cwd: process.cwd(), encoding: 'utf8', timeout: 30000, windowsHide: true});
const log = {cwd: process.cwd(), argv, startedAt, finishedAt: new Date().toISOString(), exit: result.status,
    stdout: result.stdout ?? '', stderr: result.stderr ?? '', error: result.error?.message ?? null};
fs.writeFileSync(path.resolve('../checks', label + '.json'), JSON.stringify(log, null, 2));
process.stdout.write(log.stdout);
process.stderr.write(log.stderr);
if (log.error) process.stderr.write(log.error + '\n');
process.exitCode = result.status ?? 1;
