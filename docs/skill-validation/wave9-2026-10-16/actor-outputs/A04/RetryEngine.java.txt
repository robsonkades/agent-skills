public final class RetryEngine {
    public static Ports.Result execute(Ports.Intent intent, Ports.Policy p, Ports.Store store,
            Ports.Operation operation, Ports.Runtime runtime) throws InterruptedException {
        Ports.Snapshot initial = store.snapshot();
        if (initial.state() == Ports.State.APPLIED || initial.state() == Ports.State.REJECTED) {
            return new Ports.Result(initial, 0);
        }

        boolean unresolved = initial.state() == Ports.State.UNKNOWN;
        int attempts = 0;
        while (true) {
            checkInterrupted();
            if ((unresolved && !Contract.REPLAY_SAFE) || runtime.cancelled()
                    || !fits(runtime.remainingNanos(), 0, p)) {
                return stopped(store, unresolved, attempts);
            }

            Ports.Kind kind;
            String receipt;
            long wait;
            checkInterrupted();
            attempts++;
            try (Ports.Reply reply = operation.call(intent, p.attemptNanos())) {
                kind = reply.kind();
                receipt = kind == Ports.Kind.OK ? reply.receipt() : null;
                wait = Math.max(p.localWaitNanos(), reply.serverMinimumNanos());
            } catch (InterruptedException interrupted) {
                // Dispatch may already have applied; cancellation does not settle that outcome.
                store.unknown();
                throw interrupted;
            }

            if (kind == Ports.Kind.OK) {
                store.applied(receipt);
                return new Ports.Result(store.snapshot(), attempts);
            }
            if (kind == Ports.Kind.UNKNOWN) {
                unresolved = true;
                store.unknown();
            }
            if (kind == Ports.Kind.PERMANENT || attempts >= p.maxAttempts()
                    || (unresolved && !Contract.REPLAY_SAFE)) {
                return stopped(store, unresolved, attempts);
            }

            checkInterrupted();
            if (runtime.cancelled() || !fits(runtime.remainingNanos(), wait, p)
                    || !runtime.acquireRetry()) {
                return stopped(store, unresolved, attempts);
            }
            checkInterrupted();
            if (runtime.cancelled()) return stopped(store, unresolved, attempts);
            runtime.sleep(wait);
            // Sleep may overrun or be cancelled: admission is checked again before dispatch.
        }
    }

    private static boolean fits(long remaining, long wait, Ports.Policy p) {
        // Subtract only after comparison so even Long.MAX_VALUE inputs cannot overflow.
        if (wait > remaining) return false;
        remaining -= wait;
        if (p.reserveNanos() > remaining) return false;
        return p.attemptNanos() <= remaining - p.reserveNanos();
    }

    private static Ports.Result stopped(Ports.Store store, boolean unresolved, int attempts) {
        if (unresolved) store.unknown();
        else store.rejected();
        return new Ports.Result(store.snapshot(), attempts);
    }

    private static void checkInterrupted() throws InterruptedException {
        if (Thread.currentThread().isInterrupted()) throw new InterruptedException();
    }
}
