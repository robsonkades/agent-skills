public final class RetryEngine {
    public static Ports.Result execute(Ports.Intent intent, Ports.Policy p, Ports.Store store,
            Ports.Operation operation, Ports.Runtime runtime) throws InterruptedException {
        Ports.Snapshot initial = store.snapshot();
        if (initial.state() == Ports.State.APPLIED || initial.state() == Ports.State.REJECTED) {
            return new Ports.Result(initial, 0);
        }

        boolean unresolved = initial.state() == Ports.State.UNKNOWN;
        int attempts = 0;
        while (attempts < p.maxAttempts()) {
            checkInterrupted();
            if (runtime.cancelled() || (unresolved && !Contract.REPLAY_SAFE)
                    || !fits(runtime.remainingNanos(), 0, p)) {
                break;
            }

            Ports.Kind kind;
            long serverMinimum;
            attempts++;
            try (Ports.Reply reply = operation.call(intent, p.attemptNanos())) {
                kind = reply.kind();
                if (kind == Ports.Kind.OK) {
                    store.applied(reply.receipt());
                    return new Ports.Result(store.snapshot(), attempts);
                }
                if (kind == Ports.Kind.UNKNOWN) {
                    unresolved = true;
                    store.unknown();
                }
                serverMinimum = reply.serverMinimumNanos();
            } catch (InterruptedException interrupted) {
                // A call interrupted after dispatch has no authoritative effect outcome.
                store.unknown();
                throw interrupted;
            }

            if (kind == Ports.Kind.PERMANENT || attempts == p.maxAttempts()
                    || (unresolved && !Contract.REPLAY_SAFE)) {
                break;
            }
            checkInterrupted();
            long wait = Math.max(p.localWaitNanos(), serverMinimum);
            if (runtime.cancelled() || !fits(runtime.remainingNanos(), wait, p)
                    || !runtime.acquireRetry()) {
                break;
            }
            runtime.sleep(wait);
            // The loop refreshes admission after sleep, including oversleep/cancellation.
        }

        if (unresolved) {
            store.unknown();
        } else {
            store.rejected();
        }
        return new Ports.Result(store.snapshot(), attempts);
    }

    private static boolean fits(long remaining, long wait, Ports.Policy p) {
        // Subtract only after comparison: valid long inputs may overflow when summed.
        return remaining >= p.reserveNanos()
                && remaining - p.reserveNanos() >= p.attemptNanos()
                && remaining - p.reserveNanos() - p.attemptNanos() >= wait;
    }

    private static void checkInterrupted() throws InterruptedException {
        if (Thread.currentThread().isInterrupted()) {
            throw new InterruptedException("Retry execution interrupted before dispatch");
        }
    }
}
