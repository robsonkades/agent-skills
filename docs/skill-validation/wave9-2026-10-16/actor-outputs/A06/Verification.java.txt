import java.util.ArrayList;
import java.util.List;

public final class Verification {
    private static final Ports.Intent INTENT = new Ports.Intent("purchase-1", "same payload");
    private static final Ports.Policy POLICY = new Ports.Policy(3, 10, 5, 4);
    private static int checks;

    public static void main(String[] args) throws Exception {
        stableReplay();
        terminalReplay();
        outcomeHistory();
        admissionBoundaries();
        retryAdmission();
        cancellation();
        interruption();
        System.out.println("Verification passed: " + checks + " scenarios");
    }

    private static void stableReplay() throws Exception {
        Fixture f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.TRANSIENT, 7).reply(Ports.Kind.OK, Long.MAX_VALUE);
        expect(f.run(POLICY), Ports.State.APPLIED, 2);
        eq(f.rt.waits, List.of(7L), "server minimum");
        eq(f.rt.tokens, 1, "one retry token");
        eq(f.store.snapshot().receipt(), "original-receipt", "authoritative receipt");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.TRANSIENT, 1).reply(Ports.Kind.OK, 0);
        expect(f.run(POLICY), Ports.State.APPLIED, 2);
        eq(f.rt.waits, List.of(4L), "local delay");
        f.closed();
        checks++;
    }

    private static void terminalReplay() throws Exception {
        for (Ports.State state : List.of(Ports.State.APPLIED, Ports.State.REJECTED)) {
            Fixture f = new Fixture(state, 0);
            Ports.Snapshot original = f.store.snapshot();
            expect(f.run(POLICY), state, 0);
            eq(f.store.snapshot(), original, "terminal snapshot");
            eq(f.store.writes, 0, "terminal writes");
            eq(f.calls, 0, "terminal dispatch");
            eq(f.rt.tokens, 0, "terminal tokens");
            checks++;
        }
    }

    private static void outcomeHistory() throws Exception {
        for (Ports.State initial : List.of(Ports.State.PENDING, Ports.State.UNKNOWN)) {
            for (Ports.Kind kind : Ports.Kind.values()) {
                Fixture f = new Fixture(initial, 100);
                f.reply(kind, Long.MAX_VALUE);
                Ports.State expected = kind == Ports.Kind.OK ? Ports.State.APPLIED
                        : initial == Ports.State.UNKNOWN || kind == Ports.Kind.UNKNOWN
                                ? Ports.State.UNKNOWN : Ports.State.REJECTED;
                expect(f.run(new Ports.Policy(1, 10, 5, 4)), expected, 1);
                eq(f.rt.tokens, 0, "no token at exhaustion/terminal result");
                eq(f.rt.waits.size(), 0, "no wait at exhaustion/terminal result");
                f.closed();
                checks++;
            }
        }
        Fixture f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.UNKNOWN, 9).reply(Ports.Kind.TRANSIENT, 0)
                .reply(Ports.Kind.PERMANENT, 0);
        expect(f.run(POLICY), Ports.State.UNKNOWN, 3);
        eq(f.rt.waits, List.of(9L, 4L), "unknown retry still observes minimum");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.UNKNOWN, 100);
        f.reply(Ports.Kind.TRANSIENT, 0).reply(Ports.Kind.OK, 0);
        expect(f.run(POLICY), Ports.State.APPLIED, 2);
        f.closed();
        checks++;
    }

    private static void admissionBoundaries() throws Exception {
        long[][] cases = {
            {14, 10, 5, 0}, {15, 10, 5, 1},
            {Long.MAX_VALUE, Long.MAX_VALUE, 1, 0},
            {Long.MAX_VALUE, 1, Long.MAX_VALUE, 0},
            {Long.MAX_VALUE, Long.MAX_VALUE, 0, 1},
            {Long.MAX_VALUE, 1, Long.MAX_VALUE - 1, 1}
        };
        for (long[] c : cases) {
            Fixture f = new Fixture(Ports.State.PENDING, c[0]);
            f.reply(Ports.Kind.OK, 0);
            expect(f.run(new Ports.Policy(2, c[1], c[2], 0)),
                    c[3] == 1 ? Ports.State.APPLIED : Ports.State.REJECTED, (int)c[3]);
            eq(f.rt.tokens, 0, "initial call needs no token");
            f.closed();
            checks++;
        }
        for (Ports.State state : List.of(Ports.State.PENDING, Ports.State.UNKNOWN)) {
            Fixture f = new Fixture(state, 0);
            expect(f.run(POLICY), state == Ports.State.UNKNOWN ? state : Ports.State.REJECTED, 0);
            checks++;
        }
    }

    private static void retryAdmission() throws Exception {
        for (Ports.Kind kind : List.of(Ports.Kind.TRANSIENT, Ports.Kind.UNKNOWN)) {
            Fixture f = new Fixture(Ports.State.PENDING, 21);
            f.reply(kind, 7);
            expect(f.run(POLICY), kind == Ports.Kind.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.REJECTED, 1);
            eq(f.rt.tokens, 0, "refused wait does not consume token");
            eq(f.rt.waits.size(), 0, "minimum cannot be truncated");
            f.closed();
            checks++;
        }
        Fixture f = new Fixture(Ports.State.PENDING, 22);
        f.reply(Ports.Kind.TRANSIENT, 7).reply(Ports.Kind.OK, 0);
        expect(f.run(POLICY), Ports.State.APPLIED, 2);
        eq(f.rt.waits, List.of(7L), "exact retry boundary");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, Long.MAX_VALUE);
        f.reply(Ports.Kind.TRANSIENT, Long.MAX_VALUE);
        expect(f.run(new Ports.Policy(2, 1, 0, 0)), Ports.State.REJECTED, 1);
        eq(f.rt.tokens, 0, "wait sum cannot overflow");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, Long.MAX_VALUE);
        f.reply(Ports.Kind.TRANSIENT, Long.MAX_VALUE - 1).reply(Ports.Kind.OK, 0);
        expect(f.run(new Ports.Policy(2, 1, 0, 0)), Ports.State.APPLIED, 2);
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.UNKNOWN, 0);
        f.rt.allowToken = false;
        expect(f.run(POLICY), Ports.State.UNKNOWN, 1);
        eq(f.rt.tokens, 1, "denied token request");
        eq(f.rt.waits.size(), 0, "no sleep after denied token");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, 22);
        f.reply(Ports.Kind.UNKNOWN, 7);
        f.rt.oversleep = 1;
        expect(f.run(POLICY), Ports.State.UNKNOWN, 1);
        eq(f.rt.tokens, 1, "oversleep consumes admission only");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.TRANSIENT, 0);
        f.callCost = 86;
        expect(f.run(POLICY), Ports.State.REJECTED, 1);
        eq(f.rt.tokens, 0, "fresh budget after call");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.TRANSIENT, 0).reply(Ports.Kind.OK, 0);
        expect(f.run(new Ports.Policy(2, 10, 0, 0)), Ports.State.APPLIED, 2);
        eq(f.rt.tokens, 1, "zero delay still requires admission");
        f.closed();
        checks++;
    }

    private static void cancellation() throws Exception {
        for (Ports.State state : List.of(Ports.State.PENDING, Ports.State.UNKNOWN)) {
            Fixture f = new Fixture(state, 100);
            f.rt.cancelled = true;
            expect(f.run(POLICY), state == Ports.State.UNKNOWN ? state : Ports.State.REJECTED, 0);
            eq(f.rt.tokens, 0, "cancelled before dispatch");
            checks++;
        }
        Fixture f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.UNKNOWN, 0);
        f.cancelAfterCall = true;
        expect(f.run(POLICY), Ports.State.UNKNOWN, 1);
        eq(f.rt.tokens, 0, "cancelled before retry admission");
        f.closed();
        checks++;
        f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.UNKNOWN, 0);
        f.rt.cancelAfterSleep = true;
        expect(f.run(POLICY), Ports.State.UNKNOWN, 1);
        eq(f.rt.tokens, 1, "cancelled after admitted wait");
        f.closed();
        checks++;
    }

    private static void interruption() throws Exception {
        for (Ports.State state : List.of(Ports.State.PENDING, Ports.State.UNKNOWN)) {
            Fixture f = new Fixture(state, 100);
            Thread.currentThread().interrupt();
            try {
                interrupted(f);
                eq(f.calls, 0, "pre-interrupted dispatch");
                eq(f.store.snapshot().state(), state, "exception is not normal refusal");
            } finally {
                Thread.interrupted();
            }
            checks++;
        }
        Fixture f = new Fixture(Ports.State.PENDING, 100);
        f.interruptCall = true;
        interrupted(f);
        eq(f.store.snapshot().state(), Ports.State.UNKNOWN, "interrupted call is uncertain");
        eq(f.calls, 1, "interrupted attempt counted as dispatch");
        Thread.interrupted();
        checks++;
        for (Ports.Kind kind : List.of(Ports.Kind.UNKNOWN, Ports.Kind.TRANSIENT)) {
            f = new Fixture(Ports.State.PENDING, 100);
            f.reply(kind, 0);
            f.rt.interruptSleep = true;
            interrupted(f);
            eq(f.store.snapshot().state(), kind == Ports.Kind.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.PENDING,
                    "sleep interruption preserves evidence");
            f.closed();
            Thread.interrupted();
            checks++;
        }
        f = new Fixture(Ports.State.PENDING, 100);
        f.reply(Ports.Kind.UNKNOWN, 0);
        f.interruptAfterCall = true;
        try {
            interrupted(f);
            eq(f.calls, 1, "interrupted before retry");
            eq(f.rt.tokens, 0, "interrupted retry token");
            eq(f.store.snapshot().state(), Ports.State.UNKNOWN, "unknown recorded before interrupt propagation");
            f.closed();
        } finally {
            Thread.interrupted();
        }
        checks++;
    }

    private static void interrupted(Fixture f) throws Exception {
        try {
            f.run(POLICY);
            throw new AssertionError("InterruptedException expected");
        } catch (InterruptedException expected) {
            // Propagation, not success or a fabricated failure result.
        }
    }

    private static void expect(Ports.Result r, Ports.State state, int attempts) {
        eq(r.snapshot().state(), state, "result state");
        eq(r.attempts(), attempts, "attempt count");
    }

    private static void eq(Object actual, Object expected, String message) {
        if (!java.util.Objects.equals(actual, expected))
            throw new AssertionError(message + ": expected " + expected + ", got " + actual);
    }

    private static final class MemoryStore implements Ports.Store {
        Ports.Snapshot snapshot;
        int writes;
        MemoryStore(Ports.State state) {
            snapshot = new Ports.Snapshot(state, state == Ports.State.APPLIED ? "stored-receipt" : null);
        }
        public Ports.Snapshot snapshot() { return snapshot; }
        public void unknown() { writes++; snapshot = new Ports.Snapshot(Ports.State.UNKNOWN, null); }
        public void applied(String receipt) { writes++; snapshot = new Ports.Snapshot(Ports.State.APPLIED, receipt); }
        public void rejected() { writes++; snapshot = new Ports.Snapshot(Ports.State.REJECTED, null); }
    }

    private static final class TestReply implements Ports.Reply {
        final Ports.Kind kind;
        final long minimum;
        int closes;
        TestReply(Ports.Kind kind, long minimum) { this.kind = kind; this.minimum = minimum; }
        public Ports.Kind kind() { return kind; }
        public String receipt() { return "original-receipt"; }
        public long serverMinimumNanos() { return minimum; }
        public void close() { closes++; eq(closes, 1, "reply closed exactly once"); }
    }

    private static final class TestRuntime implements Ports.Runtime {
        final Fixture fixture;
        long remaining;
        long oversleep;
        int tokens;
        boolean allowToken = true;
        boolean cancelled;
        boolean cancelAfterSleep;
        boolean interruptSleep;
        final List<Long> waits = new ArrayList<>();
        TestRuntime(Fixture fixture, long remaining) { this.fixture = fixture; this.remaining = remaining; }
        public long remainingNanos() { return remaining; }
        public boolean cancelled() { return cancelled; }
        public boolean acquireRetry() { fixture.closed(); fixture.uncertaintyPersisted(); tokens++; return allowToken; }
        public void sleep(long nanos) throws InterruptedException {
            fixture.closed();
            fixture.uncertaintyPersisted();
            waits.add(nanos);
            if (interruptSleep) throw new InterruptedException("sleep interrupted");
            remaining = Math.max(0, remaining - nanos);
            remaining = Math.max(0, remaining - oversleep);
            if (cancelAfterSleep) cancelled = true;
        }
    }

    private static final class Fixture {
        final MemoryStore store;
        final TestRuntime rt;
        final List<TestReply> script = new ArrayList<>();
        final List<TestReply> returned = new ArrayList<>();
        int calls;
        long callCost;
        boolean unresolved;
        boolean cancelAfterCall;
        boolean interruptAfterCall;
        boolean interruptCall;
        Fixture(Ports.State state, long remaining) {
            store = new MemoryStore(state);
            unresolved = state == Ports.State.UNKNOWN;
            rt = new TestRuntime(this, remaining);
        }
        Fixture reply(Ports.Kind kind, long minimum) { script.add(new TestReply(kind, minimum)); return this; }
        Ports.Result run(Ports.Policy policy) throws InterruptedException {
            return RetryEngine.execute(INTENT, policy, store, (intent, timeout) -> {
                closed();
                uncertaintyPersisted();
                eq(intent, INTENT, "same intent ID and payload on every dispatch");
                eq(timeout, policy.attemptNanos(), "configured attempt timeout");
                if (rt.cancelled || Thread.currentThread().isInterrupted())
                    throw new AssertionError("dispatch after cancellation/interruption");
                if (rt.remaining < policy.attemptNanos()
                        || rt.remaining - policy.attemptNanos() < policy.reserveNanos())
                    throw new AssertionError("dispatch without full budget");
                calls++;
                rt.remaining = Math.max(0, rt.remaining - callCost);
                if (interruptCall) throw new InterruptedException("call interrupted");
                if (calls > script.size()) throw new AssertionError("unexpected extra dispatch");
                TestReply reply = script.get(calls - 1);
                returned.add(reply);
                if (reply.kind == Ports.Kind.UNKNOWN) unresolved = true;
                if (reply.kind == Ports.Kind.OK) unresolved = false;
                if (cancelAfterCall) rt.cancelled = true;
                if (interruptAfterCall) Thread.currentThread().interrupt();
                return reply;
            }, rt);
        }
        void closed() { for (TestReply reply : returned) eq(reply.closes, 1, "release before next action"); }
        void uncertaintyPersisted() {
            if (unresolved) eq(store.snapshot().state(), Ports.State.UNKNOWN, "persist unknown before retry/wait");
        }
    }
}
