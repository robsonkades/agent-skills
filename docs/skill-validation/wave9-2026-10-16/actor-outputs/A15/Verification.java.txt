import java.util.ArrayList;
import java.util.List;

public final class Verification {
    private static final Ports.Intent INTENT = new Ports.Intent("intent-original", "payload-original");
    private static final Ports.Policy POLICY = new Ports.Policy(3, 10, 5, 4);
    private static int passed;
    private static int failed;

    @FunctionalInterface private interface Check { void run() throws Exception; }
    @FunctionalInterface private interface Hook { void run() throws InterruptedException; }

    private static final class Store implements Ports.Store {
        Ports.Snapshot value;
        int writes;
        Store(Ports.State state) { value = new Ports.Snapshot(state, state == Ports.State.APPLIED ? "original-receipt" : null); }
        public Ports.Snapshot snapshot() { return value; }
        public void unknown() { writes++; value = new Ports.Snapshot(Ports.State.UNKNOWN, null); }
        public void applied(String receipt) { writes++; value = new Ports.Snapshot(Ports.State.APPLIED, receipt); }
        public void rejected() { writes++; value = new Ports.Snapshot(Ports.State.REJECTED, null); }
    }

    private static final class Reply implements Ports.Reply {
        final Ports.Kind kind;
        final long minimum;
        int closes;
        Reply(Ports.Kind kind, long minimum) { this.kind = kind; this.minimum = minimum; }
        public Ports.Kind kind() { return kind; }
        public String receipt() { return "authoritative-receipt"; }
        public long serverMinimumNanos() { return minimum; }
        public void close() { closes++; require(closes == 1, "reply closed twice"); }
    }

    private static final class Fixture implements Ports.Runtime, Ports.Operation {
        final Store store;
        final List<Reply> replies = new ArrayList<>();
        final List<Ports.Intent> intents = new ArrayList<>();
        final List<Long> timeouts = new ArrayList<>();
        final List<Long> sleeps = new ArrayList<>();
        long remaining = 1000;
        long callCost;
        int grants;
        boolean cancelled;
        boolean token = true;
        Hook inCall = () -> {};
        Hook inSleep = () -> {};
        Fixture() { this(Ports.State.PENDING); }
        Fixture(Ports.State state) { store = new Store(state); }
        Fixture add(Ports.Kind kind, long minimum) { replies.add(new Reply(kind, minimum)); return this; }
        public long remainingNanos() { return remaining; }
        public boolean cancelled() { return cancelled; }
        public boolean acquireRetry() { grants++; return token; }
        public void sleep(long nanos) throws InterruptedException {
            assertPreviousClosed();
            sleeps.add(nanos);
            require(nanos >= 0 && nanos <= remaining, "invalid sleep budget");
            remaining -= nanos;
            inSleep.run();
        }
        public Ports.Reply call(Ports.Intent intent, long timeout) throws InterruptedException {
            assertPreviousClosed();
            intents.add(intent);
            timeouts.add(timeout);
            remaining = Math.max(0, remaining - callCost);
            inCall.run();
            require(intents.size() <= replies.size(), "unexpected dispatch");
            return replies.get(intents.size() - 1);
        }
        void assertPreviousClosed() {
            for (int i = 0; i < intents.size() && i < replies.size(); i++) {
                require(replies.get(i).closes == 1, "reply must close before next call or sleep");
            }
        }
        Ports.Result run() throws InterruptedException { return run(POLICY); }
        Ports.Result run(Ports.Policy policy) throws InterruptedException {
            Ports.Result result = RetryEngine.execute(INTENT, policy, store, this, this);
            assertPreviousClosed();
            require(result.attempts() == intents.size(), "returned attempt count");
            require(result.snapshot().equals(store.snapshot()), "returned durable outcome");
            for (Ports.Intent actual : intents) require(actual.equals(INTENT), "stable intent and payload required");
            for (long actual : timeouts) require(actual == policy.attemptNanos(), "propagated attempt timeout");
            return result;
        }
    }

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }
    private static void state(Ports.Result result, Ports.State state, int attempts) {
        require(result.snapshot().state() == state && result.attempts() == attempts,
                "expected " + state + "/" + attempts + ", got " + result);
    }
    private static void test(String name, Check check) {
        try { check.run(); passed++; System.out.println("PASS " + name); }
        catch (Throwable failure) { failed++; System.out.println("FAIL " + name + ": " + failure); }
        finally { Thread.interrupted(); }
    }
    private static void interrupted(Fixture f, InterruptedException expected) throws Exception {
        try { f.run(); throw new AssertionError("interruption swallowed"); }
        catch (InterruptedException actual) { if (expected != null) require(actual == expected, "original exception propagated"); }
    }

    public static void main(String[] args) throws Exception {
        require(!Contract.REPLAY_SAFE, "this verification targets the supplied false replay guarantee");
        for (Ports.State terminal : List.of(Ports.State.APPLIED, Ports.State.REJECTED)) {
            test("terminal replay " + terminal, () -> {
                Fixture f = new Fixture(terminal);
                Ports.Snapshot initial = f.store.snapshot();
                state(f.run(), terminal, 0);
                require(f.store.writes == 0 && initial.equals(f.store.snapshot()), "terminal unchanged");
            });
        }
        test("successful original identity, timeout and receipt", () -> {
            Fixture f = new Fixture().add(Ports.Kind.OK, Long.MAX_VALUE);
            Ports.Result result = f.run(); state(result, Ports.State.APPLIED, 1);
            require("authoritative-receipt".equals(result.snapshot().receipt()), "original receipt");
            require(f.grants == 0 && f.sleeps.isEmpty(), "initial call requires no retry admission");
        });
        test("transient retry obeys server minimum", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 9).add(Ports.Kind.OK, 0);
            state(f.run(), Ports.State.APPLIED, 2);
            require(f.sleeps.equals(List.of(9L)) && f.grants == 1, "max delay and one token");
        });
        test("transient retry obeys local minimum", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 1).add(Ports.Kind.OK, 0);
            state(f.run(), Ports.State.APPLIED, 2);
            require(f.sleeps.equals(List.of(4L)), "local delay must not shrink");
        });
        test("unknown after dispatch persists and stops", () -> {
            Fixture f = new Fixture().add(Ports.Kind.UNKNOWN, 9);
            state(f.run(), Ports.State.UNKNOWN, 1);
            require(f.grants == 0 && f.sleeps.isEmpty(), "unsafe replay refused");
        });
        test("initial unknown survives invocation restart", () -> {
            Fixture f = new Fixture(Ports.State.UNKNOWN);
            state(f.run(), Ports.State.UNKNOWN, 0);
            require(f.grants == 0 && f.sleeps.isEmpty(), "no unsafe redispatch");
        });
        test("permanent known nonapplication", () -> {
            Fixture f = new Fixture().add(Ports.Kind.PERMANENT, Long.MAX_VALUE);
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 0 && f.sleeps.isEmpty(), "permanent stop must not wait");
        });
        test("exhaustion does not buy unused final token", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0).add(Ports.Kind.TRANSIENT, 0).add(Ports.Kind.TRANSIENT, 0);
            state(f.run(), Ports.State.REJECTED, 3);
            require(f.grants == 2 && f.sleeps.size() == 2, "only between-attempt admissions");
        });
        test("insufficient initial budget", () -> {
            Fixture f = new Fixture(); f.remaining = 14;
            state(f.run(), Ports.State.REJECTED, 0);
        });
        test("exact initial budget", () -> {
            Fixture f = new Fixture().add(Ports.Kind.OK, 0); f.remaining = 15;
            state(f.run(), Ports.State.APPLIED, 1);
        });
        test("overflowing attempt plus reserve refused", () -> {
            Fixture f = new Fixture(); f.remaining = Long.MAX_VALUE;
            state(f.run(new Ports.Policy(1, Long.MAX_VALUE, 1, 0)), Ports.State.REJECTED, 0);
        });
        test("largest representable exact attempt accepted", () -> {
            Fixture f = new Fixture().add(Ports.Kind.OK, 0); f.remaining = Long.MAX_VALUE;
            state(f.run(new Ports.Policy(1, Long.MAX_VALUE, 0, 0)), Ports.State.APPLIED, 1);
        });
        test("overflowing retry wait refused before token", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, Long.MAX_VALUE); f.remaining = Long.MAX_VALUE;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 0 && f.sleeps.isEmpty(), "impossible wait not admitted");
        });
        test("server minimum cannot be truncated to fit", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 10); f.remaining = 24;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 0 && f.sleeps.isEmpty(), "wait requires attempt and reserve too");
        });
        test("exact retry wait plus call budget", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 9).add(Ports.Kind.OK, 0); f.remaining = 24;
            state(f.run(), Ports.State.APPLIED, 2);
            require(f.sleeps.equals(List.of(9L)), "exact-fit wait");
        });
        test("budget consumed by call prevents wait", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.remaining = 20; f.callCost = 10;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 0 && f.sleeps.isEmpty(), "fresh remaining budget required");
        });
        test("oversleep burns grant but prevents next call", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.inSleep = () -> f.remaining = 14;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 1 && f.sleeps.size() == 1, "token is admission, not attempt");
        });
        test("retry token denial closes reply", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.token = false;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 1 && f.sleeps.isEmpty(), "denial prevents sleep");
        });
        test("cancellation before initial call", () -> {
            Fixture f = new Fixture(); f.cancelled = true;
            state(f.run(), Ports.State.REJECTED, 0);
        });
        test("cancellation after known nonapplication", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.inCall = () -> f.cancelled = true;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 0 && f.sleeps.isEmpty(), "cancellation checked before retry admission");
        });
        test("cancellation during sleep", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.inSleep = () -> f.cancelled = true;
            state(f.run(), Ports.State.REJECTED, 1);
            require(f.grants == 1, "one admitted retry but no dispatch");
        });
        test("pre-interruption prevents dispatch", () -> {
            Fixture f = new Fixture(); Thread.currentThread().interrupt(); interrupted(f, null);
            require(f.intents.isEmpty() && f.store.snapshot().state() == Ports.State.PENDING, "no attempt or normal-stop write");
        });
        test("interrupted call records unknown and rethrows", () -> {
            Fixture f = new Fixture(); InterruptedException signal = new InterruptedException("call interrupted");
            f.inCall = () -> { throw signal; }; interrupted(f, signal);
            require(f.intents.size() == 1 && f.store.snapshot().state() == Ports.State.UNKNOWN, "ambiguous call persisted");
        });
        test("interrupted sleep closes returned reply", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0);
            InterruptedException signal = new InterruptedException("sleep interrupted");
            f.inSleep = () -> { throw signal; }; interrupted(f, signal);
            require(f.intents.size() == 1 && f.replies.get(0).closes == 1, "closed before interrupted wait");
            require(f.store.snapshot().state() == Ports.State.PENDING, "known nonapplication is not unknown");
        });
        test("interrupt flag after sleep prevents dispatch", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.inSleep = () -> Thread.currentThread().interrupt();
            interrupted(f, null); require(f.intents.size() == 1, "post-sleep interruption checked");
        });
        test("zero-delay retry still needs one token", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0).add(Ports.Kind.OK, 0);
            state(f.run(new Ports.Policy(2, 10, 0, 0)), Ports.State.APPLIED, 2);
            require(f.grants == 1 && f.sleeps.equals(List.of(0L)), "zero-delay admission");
        });
        System.out.println("Verification: " + passed + " passed, " + failed + " failed");
        if (failed != 0) throw new AssertionError("Verification failures: " + failed);
    }
}
