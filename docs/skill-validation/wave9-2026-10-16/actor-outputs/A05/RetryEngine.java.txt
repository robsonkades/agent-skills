public final class RetryEngine {
    public static Ports.Result execute(Ports.Intent intent, Ports.Policy p, Ports.Store store,
            Ports.Operation operation, Ports.Runtime runtime) throws InterruptedException {
        Ports.Snapshot initial = store.snapshot();
        if (initial.state() == Ports.State.APPLIED || initial.state() == Ports.State.REJECTED) {
            return new Ports.Result(initial, 0);
        }
        boolean unresolved = initial.state() == Ports.State.UNKNOWN;
        int attempts = 0;
        while (attempts < p.maxAttempts()) {
            checkInterrupted();
            if (runtime.cancelled() || (unresolved && !Contract.REPLAY_SAFE)
                    || !fits(runtime.remainingNanos(), 0, p)) break;

            attempts++;
            Ports.Reply response;
            try {
                response = operation.call(intent, p.attemptNanos());
            } catch (InterruptedException interrupted) {
                store.unknown();
                throw interrupted;
            }
            long wait;
            try (Ports.Reply reply = response) {
                Ports.Kind kind = reply.kind();
                if (kind == Ports.Kind.OK) {
                    store.applied(reply.receipt());
                    return new Ports.Result(store.snapshot(), attempts);
                }
                if (kind == Ports.Kind.UNKNOWN) {
                    unresolved = true;
                    store.unknown();
                }
                if (kind == Ports.Kind.PERMANENT) {
                    return stop(store, attempts, unresolved);
                }
                wait = Math.max(p.localWaitNanos(), reply.serverMinimumNanos());
            }
            if (attempts >= p.maxAttempts() || (unresolved && !Contract.REPLAY_SAFE)) break;
            checkInterrupted();
            if (runtime.cancelled() || !fits(runtime.remainingNanos(), wait, p)) break;
            if (!runtime.acquireRetry()) break;
            checkInterrupted();
            if (runtime.cancelled()) break;
            runtime.sleep(wait);
        }
        return stop(store, attempts, unresolved);
    }

    private static boolean fits(long remaining, long wait, Ports.Policy p) {
        // Subtract only after each bound check; sums may exceed Long.MAX_VALUE.
        return wait <= remaining && p.reserveNanos() <= remaining - wait
                && p.attemptNanos() <= remaining - wait - p.reserveNanos();
    }

    private static void checkInterrupted() throws InterruptedException {
        if (Thread.currentThread().isInterrupted()) throw new InterruptedException();
    }

    private static Ports.Result stop(Ports.Store store, int attempts, boolean unresolved) {
        if (unresolved) store.unknown();
        else store.rejected();
        return new Ports.Result(store.snapshot(), attempts);
    }
}
