import java.math.BigInteger;
import java.util.function.LongUnaryOperator;

public final class Verification {
    private static int checks;
    private static final long MAX = Long.MAX_VALUE;
    private static final BigInteger BIG_MAX = BigInteger.valueOf(MAX);

    public static void main(String[] args) {
        examples();
        matrix();
        System.out.println("Verification passed: " + checks + " plans checked");
    }

    private static void examples() {
        expect("initial ignores advice and randomness", new RetryPlan.Plan(true, 0, 80),
                RetryPlan.next(0, 0, 100, 20, 10, 100, 2, 8, 0, 3, MAX, noDraw()));
        expect("attempt limit reached", rejected(),
                RetryPlan.next(0, 0, 100, 0, 1, 100, 2, 8, 3, 3, 0, noDraw()));
        expect("attempt limit exceeded", rejected(),
                RetryPlan.next(0, 0, 100, 0, 1, 100, 2, 8, 4, 3, 0, noDraw()));
        expect("deadline reached", rejected(),
                RetryPlan.next(-10, 0, 10, 0, 1, 100, 2, 8, 1, 3, 0, noDraw()));
        expect("expired with maximum elapsed", rejected(),
                RetryPlan.next(8, 8 + MAX, 1, 0, 1, 1, 1, MAX, 1, 3, 0, noDraw()));
        expect("oversized reserve", rejected(),
                RetryPlan.next(0, 0, 100, MAX, 1, 100, 2, 8, 1, 3, 0, noDraw()));
        expect("reserve plus minimum would overflow", rejected(),
                RetryPlan.next(0, 0, MAX, MAX - 1, 2, 2, 2, 8, 1, 3, 0, noDraw()));
        expect("server advice is not truncated", rejected(),
                RetryPlan.next(0, 0, MAX, 1, 1, MAX, 2, 8, 1, 3, MAX, noDraw()));
        expect("initial exact fit", new RetryPlan.Plan(true, 0, 10),
                RetryPlan.next(0, 0, 30, 20, 10, 100, 2, 8, 0, 1, MAX, noDraw()));
        expect("first retry uses base and greater server delay", new RetryPlan.Plan(true, 10, 70),
                RetryPlan.next(0, 0, 100, 20, 10, 100, 8, 100, 1, 3, 10, draw(8, 7)));
        expect("local draw wins and attempt is capped", new RetryPlan.Plan(true, 15, 30),
                RetryPlan.next(0, 0, 100, 20, 10, 30, 8, 100, 2, 3, 2, draw(16, 15)));
        expect("retry exact fit", new RetryPlan.Plan(true, 15, 10),
                RetryPlan.next(0, 0, 45, 20, 10, 100, 8, 100, 2, 3, 0, draw(16, 15)));
        expect("retry one nanosecond short", rejected(),
                RetryPlan.next(0, 0, 44, 20, 10, 100, 8, 100, 2, 3, 0, draw(16, 15)));
        expect("signed wrap and shrinking remaining", new RetryPlan.Plan(true, 0, 70),
                RetryPlan.next(MAX - 4, Long.MIN_VALUE + 5, 100, 20, 1, 100, 1, 1, 0, 1, 0, noDraw()));
        expect("largest supported initial allocation", new RetryPlan.Plan(true, 0, MAX),
                RetryPlan.next(Long.MIN_VALUE, Long.MIN_VALUE, MAX, 0, MAX, MAX, MAX, MAX,
                        0, 1, MAX, noDraw()));
        expect("large retry delay exact fit", new RetryPlan.Plan(true, MAX - 1, 1),
                RetryPlan.next(0, 0, MAX, 0, 1, MAX, MAX, MAX, 1, 2, 0, draw(MAX, MAX - 1)));
        expect("exponential multiplication saturates", new RetryPlan.Plan(true, 0, MAX),
                RetryPlan.next(0, 0, MAX, 0, 1, MAX, MAX, MAX, 2, 3, 0, draw(MAX, 0)));
        expect("huge exponent is not masked", new RetryPlan.Plan(true, 16, 84),
                RetryPlan.next(0, 0, 100, 0, 1, 100, 1, 17, Integer.MAX_VALUE - 1,
                        Integer.MAX_VALUE, 0, draw(17, 16)));
    }

    private static void matrix() {
        long[][] budgets = {
            {1, 0, 0, 1, 1}, {20, 0, 0, 1, 20}, {20, 3, 4, 2, 30},
            {20, 20, 0, 1, 10}, {20, 21, 0, 1, 10}, {20, MAX, 0, 1, 1},
            {MAX, 0, 0, 1, MAX}, {MAX, 0, MAX - 1, 1, MAX},
            {MAX, 0, MAX - 1, 2, MAX}, {MAX, MAX - 1, 0, 1, MAX},
            {MAX, 0, 0, MAX, MAX}, {MAX, 1, 0, MAX, MAX}, {50, 0, MAX, 1, MAX}
        };
        long[] starts = {0, -100, Long.MIN_VALUE, MAX - 2};
        long[] bases = {1, 2, 9, 1L << 62, MAX};
        long[] caps = {1, 7, 1L << 62, MAX};
        long[] advice = {0, 1, 11, MAX};
        int[] completedValues = {0, 1, 2, 3, 62, 63, 64, 65, Integer.MAX_VALUE - 1};
        for (long[] b : budgets) {
            for (long start : starts) {
                for (long base : bases) {
                    for (long cap : caps) {
                        for (long server : advice) {
                            for (int completed : completedValues) {
                                for (int mode = 0; mode < 3; mode++) {
                                    verify(start, b, base, cap, completed, Integer.MAX_VALUE, server, mode);
                                }
                            }
                        }
                    }
                }
            }
        }
        for (int maxAttempts : new int[] {1, 2, 3, Integer.MAX_VALUE}) {
            for (int completed : new int[] {0, maxAttempts - 1, maxAttempts}) {
                verify(0, new long[] {MAX, 0, 0, 1, MAX}, 2, MAX,
                        completed, maxAttempts, 0, 2);
            }
        }
    }

    private static void verify(long start, long[] b, long base, long cap, int completed,
            int maxAttempts, long server, int mode) {
        long total = b[0], elapsed = b[1], reserve = b[2], minimum = b[3], attemptCap = b[4];
        BigInteger window = BigInteger.valueOf(base);
        if (completed > 0) {
            // Beyond 63 doublings, every positive base exceeds every valid signed-long cap.
            window = window.shiftLeft(Math.min(completed - 1, 63));
        }
        long bound = window.min(BigInteger.valueOf(cap)).longValueExact();
        long sample = mode == 0 ? 0 : mode == 1 ? bound / 2 : bound - 1;
        long wait = completed == 0 ? 0 : Math.max(server, sample);
        BigInteger left = BigInteger.valueOf(total).subtract(BigInteger.valueOf(elapsed))
                .subtract(BigInteger.valueOf(reserve)).subtract(BigInteger.valueOf(wait));
        boolean admitted = completed < maxAttempts && left.compareTo(BigInteger.valueOf(minimum)) >= 0;
        RetryPlan.Plan expected = admitted
                ? new RetryPlan.Plan(true, wait, left.min(BigInteger.valueOf(attemptCap)).longValueExact())
                : rejected();
        int[] draws = {0};
        RetryPlan.Plan actual = RetryPlan.next(start, start + elapsed, total, reserve, minimum,
                attemptCap, base, cap, completed, maxAttempts, server, suppliedBound -> {
                    if (completed == 0) throw new AssertionError("initial call drew randomness");
                    if (++draws[0] != 1) throw new AssertionError("multiple draws");
                    if (suppliedBound != bound || suppliedBound <= 0) {
                        throw new AssertionError("bound " + suppliedBound + " != " + bound);
                    }
                    return sample;
                });
        expect("matrix completed=" + completed + " base=" + base + " cap=" + cap, expected, actual);
        if (admitted && completed > 0 && draws[0] != 1) throw new AssertionError("missing retry draw");
        if (actual.admitted()) {
            BigInteger spent = BigInteger.valueOf(elapsed).add(BigInteger.valueOf(reserve))
                    .add(BigInteger.valueOf(actual.waitNanos())).add(BigInteger.valueOf(actual.attemptNanos()));
            if (spent.compareTo(BigInteger.valueOf(total)) > 0 || spent.compareTo(BIG_MAX) > 0) {
                throw new AssertionError("budget exceeded");
            }
        }
    }

    private static RetryPlan.Plan rejected() {
        return new RetryPlan.Plan(false, 0, 0);
    }

    private static LongUnaryOperator noDraw() {
        return bound -> { throw new AssertionError("unexpected draw"); };
    }

    private static LongUnaryOperator draw(long expectedBound, long value) {
        return bound -> {
            if (bound != expectedBound) throw new AssertionError("bound " + bound + " != " + expectedBound);
            return value;
        };
    }

    private static void expect(String label, RetryPlan.Plan expected, RetryPlan.Plan actual) {
        checks++;
        if (!expected.equals(actual)) throw new AssertionError(label + ": " + actual + " != " + expected);
    }
}
