import java.util.ArrayList;
import java.util.List;

public final class Verification {
    private static int checks;
    private static final Ports.Intent INTENT = new Ports.Intent("stable-intent", "fixed-payload");
    private static final Ports.Policy POLICY = new Ports.Policy(3, 10, 5, 4);

    private static void require(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private static final class Memory implements Ports.Store {
        Ports.Snapshot value;
        int writes;
        Memory(Ports.State state) { value = new Ports.Snapshot(state, "stored-receipt"); }
        public Ports.Snapshot snapshot() { return value; }
        public void unknown() { writes++; value = new Ports.Snapshot(Ports.State.UNKNOWN, null); }
        public void applied(String receipt) { writes++; value = new Ports.Snapshot(Ports.State.APPLIED, receipt); }
        public void rejected() { writes++; value = new Ports.Snapshot(Ports.State.REJECTED, null); }
    }

    private static final class Response implements Ports.Reply {
        final Ports.Kind kind;
        final long minimum;
        int closes;
        Response(Ports.Kind kind, long minimum) { this.kind = kind; this.minimum = minimum; }
        public Ports.Kind kind() { return kind; }
        public String receipt() { return "authoritative-receipt"; }
        public long serverMinimumNanos() { return minimum; }
        public void close() { closes++; }
    }

    private static final class Fixture implements Ports.Runtime, Ports.Operation {
        final Memory store;
        final Ports.Kind[] script;
        final List<Response> replies = new ArrayList<>();
        final List<Long> waits = new ArrayList<>();
        long remaining = 1000;
        long minimum;
        long cost;
        long oversleep;
        boolean cancelled;
        boolean cancelAfterReply;
        boolean cancelInSleep;
        boolean interruptInSleep;
        boolean flagInSleep;
        boolean interruptAfterReply;
        int interruptCall = -1;
        int tokens = 10;
        int grants;
        int admissions;
        int calls;
        long expectedTimeout = 10;
        long expectedLocalWait = 4;
        boolean unresolved;

        Fixture(Ports.State state, Ports.Kind... script) {
            store = new Memory(state);
            unresolved = state == Ports.State.UNKNOWN;
            this.script = script;
        }
        public long remainingNanos() { return remaining; }
        public boolean cancelled() { return cancelled; }
        public boolean acquireRetry() {
            closed();
            persisted();
            admissions++;
            if (tokens == 0) return false;
            tokens--;
            grants++;
            return true;
        }
        public void sleep(long nanos) throws InterruptedException {
            closed();
            persisted();
            waits.add(nanos);
            require(nanos >= minimum && nanos >= expectedLocalWait, "minimum wait");
            remaining = Math.max(0, remaining - nanos);
            remaining = Math.max(0, remaining - oversleep);
            if (cancelInSleep) cancelled = true;
            if (flagInSleep) Thread.currentThread().interrupt();
            if (interruptInSleep) throw new InterruptedException("sleep interrupted");
        }
        public Ports.Reply call(Ports.Intent intent, long timeout) throws InterruptedException {
            closed();
            persisted();
            require(intent == INTENT, "intent and payload must be reused unchanged");
            require(timeout == expectedTimeout, "timeout propagation");
            require(!cancelled && !Thread.currentThread().isInterrupted(), "late/cancelled call");
            int current = calls++;
            if (current == interruptCall) throw new InterruptedException("call interrupted");
            require(current < script.length, "unexpected extra call");
            remaining = Math.max(0, remaining - cost);
            Response reply = new Response(script[current], minimum);
            replies.add(reply);
            if (reply.kind == Ports.Kind.UNKNOWN) unresolved = true;
            if (cancelAfterReply) cancelled = true;
            if (interruptAfterReply) Thread.currentThread().interrupt();
            return reply;
        }
        void closed() {
            for (Response reply : replies) require(reply.closes == 1, "reply close exactly once before continuation");
        }
        void persisted() {
            if (unresolved) require(store.snapshot().state() == Ports.State.UNKNOWN, "UNKNOWN must be durable before retry/wait");
        }
        Ports.Result execute(Ports.Policy policy) throws InterruptedException {
            expectedTimeout = policy.attemptNanos();
            expectedLocalWait = policy.localWaitNanos();
            return RetryEngine.execute(INTENT, policy, store, this, this);
        }
        void expect(Ports.Policy policy, Ports.State state, int attempts, int sleeps, int budgetChecks)
                throws InterruptedException {
            Ports.Result result = execute(policy);
            closed();
            require(result.snapshot().state() == state, "wrong outcome: " + result);
            require(result.snapshot().equals(store.snapshot()), "returned snapshot differs from store");
            require(result.attempts() == attempts && calls == attempts, "wrong attempt count");
            require(waits.size() == sleeps && admissions == budgetChecks, "unexpected sleep or token acquisition");
            if (state == Ports.State.APPLIED && attempts > 0) {
                require("authoritative-receipt".equals(result.snapshot().receipt()), "invented receipt");
            }
            checks++;
        }
    }

    private static void interrupted(Fixture f, Ports.State expected, int calls) throws Exception {
        try {
            f.execute(POLICY);
            throw new AssertionError("interruption swallowed");
        } catch (InterruptedException expectedException) {
            f.closed();
            require(f.calls == calls, "calls after interruption");
            require(f.store.snapshot().state() == expected, "interruption erased uncertainty");
            checks++;
        } finally {
            Thread.interrupted();
        }
    }

    private static void sequences() throws Exception {
        // Enumerate reply histories independently of engine internals, including
        // UNKNOWN followed by known non-application and authoritative recovery.
        for (Ports.State initial : new Ports.State[] {Ports.State.PENDING, Ports.State.UNKNOWN}) {
            for (Ports.Kind first : Ports.Kind.values()) {
                for (Ports.Kind second : Ports.Kind.values()) {
                    for (Ports.Kind third : Ports.Kind.values()) {
                        Ports.Kind[] script = {first, second, third};
                        boolean unknown = initial == Ports.State.UNKNOWN;
                        Ports.State expected = unknown ? Ports.State.UNKNOWN : Ports.State.REJECTED;
                        int calls = 0;
                        if (!unknown || Contract.REPLAY_SAFE) {
                            for (Ports.Kind kind : script) {
                                calls++;
                                if (kind == Ports.Kind.OK) { expected = Ports.State.APPLIED; break; }
                                unknown |= kind == Ports.Kind.UNKNOWN;
                                expected = unknown ? Ports.State.UNKNOWN : Ports.State.REJECTED;
                                if (kind == Ports.Kind.PERMANENT || (unknown && !Contract.REPLAY_SAFE)) break;
                            }
                        }
                        int retries = Math.max(0, calls - 1);
                        new Fixture(initial, script).expect(POLICY, expected, calls, retries, retries);
                    }
                }
            }
        }
    }

    public static void main(String[] args) throws Exception {
        sequences();
        for (Ports.State state : new Ports.State[] {Ports.State.APPLIED, Ports.State.REJECTED}) {
            Fixture f = new Fixture(state);
            f.cancelled = true;
            f.remaining = 0;
            f.expect(POLICY, state, 0, 0, 0);
            require(f.store.writes == 0 && "stored-receipt".equals(f.store.snapshot().receipt()), "terminal snapshot changed");
        }
        for (Ports.State initial : new Ports.State[] {Ports.State.PENDING, Ports.State.UNKNOWN}) {
            Ports.State stopped = initial == Ports.State.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.REJECTED;
            Fixture cancelled = new Fixture(initial);
            cancelled.cancelled = true;
            cancelled.expect(POLICY, stopped, 0, 0, 0);
            Fixture tooShort = new Fixture(initial);
            tooShort.remaining = 14;
            tooShort.expect(POLICY, stopped, 0, 0, 0);
            Fixture preInterrupted = new Fixture(initial);
            Thread.currentThread().interrupt();
            interrupted(preInterrupted, initial, 0);
        }
        Fixture exact = new Fixture(Ports.State.PENDING, Ports.Kind.OK);
        exact.remaining = 15;
        exact.expect(POLICY, Ports.State.APPLIED, 1, 0, 0);
        Fixture overflow = new Fixture(Ports.State.PENDING);
        overflow.remaining = Long.MAX_VALUE;
        overflow.expect(new Ports.Policy(3, Long.MAX_VALUE, 1, 4), Ports.State.REJECTED, 0, 0, 0);
        Fixture hugeExact = new Fixture(Ports.State.PENDING, Ports.Kind.OK);
        hugeExact.remaining = Long.MAX_VALUE;
        hugeExact.expect(new Ports.Policy(1, Long.MAX_VALUE, 0, 4), Ports.State.APPLIED, 1, 0, 0);

        for (Ports.Kind first : new Ports.Kind[] {Ports.Kind.TRANSIENT, Ports.Kind.UNKNOWN}) {
            Ports.State stopped = first == Ports.Kind.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.REJECTED;
            Fixture denied = new Fixture(Ports.State.PENDING, first);
            denied.tokens = 0;
            denied.expect(POLICY, stopped, 1, 0, 1);
            Fixture count = new Fixture(Ports.State.PENDING, first);
            count.expect(new Ports.Policy(1, 10, 5, 4), stopped, 1, 0, 0);
            Fixture minimum = new Fixture(Ports.State.PENDING, first, Ports.Kind.OK);
            minimum.minimum = 20;
            minimum.remaining = 35;
            minimum.expect(POLICY, Ports.State.APPLIED, 2, 1, 1);
            require(minimum.waits.equals(List.of(20L)), "server minimum truncated");
            Fixture local = new Fixture(Ports.State.PENDING, first, Ports.Kind.OK);
            local.minimum = 1;
            local.expect(POLICY, Ports.State.APPLIED, 2, 1, 1);
            require(local.waits.equals(List.of(4L)), "local delay changed");
            Fixture noRoom = new Fixture(Ports.State.PENDING, first);
            noRoom.remaining = 34;
            noRoom.minimum = 20;
            noRoom.expect(POLICY, stopped, 1, 0, 0);
            Fixture hugeWait = new Fixture(Ports.State.PENDING, first);
            hugeWait.remaining = Long.MAX_VALUE;
            hugeWait.minimum = Long.MAX_VALUE;
            hugeWait.expect(POLICY, stopped, 1, 0, 0);
            Fixture spent = new Fixture(Ports.State.PENDING, first);
            spent.remaining = 30;
            spent.cost = 12;
            spent.expect(POLICY, stopped, 1, 0, 0);
            Fixture over = new Fixture(Ports.State.PENDING, first);
            over.remaining = 20;
            over.oversleep = 2;
            over.expect(POLICY, stopped, 1, 1, 1);
            require(over.grants == 1, "spent admission must not be refunded");
            Fixture after = new Fixture(Ports.State.PENDING, first);
            after.cancelAfterReply = true;
            after.expect(POLICY, stopped, 1, 0, 0);
            Fixture waking = new Fixture(Ports.State.PENDING, first);
            waking.cancelInSleep = true;
            waking.expect(POLICY, stopped, 1, 1, 1);
            Fixture sleepInterrupt = new Fixture(Ports.State.PENDING, first);
            sleepInterrupt.interruptInSleep = true;
            interrupted(sleepInterrupt, first == Ports.Kind.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.PENDING, 1);
            Fixture flagged = new Fixture(Ports.State.PENDING, first);
            flagged.flagInSleep = true;
            interrupted(flagged, first == Ports.Kind.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.PENDING, 1);
            Fixture replyInterrupted = new Fixture(Ports.State.PENDING, first);
            replyInterrupted.interruptAfterReply = true;
            interrupted(replyInterrupted, first == Ports.Kind.UNKNOWN ? Ports.State.UNKNOWN : Ports.State.PENDING, 1);
        }
        Fixture callInterrupt = new Fixture(Ports.State.PENDING);
        callInterrupt.interruptCall = 0;
        interrupted(callInterrupt, Ports.State.UNKNOWN, 1);
        Fixture retryInterrupt = new Fixture(Ports.State.UNKNOWN, Ports.Kind.TRANSIENT);
        retryInterrupt.interruptCall = 1;
        interrupted(retryInterrupt, Ports.State.UNKNOWN, 2);
        Fixture zero = new Fixture(Ports.State.PENDING, Ports.Kind.TRANSIENT, Ports.Kind.OK);
        zero.expect(new Ports.Policy(2, 10, 0, 0), Ports.State.APPLIED, 2, 1, 1);
        require(zero.waits.equals(List.of(0L)), "zero wait changed");
        System.out.println("Verification passed: " + checks + " scenarios; replaySafe=" + Contract.REPLAY_SAFE);
    }
}
