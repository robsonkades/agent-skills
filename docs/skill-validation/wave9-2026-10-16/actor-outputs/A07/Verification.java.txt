import java.math.BigInteger;
import java.util.Random;

/** Deterministic checks of the actual RetryPlan, with an arbitrary-precision oracle. */
public final class Verification {
    private static int cases;
    private static int retryDraws;

    public static void main(String[] args) {
        check("initial keeps reserve and ignores server advice", 0, 0, 100, 25, 10,
                100, 2, 8, 0, 3, Long.MAX_VALUE, 0);
        check("first retry uses base window", 0, 0, 100, 5, 10,
                100, 7, 100, 1, 3, 0, 6);
        check("local draw dominates", 0, 0, 100, 5, 10,
                100, 7, 100, 1, 3, 2, 6);
        check("server minimum dominates", 0, 0, 100, 5, 10,
                100, 7, 100, 1, 3, 20, 6);
        check("server advice is not capped by local delay cap", 0, 0, 100, 5, 10,
                100, 2, 3, 1, 3, 50, 1);
        check("attempt cap remains exact", 0, 0, 100, 5, 10,
                12, 7, 100, 1, 3, 20, 6);
        check("exact work wait reserve fit", 0, 0, 30, 5, 10,
                100, 7, 100, 1, 3, 15, 6);
        check("one nanosecond short", 0, 0, 29, 5, 10,
                100, 7, 100, 1, 3, 15, 6);
        check("server advice too long is not truncated", 0, 0, 30, 5, 10,
                100, 7, 100, 1, 3, Long.MAX_VALUE, 0);
        check("reached total call limit", 0, 0, 100, 0, 1,
                100, 1, 8, 3, 3, 0, 0);
        check("beyond total call limit", 0, 0, 100, 0, 1,
                100, 1, 8, 4, 3, 0, 0);
        check("initial allowed with one total call", 0, 0, 1, 0, 1,
                1, 1, 1, 0, 1, Long.MAX_VALUE, 0);
        check("expired exactly", -100, 50, 50, 0, 1,
                10, 1, 1, 0, 1, 0, 0);
        check("expired beyond budget", -100, 51, 50, 0, 1,
                10, 1, 1, 1, 2, 0, 0);
        check("negative clock readings", -100, 20, 100, 10, 1,
                100, 1, 1, 0, 1, 0, 0);
        check("signed clock wrap", Long.MAX_VALUE - 2, 10, 100, 5, 1,
                100, 5, 100, 1, 3, 2, 4);
        check("maximum elapsed duration", Long.MAX_VALUE - 2, Long.MAX_VALUE,
                Long.MAX_VALUE, 0, 1, 1, 1, 1, 0, 1, 0, 0);
        check("maximum budget with one nanosecond left", Long.MIN_VALUE,
                Long.MAX_VALUE - 1, Long.MAX_VALUE, 0, 1, 1, 1, 1, 0, 1, 0, 0);
        check("maximum work exact fit", -1, 0, Long.MAX_VALUE, 0, Long.MAX_VALUE,
                Long.MAX_VALUE, 1, 1, 0, 1, 0, 0);
        check("reserve plus minimum would overflow", 0, 0, Long.MAX_VALUE, Long.MAX_VALUE,
                1, Long.MAX_VALUE, 1, 1, 0, 1, 0, 0);
        check("huge reserve leaves exact work", 0, 0, Long.MAX_VALUE, Long.MAX_VALUE - 1,
                1, Long.MAX_VALUE, 1, 1, 0, 1, 0, 0);
        check("wait plus reserve would overflow", 0, 0, Long.MAX_VALUE, 2, 1,
                Long.MAX_VALUE, 1, 1, 1, 2, Long.MAX_VALUE, 0);
        check("large untruncated wait exact fit", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, 1, 1, 1, 2, Long.MAX_VALUE - 1, 0);
        check("base already beyond cap", 0, 0, 100, 0, 1,
                100, Long.MAX_VALUE, 10, 1, 3, 0, 9);
        check("multiplication saturation", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, Long.MAX_VALUE, Long.MAX_VALUE, 2, 3, 0, 7);
        check("power sixty two stays representable", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, 1, Long.MAX_VALUE, 63, 64, 0, Long.MAX_VALUE);
        check("power sixty three saturates", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, 1, Long.MAX_VALUE, 64, 65, 0, Long.MAX_VALUE);
        check("shift distance does not wrap at sixty four", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, 1, Long.MAX_VALUE, 65, 66, 0, Long.MAX_VALUE);
        check("largest admissible attempt count", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, 1, 1000, Integer.MAX_VALUE - 1, Integer.MAX_VALUE, 0, 999);
        check("largest completed count stops", 0, 0, Long.MAX_VALUE, 0, 1,
                Long.MAX_VALUE, 1, 1000, Integer.MAX_VALUE, Integer.MAX_VALUE, 0, 999);

        // Exhaustive small-domain budgets include zero/last draw, exact fit and rejection.
        for (int total = 1; total <= 8; total++) {
            for (int elapsed = 0; elapsed <= 9; elapsed++) {
                for (int reserve = 0; reserve <= 9; reserve++) {
                    for (int minimum = 1; minimum <= 3; minimum++) {
                        for (int completed = 0; completed <= 3; completed++) {
                            for (int server = 0; server <= 9; server += 3) {
                                for (long sample : new long[] {0, Long.MAX_VALUE}) {
                                    check("small domain", Long.MAX_VALUE - 3, elapsed, total,
                                            reserve, minimum, 5, 2, 5, completed, 3, server, sample);
                                }
                            }
                        }
                    }
                }
            }
        }

        Random random = new Random(0xA07C0FFEE17L);
        for (int i = 0; i < 12000; i++) {
            long total = positive(random);
            long elapsed = switch (i % 4) {
                case 0 -> nonnegative(random) % total;
                case 1 -> total - 1;
                case 2 -> total;
                default -> nonnegative(random);
            };
            long cap = positive(random);
            long minimum = 1 + nonnegative(random) % cap;
            int[] counts = {0, 1, 2, 62, 63, 64, 65, 1024, Integer.MAX_VALUE - 1};
            int completed = counts[random.nextInt(counts.length)];
            int maxAttempts = i % 7 == 0 ? Math.max(1, completed) : completed + 1;
            check("seeded boundaries " + i, random.nextLong(), elapsed, total,
                    i % 3 == 0 ? 0 : nonnegative(random), minimum, cap,
                    positive(random), positive(random), completed, maxAttempts,
                    i % 3 == 0 ? 0 : nonnegative(random), nonnegative(random));
        }
        System.out.println("Verification passed: " + cases + " plans, " + retryDraws
                + " retry draws; explicit, exhaustive-small and seeded-large cases.");
    }

    private static void check(String label, long start, long elapsed, long total,
            long reserve, long minimumWork, long attemptCap, long base, long delayCap,
            int completed, int maxAttempts, long serverMinimum, long sample) {
        // BigInteger keeps the model's sum/difference independent of long overflow.
        // For any positive base, exponent >= 63 necessarily exceeds every long cap.
        long expectedBound = BigInteger.valueOf(base)
                .shiftLeft(Math.min(Math.max(completed - 1, 0), 63))
                .min(BigInteger.valueOf(delayCap)).longValueExact();
        long drawn = sample == Long.MAX_VALUE ? expectedBound - 1 : sample % expectedBound;
        long wait = completed == 0 ? 0 : Math.max(drawn, serverMinimum);
        BigInteger available = BigInteger.valueOf(total).subtract(BigInteger.valueOf(elapsed))
                .subtract(BigInteger.valueOf(reserve)).subtract(BigInteger.valueOf(wait));
        boolean admitted = completed < maxAttempts
                && available.compareTo(BigInteger.valueOf(minimumWork)) >= 0;
        RetryPlan.Plan expected = admitted
                ? new RetryPlan.Plan(true, wait,
                        available.min(BigInteger.valueOf(attemptCap)).longValueExact())
                : new RetryPlan.Plan(false, 0, 0);
        int[] calls = {0};
        RetryPlan.Plan actual = RetryPlan.next(start, start + elapsed, total, reserve,
                minimumWork, attemptCap, base, delayCap, completed, maxAttempts,
                serverMinimum, bound -> {
                    if (completed == 0) throw new AssertionError(label + ": drew on initial call");
                    if (++calls[0] != 1) throw new AssertionError(label + ": multiple draws");
                    if (bound != expectedBound) throw new AssertionError(label
                            + ": jitter bound expected " + expectedBound + " but got " + bound);
                    retryDraws++;
                    return drawn;
                });
        if (!actual.equals(expected)) {
            throw new AssertionError(label + ": expected " + expected + " but got " + actual);
        }
        if (admitted && completed > 0 && calls[0] != 1) {
            throw new AssertionError(label + ": admitted retry must use supplied draw");
        }
        cases++;
    }

    private static long nonnegative(Random random) {
        long[] edges = {0, 1, 2, 3, 7, 63, Long.MAX_VALUE - 1, Long.MAX_VALUE};
        return random.nextBoolean() ? edges[random.nextInt(edges.length)]
                : random.nextLong() & Long.MAX_VALUE;
    }

    private static long positive(Random random) {
        long value = nonnegative(random);
        return value == 0 ? 1 : value;
    }
}
