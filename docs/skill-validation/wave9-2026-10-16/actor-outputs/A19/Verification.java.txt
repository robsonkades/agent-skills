import java.math.BigInteger;
import java.util.Random;
import java.util.function.LongUnaryOperator;

public final class Verification {
    private static int checks;
    private static final LongUnaryOperator NO_DRAW = bound -> {
        throw new AssertionError("Randomness must not be drawn here");
    };

    public static void main(String[] args) {
        examples();
        boundaryMatrix();
        generatedCases();
        System.out.println("Verification passed: " + checks + " cases");
    }

    private static void examples() {
        expect("initial ignores server advice and keeps reserve",
                new RetryPlan.Plan(true, 0, 80),
                RetryPlan.next(-20, -10, 100, 10, 1, 100, 2, 8, 0, 3,
                        Long.MAX_VALUE, NO_DRAW));
        expect("attempt limit includes the first call", new RetryPlan.Plan(false, 0, 0),
                RetryPlan.next(0, 0, 100, 0, 1, 10, 2, 8, 3, 3, 0, NO_DRAW));
        expect("past attempt limit", new RetryPlan.Plan(false, 0, 0),
                RetryPlan.next(0, 0, 100, 0, 1, 10, 2, 8, 4, 3, 0, NO_DRAW));
        expect("elapsed exact expiry", new RetryPlan.Plan(false, 0, 0),
                RetryPlan.next(30, 130, 100, 0, 1, 10, 2, 8, 1, 3, 0, NO_DRAW));
        expect("huge reserve rejects without overflowing", new RetryPlan.Plan(false, 0, 0),
                RetryPlan.next(0, 0, 100, Long.MAX_VALUE, 10, 10, 2, 8,
                        1, 3, 0, NO_DRAW));
        expect("server advice cannot fit", new RetryPlan.Plan(false, 0, 0),
                RetryPlan.next(0, 0, Long.MAX_VALUE, 0, 1, Long.MAX_VALUE, 2, 8,
                        1, 3, Long.MAX_VALUE, NO_DRAW));
        compare("first retry and server minimum", 0, 0, 100, 10, 20, 100,
                8, 100, 1, 3, 12, 7);
        compare("local draw wins", 0, 0, 100, 10, 20, 100,
                8, 100, 2, 3, 2, 15);
        compare("exact fit admitted", 0, 0, 40, 7, 20, 100,
                8, 100, 1, 3, 13, 0);
        compare("one nanosecond short", 0, 0, 39, 7, 20, 100,
                8, 100, 1, 3, 13, 0);
        compare("attempt cap wins", 0, 0, 100, 10, 2, 5,
                8, 100, 1, 3, 0, 7);
        compare("zero jitter remains zero", 0, 0, 100, 10, 20, 100,
                8, 100, 1, 3, 0, 0);
        compare("signed clock wrap", Long.MAX_VALUE - 5, 20, 100, 10, 20, 100,
                8, 100, 1, 3, 0, 7);
        compare("maximum elapsed expires", 1, Long.MAX_VALUE, Long.MAX_VALUE,
                0, 1, 1, 1, 1, 0, 1, 0, 0);
        compare("large reserve exact fit", -1, 0, Long.MAX_VALUE,
                Long.MAX_VALUE - 1, 1, 1, 1, 1, 0, 1, Long.MAX_VALUE, 0);
        compare("large server wait exact fit", -1, 0, Long.MAX_VALUE,
                0, 1, Long.MAX_VALUE, 1, 1, 1, 2, Long.MAX_VALUE - 1, 0);
        compare("jitter cannot be shortened to admit", 0, 0, 20,
                0, 10, 10, 16, 16, 1, 3, 0, 15);
    }

    private static void boundaryMatrix() {
        long[] sizes = {1, 2, 3, 7, 1L << 31, 1L << 62, Long.MAX_VALUE};
        int[] attempts = {1, 2, 3, 31, 32, 62, 63, 64, 65, 127,
                Integer.MAX_VALUE - 1};
        for (long base : sizes) {
            for (long cap : sizes) {
                for (int completed : attempts) {
                    long bound = window(base, cap, completed);
                    for (long draw : new long[] {0, bound / 2, bound - 1}) {
                        compare("window boundaries", Long.MAX_VALUE - 2, 3,
                                Long.MAX_VALUE, 0, 1, Long.MAX_VALUE,
                                base, cap, completed, Integer.MAX_VALUE, 0, draw);
                    }
                }
            }
        }
    }

    private static void generatedCases() {
        Random random = new Random(190019L);
        long[] edges = {0, 1, 2, 10, (1L << 62) - 1, 1L << 62,
                Long.MAX_VALUE - 1, Long.MAX_VALUE};
        int[] attempts = {0, 1, 2, 3, 31, 62, 63, 64, 65,
                Integer.MAX_VALUE - 1, Integer.MAX_VALUE};
        for (int i = 0; i < 12000; i++) {
            long start = random.nextLong();
            long elapsed = nonnegative(random, edges);
            long total = positive(random, edges);
            long reserve = nonnegative(random, edges);
            long attemptCap = positive(random, edges);
            long minimum = Math.min(attemptCap, positive(random, edges));
            long base = positive(random, edges);
            long cap = positive(random, edges);
            int completed = attempts[random.nextInt(attempts.length)];
            int max = Math.max(1, attempts[random.nextInt(attempts.length)]);
            long server = nonnegative(random, edges);
            long bound = completed == 0 ? 1 : window(base, cap, completed);
            long draw = (random.nextLong() & Long.MAX_VALUE) % bound;
            compare("generated " + i, start, elapsed, total, reserve, minimum,
                    attemptCap, base, cap, completed, max, server, draw);
        }
    }

    private static long nonnegative(Random random, long[] edges) {
        return random.nextBoolean() ? edges[random.nextInt(edges.length)]
                : random.nextLong() & Long.MAX_VALUE;
    }

    private static long positive(Random random, long[] edges) {
        return Math.max(1, nonnegative(random, edges));
    }

    private static long window(long base, long cap, int completed) {
        // All positive long caps are exceeded by 2^63; larger exponents have the same minimum.
        return BigInteger.valueOf(base).shiftLeft(Math.min(completed - 1, 63))
                .min(BigInteger.valueOf(cap)).longValueExact();
    }

    private static void compare(String label, long start, long elapsed, long total,
            long reserve, long minimum, long attemptCap, long base, long cap,
            int completed, int max, long server, long draw) {
        BigInteger remaining = BigInteger.valueOf(total).subtract(BigInteger.valueOf(elapsed));
        long wait = completed == 0 ? 0 : Math.max(draw, server);
        BigInteger allocation = remaining.subtract(BigInteger.valueOf(wait))
                .subtract(BigInteger.valueOf(reserve)).min(BigInteger.valueOf(attemptCap));
        boolean admitted = completed < max && allocation.compareTo(BigInteger.valueOf(minimum)) >= 0;
        RetryPlan.Plan expected = admitted ? new RetryPlan.Plan(true, wait, allocation.longValueExact())
                : new RetryPlan.Plan(false, 0, 0);
        int[] draws = {0};
        LongUnaryOperator source = bound -> {
            if (completed == 0) throw new AssertionError(label + ": initial draw");
            if (bound != window(base, cap, completed)) {
                throw new AssertionError(label + ": wrong jitter bound " + bound);
            }
            if (draw < 0 || draw >= bound) throw new AssertionError("Invalid test draw");
            if (++draws[0] > 1) throw new AssertionError(label + ": multiple draws");
            return draw;
        };
        RetryPlan.Plan actual = RetryPlan.next(start, start + elapsed, total, reserve,
                minimum, attemptCap, base, cap, completed, max, server, source);
        expect(label, expected, actual);
        if (admitted && completed > 0 && draws[0] != 1) {
            throw new AssertionError(label + ": admitted retry did not draw exactly once");
        }
    }

    private static void expect(String label, RetryPlan.Plan expected, RetryPlan.Plan actual) {
        if (!expected.equals(actual)) {
            throw new AssertionError(label + ": expected " + expected + ", got " + actual);
        }
        checks++;
    }
}
