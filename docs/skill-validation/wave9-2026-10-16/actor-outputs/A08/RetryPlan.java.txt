import java.util.function.LongUnaryOperator;
public final class RetryPlan {
    public record Plan(boolean admitted, long waitNanos, long attemptNanos) {}
    public static Plan next(long start, long now, long total, long reserve, long minimumWork,
            long attemptCap, long base, long delayCap, int completed, int maxAttempts,
            long serverMinimum, LongUnaryOperator draw) {
        if (completed >= maxAttempts) return new Plan(false, 0, 0);

        // Subtraction preserves elapsed time across nanoTime's signed wrap.
        long elapsed = now - start;
        if (elapsed >= total) return new Plan(false, 0, 0);
        long remaining = total - elapsed;
        if (reserve > remaining) return new Plan(false, 0, 0);
        long available = remaining - reserve;
        if (available < minimumWork) return new Plan(false, 0, 0);

        long wait = 0;
        if (completed > 0) {
            long maximumWait = available - minimumWork;
            if (serverMinimum > maximumWait) return new Plan(false, 0, 0);

            int exponent = completed - 1;
            // Compare before shifting; Java masks large shift counts and can overflow.
            long window = exponent >= 63 || base > (delayCap >> exponent)
                    ? delayCap : base << exponent;
            wait = Math.max(draw.applyAsLong(window), serverMinimum);
            if (wait > maximumWait) return new Plan(false, 0, 0);
        }
        return new Plan(true, wait, Math.min(attemptCap, available - wait));
    }
}
