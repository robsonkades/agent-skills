import java.math.BigInteger;
import java.util.function.LongUnaryOperator;

public final class Verification {
    private static final BigInteger TWO_64 = BigInteger.ONE.shiftLeft(64);
    private static long checked;

    private record Input(long start, long elapsed, long total, long reserve,
            long minimumWork, long attemptCap, long base, long delayCap,
            int completed, int maxAttempts, long serverMinimum) {}

    public static void main(String[] args) {
        explicitExamples();
        smallDomain();
        largeBoundaries();
        System.out.println("Verification passed: " + checked
                + " actual RetryPlan calls; exact plans, draw bounds, and budget invariants checked");
    }

    private static void explicitExamples() {
        expect(new RetryPlan.Plan(true, 0, 70), 0,
                new Input(-100, 20, 100, 10, 10, 90, 2, 8, 0, 3, Long.MAX_VALUE), 2);
        expect(new RetryPlan.Plan(true, 7, 83), 1,
                new Input(0, 0, 100, 10, 10, 90, 4, 100, 1, 3, 7), 2);
        expect(new RetryPlan.Plan(true, 7, 83), 1,
                new Input(0, 0, 100, 10, 10, 90, 8, 100, 1, 3, 2), 2);
        expect(new RetryPlan.Plan(true, 7, 10), 1,
                new Input(0, 0, 20, 3, 10, 100, 4, 8, 1, 3, 7), 0);
        expect(new RetryPlan.Plan(false, 0, 0), 0,
                new Input(0, 0, 19, 3, 10, 100, 4, 8, 1, 3, 7), 0);
        expect(new RetryPlan.Plan(false, 0, 0), 1,
                new Input(0, 0, 20, 3, 10, 100, 9, 9, 1, 3, 0), 2);
        expect(new RetryPlan.Plan(false, 0, 0), 0,
                new Input(0, 0, 100, 0, 1, 10, 1, 10, 3, 3, 0), 0);
        expect(new RetryPlan.Plan(false, 0, 0), 0,
                new Input(0, 0, 100, Long.MAX_VALUE, 1, 10, 1, 10, 1, 3, 1), 0);
        expect(new RetryPlan.Plan(false, 0, 0), 0,
                new Input(0, 0, Long.MAX_VALUE, 0, 1, 10, 1, 10, 1, 3, Long.MAX_VALUE), 0);
        expect(new RetryPlan.Plan(true, 0, 17), 0,
                new Input(Long.MAX_VALUE - 2, 5, 25, 3, 1, 30, 1, 10, 0, 3, 0), 0);
        expect(new RetryPlan.Plan(true, 0, 1), 0,
                new Input(Long.MIN_VALUE, Long.MAX_VALUE - 1, Long.MAX_VALUE,
                        0, 1, Long.MAX_VALUE, 1, 1, 0, 1, 0), 0);
        expect(new RetryPlan.Plan(true, Long.MAX_VALUE - 1, 1), 1,
                new Input(-1, 0, Long.MAX_VALUE, 0, 1, Long.MAX_VALUE,
                        1, Long.MAX_VALUE, Integer.MAX_VALUE - 1, Integer.MAX_VALUE, 0), 2);
        expect(new RetryPlan.Plan(false, 0, 0), 0,
                new Input(Long.MAX_VALUE, Long.MAX_VALUE, Long.MAX_VALUE,
                        0, 1, 1, 1, 1, 1, 2, 0), 0);
    }

    private static void smallDomain() {
        for (long total : new long[] {1, 8, 19, 64})
        for (long elapsed : new long[] {0, 1, 8, 65})
        for (long reserve : new long[] {0, 3, 63})
        for (long minimum : new long[] {1, 4})
        for (long cap : new long[] {minimum, 20})
        for (long base : new long[] {1, 3, 9})
        for (long delayCap : new long[] {1, 5, 17})
        for (int completed : new int[] {0, 1, 2, 3, 4})
        for (long advice : new long[] {0, 2, 18})
        for (int drawMode = 0; drawMode < 3; drawMode++) {
            verify(new Input(-7, elapsed, total, reserve, minimum, cap,
                    base, delayCap, completed, 4, advice), drawMode, null, -1);
        }
    }

    private static void largeBoundaries() {
        long[] positive = {1, 2, 3, (1L << 31) - 1, 1L << 31,
                (1L << 62) - 1, 1L << 62, (1L << 62) + 1, Long.MAX_VALUE};
        int[] completedValues = {1, 2, 31, 32, 62, 63, 64, 65, 127, 128,
                Integer.MAX_VALUE - 1};
        for (long base : positive)
        for (long cap : positive)
        for (int completed : completedValues)
        for (int mode = 0; mode < 3; mode++) {
            verify(new Input(Long.MAX_VALUE, 0, Long.MAX_VALUE, 0, 1,
                    Long.MAX_VALUE, base, cap, completed, Integer.MAX_VALUE, 0),
                    mode, null, -1);
        }

        long[] starts = {Long.MIN_VALUE, Long.MIN_VALUE + 1, -17, -1, 0,
                Long.MAX_VALUE - 2, Long.MAX_VALUE};
        long[] elapsedValues = {0, 1, 3, 1L << 62, Long.MAX_VALUE - 1, Long.MAX_VALUE};
        long[] reserves = {0, 1, 1L << 62, Long.MAX_VALUE - 1, Long.MAX_VALUE};
        for (long start : starts)
        for (long elapsed : elapsedValues)
        for (long total : positive)
        for (long reserve : reserves)
        for (long minimum : positive)
        for (int completed : new int[] {0, 1})
        for (long advice : new long[] {0, Long.MAX_VALUE - 1, Long.MAX_VALUE}) {
            verify(new Input(start, elapsed, total, reserve, minimum,
                    Long.MAX_VALUE, Long.MAX_VALUE, Long.MAX_VALUE,
                    completed, 2, advice), 2, null, -1);
        }
    }

    private static void expect(RetryPlan.Plan plan, int draws, Input input, int mode) {
        verify(input, mode, plan, draws);
    }

    private static void verify(Input in, int mode, RetryPlan.Plan explicit, int draws) {
        long now = in.start() + in.elapsed();
        long bound = expectedBound(in);
        int[] calls = {0};
        LongUnaryOperator draw = actualBound -> {
            calls[0]++;
            if (actualBound <= 0 || actualBound != bound) {
                throw new AssertionError("Wrong draw bound: " + actualBound + " expected " + bound + " " + in);
            }
            return sample(mode, actualBound);
        };
        RetryPlan.Plan actual = RetryPlan.next(in.start(), now, in.total(), in.reserve(),
                in.minimumWork(), in.attemptCap(), in.base(), in.delayCap(),
                in.completed(), in.maxAttempts(), in.serverMinimum(), draw);
        RetryPlan.Plan expected = reference(in, now, mode, bound);
        if (!actual.equals(expected) || (explicit != null && !actual.equals(explicit))) {
            throw new AssertionError(in + " mode=" + mode + " actual=" + actual
                    + " expected=" + expected + " explicit=" + explicit);
        }
        if (calls[0] > 1 || (in.completed() == 0 && calls[0] != 0)
                || (actual.admitted() && in.completed() > 0 && calls[0] != 1)
                || (draws >= 0 && calls[0] != draws)) {
            throw new AssertionError("Unexpected draw count " + calls[0] + " " + in);
        }
        if (actual.admitted()) {
            BigInteger consumed = bi(in.elapsed()).add(bi(in.reserve()))
                    .add(bi(actual.waitNanos())).add(bi(actual.attemptNanos()));
            if (consumed.compareTo(bi(in.total())) > 0 || actual.waitNanos() < 0
                    || actual.attemptNanos() < in.minimumWork()
                    || actual.attemptNanos() > in.attemptCap()) {
                throw new AssertionError("Budget invariant failed " + in);
            }
        }
        checked++;
    }

    private static RetryPlan.Plan reference(Input in, long now, int mode, long bound) {
        if (in.completed() >= in.maxAttempts()) return new RetryPlan.Plan(false, 0, 0);
        // Compute elapsed using unbounded arithmetic modulo the unsigned clock range.
        BigInteger elapsed = bi(now).subtract(bi(in.start())).mod(TWO_64);
        long wait = in.completed() == 0 ? 0 : Math.max(sample(mode, bound), in.serverMinimum());
        BigInteger work = bi(in.total()).subtract(elapsed).subtract(bi(in.reserve())).subtract(bi(wait));
        if (work.compareTo(bi(in.minimumWork())) < 0) return new RetryPlan.Plan(false, 0, 0);
        return new RetryPlan.Plan(true, wait, work.min(bi(in.attemptCap())).longValueExact());
    }

    private static long expectedBound(Input in) {
        if (in.completed() == 0) return 1; // Never observed on the initial call.
        int exponent = in.completed() - 1;
        // Every positive base times 2^63 already exceeds every valid delay cap.
        if (exponent >= 63) return in.delayCap();
        return bi(in.base()).shiftLeft(exponent).min(bi(in.delayCap())).longValueExact();
    }

    private static long sample(int mode, long bound) {
        return mode == 0 ? 0 : mode == 1 ? bound / 2 : bound - 1;
    }

    private static BigInteger bi(long value) {
        return BigInteger.valueOf(value);
    }
}
