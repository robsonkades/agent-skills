import java.util.ArrayList;
import java.util.List;

public final class Verification {
    static final Ports.Intent INTENT = new Ports.Intent("intent-1", "stable-payload");
    static final Ports.Policy POLICY = new Ports.Policy(3, 10, 5, 4);
    static int passed;
    static int failed;

    interface Check { void run() throws Exception; }
    static void test(String name, Check check) {
        try { check.run(); passed++; System.out.println("PASS " + name); }
        catch (Throwable e) { failed++; System.out.println("FAIL " + name + ": " + e); }
        finally { Thread.interrupted(); }
    }
    static void require(boolean ok, String message) {
        if (!ok) throw new AssertionError(message);
    }

    static final class Fixture implements Ports.Store, Ports.Runtime, Ports.Operation {
        Ports.Snapshot state = new Ports.Snapshot(Ports.State.PENDING, null);
        final List<Ports.Kind> kinds = new ArrayList<>();
        final List<Long> advice = new ArrayList<>();
        final List<Long> waits = new ArrayList<>();
        final List<String> events = new ArrayList<>();
        long remaining = 1000;
        long oversleep;
        int calls, closes, open, grants, mutations;
        boolean cancelled, denyToken, cancelOnSleep, cancelOnGrant;
        boolean interruptCall, interruptSleep, interruptAfterReply, effectApplied;
        long callCost;
        long expectedTimeout = 10;

        Fixture add(Ports.Kind kind, long minimum) { kinds.add(kind); advice.add(minimum); return this; }
        public Ports.Snapshot snapshot() { return state; }
        public void unknown() { mutations++; events.add("UNKNOWN"); state = new Ports.Snapshot(Ports.State.UNKNOWN, null); }
        public void applied(String receipt) { mutations++; events.add("APPLIED"); state = new Ports.Snapshot(Ports.State.APPLIED, receipt); }
        public void rejected() { mutations++; events.add("REJECTED"); state = new Ports.Snapshot(Ports.State.REJECTED, null); }
        public long remainingNanos() { return remaining; }
        public boolean cancelled() { return cancelled; }
        public boolean acquireRetry() {
            require(open == 0, "response open at token acquisition");
            grants++; events.add("token");
            if (cancelOnGrant) cancelled = true;
            return !denyToken;
        }
        public void sleep(long nanos) throws InterruptedException {
            require(open == 0, "response open during sleep");
            waits.add(nanos); events.add("sleep");
            if (interruptSleep) throw new InterruptedException("sleep interrupted");
            remaining = nanos >= remaining ? 0 : remaining - nanos;
            remaining = oversleep >= remaining ? 0 : remaining - oversleep;
            if (cancelOnSleep) cancelled = true;
        }
        public Ports.Reply call(Ports.Intent intent, long timeout) throws InterruptedException {
            require(open == 0, "response open at next call");
            require(!cancelled, "called while cancelled");
            require(!Thread.currentThread().isInterrupted(), "called while interrupted");
            require(INTENT.equals(intent), "intent identity/payload changed");
            require(timeout == expectedTimeout, "wrong attempt timeout");
            int index = calls++;
            events.add("call");
            require(index < kinds.size() || interruptCall, "unexpected call");
            remaining = callCost >= remaining ? 0 : remaining - callCost;
            if (interruptCall) { effectApplied = true; throw new InterruptedException("lost reply after apply"); }
            open++;
            if (interruptAfterReply) Thread.currentThread().interrupt();
            return new Ports.Reply() {
                boolean closed;
                public Ports.Kind kind() { return kinds.get(index); }
                public String receipt() { return "original-receipt"; }
                public long serverMinimumNanos() { return advice.get(index); }
                public void close() {
                    require(!closed, "double close"); closed = true; closes++; open--; events.add("close");
                }
            };
        }
        Ports.Result run() throws Exception { return run(POLICY); }
        Ports.Result run(Ports.Policy policy) throws Exception {
            Ports.Result result = RetryEngine.execute(INTENT, policy, this, this, this);
            require(result.attempts() == calls, "incorrect invocation attempt count");
            require(result.snapshot().equals(state), "result/store mismatch");
            require(open == 0 && closes == calls, "unclosed response on normal return");
            return result;
        }
        void state(Ports.State expected) { require(state.state() == expected, "expected " + expected + ", got " + state); }
    }

    public static void main(String[] args) {
        test("authoritative OK and stable identity", () -> {
            Fixture f = new Fixture().add(Ports.Kind.OK, Long.MAX_VALUE); f.run();
            f.state(Ports.State.APPLIED); require("original-receipt".equals(f.state.receipt()), "receipt lost");
            require(f.grants == 0 && f.waits.isEmpty(), "success retried");
        });
        test("transient retry preserves identity and max delay", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 9).add(Ports.Kind.OK, 0); f.run();
            f.state(Ports.State.APPLIED); require(f.calls == 2 && f.grants == 1, "retry not admitted");
            require(f.waits.equals(List.of(9L)), "server minimum not honored");
        });
        test("local delay can dominate server advice", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 1).add(Ports.Kind.OK, 0); f.run();
            require(f.waits.equals(List.of(4L)), "local wait not honored");
        });
        test("permanent nonapplication stops", () -> {
            Fixture f = new Fixture().add(Ports.Kind.PERMANENT, 99); f.run(); f.state(Ports.State.REJECTED);
            require(f.grants == 0 && f.waits.isEmpty(), "permanent reply retried");
        });
        test("unknown effect is durable and cannot be replayed", () -> {
            require(!Contract.REPLAY_SAFE, "this regression targets the supplied unsafe peer");
            Fixture f = new Fixture().add(Ports.Kind.UNKNOWN, 99); f.effectApplied = true; f.run();
            f.state(Ports.State.UNKNOWN); require(f.calls == 1 && f.grants == 0 && f.waits.isEmpty(), "unsafe replay");
            Ports.Result next = RetryEngine.execute(INTENT, POLICY, f, f, f);
            require(next.attempts() == 0 && f.calls == 1, "unknown replayed in next invocation");
        });
        test("previous UNKNOWN is preserved", () -> {
            Fixture f = new Fixture(); f.state = new Ports.Snapshot(Ports.State.UNKNOWN, null); f.run();
            f.state(Ports.State.UNKNOWN); require(f.calls == 0 && f.grants == 0, "unknown dispatched");
        });
        for (Ports.State terminal : List.of(Ports.State.APPLIED, Ports.State.REJECTED)) {
            test("terminal " + terminal + " is returned unchanged", () -> {
                Fixture f = new Fixture(); f.state = new Ports.Snapshot(terminal, "stored");
                Ports.Snapshot original = f.state; f.run();
                require(f.state == original && f.mutations == 0 && f.calls == 0, "terminal changed");
            });
        }
        test("initial insufficient budget rejects without call", () -> {
            Fixture f = new Fixture(); f.remaining = 14; f.run(); f.state(Ports.State.REJECTED);
            require(f.calls == 0, "inadmissible call");
        });
        test("exact initial budget admits call", () -> {
            Fixture f = new Fixture().add(Ports.Kind.OK, 0); f.remaining = 15; f.run(); f.state(Ports.State.APPLIED);
        });
        test("attempt plus reserve cannot overflow", () -> {
            Fixture f = new Fixture(); f.remaining = Long.MAX_VALUE;
            f.run(new Ports.Policy(2, Long.MAX_VALUE, 1, 0)); f.state(Ports.State.REJECTED);
            require(f.calls == 0, "overflow admitted call");
        });
        test("unrepresentable wait total refuses before token", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, Long.MAX_VALUE); f.remaining = Long.MAX_VALUE;
            f.run(); f.state(Ports.State.REJECTED);
            require(f.grants == 0 && f.waits.isEmpty(), "overflow wait admitted");
        });
        test("server minimum cannot be truncated to fit", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 6); f.remaining = 20; f.run();
            f.state(Ports.State.REJECTED); require(f.grants == 0 && f.waits.isEmpty(), "oversized minimum waited");
        });
        test("exact retry budget fits", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 5).add(Ports.Kind.OK, 0); f.remaining = 20; f.run();
            require(f.calls == 2 && f.waits.equals(List.of(5L)), "exact retry fit rejected");
        });
        test("fresh budget includes time spent calling", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.remaining = 30; f.callCost = 12;
            f.run(); f.state(Ports.State.REJECTED); require(f.grants == 0, "stale budget admitted retry");
        });
        test("oversleep forbids next call; grant is consumed", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.remaining = 20; f.oversleep = 2;
            f.run(); f.state(Ports.State.REJECTED); require(f.calls == 1 && f.grants == 1, "oversleep admission wrong");
        });
        test("retry budget denial closes reply", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.denyToken = true; f.run();
            f.state(Ports.State.REJECTED); require(f.grants == 1 && f.waits.isEmpty(), "denied retry waited");
        });
        test("last attempt neither grants nor sleeps", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.run(new Ports.Policy(1, 10, 5, 4));
            f.state(Ports.State.REJECTED); require(f.grants == 0 && f.waits.isEmpty(), "wait after exhaustion");
        });
        test("count exhaustion preserves bounded retries", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0).add(Ports.Kind.TRANSIENT, 0).add(Ports.Kind.TRANSIENT, 0);
            f.run(); f.state(Ports.State.REJECTED); require(f.calls == 3 && f.grants == 2 && f.waits.size() == 2, "attempt bound wrong");
        });
        test("zero wait is permitted", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0).add(Ports.Kind.OK, 0);
            f.run(new Ports.Policy(2, 10, 0, 0)); require(f.calls == 2 && f.grants == 1, "zero wait disallowed");
        });
        test("initial cancellation prevents call", () -> {
            Fixture f = new Fixture(); f.cancelled = true; f.run(); f.state(Ports.State.REJECTED); require(f.calls == 0, "cancel dispatched");
        });
        test("cancellation during sleep prevents next call", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.cancelOnSleep = true; f.run();
            f.state(Ports.State.REJECTED); require(f.calls == 1, "cancel after sleep dispatched");
        });
        test("cancellation after grant stops before sleep", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.cancelOnGrant = true; f.run();
            f.state(Ports.State.REJECTED); require(f.waits.isEmpty() && f.calls == 1 && f.grants == 1, "grant cancellation ignored");
        });
        test("pre-interruption propagates without dispatch", () -> {
            Fixture f = new Fixture(); Thread.currentThread().interrupt();
            try { f.run(); throw new AssertionError("interrupt swallowed"); }
            catch (InterruptedException expected) { require(f.calls == 0, "preinterrupt dispatched"); }
        });
        test("interrupted dispatch preserves unknown effect", () -> {
            Fixture f = new Fixture(); f.interruptCall = true;
            try { f.run(); throw new AssertionError("interrupt swallowed"); }
            catch (InterruptedException expected) {
                f.state(Ports.State.UNKNOWN); require(f.effectApplied && f.calls == 1 && f.grants == 0, "unknown lost");
            }
        });
        test("sleep interruption propagates after close", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.interruptSleep = true;
            try { f.run(); throw new AssertionError("interrupt swallowed"); }
            catch (InterruptedException expected) { require(f.calls == 1 && f.closes == 1 && f.open == 0, "resource leaked"); }
        });
        test("interrupt after reply prevents retry", () -> {
            Fixture f = new Fixture().add(Ports.Kind.TRANSIENT, 0); f.interruptAfterReply = true;
            try { f.run(); throw new AssertionError("interrupt swallowed"); }
            catch (InterruptedException expected) { require(f.calls == 1 && f.closes == 1 && f.grants == 0, "interruption ignored"); }
        });
        System.out.println("Verification: " + passed + " passed, " + failed + " failed");
        if (failed != 0) throw new AssertionError("regressions failed");
    }
}
