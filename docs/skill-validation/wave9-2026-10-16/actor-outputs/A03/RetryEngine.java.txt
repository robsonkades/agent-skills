public final class RetryEngine {
    public static Ports.Result execute(Ports.Intent intent, Ports.Policy p, Ports.Store store,
            Ports.Operation operation, Ports.Runtime runtime) throws InterruptedException {
        Ports.Snapshot initial = store.snapshot();
        if (initial.state() == Ports.State.APPLIED || initial.state() == Ports.State.REJECTED) {
            return new Ports.Result(initial, 0);
        }
        boolean unresolved = initial.state() == Ports.State.UNKNOWN;
        int attempts = 0;
        while (attempts < p.maxAttempts()) {
            checkInterrupted();
            if ((unresolved && !Contract.REPLAY_SAFE) || runtime.cancelled()
                    || !fits(runtime.remainingNanos(), p, 0)) {
                break;
            }

            Ports.Reply reply;
            attempts++;
            try {
                reply = operation.call(intent, p.attemptNanos());
            } catch (InterruptedException | RuntimeException failure) {
                // A thrown call supplies no evidence that its effect did not apply.
                store.unknown();
                throw failure;
            }

            Ports.Kind kind;
            String receipt;
            long wait;
            try (reply) {
                kind = reply.kind();
                receipt = kind == Ports.Kind.OK ? reply.receipt() : null;
                wait = Math.max(p.localWaitNanos(), reply.serverMinimumNanos());
            }
            if (kind == Ports.Kind.OK) {
                store.applied(receipt);
                return new Ports.Result(store.snapshot(), attempts);
            }
            if (kind == Ports.Kind.UNKNOWN) {
                unresolved = true;
                store.unknown();
            }
            if (kind == Ports.Kind.PERMANENT) break;

            checkInterrupted();
            if ((unresolved && !Contract.REPLAY_SAFE) || attempts >= p.maxAttempts()
                    || runtime.cancelled() || !fits(runtime.remainingNanos(), p, wait)) {
                break;
            }
            if (!runtime.acquireRetry()) break;
            checkInterrupted();
            if (runtime.cancelled()) break;
            runtime.sleep(wait);
            // Admission is rechecked at the loop head, including after an oversleep.
        }
        if (!unresolved) store.rejected();
        return new Ports.Result(store.snapshot(), attempts);
    }

    private static boolean fits(long remaining, Ports.Policy p, long wait) {
        // Compare before subtraction; adding the three policy durations can overflow.
        return remaining >= p.reserveNanos()
                && remaining - p.reserveNanos() >= p.attemptNanos()
                && remaining - p.reserveNanos() - p.attemptNanos() >= wait;
    }

    private static void checkInterrupted() throws InterruptedException {
        if (Thread.currentThread().isInterrupted()) {
            throw new InterruptedException("Retry execution interrupted");
        }
    }
}
