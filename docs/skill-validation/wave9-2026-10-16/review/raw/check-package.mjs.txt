import { readFileSync, writeFileSync, readdirSync, existsSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { createHash } from 'node:crypto';
import { parseManifest } from 'file:///C:/git/agent-skills/packages/core/src/domain/manifest.ts';
import { parseSkillDocument, documentDescription, documentName } from 'file:///C:/git/agent-skills/packages/core/src/domain/skill-document.ts';
import * as prettier from 'file:///C:/git/agent-skills/node_modules/prettier/index.mjs';

const repo = 'C:/git/agent-skills';
const owned = join(repo, 'skills/retries-and-backoff');
const scratch = 'C:/Users/robso/AppData/Local/Temp/skills-wave9-93d1c7a4/review';
const snapshot = 'C:/Users/robso/AppData/Local/Temp/skills-wave9-93d1c7a4/snapshot';
const baseline = JSON.parse(readFileSync(join(repo, 'docs/skill-validation/wave9-2026-10-16/baseline.json'), 'utf8')).skill;
const checks = [];
function check(name, pass, details) { checks.push({ name, pass, details }); }
const document = parseSkillDocument(readFileSync(join(owned, 'SKILL.md'), 'utf8'));
const parsed = parseManifest(readFileSync(join(owned, 'skill.yaml'), 'utf8'), { strict: true });
check('repository manifest parser, strict mode', parsed.issues.length === 0, parsed.issues);
check('frontmatter matches manifest', documentName(document) === parsed.manifest.name
  && documentDescription(document) === parsed.manifest.description);
check('version increased exactly once', baseline.version === '1.1.4' && parsed.manifest.version === '1.1.5',
  { assignment: baseline.version, final: parsed.manifest.version });
const paths = readdirSync(owned, { recursive: true, withFileTypes: true })
  .filter((entry) => entry.isFile()).map((entry) => join(entry.parentPath, entry.name));
check('only original four shipped files', paths.length === 4, paths);
const hashes = [];
for (const item of baseline.files) {
  const before = readFileSync(join(snapshot, item.path));
  const after = readFileSync(join(owned, item.path));
  const beforeHash = createHash('sha256').update(before).digest('hex');
  const afterHash = createHash('sha256').update(after).digest('hex');
  check(`starting snapshot matches recorded hash: ${item.path}`, beforeHash === item.sha256);
  if (item.path === 'SKILL.md') check('main skill remains byte-identical', beforeHash === afterHash);
  hashes.push({ path: item.path, beforeSha256: beforeHash, afterSha256: afterHash, changed: beforeHash !== afterHash });
}
for (const path of paths.filter((path) => path.endsWith('.md'))) {
  const text = readFileSync(path, 'utf8');
  for (const match of text.matchAll(/\]\(([^)]+)\)/g)) {
    const link = match[1];
    if (!link.startsWith('http') && !link.startsWith('<http') && !link.startsWith('#')) {
      check(`local link ${link}`, existsSync(join(dirname(path), link)));
    }
  }
}
for (const reference of ['references/retry-in-java.md', 'references/retry-failure-modes.md']) {
  check(`reference routed from main: ${reference}`, document.body.includes(`](${reference})`));
}
const formatConfig = JSON.parse(readFileSync(join(repo, '.prettierrc.json'), 'utf8'));
for (const item of baseline.files) {
  const path = join(owned, item.path);
  const content = readFileSync(path, 'utf8');
  const before = readFileSync(join(snapshot, item.path), 'utf8');
  const currentFormatted = await prettier.check(content, { ...formatConfig, filepath: path });
  const baselineFormatted = await prettier.check(before, { ...formatConfig, filepath: path });
  // Git checkout uses CRLF in this environment; also check normalized contents to expose substantive formatting drift.
  const normalizedCurrent = await prettier.check(content.replaceAll('\r\n', '\n'), { ...formatConfig, filepath: path });
  const normalizedBaseline = await prettier.check(before.replaceAll('\r\n', '\n'), { ...formatConfig, filepath: path });
  checks.push({ name: `prettier API: ${item.path}`, pass: currentFormatted,
    details: { baselineFormatted, normalizedCurrent, normalizedBaseline } });
}
writeFileSync(join(scratch, 'package-checks.json'), JSON.stringify({ checks, hashes }, null, 2) + '\n');
for (const result of checks) console.log(`${result.pass ? 'PASS' : 'FAIL'} ${result.name}${result.details === undefined ? '' : ': ' + JSON.stringify(result.details)}`);
process.exitCode = checks.some((result) => !result.pass) ? 1 : 0;
