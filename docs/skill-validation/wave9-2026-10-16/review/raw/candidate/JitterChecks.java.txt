import java.time.Duration;
import java.util.concurrent.ThreadLocalRandom;

public class JitterChecks {
static Duration fullJitter(int attempt, Duration base, Duration cap) {
    if (attempt < 0 || base.isNegative() || base.isZero()
            || cap.isNegative() || cap.isZero()) {
        throw new IllegalArgumentException("positive base/cap and non-negative attempt required");
    }
    long baseNanos = base.toNanos(); // ArithmeticException rejects an unrepresentable policy
    long capNanos = cap.toNanos();
    long exponential = attempt >= 63 || baseNanos > (Long.MAX_VALUE >> attempt)
            ? Long.MAX_VALUE : baseNanos << attempt;
    long window = Math.min(capNanos, exponential);
    return Duration.ofNanos(ThreadLocalRandom.current().nextLong(window)); // [0, window)
}

  static void check(boolean ok, String message) {
    if (!ok) throw new AssertionError(message);
  }
  public static void main(String[] args) {
    long[][] cases = {{0,1,1,1}, {0,5,100,5}, {1,5,100,10}, {62,1,Long.MAX_VALUE,1L << 62},
      {63,1,Long.MAX_VALUE,Long.MAX_VALUE}, {Integer.MAX_VALUE,10,100,100},
      {62,3,1000,1000}, {0,Long.MAX_VALUE,100,100}};
    int samples = 0;
    for (long[] c : cases) for (int i = 0; i < 2000; i++) {
      long draw = fullJitter((int)c[0], Duration.ofNanos(c[1]), Duration.ofNanos(c[2])).toNanos();
      check(draw >= 0 && draw < c[3], "out-of-window draw " + draw);
      samples++;
    }
    int rejected = 0;
    try { fullJitter(-1, Duration.ofNanos(1), Duration.ofNanos(1)); }
    catch (IllegalArgumentException expected) { rejected++; }
    try { fullJitter(0, Duration.ZERO, Duration.ofNanos(1)); }
    catch (IllegalArgumentException expected) { rejected++; }
    try { fullJitter(0, Duration.ofNanos(1), Duration.ofNanos(-1)); }
    catch (IllegalArgumentException expected) { rejected++; }
    try { fullJitter(0, Duration.ofSeconds(Long.MAX_VALUE), Duration.ofNanos(1)); }
    catch (ArithmeticException expected) { rejected++; }
    check(rejected == 4, "invalid policy accepted");
    System.out.println("PASS fullJitter: " + samples + " range/overflow samples; 4 invalid policies rejected");
  }
}
