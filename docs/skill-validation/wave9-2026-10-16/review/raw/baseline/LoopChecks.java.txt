import java.time.Duration;
import java.util.concurrent.ThreadLocalRandom;

public class LoopChecks {
static Duration fullJitter(int attempt, Duration base, Duration cap) {
    if (attempt < 0 || base.isNegative() || base.isZero()
            || cap.isNegative() || cap.isZero()) {
        throw new IllegalArgumentException("positive base/cap and non-negative attempt required");
    }
    long baseNanos = base.toNanos(); // ArithmeticException rejects an unrepresentable policy
    long capNanos = cap.toNanos();
    long exponential = attempt >= 63 || baseNanos > (Long.MAX_VALUE >> attempt)
            ? Long.MAX_VALUE : baseNanos << attempt;
    long window = Math.min(capNanos, exponential);
    return Duration.ofNanos(ThreadLocalRandom.current().nextLong(window)); // [0, window)
}

// Record patterns in the switch below require Java 21; use instanceof/visitor on Java 17.
public sealed interface Outcome<T> {
    record Ok<T>(T value) implements Outcome<T> {}
    record Transient<T>(String code, Duration advisedDelay) implements Outcome<T> {}  // ZERO = none
    record Permanent<T>(String code) implements Outcome<T> {}
    record Ambiguous<T>(String code) implements Outcome<T> {}   // may or may not have been applied
}

// Conceptual: no metrics, no per-endpoint budget scoping.
<T> T execute(Op<T> op, Policy policy, Deadline deadline, boolean replaySafe)
        throws InterruptedException {
    boolean unresolved = false;
    for (int attempt = 0; ; attempt++) {
        if (Thread.currentThread().isInterrupted()) throw new InterruptedException();
        Duration remaining = deadline.remaining();
        if (remaining.isNegative() || remaining.isZero()
                || remaining.compareTo(policy.expectedCost()) < 0) {
            throw stopped("deadline-exhausted", unresolved);
        }
        Outcome<T> outcome = op.call(deadline);
        Duration advised = Duration.ZERO;
        switch (outcome) {
            case Outcome.Ok<T>(T value) -> { budget.recordSuccess(); return value; }
            case Outcome.Permanent<T> p -> throw stopped(p.code(), unresolved);
            case Outcome.Ambiguous<T> a -> {
                unresolved = true;
                if (!replaySafe) throw stopped(a.code(), true); // pending/unknown, not definite failure
            }
            case Outcome.Transient<T> t -> advised = t.advisedDelay();
        }
        if (attempt + 1 >= policy.maxAttempts()) throw stopped("attempts-exhausted", unresolved);

        Duration local = fullJitter(attempt, policy.base(), policy.cap());
        Duration wait = advised.compareTo(local) > 0 ? advised : local;
        if (deadline.remaining().minus(wait).compareTo(policy.expectedCost()) < 0) {
            throw stopped("deadline-would-be-exceeded", unresolved);
        }
        long waitNanos = wait.toNanos(); // validate conversion before reserving retry budget
        if (!budget.tryAcquire()) throw stopped("retry-budget-exhausted", unresolved);
        TimeUnit.NANOSECONDS.sleep(waitNanos); // Java 17 API; unmounting depends on JDK/context
    }
}

  static long now;
  static long oversleep;
  static boolean interruptSleep;
  static int sleeps;
  static int calls;
  static boolean durableUnknown;
  final Budget budget = new Budget();
  record Policy(int maxAttempts, Duration expectedCost, Duration base, Duration cap) {}
  record Deadline(long budgetNanos) {
    Duration remaining() { return Duration.ofNanos(budgetNanos - now); }
  }
  interface Op<T> { Outcome<T> call(Deadline deadline) throws InterruptedException; }
  static final class Budget {
    int tokens = 1;
    int successes;
    boolean tryAcquire() { if (tokens < 1) return false; tokens--; return true; }
    void recordSuccess() { successes++; }
  }
  enum TimeUnit {
    NANOSECONDS;
    void sleep(long nanos) throws InterruptedException {
      sleeps++;
      if (interruptSleep) throw new InterruptedException("injected during delay");
      now += nanos + oversleep;
    }
  }
  static final class Stopped extends RuntimeException {
    final String reason;
    final boolean unresolved;
    Stopped(String reason, boolean unresolved) { this.reason = reason; this.unresolved = unresolved; }
  }
  static Stopped stopped(String reason, boolean unresolved) { return new Stopped(reason, unresolved); }
  static Policy policy(int attempts) {
    return new Policy(attempts, Duration.ofNanos(1), Duration.ofNanos(1), Duration.ofNanos(1));
  }
  @SuppressWarnings("unchecked")
  static Outcome<String> ambiguous(long advice) {
    durableUnknown = true; // fixture operation owner, not a durable storage implementation
    try {
      var ctor = Outcome.Ambiguous.class.getDeclaredConstructors()[0];
      return (Outcome<String>) (ctor.getParameterCount() == 1
        ? ctor.newInstance("503") : ctor.newInstance("503", Duration.ofNanos(advice)));
    } catch (ReflectiveOperationException e) { throw new AssertionError(e); }
  }
  static Outcome<String> ok() { durableUnknown = false; return new Outcome.Ok<>("same-intent receipt"); }
  static void check(boolean ok, String message) { if (!ok) throw new AssertionError(message); }
  interface Checked { void run() throws Exception; }
  static void expectStopped(String reason, boolean unresolved, Checked action) throws Exception {
    try { action.run(); throw new AssertionError("expected stop " + reason); }
    catch (Stopped s) { check(s.reason.equals(reason) && s.unresolved == unresolved,
      "actual stop " + s.reason + ", unresolved=" + s.unresolved); }
  }
  static void reset() {
    now = 0; oversleep = 0; sleeps = 0; calls = 0; interruptSleep = false; durableUnknown = false;
    Thread.interrupted();
  }
  record Case(String name, Checked body) {}
  public static void main(String[] args) throws Exception {
    Case[] cases = {
      new Case("ambiguous safe replay respects server minimum", () -> {
        var h = new LoopChecks();
        String result = h.execute(d -> ++calls == 1 ? ambiguous(100) : ok(), policy(2), new Deadline(200), true);
        check(calls == 2 && now >= 100 && result.equals("same-intent receipt"), "server minimum lost: " + now);
        check(h.budget.successes == 1 && !durableUnknown, "same-intent success did not resolve");
      }),
      new Case("unfittable ambiguous guidance stops unknown before token/sleep", () -> {
        var h = new LoopChecks();
        expectStopped("deadline-would-be-exceeded", true, () ->
          h.execute(d -> ++calls == 1 ? ambiguous(100) : ok(), policy(2), new Deadline(50), true));
        check(calls == 1 && sleeps == 0 && h.budget.tokens == 1 && durableUnknown, "attempt/token/state leak");
      }),
      new Case("unsafe ambiguous replay stops after one attempt", () -> {
        var h = new LoopChecks();
        expectStopped("503", true, () -> h.execute(d -> { calls++; return ambiguous(10); }, policy(3), new Deadline(200), false));
        check(calls == 1 && sleeps == 0 && durableUnknown, "unsafe replay started");
      }),
      new Case("retry budget exhaustion preserves unknown", () -> {
        var h = new LoopChecks(); h.budget.tokens = 0;
        expectStopped("retry-budget-exhausted", true, () -> h.execute(d -> { calls++; return ambiguous(10); }, policy(3), new Deadline(200), true));
        check(calls == 1 && sleeps == 0 && durableUnknown, "budget/state leak");
      }),
      new Case("attempt exhaustion preserves unknown", () -> {
        var h = new LoopChecks();
        expectStopped("attempts-exhausted", true, () -> h.execute(d -> { calls++; return ambiguous(10); }, policy(1), new Deadline(200), true));
        check(calls == 1 && sleeps == 0 && durableUnknown, "attempt limit/state leak");
      }),
      new Case("later terminal attempt cannot erase earlier ambiguity", () -> {
        var h = new LoopChecks();
        expectStopped("denied", true, () -> h.execute(d -> ++calls == 1 ? ambiguous(10) : new Outcome.Permanent<String>("denied"), policy(2), new Deadline(200), true));
        check(calls == 2 && durableUnknown, "lost prior unknown");
      }),
      new Case("expired entry starts no attempt", () -> {
        var h = new LoopChecks();
        expectStopped("deadline-exhausted", false, () -> h.execute(d -> { calls++; return ok(); }, policy(2), new Deadline(0), true));
        check(calls == 0, "started expired operation");
      }),
      new Case("pre-interrupted entry starts no attempt", () -> {
        var h = new LoopChecks(); Thread.currentThread().interrupt();
        try { h.execute(d -> { calls++; return ok(); }, policy(2), new Deadline(200), true); throw new AssertionError("did not interrupt"); }
        catch (InterruptedException expected) { check(calls == 0, "started interrupted operation"); }
        finally { Thread.interrupted(); }
      }),
      new Case("timer oversleep starts no late attempt", () -> {
        var h = new LoopChecks(); oversleep = 100;
        expectStopped("deadline-exhausted", true, () -> h.execute(d -> { calls++; return ambiguous(10); }, policy(2), new Deadline(50), true));
        check(calls == 1 && durableUnknown, "started late attempt or lost unknown");
      }),
      new Case("interrupted sleep starts no further attempt", () -> {
        var h = new LoopChecks(); interruptSleep = true;
        try { h.execute(d -> { calls++; return ambiguous(10); }, policy(2), new Deadline(200), true); throw new AssertionError("did not interrupt"); }
        catch (InterruptedException expected) { check(calls == 1 && durableUnknown, "replayed or lost owner state"); }
      }),
      new Case("known rejection retains delay without manufacturing ambiguity", () -> {
        var h = new LoopChecks();
        expectStopped("denied", false, () -> h.execute(d -> ++calls == 1 ? new Outcome.Transient<String>("429", Duration.ofNanos(20)) : new Outcome.Permanent<String>("denied"), policy(2), new Deadline(200), false));
        check(calls == 2 && now >= 20 && !durableUnknown, "rejection advice/state mismatch");
      })
    };
    int failures = 0;
    for (Case c : cases) {
      reset();
      try { c.body.run(); System.out.println("PASS " + c.name); }
      catch (Throwable failure) { failures++; System.out.println("FAIL " + c.name + ": " + failure); }
    }
    System.out.println("Loop cases: " + (cases.length - failures) + "/" + cases.length + " passed");
    if (failures != 0) System.exit(1);
  }
}
