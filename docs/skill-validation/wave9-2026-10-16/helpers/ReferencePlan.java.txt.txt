import java.util.function.LongUnaryOperator;
public final class RetryPlan {
    public record Plan(boolean admitted, long waitNanos, long attemptNanos) {}
    public static Plan next(long start,long now,long total,long reserve,long minimumWork,
            long attemptCap,long base,long delayCap,int completed,int maxAttempts,
            long serverMinimum,LongUnaryOperator draw) {
        Plan rejected = new Plan(false,0,0);
        long elapsed = now - start;
        if(completed >= maxAttempts || elapsed >= total) return rejected;
        long remaining = total - elapsed;
        long wait = 0;
        if(completed > 0) {
            int exponent = completed - 1;
            long grown = exponent >= 63 || base > (Long.MAX_VALUE >> exponent) ? Long.MAX_VALUE : base << exponent;
            long window = Math.min(delayCap,grown);
            wait = Math.max(serverMinimum,draw.applyAsLong(window));
        }
        if(remaining < wait || remaining-wait < reserve || remaining-wait-reserve < minimumWork) return rejected;
        return new Plan(true,wait,Math.min(attemptCap,remaining-wait-reserve));
    }
}
