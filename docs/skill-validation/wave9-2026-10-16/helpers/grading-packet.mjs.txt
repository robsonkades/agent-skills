import fs from 'node:fs';
import crypto from 'node:crypto';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave9-93d1c7a4',report='C:/git/agent-skills/docs/skill-validation/wave9-2026-10-16';
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const grades=JSON.parse(fs.readFileSync(report+'/root-grades.json','utf8'));
const actors=JSON.parse(fs.readFileSync(report+'/actors.frozen.json.txt','utf8'));
if(grades.length!==24||actors.some(a=>!grades.find(g=>g.id===a.id)))throw Error('Root grading incomplete');
if(fs.existsSync(report+'/root-grades-seal.json'))throw Error('Root grades already sealed');

fs.writeFileSync(report+'/root-grades-seal.json',JSON.stringify({at:new Date().toISOString(),sha256:sha(report+'/root-grades.frozen.json.txt'),note:'Sealed before the independent actor judgments; formatting of the friendly JSON is permitted, substantive changes require separate adjudication'},null,2)+'\n');
const dir=root+'/masked';fs.mkdirSync(dir,{recursive:true});const map=[];
for(let i=0;i<24;i++){
 const a=actors[(i*7+5)%24],id='G'+String(i+1).padStart(2,'0'),out=a.run+'/output';
 const impl=a.caseId==='R3'?'RetryPlan.java':'RetryEngine.java';
 const p={id,caseId:a.caseId,review:fs.readFileSync(out+'/REVIEW.md','utf8'),result:JSON.parse(fs.readFileSync(out+'/result.json','utf8')),implementation:a.caseId==='R4'?null:fs.readFileSync(out+'/'+impl,'utf8')};
 const path=dir+'/'+id+'.json';fs.writeFileSync(path,JSON.stringify(p,null,2).replace(new RegExp('\\b'+a.id+'\\b','g'),id)+'\n');map.push({maskedId:id,actorId:a.id,file:id+'.json',sha256:sha(path),masking:'Own actor identifier replaced by masked identifier; other wording is preserved and may disclose skill exposure.'});
}
const criteria=JSON.parse(fs.readFileSync(root+'/frozen/criteria.json','utf8'));
fs.writeFileSync(dir+'/criteria.json',JSON.stringify({criteria:criteria.criteria,cases:criteria.cases,scope:'Evaluate C2, C3Text (claims and bounded evidence only) and C4Text (scope and honesty apparent in text). C1 and actual verifier sensitivity are not independently executed in this review. Implementation is included for semantic reading; Verification.java, logs, journals, conditions, model traces and root grades are withheld. Masking is partial: wording can disclose context; record any such disclosure.'},null,2)+'\n');
fs.writeFileSync(dir+'/manifest.json',JSON.stringify(map.map(({actorId,...rest})=>rest),null,2)+'\n');
fs.writeFileSync(report+'/grading-map.json',JSON.stringify(map,null,2)+'\n');
console.log(JSON.stringify({sealed:true,packets:map.length,dir,rootSeal:report+'/root-grades-seal.json'}));
