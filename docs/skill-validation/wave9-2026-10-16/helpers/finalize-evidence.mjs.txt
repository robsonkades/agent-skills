import fs from 'node:fs';
import crypto from 'node:crypto';
import {execFileSync} from 'node:child_process';
import * as prettier from 'file:///C:/git/agent-skills/node_modules/prettier/index.mjs';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave9-93d1c7a4',repo='C:/git/agent-skills',report=repo+'/docs/skill-validation/wave9-2026-10-16';
const phase=process.argv[2];
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const write=(p,v)=>fs.writeFileSync(p,JSON.stringify(v,null,2)+'\n');
function files(dir,prefix=''){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?files(dir+'/'+e.name,prefix+e.name+'/'):[prefix+e.name]).sort();}
function raw(src,dest){fs.mkdirSync(dest.slice(0,dest.lastIndexOf('/')),{recursive:true});fs.copyFileSync(src,dest);}
if(phase==='archive'){
 for(const p of files(root).filter(p=>!p.includes('/')&&/\.(mjs|java|txt|json)$/.test(p)))raw(root+'/'+p,report+'/helpers/'+p+'.txt');
 for(const p of files(root+'/review').filter(p=>!p.includes('/classes/')&&!p.endsWith('.class')))raw(root+'/review/'+p,report+'/review/raw/'+p+'.txt');
 if(fs.existsSync(root+'/masked'))for(const p of files(root+'/masked'))raw(root+'/masked/'+p,report+'/review/masked/'+p+'.txt');
 if(fs.existsSync(root+'/review/actor-grades.json'))raw(root+'/review/actor-grades.json',report+'/review/actor-grades.json.txt');
 const rechecks=fs.readdirSync(report+'/rechecks').filter(p=>p.endsWith('.json')).sort().map(p=>JSON.parse(fs.readFileSync(report+'/rechecks/'+p,'utf8')));
 write(report+'/rechecks.json',rechecks);
 console.log(JSON.stringify({archived:true,rechecks:rechecks.length}));
}else if(phase==='format'){
 const owned=files(report).filter(p=>/\.(json|md)$/.test(p)).map(p=>report+'/'+p).concat(repo+'/docs/skills-evolution-inventory.md');
 let changed=0;for(const p of owned){const before=fs.readFileSync(p,'utf8'),options=await prettier.resolveConfig(p),after=await prettier.format(before,{...options,filepath:p});if(before!==after){fs.writeFileSync(p,after);changed++;}}
 console.log(JSON.stringify({formatted:owned.length,changed}));
}else if(phase==='integrity'){
 const baseline=JSON.parse(fs.readFileSync(report+'/baseline.json','utf8')),freeze=JSON.parse(fs.readFileSync(report+'/freeze.json','utf8')),actors=JSON.parse(fs.readFileSync(report+'/actors.frozen.json.txt','utf8'));
 const allowed=new Set(['skills/retries-and-backoff/references/retry-in-java.md','skills/retries-and-backoff/references/retry-failure-modes.md','skills/retries-and-backoff/skill.yaml','registry/skills.yaml','docs/skills-evolution-inventory.md']);
 const changes=baseline.allFiles.filter(f=>!fs.existsSync(repo+'/'+f.path)||sha(repo+'/'+f.path)!==f.sha256).map(f=>f.path);
 const unexpected=changes.filter(p=>!allowed.has(p));
 const current=execFileSync('git',['ls-files','-z','--cached','--others','--exclude-standard'],{cwd:repo,encoding:'utf8',maxBuffer:64*1024*1024}).split('\0').filter(Boolean),before=new Set(baseline.allFiles.map(f=>f.path));
 const unexpectedNew=current.filter(p=>!before.has(p)&&!p.startsWith('docs/skill-validation/wave9-2026-10-16/'));
 write(report+'/preservation-final.json',{at:new Date().toISOString(),checked:baseline.allFiles.length,changes,unexpected,unexpectedNew,pass:unexpected.length===0&&unexpectedNew.length===0,note:'Tracked and nonignored files present in the initial working tree; generated ignored build products are outside this comparison'});
 const failures=[];let inputChecks=0;
 for(const a of actors)for(const f of a.inputFiles){inputChecks++;if(sha(a.run+'/'+f.path)!==f.sha256||sha(report+'/actor-inputs/'+a.id+'/'+f.path+'.txt')!==f.sha256)failures.push('input '+a.id+'/'+f.path);}
 for(const f of freeze.snapshots)if(sha(root+'/frozen/'+f.path)!==f.sha256||sha(report+'/frozen/'+f.path+'.txt')!==f.sha256)failures.push('snapshot '+f.path);
 for(const f of freeze.evaluator)if(sha(root+'/'+f.path)!==f.sha256)failures.push('current evaluator '+f.path);
 const handoff=JSON.parse(fs.readFileSync(report+'/review/research-handoff.json.txt','utf8'));
 for(const f of handoff.files)if(sha(repo+'/skills/retries-and-backoff/'+f.path)!==f.afterSha256)failures.push('candidate '+f.path);
 for(const a of actors){const r=JSON.parse(fs.readFileSync(report+'/rechecks/'+a.id+'.json','utf8'));for(const f of r.outputHashes)if(sha(a.run+'/output/'+f.path)!==f.sha256||sha(report+'/actor-outputs/'+a.id+'/'+f.path+'.txt')!==f.sha256)failures.push('output '+a.id+'/'+f.path);}
 const seal=JSON.parse(fs.readFileSync(report+'/root-grades-seal.json','utf8'));if(sha(report+'/root-grades.frozen.json.txt')!==seal.sha256)failures.push('root grades seal');
 const masked=JSON.parse(fs.readFileSync(report+'/grading-map.json','utf8'));
 for(const m of masked)if(sha(root+'/masked/'+m.file)!==m.sha256||sha(report+'/review/masked/'+m.file+'.txt')!==m.sha256)failures.push('masked packet '+m.maskedId);
 if(sha(root+'/review/actor-grades.json')!==sha(report+'/review/actor-grades.json.txt'))failures.push('independent grades archive');
 if(sha(root+'/review/calibration-inputs.json')!==freeze.calibration.graded.inputSha256)failures.push('calibration inputs');
 for(const p of ['actors','criteria','root-grades'])if(JSON.stringify(JSON.parse(fs.readFileSync(report+'/'+p+'.json','utf8')))!==JSON.stringify(JSON.parse(fs.readFileSync(report+'/'+p+'.frozen.json.txt','utf8'))))failures.push('friendly JSON drift '+p);
 const rawFiles=files(report).filter(p=>p.endsWith('.txt')).map(path=>({path,bytes:fs.statSync(report+'/'+path).size,sha256:sha(report+'/'+path)}));
 write(report+'/artifact-manifest.json',{at:new Date().toISOString(),note:'Raw authored artifacts and execution logs archived byte-for-byte; binaries and third-party full sources omitted',files:rawFiles});
 write(report+'/integrity-final.json',{at:new Date().toISOString(),protectedInputs:inputChecks,frozenFiles:freeze.snapshots.length,maskedPackets:masked.length,rawArtifacts:rawFiles.length,failures,pass:failures.length===0});
 console.log(JSON.stringify({preservation:unexpected.length===0&&unexpectedNew.length===0,changes,unexpected,unexpectedNew,integrity:failures.length===0,inputChecks,rawArtifacts:rawFiles.length,failures}));
 if(unexpected.length||unexpectedNew.length||failures.length)process.exitCode=1;
}else throw Error('Expected archive, format or integrity');
