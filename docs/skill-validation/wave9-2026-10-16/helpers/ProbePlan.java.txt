import java.math.BigInteger;
import java.util.SplittableRandom;
public final class ProbePlan {
    static int scenarios;
    static void check(boolean b,String message){if(!b)throw new AssertionError(message);}
    static void test(long start,long elapsed,long total,long reserve,long minimum,long attemptCap,long base,long cap,int completed,int max,long advice,boolean high){
        long now=start+elapsed;
        BigInteger remain=BigInteger.valueOf(total).subtract(BigInteger.valueOf(elapsed));
        BigInteger window=BigInteger.valueOf(base).shiftLeft(Math.min(63,Math.max(0,completed-1))).min(BigInteger.valueOf(cap));
        long draw=high?window.longValueExact()-1:0;
        long wait=completed==0?0:Math.max(draw,advice);
        BigInteger available=remain.subtract(BigInteger.valueOf(wait)).subtract(BigInteger.valueOf(reserve));
        boolean admitted=completed<max&&available.compareTo(BigInteger.valueOf(minimum))>=0;
        long timeout=admitted?available.min(BigInteger.valueOf(attemptCap)).longValueExact():0;
        int[] draws={0};
        RetryPlan.Plan p=RetryPlan.next(start,now,total,reserve,minimum,attemptCap,base,cap,completed,max,advice,b->{draws[0]++;check(b==window.longValueExact(),"full-jitter window saturation/ordinal");return high?b-1:0;});
        check(p.admitted()==admitted,"admission case "+scenarios+" expected "+admitted+" got "+p);
        check(p.waitNanos()==(admitted?wait:0)&&p.attemptNanos()==timeout,"wait/reserve/timeout case "+scenarios+" got "+p);
        if(completed==0)check(draws[0]==0,"initial call does not draw jitter");
        if(admitted&&completed>0)check(draws[0]==1,"retry uses supplied jitter source");
        scenarios++;
    }
    public static void main(String[] args){
        long[] origins={0,-50,Long.MAX_VALUE-20,Long.MIN_VALUE+20};
        for(long start:origins)for(int completed:new int[]{0,1,2,3,63,64,Integer.MAX_VALUE-1})for(long advice:new long[]{0,7,100,Long.MAX_VALUE})for(boolean high:new boolean[]{false,true})
            test(start,25,100,5,10,40,4,50,completed,completed==Integer.MAX_VALUE-1?Integer.MAX_VALUE:4,advice,high);
        test(Long.MAX_VALUE-20,10,100,5,10,40,4,50,0,3,0,false);
        test(0,0,Long.MAX_VALUE,0,1,100,(1L<<53)+1,Long.MAX_VALUE,1,2,0,true);
        test(0,0,15,5,10,40,4,50,0,3,0,false);
        test(0,0,15,5,10,40,4,50,1,3,0,false);
        test(0,0,Long.MAX_VALUE,Long.MAX_VALUE-3,10,10,4,50,1,3,Long.MAX_VALUE,false);
        test(0,0,Long.MAX_VALUE,0,10,Long.MAX_VALUE,Long.MAX_VALUE/2+1,Long.MAX_VALUE,3,4,0,true);
        test(Long.MAX_VALUE-10,101,100,0,1,10,1,5,0,1,0,false);
        test(-100,0,100,0,1,10,1,5,1,1,0,false);
        SplittableRandom rng=new SplittableRandom(920260916L);
        for(int i=0;i<300;i++){
            long total=1+rng.nextInt(2000),elapsed=rng.nextInt(2500),reserve=rng.nextInt(200),minimum=1+rng.nextInt(100),attemptCap=minimum+rng.nextInt(300),base=1+rng.nextInt(500),cap=1+rng.nextInt(800),advice=rng.nextInt(1000);
            int completed=rng.nextInt(8),max=1+rng.nextInt(8);
            test(rng.nextLong(),elapsed,total,reserve,minimum,attemptCap,base,cap,completed,max,advice,rng.nextBoolean());
        }
        System.out.println("{\"status\":\"PASS\",\"scenarios\":"+scenarios+",\"oracle\":\"BigInteger mathematical budget and window; finite deterministic cases\"}");
    }
}
