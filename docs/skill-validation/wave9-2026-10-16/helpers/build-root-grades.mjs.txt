if(fs.existsSync(report+'/root-grades-seal.json'))throw Error('Already sealed; use separate adjudication');
const grades=actors.map(a=>{
 const n=notes[a.id],r=read(report+'/rechecks/'+a.id+'.json'),java=a.caseId!=='R4';
 if(!n?.reviewRead||!n.resultRead||!n.journalDeclarationsRead||(java&&(!n.codeRead||!n.verificationRead)))throw Error('Incomplete actual root reading '+a.id);
 const words=fs.readFileSync(a.run+'/output/REVIEW.md','utf8').trim().split(/\s+/).length;
 const protectedPass=r.protectedCopies.every(f=>f.expected===f.actual)&&r.inputChecks.every(f=>f.sha256===f.actual);
 const C1=r.probePass;
 const C2=n.C2===true;
 const C3=java?r.verificationPass&&r.counterfactualSensitive:n.C3Text===true;
 const C4=protectedPass&&r.allWritesStopped&&words<=750;
 return {id:a.id,caseId:a.caseId,condition:a.condition,repetition:a.repetition,gradedAt:new Date().toISOString(),C1,C2,C3,C4,pass:C1&&C2&&C3&&C4,reviewWords:words,
  rationale:{C1:java?'Actual implementation compiled with --release 17 and passed the pre-frozen private probe on VM25.':'Five pre-frozen numerical metrics passed against actual result.json.',C2:n.note,C3:java?'Root read actual Verification.java, reran it on the delivered code, then compiled the incorrect original replacement and observed an assertion failure. This establishes sensitivity to that replacement, not every possible defect.':n.note,C4:'Required artifacts read; signatures inspected in Java source and exercised by compilation; immutable copies and input hashes matched; journal declares local-only scope/stopped writers. This is procedural evidence, not OS-enforced access isolation.'},
  independentExecution:java?{positiveExit:r.verification.exit,negativeCompileExit:r.counterfactual.compile.exit,negativeRunExit:r.counterfactual.run.exit,firstFailure:r.counterfactual.run.stderr.split(/\r?\n/).find(s=>s.includes('AssertionError'))??r.counterfactual.run.stdout.split(/\r?\n/).find(s=>s.includes('FAIL'))}:null,
  sourceRead:{implementation:java,verification:java,review:true,result:true,journalDeclarations:true},recheck:'rechecks/'+a.id+'.json'};
});
fs.writeFileSync(report+'/root-grades.json',JSON.stringify(grades,null,2)+'\n');
