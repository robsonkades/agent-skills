import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const root='C:/Users/robso/AppData/Local/Temp/skills-wave6-0tHxwh',report='C:/git/agent-skills/docs/skill-validation/wave6-2026-10-16';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const put=(p,b)=>{fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,b);};
const walk=(d,s='')=>fs.readdirSync(d+'/'+s,{withFileTypes:true}).flatMap(e=>{const p=s?s+'/'+e.name:e.name;return e.isDirectory()?walk(d,p):[p];});
const entries=[];
function archive(file,target,authoredHelper=false){
 const b=fs.readFileSync(file);const text=b.subarray(0,6000).toString('utf8');
 const denied=/\.(class|jar|exe|dll|zip|html|pdf)$/i.test(file)||b.length>1048576||(!authoredHelper&&(/<!doctype\s+html|<html(?:\s|>)/i.test(text)||/\[wordlim:/.test(text)||/copyright|SPDX-License-Identifier|licensed under/i.test(text)));
 const e={path:file,bytes:b.length,sha256:sha(b),archived:!denied,authoredHelper};
 if(!denied){e.target=target;put(report+'/'+target,b);}else e.reason='Binary, large or third-party document retained by reference/hash; not vendored.';
 entries.push(e);
}
const m=JSON.parse(fs.readFileSync(report+'/actors.json','utf8'));
for(const r of m.runs){
 assert.equal(JSON.parse(fs.readFileSync(r.directory+'/output/execution.json','utf8')).allWritesStopped,true);
 for(const rel of walk(r.directory+'/output'))archive(r.directory+'/output/'+rel,'actor-outputs/'+r.id+'/'+rel+'.txt');
}
for(const name of fs.readdirSync(root).filter(n=>/\.(mjs|cjs|ps1)$/.test(n)))archive(root+'/'+name,'helpers/'+name+'.txt',true);
for(const name of walk(root+'/review').filter(n=>/\.(json|md|mjs|cjs|ps1|txt|log)$/.test(n)&&n!=='masked-outputs.json'))archive(root+'/review/'+name,'review/'+name+'.txt');
for(const name of ['criteria-draft.json','calibration-expected.json'])archive(root+'/'+name,'review/'+name+'.txt');
for(const condition of ['previous','current'])for(const rel of walk(root+'/snapshots/'+condition))archive(root+'/snapshots/'+condition+'/'+rel,'snapshots/'+condition+'/'+rel+'.txt');
put(report+'/artifact-manifest.json',JSON.stringify({at:new Date().toISOString(),policy:'Raw authored text copied without reformatting. Compilation binaries and full third-party source documents retained as references only. Not proof of content correctness or complete tool traces.',entries},null,2)+'\n');
console.log(JSON.stringify({artifacts:entries.length,archived:entries.filter(e=>e.archived).length,referencesOnly:entries.filter(e=>!e.archived).length}));
