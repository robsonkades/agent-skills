import fs from 'node:fs';

const args=process.argv.slice(2),journal=args.includes('--journal');
function compact(v,k=''){
 if(Array.isArray(v))return v.map(x=>compact(x,k));
 if(v&&typeof v==='object')return Object.fromEntries(Object.entries(v).filter(([key])=>!['content','output','stdout','stderr','hashOnlyReads','hashesBefore','hashesAfter'].includes(key)).map(([key,val])=>[key,compact(val,key)]));
 return typeof v==='string'&&v.length>1600?v.slice(0,1600)+' [display clipped; raw retained]':v;
}
for(const id of args.filter(a=>a!=='--journal')){

 if(journal&&fs.existsSync(d+'/execution.json')){
  const e=JSON.parse(fs.readFileSync(d+'/execution.json','utf8'));
  console.log(JSON.stringify({id,execution:compact(e)},null,2));
 }
