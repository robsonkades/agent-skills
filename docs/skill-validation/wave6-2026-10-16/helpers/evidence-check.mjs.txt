import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
const report='C:/git/agent-skills/docs/skill-validation/wave6-2026-10-16';
const manifest=JSON.parse(fs.readFileSync(report+'/actors.json','utf8'));
const a=JSON.parse(fs.readFileSync(report+'/journal-audit.json','utf8'));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const norm=s=>String(s).replaceAll('\\','/');
const records=[];
for(const r of a.records){
 const run=manifest.runs.find(x=>x.id===r.id),declared=[],mismatches=[];
 for(const p of r.pathRecords){
  const rawPath=p.path??p.file??p.relativePath??p.filePath??p.name;if(!rawPath)continue;
  const rel=norm(rawPath).replace(run.directory+'/','');
  const original=run.inputs.find(x=>x.path===rel);
  if(!original)continue;
  const declaredHash=p.sha256??p.hash;
  if(typeof declaredHash==='string'&&/^[0-9a-f]{64}$/i.test(declaredHash)&&declaredHash.toLowerCase()!==original.sha256)mismatches.push({path:rel,declaredHash,expected:original.sha256});
  if(rel.startsWith('skill/'))declared.push({...p,path:rel});
 }
 const badCwds=r.commands.filter(c=>{const d=norm(path.win32.resolve(run.directory,norm(c.cwd??c.workdir??c.workingDirectory)));return d!==run.directory&&!d.startsWith(run.directory+'/output');}).map(c=>({at:c.at,cwd:c.cwd??c.workdir??c.workingDirectory}));
 const files=[];
 function walk(d,s=''){for(const e of fs.readdirSync(d+'/'+s,{withFileTypes:true})){const p=s?s+'/'+e.name:e.name;if(e.isDirectory()){if(e.name!=='.build')walk(d,p);}else if(/\.(json|log|out|txt)$/.test(p)&&!['assessment.json','execution.json'].includes(p))files.push(p);}}
 walk(run.directory+'/output');
 const checks=[];
 function findCommands(v,file,at=''){
  if(Array.isArray(v))return v.forEach((x,i)=>findCommands(x,file,at+'/'+i));
  if(!v||typeof v!=='object')return;
  if(('stdout'in v||'output'in v||'outputObserved'in v)&&['exit','exitCode','status'].some(k=>k in v)){
   const out=v.stdout??v.output??v.outputObserved;
   if(typeof out==='string'&&out.length<2500)checks.push({file,at,program:v.program??v.command??v.executable??v.cmd,args:v.args??v.arguments,exit:v.exit??v.exitCode??v.status,stdout:out,stderr:v.stderr});
  }
  for(const [k,x] of Object.entries(v))if(!['content','output','stdout','stderr'].includes(k))findCommands(x,file,at+'/'+k);
 }
 const exec=JSON.parse(fs.readFileSync(run.directory+'/output/execution.json','utf8'));findCommands(exec,'execution.json');
 for(const f of files.filter(f=>f.endsWith('.json'))){try{findCommands(JSON.parse(fs.readFileSync(run.directory+'/output/'+f,'utf8')),f);}catch{}}
 const textLogs=files.filter(f=>/\.(log|out|txt)$/.test(f)).map(f=>({file:f,text:fs.readFileSync(run.directory+'/output/'+f,'utf8')})).filter(x=>x.text.length<2000);
 const row={run:r.id,condition:run.condition,skillReadDeclarations:declared,declaredProtectedHashMismatches:mismatches,outsideDeclaredWorkingDirectory:badCwds,sourceUrls:[...new Set(r.sources.map(s=>s.url??s.sourceUrl).filter(Boolean))],checkRecords:checks,textLogs,logs:files,executionSha256:r.executionSha256};records.push(row);
}
const result={at:new Date().toISOString(),scope:'Mechanical comparisons against frozen file hashes, plus extraction of actor-authored read/command declarations. Does not independently certify all tool history, access isolation or actual model attention. Coordinator reviews semantics separately.',records};
fs.writeFileSync(report+'/execution-audit.json',JSON.stringify(result,null,2)+'\n');
for(const r of records)console.log(JSON.stringify({run:r.run,condition:r.condition,skillReads:r.skillReadDeclarations.map(({at,path,...p})=>({path,details:Object.fromEntries(Object.entries(p).filter(([k])=>!/hash|sha256/i.test(k)))})),hashMismatches:r.declaredProtectedHashMismatches,badCwds:r.outsideDeclaredWorkingDirectory,checks:r.checkRecords.filter(c=>/^(sequential-|while-|synchronous provider|before-recycle|invalid-input)/.test(c.stdout?.trim())).map(c=>({file:c.file,exit:c.exit,output:c.stdout})),logs:r.textLogs.filter(l=>/^(sequential-|while-|synchronous provider|before-recycle|invalid-input)/.test(l.text.trim()))}));
