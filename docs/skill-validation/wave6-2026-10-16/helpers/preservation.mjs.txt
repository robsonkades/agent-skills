import fs from 'node:fs';
import crypto from 'node:crypto';
import {execFileSync} from 'node:child_process';
const repo='C:/git/agent-skills',report=repo+'/docs/skill-validation/wave6-2026-10-16';
const base=JSON.parse(fs.readFileSync(report+'/baseline.json','utf8'));
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const allowed=new Set(['skills/java-memory-model/references/happens-before.md','skills/java-memory-model/skill.yaml','registry/skills.yaml','docs/skills-evolution-inventory.md']);
const changes=base.allFiles.filter(f=>!fs.existsSync(repo+'/'+f.path)||sha(fs.readFileSync(repo+'/'+f.path))!==f.sha256).map(f=>({path:f.path,allowed:allowed.has(f.path),before:f.sha256,after:fs.existsSync(repo+'/'+f.path)?sha(fs.readFileSync(repo+'/'+f.path)):null}));
const files=[...new Set(execFileSync('git',['ls-files','-z','--cached','--others','--exclude-standard'],{cwd:repo,encoding:'utf8',maxBuffer:64*1024*1024}).split('\0').filter(Boolean))];
const old=new Set(base.allFiles.map(f=>f.path));
const added=files.filter(f=>!old.has(f)).map(path=>({path,allowed:path.startsWith('docs/skill-validation/wave6-2026-10-16/')}));
const unexpected=[...changes.filter(x=>!x.allowed),...added.filter(x=>!x.allowed)];
const r={at:new Date().toISOString(),baselineFiles:base.allFiles.length,preserved:base.allFiles.length-changes.length,changes,addedFiles:added.length,unexpected};
fs.writeFileSync(report+'/preservation-final.json',JSON.stringify(r,null,2)+'\n');console.log(JSON.stringify(r));process.exitCode=unexpected.length?1:0;
