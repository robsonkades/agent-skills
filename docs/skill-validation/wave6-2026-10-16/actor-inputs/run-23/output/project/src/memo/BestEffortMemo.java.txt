package memo;
import java.util.OptionalInt;
import java.util.function.ToIntFunction;
public final class BestEffortMemo {
    private final ToIntFunction<PublishedEntry> reader;
    private PublishedEntry entry;
    public BestEffortMemo() { reader = ReaderFactory.create(); }
    public void install(int value) { entry = new PublishedEntry(value); }
    public OptionalInt find() {
        PublishedEntry observed = entry;
        return observed == null ? OptionalInt.empty() : OptionalInt.of(reader.applyAsInt(observed));
    }
}
