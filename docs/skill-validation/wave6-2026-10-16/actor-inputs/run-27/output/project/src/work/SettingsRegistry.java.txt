package work;
public final class SettingsRegistry {
 private final Object lock = new Object();
 private Settings current;
 private int builds;
 public Settings settings() {
  Settings observed = current;
  if (observed == null) {
   synchronized (lock) {
    observed = current;
    if (observed == null) { observed = build(); current = observed; }
   }
  }
  return observed;
 }
 private Settings build() { builds++; return new Settings(64); }
 public int buildsForCheck() { synchronized (lock) { return builds; } }
}
