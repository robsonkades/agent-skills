import work.Message;

public final class ContractCheck {
    private static void rejects(String label, int[] input) {
        try {
            new Message(input);
            throw new AssertionError(label + " accepted");
        } catch (IllegalArgumentException expected) {
            System.out.println("PASS rejects " + label);
        }
    }

    public static void main(String[] args) {
        rejects("null", null);
        rejects("length 0", new int[0]);
        rejects("length 1", new int[1]);
        rejects("length 3", new int[3]);
        int[] buffer = {10, 20};
        Message message = new Message(buffer);
        if (message.checksum() != 30) throw new AssertionError("initial checksum");
        System.out.println("PASS initial checksum=30");
        buffer[0] = 100;
        buffer[1] = 200;
        int observed = message.checksum();
        System.out.println("snapshot expected=30 observed=" + observed);
        if (observed != 30) throw new AssertionError("producer recycling changed message");
    }
}
