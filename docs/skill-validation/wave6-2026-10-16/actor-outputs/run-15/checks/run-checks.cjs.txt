const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const { spawnSync } = require('node:child_process');

const root = process.cwd();
const destination = path.join(root, 'output/project');
const supplied = [
  'README.md', 'check.mjs', 'docs/review-request.md', 'docs/contract.md',
  'src/work/Snapshot.java', 'src/work/ResultBridge.java',
  'src/work/Provider.java', 'test/work/Smoke.java'
];
const hash = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
const log = { startedAt: new Date().toISOString(), commands: [], hashChecks: [], nodeVersion: process.version };
if (!fs.existsSync(destination)) {
  fs.cpSync(path.join(root, 'input/project'), destination, { recursive: true, force: false, errorOnExist: true });
  log.copy = 'Copied input/project to output/project; supplied files are not edited.';
} else {
  log.copy = 'Existing output/project retained; supplied files are checked against input.';
}
function checkHashes(stage) {
  for (const relative of supplied) {
    const input = `input/project/${relative}`;
    const output = `output/project/${relative}`;
    const inputSha256 = hash(path.join(root, input));
    const outputSha256 = hash(path.join(root, output));
    log.hashChecks.push({ stage, input, output, inputSha256, outputSha256, access: 'hash-only', identical: inputSha256 === outputSha256 });
    if (inputSha256 !== outputSha256) throw new Error(`Supplied file differs: ${relative}`);
  }
}
function run(program, args, timeout) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, { cwd: destination, encoding: 'utf8', timeout, windowsHide: true });
  const command = { program, arguments: args, cwd: destination, timeoutMs: timeout, startedAt, endedAt: new Date().toISOString(), exit: result.status, signal: result.signal, error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  log.commands.push(command);
  process.stdout.write(JSON.stringify(command) + '\n');
  if (result.status !== 0) throw new Error(`Command failed: ${program}`);
}
try {
  checkHashes('before');
  run('C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe', ['-version'], 15000);
  run(process.execPath, ['check.mjs'], 30000);
  checkHashes('after');
  log.subprocessesStopped = true;
} catch (error) {
  log.error = error.message;
  process.exitCode = 1;
} finally {
  log.endedAt = new Date().toISOString();
  fs.writeFileSync(path.join(root, 'output/checks/runtime.json'), JSON.stringify(log, null, 2) + '\n');
}
