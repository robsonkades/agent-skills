import work.Message;

public final class ContractCheck {
    interface View { int checksum(); }
    interface Factory { View create(int[] words); }

    // Isolated proposal control, not a change to the supplied production sources.
    static final class Snapshot implements View {
        private final int[] words;
        Snapshot(int[] words) {
            if (words == null || words.length != 2) throw new IllegalArgumentException();
            this.words = words.clone();
        }
        public int checksum() { return words[0] + words[1]; }
    }

    public static void main(String[] args) {
        Factory factory = args[0].equals("current")
            ? words -> { Message message = new Message(words); return message::checksum; }
            : Snapshot::new;
        for (int[] invalid : new int[][] { null, {}, {1}, {1, 2, 3} }) {
            try {
                factory.create(invalid);
                throw new AssertionError("invalid input accepted");
            } catch (IllegalArgumentException expected) { }
        }
        System.out.println("validation=PASS (null, lengths 0, 1, 3)");
        int[] buffer = {10, 20};
        View first = factory.create(buffer);
        if (first.checksum() != 30) throw new AssertionError("initial checksum");
        buffer[0] = 100;
        buffer[1] = 200;
        View second = factory.create(buffer);
        if (second.checksum() != 300) throw new AssertionError("new message checksum");
        int actual = first.checksum();
        System.out.println("existing-message-after-recycle=" + actual);
        if (actual != 30) throw new AssertionError("expected original checksum 30, got " + actual);
        buffer[0] = -1;
        buffer[1] = -2;
        if (first.checksum() != 30 || second.checksum() != 300)
            throw new AssertionError("message changed after repeated recycling");
        System.out.println("snapshot-contract=PASS");
    }
}
