import fs from 'node:fs';
import { spawnSync } from 'node:child_process';
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const commands = [];
function run(tool, args, expectedExit = 0) {
  const program = `${jdk}/${tool}.exe`;
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, {
    cwd: process.cwd(), encoding: 'utf8', timeout: 15000, windowsHide: true,
  });
  const command = {
    program, args, cwd: process.cwd(), startedAt,
    finishedAt: new Date().toISOString(), timeoutMs: 15000,
    exit: result.status, signal: result.signal, error: result.error?.message ?? null,
    stdout: result.stdout, stderr: result.stderr, expectedExit,
  };
  commands.push(command);
  fs.writeFileSync('checks-log.json', JSON.stringify(commands, null, 2));
  console.log(JSON.stringify(command));
  if (result.status !== expectedExit) process.exit(2);
}
fs.mkdirSync('.build', { recursive: true });
run('java', ['-version']);
run('javac', ['--release', '25', '-cp', '../project/.build', '-d', '.build', 'ContractCheck.java']);
run('java', ['-cp', '.build;../project/.build', 'ContractCheck', 'current'], 1);
