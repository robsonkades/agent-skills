import fs from 'node:fs';
import path from 'node:path';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';

const root = process.cwd();
const project = path.join(root, 'output/project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const records = [];
function run(program, args) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, {cwd: project, encoding: 'utf8', timeout: 25000, windowsHide: true});
  const record = {program, args, cwd: project, startedAt, endedAt: new Date().toISOString(), timeoutMs: 25000, exit: result.status, signal: result.signal, error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? ''};
  records.push(record);
  fs.writeFileSync(path.join(root, 'output/checks/commands.json'), JSON.stringify(records, null, 2));
  process.stdout.write(JSON.stringify(record) + '\n');
  if (result.status !== 0) process.exit(1);
}
const supplied = ['README.md','docs/review-request.md','docs/contract.md','src/memo/BestEffortMemo.java','src/memo/PublishedEntry.java','src/memo/ReaderFactory.java','test/memo/Smoke.java','check.mjs'];
const hash = name => createHash('sha256').update(fs.readFileSync(name)).digest('hex');
const hashes = supplied.map(name => ({path: name, input: hash(path.join(root, 'input/project', name)), output: hash(path.join(project, name)), mode: 'hash-only'}));
fs.writeFileSync(path.join(root, 'output/checks/hashes.json'), JSON.stringify(hashes, null, 2));
if (hashes.some(h => h.input !== h.output)) throw new Error('Copies differ');
fs.mkdirSync(path.join(project, '.build'), {recursive: true});
run(path.join(jdk, 'java.exe'), ['-version']);
run(path.join(jdk, 'javac.exe'), ['--release','25','-d','.build','src/memo/BestEffortMemo.java','src/memo/PublishedEntry.java','src/memo/ReaderFactory.java','test/memo/Smoke.java']);
run(path.join(jdk, 'java.exe'), ['-cp','.build','memo.Smoke']);
run(path.join(jdk, 'javap.exe'), ['-classpath','.build','-v','-p','memo.ReaderFactory']);
console.log('supplied-copies-identical=8; all-child-processes-exited=true');
