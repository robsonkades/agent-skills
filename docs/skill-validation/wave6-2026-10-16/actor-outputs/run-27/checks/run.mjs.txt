import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const cwd = path.resolve('output/project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const commands = [
  ['java.exe', ['-version']],
  ['javac.exe', ['--release', '25', '-d', '.build', 'src/work/Settings.java', 'src/work/SettingsRegistry.java', 'test/work/Smoke.java']],
  ['java.exe', ['-cp', '.build', 'work.Smoke']],
];
fs.mkdirSync(path.join(cwd, '.build'), { recursive: true });
const results = [];
for (const [name, args] of commands) {
  const program = path.join(jdk, name);
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, { cwd, encoding: 'utf8', timeout: 15000, windowsHide: true });
  results.push({ program, args, cwd, startedAt, endedAt: new Date().toISOString(), timeoutMs: 15000, exit: result.status, signal: result.signal, error: result.error?.message ?? null, stdout: result.stdout ?? '', stderr: result.stderr ?? '' });
  if (result.status !== 0) break;
}
fs.writeFileSync('output/checks/java-checks.json', JSON.stringify({ nodeVersion: process.version, commands: results, allChildProcessesFinished: true }, null, 2) + '\n');
console.log(JSON.stringify(results, null, 2));
if (results.length !== commands.length || results.some(result => result.exit !== 0)) process.exitCode = 1;
