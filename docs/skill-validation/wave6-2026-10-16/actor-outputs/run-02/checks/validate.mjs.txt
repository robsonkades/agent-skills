import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '../..');
const project = path.join(root, 'output/project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const files = ['README.md', 'check.mjs', 'docs/contract.md', 'docs/review-request.md',
  'src/memo/BestEffortMemo.java', 'src/memo/PublishedEntry.java',
  'src/memo/ReaderFactory.java', 'test/memo/Smoke.java'];
const hash = p => createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const log = { startedAt: new Date().toISOString(), nodeVersion: process.version,
  inheritedJavaOptions: Object.fromEntries(['JAVA_TOOL_OPTIONS', 'JDK_JAVA_OPTIONS', '_JAVA_OPTIONS'].map(k => [k, process.env[k] ?? null])),
  comparisons: [], commands: [], allChildrenExited: false };
for (const file of files) {
  const input = hash(path.join(root, 'input/project', file));
  const before = hash(path.join(project, file));
  log.comparisons.push({ file, access: 'hash-only', inputSha256: input, outputBeforeSha256: before });
  if (input !== before) throw new Error(`Working copy mismatch: ${file}`);
}
function run(program, args) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, { cwd: project, encoding: 'utf8', timeout: 15000, windowsHide: true });
  const entry = { program, arguments: args, cwd: project, startedAt,
    endedAt: new Date().toISOString(), timeoutMs: 15000, exit: result.status,
    signal: result.signal, stdout: result.stdout ?? '', stderr: result.stderr ?? '',
    error: result.error?.message ?? null };
  log.commands.push(entry);
  console.log(JSON.stringify(entry));
  if (result.status !== 0 || result.error) throw new Error('Check command failed');
}
try {
  run(path.join(jdk, 'javac.exe'), ['-version']);
  run(path.join(jdk, 'java.exe'), ['-version']);
  fs.mkdirSync(path.join(project, '.build'), { recursive: true });
  run(path.join(jdk, 'javac.exe'), ['--release', '25', '-d', '.build',
    'src/memo/BestEffortMemo.java', 'src/memo/PublishedEntry.java',
    'src/memo/ReaderFactory.java', 'test/memo/Smoke.java']);
  run(path.join(jdk, 'java.exe'), ['-cp', '.build', 'memo.Smoke']);
  for (const comparison of log.comparisons) {
    comparison.outputAfterSha256 = hash(path.join(project, comparison.file));
    comparison.unchanged = comparison.outputAfterSha256 === comparison.inputSha256;
    if (!comparison.unchanged) throw new Error(`Immutable file changed: ${comparison.file}`);
  }
  log.allChildrenExited = true;
} finally {
  log.endedAt = new Date().toISOString();
  fs.writeFileSync(path.join(root, 'output/checks/validation.json'), JSON.stringify(log, null, 2) + '\n');
}
