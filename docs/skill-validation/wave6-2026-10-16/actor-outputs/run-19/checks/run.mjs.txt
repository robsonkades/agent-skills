import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { spawnSync } from 'node:child_process';

const root = path.resolve(import.meta.dirname, '../..');
const cwd = path.join(root, 'output/project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const report = { startedAt: new Date().toISOString(), commands: [], fileHashes: [], allSubprocessesExited: false };
const provided = ['README.md', 'check.mjs', 'docs/contract.md', 'docs/review-request.md', 'src/memo/BestEffortMemo.java', 'src/memo/PublishedEntry.java', 'src/memo/ReaderFactory.java', 'test/memo/Smoke.java'];
const digest = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
for (const name of provided) {
  const original = digest(path.join(root, 'input/project', name));
  const copy = digest(path.join(cwd, name));
  report.fileHashes.push({ name, originalSha256: original, beforeSha256: copy });
  if (original !== copy) throw new Error('Working copy differs: ' + name);
}
function run(program, args) {
  const startedAt = new Date().toISOString();
  const r = spawnSync(program, args, { cwd, encoding: 'utf8', timeout: 15000, windowsHide: true });
  const entry = { program, arguments: args, cwd, startedAt, finishedAt: new Date().toISOString(), timeoutMs: 15000, exit: r.status, signal: r.signal, stdout: r.stdout ?? '', stderr: r.stderr ?? '', error: r.error?.message ?? null };
  report.commands.push(entry);
  process.stdout.write(JSON.stringify(entry) + '\n');
  if (r.status !== 0) throw new Error('Command failed: ' + program);
}
try {
  run(path.join(jdk, 'java.exe'), ['-version']);
  run(path.join(jdk, 'javac.exe'), ['-version']);
  fs.mkdirSync(path.join(cwd, '.build'), { recursive: true });
  run(path.join(jdk, 'javac.exe'), ['--release', '25', '-d', '.build', 'src/memo/BestEffortMemo.java', 'src/memo/PublishedEntry.java', 'src/memo/ReaderFactory.java', 'test/memo/Smoke.java']);
  run(path.join(jdk, 'java.exe'), ['-cp', '.build', 'memo.Smoke']);
  report.allSubprocessesExited = true;
} finally {
  for (const record of report.fileHashes) record.afterSha256 = digest(path.join(cwd, record.name));
  report.providedFilesUnchanged = report.fileHashes.every(record => record.originalSha256 === record.beforeSha256 && record.beforeSha256 === record.afterSha256);
  report.finishedAt = new Date().toISOString();
  fs.writeFileSync(path.join(root, 'output/checks/results.json'), JSON.stringify(report, null, 2) + '\n');
}
