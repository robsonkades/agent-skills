import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { spawnSync } from 'node:child_process';

const root = process.cwd();
const project = path.join(root, 'output/project');
const checks = path.join(root, 'output/checks');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const sha = bytes => crypto.createHash('sha256').update(bytes).digest('hex');
const report = { startedAt: new Date().toISOString(), commands: [], providedFiles: [] };
const files = ['README.md', 'check.mjs', 'docs/review-request.md', 'docs/contract.md',
  'src/memo/BestEffortMemo.java', 'src/memo/PublishedEntry.java',
  'src/memo/ReaderFactory.java', 'test/memo/Smoke.java'];
for (const file of files) {
  const original = fs.readFileSync(path.join(root, 'input/project', file));
  const copy = fs.readFileSync(path.join(project, file));
  report.providedFiles.push({ file, inputSha256: sha(original), outputSha256: sha(copy),
    identical: original.equals(copy), access: 'complete byte read for equality and hash' });
}
function run(name, program, args) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, { cwd: project, encoding: 'utf8', timeout: 15000, windowsHide: true });
  const log = `output/checks/${name}.log`;
  fs.writeFileSync(path.join(root, log), `STDOUT\n${result.stdout || ''}\nSTDERR\n${result.stderr || ''}`);
  report.commands.push({ program, args, cwd: project, startedAt, finishedAt: new Date().toISOString(),
    timeoutMs: 15000, exit: result.status, signal: result.signal, error: result.error?.message ?? null, log });
  console.log(`${name}: exit=${result.status}; ${((result.stdout || '') + (result.stderr || '')).trim().split('\n')[0] || '(no output)'}`);
  if (result.status !== 0) throw new Error(`${name} failed`);
  return result.stdout || '';
}
try {
  if (!report.providedFiles.every(file => file.identical)) throw new Error('Provided copies differ');
  run('node-version', process.execPath, ['--version']);
  run('java-version', path.join(jdk, 'java.exe'), ['-version']);
  run('javac-version', path.join(jdk, 'javac.exe'), ['-version']);
  fs.mkdirSync(path.join(project, '.build'), { recursive: true });
  run('compile', path.join(jdk, 'javac.exe'), ['--release', '25', '-d', '.build',
    'src/memo/BestEffortMemo.java', 'src/memo/PublishedEntry.java', 'src/memo/ReaderFactory.java', 'test/memo/Smoke.java']);
  run('smoke', path.join(jdk, 'java.exe'), ['-cp', '.build', 'memo.Smoke']);
  const bytecode = run('reader-bytecode', path.join(jdk, 'javap.exe'), ['-cp', '.build', '-p', '-c', '-v', 'memo.ReaderFactory']);
  console.log(bytecode.split('\n').filter(line => line.includes('REF_invokeVirtual memo/PublishedEntry.legacyValue')).join('\n'));
  report.unchangedAfterChecks = files.every(file => fs.readFileSync(path.join(root, 'input/project', file)).equals(fs.readFileSync(path.join(project, file))));
} finally {
  report.finishedAt = new Date().toISOString();
  report.allChildProcessesExited = report.commands.every(command => command.exit !== null && command.signal === null);
  fs.writeFileSync(path.join(checks, 'runtime.json'), JSON.stringify(report, null, 2) + '\n');
}
