import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const cwd = path.resolve('output/project');
const records = [];
function run(program, args) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, { cwd, encoding: 'utf8', timeout: 30000, windowsHide: true });
  const record = { program, args, cwd, startedAt, endedAt: new Date().toISOString(), timeoutMs: 30000,
    exit: result.status, signal: result.signal, stdout: result.stdout ?? '', stderr: result.stderr ?? '',
    error: result.error?.message ?? null };
  records.push(record);
  process.stdout.write(JSON.stringify(record) + '\n');
  if (result.status !== 0) process.exitCode = 1;
}
run('C:/Users/robso/.jdks/temurin-25.0.3/bin/java.exe', ['-version']);
run('C:/Users/robso/.jdks/temurin-25.0.3/bin/javac.exe', ['-version']);
run(process.execPath, ['check.mjs']);
fs.writeFileSync('output/checks/smoke-execution.json', JSON.stringify({ records, allChildrenExited: records.every(r => r.exit !== null) }, null, 2) + '\n');
