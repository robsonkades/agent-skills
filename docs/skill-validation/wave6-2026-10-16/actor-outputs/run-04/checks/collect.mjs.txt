import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';

const root = process.cwd();
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const log = { startedAt: new Date().toISOString(), commands: [], fileHashes: [], comparisons: [] };
const contentReads = [
  'TASK.md', 'skill/SKILL.md', 'skill/references/happens-before.md',
  'skill/references/review-checklist.md',
  'input/project/docs/review-request.md', 'input/project/docs/contract.md',
  'input/project/README.md', 'input/project/check.mjs',

];
function digest(file, mode) {
  const bytes = fs.readFileSync(path.join(root, file));
  const sha256 = createHash('sha256').update(bytes).digest('hex');
  log.fileHashes.push({ file, sha256, mode, complete: true, truncated: false });
  return sha256;
}
for (const file of contentReads) {
  const expected = digest(file, 'content reviewed in full via Get-Content; re-read here for hash only');
  if (file.startsWith('input/project/')) {
    const counterpart = file.replace('input/', 'output/');
    const actual = digest(counterpart, 'hash only; not a separate content review');
    log.comparisons.push({ input: file, output: counterpart, identical: actual === expected });
    if (actual !== expected) throw new Error('Provided copy differs: ' + file);
  }
}
function run(tool, args) {
  const command = { program: path.join(jdk, tool), arguments: args, cwd: path.join(root, 'output/project'), timeoutMs: 15000, startedAt: new Date().toISOString() };
  const result = spawnSync(command.program, args, { cwd: command.cwd, encoding: 'utf8', timeout: command.timeoutMs, windowsHide: true });
  Object.assign(command, { endedAt: new Date().toISOString(), exit: result.status, signal: result.signal, stdout: result.stdout || '', stderr: result.stderr || '', error: result.error?.message ?? null });
  log.commands.push(command);
  fs.writeFileSync(path.join(root, 'output/checks/evidence.json'), JSON.stringify(log, null, 2) + '\n');
  if (result.status !== 0) throw new Error(tool + ' failed');
  return result.stdout || result.stderr || '';
}
