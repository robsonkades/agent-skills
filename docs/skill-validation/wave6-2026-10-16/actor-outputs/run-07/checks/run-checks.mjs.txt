import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const root = process.cwd();
const project = path.join(root, 'output', 'project');
const jdk = 'C:/Users/robso/.jdks/temurin-25.0.3/bin';
const commands = [];
const run = (program, args, cwd, timeoutMs) => {
  const startedAt = new Date().toISOString();
  const result = spawnSync(program, args, { cwd, timeout: timeoutMs, encoding: 'utf8', windowsHide: true });
  const record = { program, args, cwd, timeoutMs, startedAt, endedAt: new Date().toISOString(), exit: result.status, signal: result.signal, stdout: result.stdout ?? '', stderr: result.stderr ?? '', error: result.error?.message ?? null };
  commands.push(record);
  process.stdout.write(JSON.stringify(record) + '\n');
  return result.status === 0;
};
let success = run(process.execPath, ['--version'], root, 5000);
success = run(path.join(jdk, 'java.exe'), ['--version'], root, 5000) && success;
success = run(path.join(jdk, 'javac.exe'), ['--version'], root, 5000) && success;
success = run(process.execPath, ['check.mjs'], project, 30000) && success;
fs.writeFileSync(path.join(root, 'output', 'checks', 'commands.json'), JSON.stringify({ commands, success, childProcessesCompleted: true }, null, 2) + '\n');
process.exitCode = success ? 0 : 1;
