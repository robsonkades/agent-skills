import assert from 'node:assert/strict';
import fs from 'node:fs';
const read = name => fs.readFileSync(new URL(name, import.meta.url));
const originals = ['README.md', 'REQUEST.md', 'schema.sql', 'seed.sql', 'current.sql', 'proposed.sql', 'fixture.mjs'];
for (const name of originals) assert.deepEqual(read(name), read('../input/' + name), `${name} must be preserved`);
const result = JSON.parse(read('result.json'));
const report = read('REVIEW.md').toString();
const words = report.trim().split(/\s+/).length;
assert.ok(words <= 750, `REVIEW.md has ${words} words`);
assert.ok(result.decision && result.evidence && result.limits);
const verified = JSON.parse(read('verification.log').toString().split('\nSTDERR:')[0]);
for (const key of ['cursorChecks', 'completePaginationWalks', 'candidateExecutions']) {
  assert.equal(result.evidence[key], verified[key], key);
}
assert.equal(verified.status, 'passed');
const solution = read('solution.sql').toString().replace(/--[^\n]*/g, '');
assert.doesNotMatch(solution, /\bOFFSET\b|\bROW_NUMBER\s*\(/i);
console.log(JSON.stringify({status: 'passed', preservedCopies: originals.length, reviewWords: words, resultMatchesVerification: true}, null, 2));
