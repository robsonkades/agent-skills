import assert from 'node:assert/strict';
import fs from 'node:fs';
import { createHash } from 'node:crypto';

const preserved = ['README.md', 'REQUEST.md', 'schema.sql', 'seed.sql', 'current.sql', 'proposed.sql', 'fixture.mjs'];
const hashes = {};
for (const name of preserved) {
  const input = fs.readFileSync(new URL(`../input/${name}`, import.meta.url));
  const output = fs.readFileSync(new URL(`./${name}`, import.meta.url));
  assert.ok(output.equals(input), `${name}: copied input bytes changed`);
  hashes[name] = createHash('sha256').update(input).digest('hex');
}
const review = fs.readFileSync(new URL('./REVIEW.md', import.meta.url), 'utf8');
const words = review.trim().split(/\s+/).length;
assert.ok(words <= 750, `Review exceeds 750 words: ${words}`);
const result = JSON.parse(fs.readFileSync(new URL('./result.json', import.meta.url), 'utf8'));
assert.ok(result.decision && result.evidence.length && result.limits.length);
const mutation = JSON.parse(fs.readFileSync(new URL('./mutation-check.log', import.meta.url), 'utf8'));
assert.equal(mutation.status, 'passed');
assert.equal(mutation.originalBytesRestored, true);
assert.equal(mutation.outcomes.length, 7);
const candidate = fs.readFileSync(new URL('./solution.sql', import.meta.url), 'utf8');
assert.match(candidate, /active IS NOT TRUE/);
console.log(JSON.stringify({ status: 'passed', preservedFiles: hashes, reviewWords: words,
  resultJsonValid: true, mutationLogVerified: true, candidateSha256: createHash('sha256').update(candidate).digest('hex') }, null, 2));
