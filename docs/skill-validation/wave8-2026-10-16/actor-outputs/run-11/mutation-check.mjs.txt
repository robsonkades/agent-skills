import assert from 'node:assert/strict';
import fs from 'node:fs';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const solutionPath = new URL('./solution.sql', import.meta.url);
const original = fs.readFileSync(solutionPath);
const source = original.toString('utf8');
const proposed = fs.readFileSync(new URL('./proposed.sql', import.meta.url), 'utf8');
const outcomes = [];
function run(name, expectPass) {
  const result = spawnSync(process.execPath, [fileURLToPath(new URL('./verify.mjs', import.meta.url))], {
    cwd: fileURLToPath(new URL('.', import.meta.url)), encoding: 'utf8', timeout: 19000,
  });
  assert.equal(result.error, undefined, `${name}: runner error`);
  assert.equal(result.signal, null, `${name}: runner signal`);
  if (expectPass) {
    assert.equal(result.status, 0, `${name}: ${result.stderr}`);
  } else {
    assert.equal(result.status, 1, `${name}: wrong candidate should fail`);
    assert.match(result.stderr, /AssertionError/);
    assert.match(result.stderr, /solution\.sql/);
  }
  outcomes.push({ name, exitCode: result.status, expected: expectPass ? 'pass' : 'assertion failure',
    detail: expectPass ? JSON.parse(result.stdout) : result.stderr.match(/AssertionError[^\n]*/)?.[0] });
}
try {
  run('approved candidate', true);
  const variants = [
    ['original proposal loses nullable-active invoices', proposed],
    ['UNION drops duplicate invoice labels', source.replace('UNION ALL', 'UNION')],
    ['unguarded UNION ALL counts overlap twice', source.replace(' AND active IS NOT TRUE', '')],
    ['one unbound branch leaks tenants', source.replace('WHERE tenant_id = :tenant AND amount', 'WHERE amount')],
    ['reversed order violates ordering', source.replace('ORDER BY customer_label;', 'ORDER BY customer_label DESC;')],
  ];
  for (const [name, sql] of variants) {
    assert.notEqual(sql, source, `${name}: mutation must change SQL`);
    fs.writeFileSync(solutionPath, sql);
    run(name, false);
  }
} finally {
  fs.writeFileSync(solutionPath, original);
}
run('restored approved candidate', true);
console.log(JSON.stringify({ status: 'passed', originalBytesRestored: fs.readFileSync(solutionPath).equals(original), outcomes }, null, 2));
