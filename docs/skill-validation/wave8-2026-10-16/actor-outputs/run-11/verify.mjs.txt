import assert from 'node:assert/strict';
import fs from 'node:fs';
import { createDb, query } from './fixture.mjs';

// The candidate is always loaded from disk; replace solution.sql to test another query.
const solution = fs.readFileSync(new URL('./solution.sql', import.meta.url), 'utf8');
const current = fs.readFileSync(new URL('./current.sql', import.meta.url), 'utf8');
const db = createDb();
let checks = 0;
const rows = labels => labels.map(customer_label => ({ customer_label }));
function check(name, tenant, expected) {
  assert.deepEqual(query(db, current, { tenant }), expected, `${name}: reference/oracle`);
  assert.deepEqual(query(db, solution, { tenant }), expected, `${name}: solution.sql`);
  checks++;
}
function replaceRows(records) {
  db.exec('DELETE FROM invoices');
  const insert = db.prepare('INSERT INTO invoices VALUES (?, ?, ?, ?, ?)');
  for (const record of records) insert.run(...record);
}
try {
  const sqlite = db.prepare('SELECT sqlite_version() AS version').get().version;
  check('seed preserves two Acme invoices and counts overlap once', 71, rows(['Acme', 'Acme', 'Bravo']));
  check('seed other tenant', 72, rows(['Other']));
  check('absent tenant', 999, []);
  check('null bound tenant', null, []);

  // Each row is isolated so equal labels cannot conceal one missing and one extra row.
  const amounts = [-9999999999.99, -1, 0, 20, 99.99, 100, 100.01, 150, 9999999999.99];
  for (const active of [1, 0, null]) {
    for (const amount of amounts) {
      replaceRows([[10, 71, 'Only', active, amount], [11, 72, 'Outside', active, amount]]);
      const expected = rows(active === 1 || amount >= 100 ? ['Only'] : []);
      check(`active=${active}, amount=${amount}`, 71, expected);
      check(`isolation active=${active}, amount=${amount}`, 73, []);
    }
  }

  replaceRows([
    [101, 71, 'Zulu', 1, 150],
    [102, 71, 'Acme', 0, 100],
    [103, 71, 'Acme', null, 100],
    [104, 71, 'Acme', 1, 100],
    [105, 71, 'Acme', 1, 20],
    [106, 71, 'Acme', null, 99.99],
    [107, 71, 'Bravo', 0, 100.01],
    [108, 71, '', null, 150],
    [109, 72, 'Acme', 1, 150],
    [110, 72, 'Outside-null', null, 100],
    [111, 72, 'Outside-false', 0, 100],
  ]);
  check('mixed duplicate labels, nulls, overlap and global ordering', 71,
    rows(['', 'Acme', 'Acme', 'Acme', 'Acme', 'Bravo', 'Zulu']));
  check('second tenant exercises both branches', 72, rows(['Acme', 'Outside-false', 'Outside-null']));
  check('mixed data absent tenant', 999, []);
  check('mixed data null bound tenant', null, []);
  replaceRows([]);
  check('empty relation', 71, []);
  console.log(JSON.stringify({ status: 'passed', checks, candidate: 'solution.sql', node: process.version, sqlite,
    scope: 'Portable result semantics only; no PostgreSQL execution or performance measurements.' }, null, 2));
} finally {
  db.close();
}
