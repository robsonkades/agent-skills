import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';

const output = new URL('./', import.meta.url);
const input = new URL('../input/', output);
const names = ['README.md', 'REQUEST.md', 'proposal.md', 'incident.json', 'preview-plan.json', 'literal-comparison.json'];
const preserved = names.map((name) => {
  const original = readFileSync(new URL(name, input));
  const copied = readFileSync(new URL(name, output));
  assert.deepEqual(copied, original, `${name} changed`);
  return { file: name, sha256: createHash('sha256').update(original).digest('hex') };
});
const parse = (name) => JSON.parse(readFileSync(new URL(name, output), 'utf8'));
const incident = parse('incident.json');
const preview = parse('preview-plan.json');
const literal = parse('literal-comparison.json');
const result = parse('result.json');
const [outer, inner] = incident.plan.children;
const expected = {
  innerExecutions: inner.loops,
  innerEmittedRows: inner.actualRows * inner.loops,
  innerInclusiveWorkMs: inner.actualTotalTimeMs * inner.loops,
  capturedExecutionTimeMs: incident.executionTimeMs,
};
assert.deepEqual(result.metrics, expected);
assert.equal(expected.innerEmittedRows, incident.plan.actualRows * incident.plan.loops);
assert.equal(inner.loops, outer.actualRows * outer.loops);
assert.equal(preview.plan.child.stoppedByParent, true);
assert.equal(preview.plan.actualRows, 20);
assert.equal(literal.preparedStatementSameParameterCaptured, false);
assert.equal(literal.tenantRows * inner.actualRows, literal.returnedRows);
for (const field of ['decision', 'evidence', 'limits', 'nextExperiment', 'metricInterpretation']) {
  assert.ok(result[field], `${field} missing`);
}
for (const value of Object.values(result.metrics)) assert.equal(typeof value, 'number');
const review = readFileSync(new URL('REVIEW.md', output), 'utf8');
const reportWords = review.trim().split(/\s+/u).length;
assert.ok(reportWords <= 750, `Report has ${reportWords} words`);
console.log(JSON.stringify({
  status: 'passed',
  nodeVersion: process.version,
  checkType: 'Local artifact arithmetic, output validation and byte preservation; no SQL execution',
  preserved,
  metrics: expected,
  reportWords,
  derived: {
    observedOrdersPassFraction: outer.actualRows / (outer.actualRows + outer.rowsRemovedByFilter),
    innerInclusiveWorkFractionOfExecution: expected.innerInclusiveWorkMs / expected.capturedExecutionTimeMs,
    requestExecutionDifferenceMs: incident.endToEndRequestMs - incident.executionTimeMs,
  },
}, null, 2));
