import assert from 'node:assert/strict';
import fs from 'node:fs';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const solutionPath = new URL('./solution.sql', import.meta.url);
const original = fs.readFileSync(solutionPath);
const sql = original.toString('utf8');
const mutants = [
  ['overlap duplicated', sql.replace(/ AND NOT \(active = true\)/, '')],
  ['duplicate labels collapsed', sql.replace('UNION ALL', 'UNION')],
  ['second branch tenant leak', sql.replace('WHERE tenant_id = :tenant AND amount', 'WHERE amount')],
  ['threshold row lost', sql.replace('amount >= 100', 'amount > 100')],
  ['global order reversed', sql.replace('ORDER BY customer_label;', 'ORDER BY customer_label DESC;')],
];
const results = [];
try {
  for (const [name, mutant] of mutants) {
    assert.notEqual(mutant, sql, `mutation applied: ${name}`);
    fs.writeFileSync(solutionPath, mutant);
    const run = spawnSync(process.execPath, [fileURLToPath(new URL('./verify.mjs', import.meta.url))], {
      encoding: 'utf8', timeout: 20000, windowsHide: true,
    });
    if (run.error) throw run.error;
    assert.equal(run.status, 1, `regression must fail for: ${name}`);
    assert.match(run.stderr, /AssertionError/);
    results.push({ name, exitCode: run.status, rejected: true });
  }
} finally {
  fs.writeFileSync(solutionPath, original);
}
assert.deepEqual(fs.readFileSync(solutionPath), original);
console.log(JSON.stringify({ status: 'passed', mutants: results, originalBytesRestored: true }, null, 2));
