import assert from 'node:assert/strict';
import fs from 'node:fs';
import {createDb} from './fixture.mjs';

// Always execute the submitted query, including when a reviewer replaces it.
const solution = fs.readFileSync(new URL('./solution.sql', import.meta.url), 'utf8');
const current = fs.readFileSync(new URL('./current.sql', import.meta.url), 'utf8').trim().replace(/;$/, '');
const proposed = fs.readFileSync(new URL('./proposed.sql', import.meta.url), 'utf8');
const db = createDb();
let cursorChecks = 0;
let traversals = 0;
let emptyChecks = 0;
const suites = [];
const proposalCounterexamples = [];

function query(sql, bindings = {}) {
  const statement = db.prepare(sql);
  statement.setReadBigInts(true);
  return statement.all(bindings).map(row => ({...row}));
}

function compare(a, b) {
  if (a.priority === null && b.priority !== null) return 1;
  if (a.priority !== null && b.priority === null) return -1;
  if (a.priority !== b.priority) return a.priority > b.priority ? -1 : 1;
  if (a.created_at !== b.created_at) return a.created_at < b.created_at ? -1 : 1;
  return a.id === b.id ? 0 : a.id > b.id ? -1 : 1;
}

function page(tenant, cursor, size) {
  return query(solution, {
    ':tenant': tenant,
    ':cursor_priority': cursor.priority,
    ':cursor_created': cursor.created_at,
    ':cursor_id': cursor.id,
    ':page_size': size,
  });
}

function exercise(name, tenants) {
  let rowCount = 0;
  for (const tenant of tenants) {
    const rows = query(current, {':tenant': tenant});
    const independent = query(
      'SELECT id, priority, created_at, label FROM tickets WHERE tenant_id = :tenant',
      {':tenant': tenant},
    ).sort(compare);
    assert.deepStrictEqual(rows, independent, `${name}: original ordering`);
    rowCount += rows.length;
    for (const size of [1, 2, 3, 7, 1000]) {
      for (let i = 0; i < rows.length; i++) {
        assert.deepStrictEqual(
          page(tenant, rows[i], size),
          rows.slice(i + 1, i + 1 + size),
          `${name}: tenant ${tenant}, cursor ${rows[i].id}, size ${size}`,
        );
        cursorChecks++;
      }
      // Bootstrap the first page from current.sql; NULL priority is a real
      // cursor value, never a sentinel meaning "first page".
      let batch = query(`${current} LIMIT :page_size`, {':tenant': tenant, ':page_size': size});
      const seen = [...batch];
      let requests = 0;
      while (batch.length) {
        assert.ok(requests++ <= rows.length, `${name}: traversal must terminate`);
        batch = page(tenant, batch.at(-1), size);
        seen.push(...batch);
      }
      assert.deepStrictEqual(seen, rows, `${name}: complete traversal, size ${size}`);
      assert.equal(new Set(seen.map(row => row.id)).size, rows.length, 'no duplicates');
      traversals++;
    }
  }
  suites.push({name, tenants, rowCount});
}

try {
  assert.deepStrictEqual(
    query(current, {':tenant': 71}).map(row => row.id),
    [2n, 1n, 3n, 4n, 6n, 5n, 7n],
  );
  exercise('provided seed', [71, 72]);

  const seedRows = query(current, {':tenant': 71});
  for (const cursorId of [1n, 3n, 6n]) {
    const index = seedRows.findIndex(row => row.id === cursorId);
    const cursor = seedRows[index];
    const actual = query(proposed, {
      ':tenant': 71, ':cursor_priority': cursor.priority,
      ':cursor_created': cursor.created_at, ':cursor_id': cursor.id, ':page_size': 20,
    });
    const expected = seedRows.slice(index + 1);
    assert.notDeepStrictEqual(actual, expected, 'proposal counterexample must reproduce');
    assert.deepStrictEqual(page(71, cursor, 20), expected, 'solution fixes counterexample');
    proposalCounterexamples.push({
      cursorId: String(cursorId),
      expectedIds: expected.map(row => String(row.id)),
      proposedIds: actual.map(row => String(row.id)),
    });
  }

  // Add hostile data only to the disposable in-memory DB; preserve all files.
  const insert = db.prepare('INSERT INTO tickets VALUES (?, ?, ?, ?, ?)');
  let id = 100;
  for (const tenant of [71, 72]) {
    for (const priority of [2147483647, 3, 2, 0, -1, -2147483648, null]) {
      for (const created of ['2025-12-31', '2026-01-01', '2026-01-03']) {
        for (let tie = 0; tie < 3; tie++) {
          const rowId = id++;
          insert.run(rowId, tenant, priority, created, `tenant=${tenant};row=${rowId};'quoted'`);
        }
      }
    }
  }
  for (const bigId of [9223372036854775807n, 9007199254740993n, -9007199254740993n, -9223372036854775808n]) {
    insert.run(bigId, 71, -2147483648, '2026-01-01', `bigint ${bigId}`);
  }
  insert.run(-1, 73, null, '2026-01-01', 'all NULL first');
  insert.run(-2, 73, null, '2026-01-01', 'all NULL tie');
  insert.run(-3, 73, null, '2026-01-02', 'all NULL later');
  insert.run(-4, 74, -2147483648, '2026-01-01', 'single row');
  exercise('negative, extrema, NULL, ties, bigint, isolation', [71, 72, 73, 74, 999]);

  // No valid previous row exists for an empty tenant, but filtering must still
  // never expose another tenant if such a bound request reaches the query.
  for (const cursor of [
    {priority: null, created_at: '2026-01-01', id: 5},
    {priority: 3, created_at: '2026-01-01', id: 1},
  ]) {
    assert.deepStrictEqual(page(999, cursor, 7), []);
    emptyChecks++;
  }
  console.log(JSON.stringify({
    status: 'passed', node: process.version,
    sqlite: query('SELECT sqlite_version() AS version')[0].version,
    cursorChecks, traversals, emptyChecks, suites, proposalCounterexamples,
    scope: 'SQLite result semantics only; no PostgreSQL execution or performance claim',
  }, null, 2));
} finally {
  db.close();
}
