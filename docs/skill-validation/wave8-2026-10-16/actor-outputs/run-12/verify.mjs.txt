import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { createDb, query } from './fixture.mjs';

// Every regression executes the on-disk candidate; no candidate SQL is embedded here.
const candidate = readFileSync(new URL('./solution.sql', import.meta.url), 'utf8');
const original = readFileSync(new URL('./current.sql', import.meta.url), 'utf8');
const db = createDb();
let checks = 0;
const expectedRows = labels => labels.map(customer_label => ({ customer_label }));
function check(name, tenant, labels) {
  const expected = expectedRows(labels);
  assert.deepEqual(query(db, original, { tenant }), expected, `${name}: original/oracle`);
  assert.deepEqual(query(db, candidate, { tenant }), expected, `${name}: solution.sql`);
  checks += 1;
  console.log(`PASS ${name}`);
}
const insert = db.prepare('INSERT INTO invoices(id, tenant_id, customer_label, active, amount) VALUES (?, ?, ?, ?, ?)');
try {
  console.log(`Node ${process.version}; SQLite ${db.prepare('SELECT sqlite_version() AS v').get().v}`);
  check('seed preserves repeated Acme and overlap once', 71, ['Acme', 'Acme', 'Bravo']);
  check('seed other tenant', 72, ['Other']);
  check('seed absent tenant', 999, []);

  insert.run(6, 71, 'NullQualified', null, 100);
  insert.run(7, 71, 'NullBelow', null, 99.99);
  insert.run(8, 72, 'ForeignNull', null, 150);
  check('nullable active qualifies exactly at threshold', 71, ['Acme', 'Acme', 'Bravo', 'NullQualified']);
  check('nullable active on other bound tenant', 72, ['ForeignNull', 'Other']);
  check('null tenant bind matches no rows', null, []);

  // Exercise every permitted active value and the amount predicate's boundary partitions.
  for (const active of [1, 0, null]) {
    for (const amount of [-9999999999.99, 0, 99.99, 100, 100.01, 9999999999.99]) {
      db.exec('DELETE FROM invoices');
      insert.run(1, 71, 'Target', active, amount);
      insert.run(2, 72, 'ForeignActive', 1, 20);
      insert.run(3, 72, 'ForeignAmount', 0, 150);
      insert.run(4, 72, 'ForeignNull', null, 150);
      const qualifies = active === 1 || amount >= 100;
      check(`active=${active}, amount=${amount}`, 71, qualifies ? ['Target'] : []);
      check(`foreign branch isolation active=${active}, amount=${amount}`, 72,
        ['ForeignActive', 'ForeignAmount', 'ForeignNull']);
    }
  }

  db.exec('DELETE FROM invoices');
  // Deliberately unsorted insertion; identical labels within and across disjoint branches.
  const rows = [
    [1, 71, 'Zulu', 1, 150],
    [2, 71, 'Acme', 0, 100],
    [3, 71, 'Acme', null, 100],
    [4, 71, 'Acme', 1, 20],
    [5, 71, 'Acme', 1, 150],
    [6, 71, 'Acme', 1, 150],
    [7, 71, 'Bravo', null, 100.01],
    [8, 71, 'Excluded', 0, 99.99],
    [9, 71, 'Excluded', null, 99.99],
    [10, 72, 'AAAA-Foreign', 1, 150],
    [11, 72, 'AAAA-Foreign', 0, 150],
  ];
  for (const row of rows) insert.run(...row);
  check('bag, overlap, projection, global ordering and tenant together', 71,
    ['Acme', 'Acme', 'Acme', 'Acme', 'Acme', 'Bravo', 'Zulu']);
  check('duplicate labels for another tenant', 72, ['AAAA-Foreign', 'AAAA-Foreign']);
  check('empty tenant with populated table', 999, []);
  db.exec('DELETE FROM invoices');
  check('empty table', 71, []);
  console.log(`PASS ${checks} semantic checks; candidate read from solution.sql`);
} finally {
  db.close();
}
