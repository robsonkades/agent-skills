import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';

const output = new URL('./', import.meta.url);
const input = new URL('../input/', import.meta.url);
const read = (base, name) => readFileSync(new URL(name, base));
const json = (base, name) => JSON.parse(read(base, name).toString('utf8'));
const digest = bytes => createHash('sha256').update(bytes).digest('hex');
const names = ['README.md', 'REQUEST.md', 'incident.json', 'preview-plan.json', 'literal-comparison.json', 'proposal.md'];
for (const name of names) assert.equal(digest(read(input, name)), digest(read(output, name)), `${name} copy changed`);
const incident = json(input, 'incident.json');
const inner = incident.plan.children[1];
const result = json(output, 'result.json');
assert.deepEqual(result.metrics, {
  innerExecutions: inner.loops,
  innerEmittedRows: inner.actualRows * inner.loops,
  innerInclusiveWorkMs: inner.actualTotalTimeMs * inner.loops,
  capturedExecutionTimeMs: incident.executionTimeMs,
});
assert.ok(result.decision.length > 0);
assert.ok(result.evidence.length > 0);
assert.ok(result.limits.length > 0);
for (const key of Object.keys(result.metrics)) assert.ok(result.metricInterpretation[key].length > 0);
assert.equal(result.nextExperiment.status, 'proposed_only_not_executed');
assert.equal(result.verification.sqlExecution, false);
const words = read(output, 'REVIEW.md').toString('utf8').trim().split(/\s+/u).length;
assert.ok(words <= 750, `REVIEW.md has ${words} words`);
console.log(JSON.stringify({passed: true, fixtureCopiesVerified: names.length, metrics: result.metrics, reviewWords: words, sqlExecuted: false}));
