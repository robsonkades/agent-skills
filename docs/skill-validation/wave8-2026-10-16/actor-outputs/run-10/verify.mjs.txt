import assert from 'node:assert/strict';
import fs from 'node:fs';
import {createDb, query} from './fixture.mjs';

// Always exercise the on-disk candidate; callers may replace solution.sql.
const candidate = fs.readFileSync(new URL('./solution.sql', import.meta.url), 'utf8');
const original = fs.readFileSync(new URL('./current.sql', import.meta.url), 'utf8');
const db = createDb();
let comparisons = 0;
const rows = labels => labels.map(customer_label => ({customer_label}));
function check(name, tenant, expected) {
  const actual = query(db, candidate, {tenant});
  assert.deepEqual(actual, expected, `${name}: candidate versus independent expected bag/order/projection`);
  assert.deepEqual(actual, query(db, original, {tenant}), `${name}: candidate versus current.sql`);
  comparisons += 1;
}

try {
  check('seed duplicate labels and overlap', 71, rows(['Acme', 'Acme', 'Bravo']));
  check('seed other tenant', 72, rows(['Other']));
  check('seed missing tenant', 999, []);
  check('null binding matches no tenant', null, []);

  db.exec('DELETE FROM invoices');
  const insert = db.prepare('INSERT INTO invoices (id, tenant_id, customer_label, active, amount) VALUES (?, ?, ?, ?, ?)');
  const cents = [-999999999999, -1, 0, 9999, 10000, 10001, 999999999999];
  const tenants = [-71, 0, 71, 72];
  const fixtureRows = [];
  let id = 1;
  for (const tenant of tenants) {
    for (const active of [false, true]) {
      for (const [index, amountCents] of cents.entries()) {
        // Reverse labels force a global sort. Two invoices share each label.
        const label = `L${cents.length - index}${active ? 'T' : 'F'}`;
        for (let duplicate = 0; duplicate < 2; duplicate += 1) {
          insert.run(id++, tenant, label, Number(active), amountCents / 100);
          fixtureRows.push({tenant, active, amountCents, label});
        }
      }
    }
  }
  for (const tenant of [...tenants, 999, null]) {
    const expected = fixtureRows
      .filter(row => row.tenant === tenant && (row.active || row.amountCents >= 10000))
      .map(row => row.label)
      .sort();
    check(`boolean/amount matrix for tenant ${tenant}`, tenant, rows(expected));
  }
  assert.throws(() => insert.run(id, 71, 'Invalid null', null, 150), /NOT NULL/);
  assert.throws(() => insert.run(id, 71, 'Invalid nonboolean', 2, 150), /CHECK/);
  db.exec('DELETE FROM invoices');
  check('empty relation', 71, []);

  console.log(JSON.stringify({
    status: 'passed', comparisons, generatedRows: fixtureRows.length,
    node: process.version, sqlite: db.prepare('SELECT sqlite_version() AS version').get().version,
    scope: 'Portable result semantics only; no PostgreSQL execution or performance claims'
  }, null, 2));
} finally {
  db.close();
}
