import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';

// Artifact checks only: this program never opens a database or executes SQL.
const output = new URL('./', import.meta.url);
const input = new URL('../input/', import.meta.url);
const names = ['README.md', 'REQUEST.md', 'proposal.md', 'incident.json', 'preview-plan.json', 'literal-comparison.json'];
for (const name of names) {
  assert.deepEqual(readFileSync(new URL(name, output)), readFileSync(new URL(name, input)), `${name} changed`);
}
const readJson = name => JSON.parse(readFileSync(new URL(name, output), 'utf8'));
const incident = readJson('incident.json');
const preview = readJson('preview-plan.json');
const literal = readJson('literal-comparison.json');
const result = readJson('result.json');
const [outer, inner] = incident.plan.children;
assert.deepEqual(result.metrics, {
  innerExecutions: inner.loops,
  innerEmittedRows: inner.actualRows * inner.loops,
  innerInclusiveWorkMs: inner.actualTotalTimeMs * inner.loops,
  capturedExecutionTimeMs: incident.executionTimeMs,
});
assert.equal(result.derivedMetrics.ordersScanMs, outer.actualTotalTimeMs * outer.loops);
assert.equal(result.derivedMetrics.ordersRetainedFraction, outer.actualRows / (outer.actualRows + outer.rowsRemovedByFilter));
assert.equal(result.derivedMetrics.joinEstimateToActualRatio, incident.plan.estimatedRows / incident.plan.actualRows);
assert.equal(result.derivedMetrics.ordersEstimateToActualRatio, outer.estimatedRows / outer.actualRows);
assert.equal(result.derivedMetrics.endToEndRequestMs, incident.endToEndRequestMs);
assert.equal(result.derivedMetrics.unattributedRequestGapMs, incident.endToEndRequestMs - incident.executionTimeMs);
assert.equal(result.derivedMetrics.incidentToLiteralResultVolumeRatio, incident.plan.actualRows / literal.returnedRows);
assert.equal(preview.plan.child.stoppedByParent, true);
assert.equal(literal.preparedStatementSameParameterCaptured, false);
assert.ok(result.evidence.length > 0 && result.limits.length > 0);
assert.equal(result.nextExperiment.status, 'proposed_not_executed');
for (const key of ['runtimeSettingChanges', 'addIndexes', 'addLimit', 'flushCaches']) {
  assert.equal(result.nextExperiment[key], false);
}
const words = readFileSync(new URL('REVIEW.md', output), 'utf8').trim().split(/\s+/u).length;
assert.ok(words <= 750, `REVIEW.md has ${words} words`);
console.log(JSON.stringify({ status: 'passed', copiedFilesPreserved: names.length, metrics: result.metrics, reviewWords: words, sqlExecuted: false }));
