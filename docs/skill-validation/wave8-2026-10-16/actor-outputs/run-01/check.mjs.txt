import assert from 'node:assert/strict';
import fs from 'node:fs';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const here = fileURLToPath(new URL('.', import.meta.url));
const protectedFiles = ['README.md', 'REQUEST.md', 'schema.sql', 'seed.sql', 'current.sql', 'proposed.sql', 'fixture.mjs'];
const integrity = [];
for (const name of protectedFiles) {
  const input = fs.readFileSync(new URL(`../input/${name}`, import.meta.url));
  const output = fs.readFileSync(new URL(name, import.meta.url));
  assert.deepEqual(output, input, `${name}: copied counterpart must be byte-identical`);
  integrity.push({ file: name, sha256: createHash('sha256').update(input).digest('hex'), matches: true });
}
function execute(logName) {
  const result = spawnSync(process.execPath, ['verify.mjs'], {
    cwd: here,
    timeout: 20000,
    encoding: 'utf8',
    windowsHide: true,
  });
  fs.writeFileSync(new URL(`${logName}.stdout.log`, import.meta.url), result.stdout || '');
  fs.writeFileSync(new URL(`${logName}.stderr.log`, import.meta.url), result.stderr || '');
  if (result.error) throw result.error;
  return result;
}
const approvedRun = execute('verification');
assert.equal(approvedRun.status, 0, approvedRun.stderr);
const approved = JSON.parse(approvedRun.stdout);
const solutionUrl = new URL('solution.sql', import.meta.url);
const original = fs.readFileSync(solutionUrl);
assert.deepEqual(original, fs.readFileSync(new URL('proposed.sql', import.meta.url)), 'retain exact proposal bytes');
let rejectedRun;
try {
  fs.writeFileSync(solutionUrl, original.toString('utf8').replace('UNION ALL', 'UNION'));
  rejectedRun = execute('external-mutation');
  assert.equal(rejectedRun.status, 1, 'replaced solution.sql must fail verification');
  assert.match(rejectedRun.stderr, /AssertionError/);
  assert.match(rejectedRun.stderr, /solution\.sql: seed\/tenant-71/);
} finally {
  fs.writeFileSync(solutionUrl, original);
}
assert.deepEqual(fs.readFileSync(solutionUrl), original, 'restore exact approved candidate');
const report = {
  verification: approved,
  positiveExit: approvedRun.status,
  externalMutation: { change: 'UNION ALL to UNION in solution.sql', exit: rejectedRun.status, rejected: true },
  restoredCandidateSha256: createHash('sha256').update(original).digest('hex'),
  protectedCopies: integrity,
  subprocessTimeoutMs: 20000,
};
fs.writeFileSync(new URL('checks.json', import.meta.url), `${JSON.stringify(report, null, 2)}\n`);
console.log(JSON.stringify(report, null, 2));
